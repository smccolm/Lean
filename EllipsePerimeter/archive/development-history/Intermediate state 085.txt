def shapeParam (a b : ℝ) : ℝ :=
  |a - b| / (a + b)

def kappa (lamPi : ℝ) : ℝ :=
  (4 * lamPi) / (1 + lamPi)^2

def omegaIntegrand (lamPi : ℝ) (t : ℝ) : ℝ :=
  Real.sqrt (1 - kappa lamPi * t) / Real.sqrt (t * (1 - t))

def Omega (lamPi : ℝ) : ℝ :=
  (1 + lamPi) * ∫ t in (0 : ℝ)..1, omegaIntegrand lamPi t

def circleIntegrand (t : ℝ) : ℝ :=
  1 / Real.sqrt (t * (1 - t))

def circleIntegral : ℝ :=
  ∫ t in (0 : ℝ)..1, circleIntegrand t

theorem shapeParam_self (a : ℝ) : shapeParam a a = 0 := by
  simp [shapeParam]

theorem shapeParam_symm (a b : ℝ) : shapeParam a b = shapeParam b a := by
  simp [shapeParam, abs_sub_comm, add_comm]

theorem kappa_zero : kappa 0 = 0 := by
  simp [kappa]

theorem kappa_nonneg {lamPi : ℝ} (h0 : 0 ≤ lamPi) : 0 ≤ kappa lamPi := by
  unfold kappa
  exact div_nonneg (by nlinarith) (sq_nonneg (1 + lamPi))

theorem kappa_le_one {lamPi : ℝ} (h0 : 0 ≤ lamPi) :
    kappa lamPi ≤ 1 := by
  unfold kappa
  have hnum : 4 * lamPi ≤ (1 + lamPi)^2 := by
    have hs : 0 ≤ (1 - lamPi)^2 := sq_nonneg (1 - lamPi)
    nlinarith
  have hdiv :
      (4 * lamPi) / (1 + lamPi)^2 ≤ (1 + lamPi)^2 / (1 + lamPi)^2 := by
    exact div_le_div_of_nonneg_right hnum (sq_nonneg (1 + lamPi))
  have hne : (1 + lamPi)^2 ≠ 0 := by
    nlinarith
  simpa [hne] using hdiv

theorem one_sub_kappa_mul_nonneg {lamPi t : ℝ}
    (h0 : 0 ≤ lamPi) (_h1 : lamPi ≤ 1) (ht0 : 0 ≤ t) (ht1 : t ≤ 1) :
    0 ≤ 1 - kappa lamPi * t := by
  have hk0 : 0 ≤ kappa lamPi := kappa_nonneg h0
  have hk1 : kappa lamPi ≤ 1 := kappa_le_one h0
  have hkt : kappa lamPi * t ≤ 1 := by
    nlinarith
  nlinarith

theorem omegaIntegrand_nonneg (lamPi t : ℝ) : 0 ≤ omegaIntegrand lamPi t := by
  unfold omegaIntegrand
  exact div_nonneg (Real.sqrt_nonneg _) (Real.sqrt_nonneg _)

theorem circleIntegrand_nonneg (t : ℝ) : 0 ≤ circleIntegrand t := by
  unfold circleIntegrand
  exact div_nonneg zero_le_one (Real.sqrt_nonneg _)

theorem omegaIntegrand_zero (t : ℝ) :
    omegaIntegrand 0 t = circleIntegrand t := by
  simp [omegaIntegrand, kappa, circleIntegrand]

theorem Omega_zero_unfold :
    Omega 0 = circleIntegral := by
  simp [Omega, omegaIntegrand, kappa, circleIntegral, circleIntegrand]

theorem Omega_circle_unfold (a : ℝ) :
    Omega (shapeParam a a) = circleIntegral := by
  simp [Omega, omegaIntegrand, kappa, shapeParam,
    circleIntegral, circleIntegrand]

theorem Omega_circle_eq_Omega_zero (a : ℝ) :
    Omega (shapeParam a a) = Omega 0 := by
  simp [shapeParam]

theorem Omega_zero_eq_pi_iff_circleIntegral_eq_pi :
    Omega 0 = Real.pi ↔ circleIntegral = Real.pi := by
  constructor <;> intro h
  · simpa [Omega_zero_unfold] using h
  · simpa [Omega_zero_unfold] using h

theorem Omega_zero_eq_pi_of_circleIntegral_eq_pi
    (h : circleIntegral = Real.pi) :
    Omega 0 = Real.pi := by
  simpa [Omega_zero_unfold] using h

theorem Omega_circle_eq_pi_iff_circleIntegral_eq_pi (a : ℝ) :
    Omega (shapeParam a a) = Real.pi ↔ circleIntegral = Real.pi := by
  constructor <;> intro h
  · simpa [Omega_circle_unfold] using h
  · simpa [Omega_circle_unfold] using h

theorem Omega_circle_eq_pi_of_circleIntegral_eq_pi
    (a : ℝ) (h : circleIntegral = Real.pi) :
    Omega (shapeParam a a) = Real.pi := by
  simpa [Omega_circle_unfold] using h

def unitCircleAntideriv (x : ℝ) : ℝ :=
  Real.arcsin x

theorem unitCircleAntideriv_hasDerivAt
    {x : ℝ} (hx₁ : x ≠ -1) (hx₂ : x ≠ 1) :
    HasDerivAt unitCircleAntideriv
      (1 / Real.sqrt (1 - x ^ 2)) x := by
  simpa [unitCircleAntideriv] using
    Real.hasDerivAt_arcsin hx₁ hx₂

theorem unitCircleAntideriv_hasDerivAt_Ioo
    {x : ℝ} (hx : x ∈ Set.Ioo (-1 : ℝ) 1) :
    HasDerivAt unitCircleAntideriv
      (1 / Real.sqrt (1 - x ^ 2)) x := by
  have hx₁ : x ≠ -1 := by
    linarith [hx.1]
  have hx₂ : x ≠ 1 := by
    linarith [hx.2]
  exact unitCircleAntideriv_hasDerivAt hx₁ hx₂

theorem unitCircleAntideriv_one :
    unitCircleAntideriv 1 = Real.pi / 2 := by
  simp [unitCircleAntideriv]

theorem unitCircleAntideriv_neg_one :
    unitCircleAntideriv (-1) = -(Real.pi / 2) := by
  simp [unitCircleAntideriv]

theorem unitCircleAntideriv_endpoint_gap :
    unitCircleAntideriv 1 - unitCircleAntideriv (-1) = Real.pi := by
  rw [unitCircleAntideriv_one, unitCircleAntideriv_neg_one]
  ring

def unitCircleIntegrand (x : ℝ) : ℝ :=
  1 / Real.sqrt (1 - x ^ 2)

theorem unitCircleIntegrand_nonneg (x : ℝ) :
    0 ≤ unitCircleIntegrand x := by
  unfold unitCircleIntegrand
  exact div_nonneg zero_le_one (Real.sqrt_nonneg _)

theorem affine_to_unitCircle_mem_Ioo {t : ℝ}
    (ht : t ∈ Set.Ioo (0 : ℝ) 1) :
    2 * t - 1 ∈ Set.Ioo (-1 : ℝ) 1 := by
  constructor <;> nlinarith [ht.1, ht.2]

theorem one_sub_affine_sq_eq_four_mul (t : ℝ) :
    1 - (2 * t - 1) ^ 2 = 4 * (t * (1 - t)) := by
  ring

theorem sqrt_one_sub_affine_sq (t : ℝ) :
    Real.sqrt (1 - (2 * t - 1) ^ 2) =
      2 * Real.sqrt (t * (1 - t)) := by
  rw [one_sub_affine_sq_eq_four_mul]
  calc
    Real.sqrt (4 * (t * (1 - t)))
        = Real.sqrt (4 : ℝ) * Real.sqrt (t * (1 - t)) := by
          have h4 : 0 ≤ (4 : ℝ) := by
            norm_num
          rw [Real.sqrt_mul h4]
    _ = 2 * Real.sqrt (t * (1 - t)) := by
          have hsqrt4 : Real.sqrt (4 : ℝ) = 2 := by
            have h : (4 : ℝ) = 2 ^ 2 := by
              norm_num
            have h2 : 0 ≤ (2 : ℝ) := by
              norm_num
            rw [h, Real.sqrt_sq h2]
          rw [hsqrt4]

theorem circleIntegrand_eq_two_mul_unitCircleIntegrand_affine
    {t : ℝ} (ht : t ∈ Set.Ioo (0 : ℝ) 1) :
    circleIntegrand t = 2 * unitCircleIntegrand (2 * t - 1) := by
  have hspos : 0 < Real.sqrt (t * (1 - t)) := by
    apply Real.sqrt_pos.2
    nlinarith [ht.1, ht.2]
  have hsneq : Real.sqrt (t * (1 - t)) ≠ 0 := by
    linarith
  unfold circleIntegrand unitCircleIntegrand
  have hsqrt := sqrt_one_sub_affine_sq t
  calc
    1 / Real.sqrt (t * (1 - t))
        = 2 / (2 * Real.sqrt (t * (1 - t))) := by
          field_simp [hsneq]
    _ = 2 / Real.sqrt (1 - (2 * t - 1) ^ 2) := by
          rw [hsqrt]
    _ = 2 * (1 / Real.sqrt (1 - (2 * t - 1) ^ 2)) := by
          ring

theorem circleIntegrand_eq_two_mul_unitCircleIntegrand_affine_Icc
    {t : ℝ} (ht : t ∈ Set.Icc (0 : ℝ) 1) :
    circleIntegrand t = 2 * unitCircleIntegrand (2 * t - 1) := by
  rcases lt_or_eq_of_le ht.1 with ht0 | rfl
  · rcases lt_or_eq_of_le ht.2 with ht1 | rfl
    · exact circleIntegrand_eq_two_mul_unitCircleIntegrand_affine ⟨ht0, ht1⟩
    · unfold circleIntegrand unitCircleIntegrand
      have h1 : Real.sqrt (1 * (1 - 1)) = 0 := by
        norm_num
      have h2 : Real.sqrt (1 - (2 * 1 - 1) ^ 2) = 0 := by
        norm_num
      rw [h1, h2]
      norm_num
  · unfold circleIntegrand unitCircleIntegrand
    have h1 : Real.sqrt (0 * (1 - 0)) = 0 := by
      norm_num
    have h2 : Real.sqrt (1 - (2 * 0 - 1) ^ 2) = 0 := by
      norm_num
    rw [h1, h2]
    norm_num

theorem unitCircleIntegrand_intervalIntegrable :
    IntervalIntegrable unitCircleIntegrand MeasureTheory.volume (-1 : ℝ) 1 := by
  apply intervalIntegral.intervalIntegrable_deriv_of_nonneg
  · have h_eq : Set.uIcc (-1 : ℝ) 1 = Set.Icc (-1 : ℝ) 1 :=
      Set.uIcc_of_le (by norm_num)
    rw [h_eq]
    exact Real.continuous_arcsin.continuousOn
  · intro x hx
    have hx_eq :
        Set.Ioo (min (-1 : ℝ) 1) (max (-1 : ℝ) 1) = Set.Ioo (-1 : ℝ) 1 := by
      congr 1 <;> norm_num
    rw [hx_eq] at hx
    exact unitCircleAntideriv_hasDerivAt_Ioo hx
  · intro x _hx
    exact unitCircleIntegrand_nonneg x

theorem unitCircleIntegral_eq_endpoint_gap :
    (∫ x in (-1 : ℝ)..1, unitCircleIntegrand x) =
      unitCircleAntideriv 1 - unitCircleAntideriv (-1) := by
  apply intervalIntegral.integral_eq_sub_of_hasDerivAt_of_le (by norm_num)
  any_goals exact unitCircleIntegrand_intervalIntegrable
  any_goals exact Real.continuous_arcsin.continuousOn
  any_goals
    intro x hx
    exact unitCircleAntideriv_hasDerivAt_Ioo hx

theorem circleIntegral_eq_unitCircleIntegral :
    circleIntegral = ∫ x in (-1 : ℝ)..1, unitCircleIntegrand x := by
  calc
    circleIntegral = ∫ t in (0 : ℝ)..1, circleIntegrand t := rfl
    _ = ∫ t in (0 : ℝ)..1, (2 : ℝ) • unitCircleIntegrand (2 * t + (-1)) := by
          apply intervalIntegral.integral_congr
          intro t ht
          have ht' : t ∈ Set.Icc (0 : ℝ) 1 := by
            have h : Set.uIcc (0 : ℝ) 1 = Set.Icc (0 : ℝ) 1 :=
              Set.uIcc_of_le (by norm_num)
            rwa [h] at ht
          have h_val := circleIntegrand_eq_two_mul_unitCircleIntegrand_affine_Icc ht'
          calc
            circleIntegrand t = 2 * unitCircleIntegrand (2 * t - 1) := h_val
            _ = (2 : ℝ) • unitCircleIntegrand (2 * t + (-1)) := rfl
    _ = (2 : ℝ) • ∫ t in (0 : ℝ)..1, unitCircleIntegrand (2 * t + (-1)) := by
          rw [intervalIntegral.integral_smul]
    _ = ∫ x in (2 * 0 + (-1) : ℝ)..(2 * 1 + (-1)), unitCircleIntegrand x := by
          exact intervalIntegral.smul_integral_comp_mul_add unitCircleIntegrand
            (a := (0 : ℝ)) (b := (1 : ℝ)) (c := (2 : ℝ)) (d := (-1 : ℝ))
    _ = ∫ x in (-1 : ℝ)..1, unitCircleIntegrand x := by
          congr 1 <;> norm_num

theorem circleIntegral_eq_pi :
    circleIntegral = Real.pi := by
  rw [circleIntegral_eq_unitCircleIntegral, unitCircleIntegral_eq_endpoint_gap]
  rw [unitCircleAntideriv_endpoint_gap]

theorem Omega_zero_eq_pi :
    Omega 0 = Real.pi := by
  exact Omega_zero_eq_pi_of_circleIntegral_eq_pi circleIntegral_eq_pi

theorem Omega_circle_eq_pi (a : ℝ) :
    Omega (shapeParam a a) = Real.pi := by
  exact Omega_circle_eq_pi_of_circleIntegral_eq_pi a circleIntegral_eq_pi

def ellipsePerimeterModel (a b : ℝ) : ℝ :=
  (a + b) * Omega (shapeParam a b)

theorem Omega_shapeParam_symm (a b : ℝ) :
    Omega (shapeParam a b) = Omega (shapeParam b a) := by
  rw [shapeParam_symm]

theorem Omega_shapeParam_self_eq_pi (a : ℝ) :
    Omega (shapeParam a a) = Real.pi := by
  simpa [shapeParam_self] using Omega_zero_eq_pi

theorem ellipsePerimeterModel_symm (a b : ℝ) :
    ellipsePerimeterModel a b = ellipsePerimeterModel b a := by
  unfold ellipsePerimeterModel
  rw [shapeParam_symm, add_comm]

theorem ellipsePerimeterModel_of_shapeParam_eq_zero {a b : ℝ}
    (h : shapeParam a b = 0) :
    ellipsePerimeterModel a b = (a + b) * Real.pi := by
  unfold ellipsePerimeterModel
  rw [h, Omega_zero_eq_pi]

theorem ellipsePerimeterModel_circle (a : ℝ) :
    ellipsePerimeterModel a a = (2 * a) * Real.pi := by
  unfold ellipsePerimeterModel
  rw [shapeParam_self, Omega_zero_eq_pi]
  ring

theorem ellipsePerimeterModel_circle' (a : ℝ) :
    ellipsePerimeterModel a a = 2 * Real.pi * a := by
  rw [ellipsePerimeterModel_circle]
  ring

theorem shapeParam_nonneg_of_pos_sum {a b : ℝ}
    (hsum : 0 < a + b) :
    0 ≤ shapeParam a b := by
  unfold shapeParam
  exact div_nonneg (abs_nonneg (a - b)) (le_of_lt hsum)

theorem shapeParam_le_one_of_nonneg {a b : ℝ}
    (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    shapeParam a b ≤ 1 := by
  unfold shapeParam
  have hab : |a - b| ≤ a + b := by
    refine abs_le.mpr ?_
    constructor <;> nlinarith
  have haux : |a - b| / (a + b) ≤ (a + b) / (a + b) := by
    exact div_le_div_of_nonneg_right hab (le_of_lt hsum)
  have hne : a + b ≠ 0 := ne_of_gt hsum
  simpa [hne] using haux

theorem shapeParam_mem_Icc_of_nonneg {a b : ℝ}
    (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    shapeParam a b ∈ Set.Icc (0 : ℝ) 1 := by
  constructor
  · exact shapeParam_nonneg_of_pos_sum hsum
  · exact shapeParam_le_one_of_nonneg ha hb hsum

theorem shapeParam_eq_zero_iff_of_pos_sum {a b : ℝ}
    (hsum : 0 < a + b) :
    shapeParam a b = 0 ↔ a = b := by
  constructor
  · intro h
    have hzero : |a - b| = 0 := by
      by_contra hne
      have hdivne : |a - b| / (a + b) ≠ 0 := by
        exact div_ne_zero hne (ne_of_gt hsum)
      exact hdivne (by simpa [shapeParam] using h)
    have hab : a - b = 0 := abs_eq_zero.mp hzero
    linarith
  · intro h
    simp [shapeParam, h]

theorem shapeParam_eq_zero_iff_of_nonneg_of_pos_sum {a b : ℝ}
    (_ha : 0 ≤ a) (_hb : 0 ≤ b) (hsum : 0 < a + b) :
    shapeParam a b = 0 ↔ a = b := by
  exact shapeParam_eq_zero_iff_of_pos_sum hsum

theorem ellipsePerimeterModel_eq_sum_mul_pi_of_eq {a b : ℝ}

    ellipsePerimeterModel a b = (a + b) * Real.pi := by

theorem ellipsePerimeterModel_eq_sum_mul_pi_of_shapeParam_zero {a b : ℝ}

    ellipsePerimeterModel a b = (a + b) * Real.pi := by

theorem shapeParam_eq_sub_div_of_le {a b : ℝ}
    (h : b ≤ a) :
    shapeParam a b = (a - b) / (a + b) := by
  unfold shapeParam
  rw [abs_of_nonneg (sub_nonneg.mpr h)]

theorem shapeParam_eq_rev_sub_div_of_le {a b : ℝ}
    (h : a ≤ b) :
    shapeParam a b = (b - a) / (a + b) := by
  unfold shapeParam
  rw [abs_of_nonpos (sub_nonpos.mpr h)]
  congr 1
  ring

theorem shapeParam_right_zero_of_pos {a : ℝ}
    (ha : 0 < a) :
    shapeParam a 0 = 1 := by
  rw [shapeParam_eq_sub_div_of_le (by linarith)]
  simp [ha.ne']

theorem shapeParam_left_zero_of_pos {b : ℝ}
    (hb : 0 < b) :
    shapeParam 0 b = 1 := by
  rw [shapeParam_eq_rev_sub_div_of_le (by linarith)]
  simp [hb.ne']

theorem ellipsePerimeterModel_right_zero_of_pos {a : ℝ}

theorem ellipsePerimeterModel_left_zero_of_pos {b : ℝ}

theorem shapeParam_scale_of_pos {a b c : ℝ}
    (hc : 0 < c) :
    shapeParam (c * a) (c * b) = shapeParam a b := by
  unfold shapeParam
  have hsub : c * a - c * b = c * (a - b) := by
    ring
  have hadd : c * a + c * b = c * (a + b) := by
    ring
  rw [hsub, hadd, abs_mul, abs_of_nonneg (le_of_lt hc)]
  by_cases hab : a + b = 0
  · simp [hab]
  · have hc0 : c ≠ 0 := ne_of_gt hc
    field_simp [hc0, hab]

theorem ellipsePerimeterModel_scale_of_pos {a b c : ℝ}
    (hc : 0 < c) :
    ellipsePerimeterModel (c * a) (c * b) = c * ellipsePerimeterModel a b := by
  unfold ellipsePerimeterModel
  rw [shapeParam_scale_of_pos hc]
  ring

def eccentricitySq (a b : ℝ) : ℝ :=
  1 - (b / a) ^ 2

theorem one_add_shapeParam_eq_two_mul_div_of_le {a b : ℝ}
    (hba : b ≤ a) (hsum : a + b ≠ 0) :
    1 + shapeParam a b = (2 * a) / (a + b) := by
  rw [shapeParam_eq_sub_div_of_le hba]
  field_simp [hsum]
  ring

theorem kappa_shapeParam_eq_one_sub_sq_div_sq_of_le {a b : ℝ}
    (ha : 0 < a) (hb : 0 ≤ b) (hba : b ≤ a) :
    kappa (shapeParam a b) = 1 - (b / a) ^ 2 := by
  have hsum : a + b ≠ 0 := by
    linarith
  have ha0 : a ≠ 0 := ne_of_gt ha
  rw [shapeParam_eq_sub_div_of_le hba, kappa]
  have hden : 1 + (a - b) / (a + b) = (2 * a) / (a + b) := by
    field_simp [hsum]
    ring
  rw [hden]
  field_simp [ha0, hsum]
  ring

theorem kappa_shapeParam_eq_sub_sq_div_sq_of_le {a b : ℝ}
    (ha : 0 < a) (hb : 0 ≤ b) (hba : b ≤ a) :
    kappa (shapeParam a b) = (a ^ 2 - b ^ 2) / a ^ 2 := by
  rw [kappa_shapeParam_eq_one_sub_sq_div_sq_of_le ha hb hba]
  have ha0 : a ≠ 0 := ne_of_gt ha
  field_simp [ha0]

theorem kappa_shapeParam_eq_eccentricitySq_of_le {a b : ℝ}
    (ha : 0 < a) (hb : 0 ≤ b) (hba : b ≤ a) :
    kappa (shapeParam a b) = eccentricitySq a b := by
  unfold eccentricitySq
  exact kappa_shapeParam_eq_one_sub_sq_div_sq_of_le ha hb hba

theorem omegaIntegrand_shapeParam_eq_eccentricitySq_of_le {a b t : ℝ}
    (ha : 0 < a) (hb : 0 ≤ b) (hba : b ≤ a) :
    omegaIntegrand (shapeParam a b) t =
      Real.sqrt (1 - eccentricitySq a b * t) /
        Real.sqrt (t * (1 - t)) := by
  unfold omegaIntegrand eccentricitySq
  rw [kappa_shapeParam_eq_one_sub_sq_div_sq_of_le ha hb hba]

theorem ellipsePerimeterModel_eq_majorAxisIntegral_of_le {a b : ℝ}
    (ha : 0 < a) (hb : 0 ≤ b) (hba : b ≤ a) :
    ellipsePerimeterModel a b =
      2 * a * ∫ t in (0 : ℝ)..1, omegaIntegrand (shapeParam a b) t := by
  have hsum : a + b ≠ 0 := by
    linarith
  unfold ellipsePerimeterModel Omega
  rw [one_add_shapeParam_eq_two_mul_div_of_le hba hsum]
  field_simp [hsum]

theorem ellipsePerimeterModel_eq_classicalIntegral_of_le {a b : ℝ}
    (ha : 0 < a) (hb : 0 ≤ b) (hba : b ≤ a) :
    ellipsePerimeterModel a b =
      2 * a *
        ∫ t in (0 : ℝ)..1,
          Real.sqrt (1 - eccentricitySq a b * t) /
            Real.sqrt (t * (1 - t)) := by
  rw [ellipsePerimeterModel_eq_majorAxisIntegral_of_le ha hb hba]
  apply congrArg (fun z : ℝ => 2 * a * z)
  apply intervalIntegral.integral_congr
  intro t ht
  exact omegaIntegrand_shapeParam_eq_eccentricitySq_of_le ha hb hba

theorem ellipsePerimeterModel_eq_classicalIntegral_of_ge {a b : ℝ}
    (ha : 0 ≤ a) (hb : 0 < b) (hab : a ≤ b) :
    ellipsePerimeterModel a b =
      2 * b *
        ∫ t in (0 : ℝ)..1,
          Real.sqrt (1 - eccentricitySq b a * t) /
            Real.sqrt (t * (1 - t)) := by
  rw [ellipsePerimeterModel_symm]
  exact ellipsePerimeterModel_eq_classicalIntegral_of_le hb ha hab

theorem eccentricitySq_nonneg_of_le {a b : ℝ}

theorem eccentricitySq_le_one_of_le {a b : ℝ}

theorem eccentricitySq_self {a : ℝ} (ha : a ≠ 0) :

theorem elliptic_integrand_subst_eq_trig_on_Ioo (k : ℝ) :
    Set.EqOn
      (fun θ =>
        (Real.sqrt (1 - k ^ 2 * sinSqSubst θ) /
          (2 * Real.sqrt (sinSqSubst θ * (1 - sinSqSubst θ)))) *
          (2 * Real.sin θ * Real.cos θ))
      (fun θ => Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2))
      (Set.Ioo (0 : ℝ) (Real.pi / 2)) := by
  intro θ hθ
  have hθ' : θ ∈ Set.Icc (0 : ℝ) (Real.pi / 2) := by
    exact ⟨le_of_lt hθ.1, le_of_lt hθ.2⟩
  have hsin : 0 < Real.sin θ := by
    have hmem : θ ∈ Set.Ioo (0 : ℝ) Real.pi := by
      constructor
      · exact hθ.1
      · nlinarith [hθ.2, Real.pi_pos]
    exact Real.sin_pos_of_mem_Ioo hmem
  have hcos : 0 < Real.cos θ := by
    have hmem : θ ∈ Set.Ioo (-(Real.pi / 2)) (Real.pi / 2) := by
      constructor
      · have hneg : -(Real.pi / 2) < (0 : ℝ) := by
          nlinarith [Real.pi_pos]
        exact lt_trans hneg hθ.1
      · exact hθ.2
    exact Real.cos_pos_of_mem_Ioo hmem
  have hden :
      2 * Real.sqrt (sinSqSubst θ * (1 - sinSqSubst θ)) =
        2 * Real.sin θ * Real.cos θ := by
    calc
      2 * Real.sqrt (sinSqSubst θ * (1 - sinSqSubst θ))
          = 2 * (Real.sin θ * Real.cos θ) := by
              rw
                [sqrt_sinSqSubst_mul_one_sub_sinSqSubst_eq_sin_mul_cos_of_mem_Icc_zero_pi_div_two
                  hθ']
      _ = 2 * Real.sin θ * Real.cos θ := by
            ring
  calc
    (Real.sqrt (1 - k ^ 2 * sinSqSubst θ) /
      (2 * Real.sqrt (sinSqSubst θ * (1 - sinSqSubst θ)))) *
        (2 * Real.sin θ * Real.cos θ)
      =
      (Real.sqrt (1 - k ^ 2 * sinSqSubst θ) /
        (2 * Real.sin θ * Real.cos θ)) *
          (2 * Real.sin θ * Real.cos θ) := by
            rw [hden]
    _ = Real.sqrt (1 - k ^ 2 * sinSqSubst θ) := by
          have hmul : 2 * Real.sin θ * Real.cos θ ≠ 0 := by
            exact mul_ne_zero (mul_ne_zero (by norm_num) hsin.ne') hcos.ne'
          exact div_mul_cancel₀ _ hmul
    _ = Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2) := by
          simp [sinSqSubst]

theorem completeEllipticEParam_integrand_nonneg_on_Ioo
    {k t : ℝ} (_hk0 : 0 ≤ k) (_hk1 : k ≤ 1)

def completeEllipticEHalfIntegrand (k t : ℝ) : ℝ :=
  Real.sqrt (1 - k ^ 2 * t) / (2 * Real.sqrt (t * (1 - t)))

def completeEllipticESubstIntegrand (k θ : ℝ) : ℝ :=
  (2 * Real.sin θ * Real.cos θ) *
    completeEllipticEHalfIntegrand k (sinSqSubst θ)

theorem completeEllipticEHalfIntegrand_eq_const_mul
    (k t : ℝ) :
    completeEllipticEHalfIntegrand k t =
      (1 / 2) *
        (Real.sqrt (1 - k ^ 2 * t) /
          Real.sqrt (t * (1 - t))) := by
  unfold completeEllipticEHalfIntegrand
  by_cases h : Real.sqrt (t * (1 - t)) = 0
  · simp [h]
  · field_simp [h]

theorem completeEllipticEModulus_eq_integral_halfIntegrand (k : ℝ) :
    completeEllipticEModulus k =
      ∫ t in (0 : ℝ)..1, completeEllipticEHalfIntegrand k t := by
  unfold completeEllipticEModulus
  unfold completeEllipticEParam
  rw [← intervalIntegral.integral_const_mul]
  apply intervalIntegral.integral_congr
  intro t _ht
  simpa using (completeEllipticEHalfIntegrand_eq_const_mul k t).symm

theorem completeEllipticESubstIntegrand_eq_trig_ae (k : ℝ) :
    ∀ᵐ θ : ℝ ∂ MeasureTheory.volume,
      θ ∈ Set.uIoc (0 : ℝ) (Real.pi / 2) →
        completeEllipticESubstIntegrand k θ =
          Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2) := by
  have hne : ∀ᵐ θ : ℝ ∂ MeasureTheory.volume, θ ≠ Real.pi / 2 := by
    rw [MeasureTheory.ae_iff]
    · simp
  filter_upwards [hne] with θ hne hθ
  have hle : (0 : ℝ) ≤ Real.pi / 2 := by
    nlinarith [Real.pi_pos]
  rw [Set.uIoc_of_le hle] at hθ
  have hθ' : θ ∈ Set.Ioo (0 : ℝ) (Real.pi / 2) := by
    constructor
    · exact hθ.1
    · exact lt_of_le_of_ne hθ.2 hne
  simpa [completeEllipticESubstIntegrand, completeEllipticEHalfIntegrand,
    mul_assoc, mul_left_comm, mul_comm] using
    elliptic_integrand_subst_eq_trig_on_Ioo k hθ'

theorem completeEllipticEHalfIntegrand_substitution (k : ℝ) :
    ∫ θ in (0 : ℝ)..(Real.pi / 2), completeEllipticESubstIntegrand k θ =
      ∫ t in (sinSqSubst 0)..(sinSqSubst (Real.pi / 2)),
        completeEllipticEHalfIntegrand k t := by
  have hf :
      ContinuousOn sinSqSubst (Set.uIcc (0 : ℝ) (Real.pi / 2)) := by
    unfold sinSqSubst
    exact (Real.continuous_sin.pow 2).continuousOn
  have hff' :
      ∀ x ∈ Set.Ioo (min (0 : ℝ) (Real.pi / 2)) (max (0 : ℝ) (Real.pi / 2)),
        HasDerivAt sinSqSubst (2 * Real.sin x * Real.cos x) x := by
    intro x _hx
    exact sinSqSubst_deriv x
  have hf' :
      ∀ x ∈ Set.Ioo (min (0 : ℝ) (Real.pi / 2)) (max (0 : ℝ) (Real.pi / 2)),
        0 ≤ 2 * Real.sin x * Real.cos x := by
    intro x hx
    have hle : (0 : ℝ) ≤ Real.pi / 2 := by
      nlinarith [Real.pi_pos]
    have hx' : x ∈ Set.Ioo (0 : ℝ) (Real.pi / 2) := by
      simpa [min_eq_left hle, max_eq_right hle] using hx
    exact le_of_lt (jacobian_sinSqSubst_pos_on_Ioo hx')
  simpa [completeEllipticESubstIntegrand, Function.comp, mul_assoc, mul_left_comm, mul_comm]
    using
      (intervalIntegral.integral_comp_mul_deriv_of_deriv_nonneg
        (a := (0 : ℝ))
        (b := (Real.pi / 2))
        (f := sinSqSubst)
        (f' := fun θ : ℝ => 2 * Real.sin θ * Real.cos θ)
        (g := completeEllipticEHalfIntegrand k)
        hf
        hff'
        hf')

/--
Trig-substitution bridge.

Shows that the parameter-form complete elliptic integral agrees with the
standard trigonometric form under `t = sin θ ^ 2`.

theorem completeEllipticEModulus_eq_trig (k : ℝ) :
    completeEllipticEModulus k = completeEllipticETrig k := by
  calc
    completeEllipticEModulus k
        = ∫ t in (0 : ℝ)..1, completeEllipticEHalfIntegrand k t := by
            exact completeEllipticEModulus_eq_integral_halfIntegrand k
    _ = ∫ θ in (0 : ℝ)..(Real.pi / 2), completeEllipticESubstIntegrand k θ := by
          symm
          simpa [sinSqSubst_zero, sinSqSubst_pi_div_two] using
            completeEllipticEHalfIntegrand_substitution k
    _ = ∫ θ in (0 : ℝ)..(Real.pi / 2),
          Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2) := by
          apply intervalIntegral.integral_congr_ae
          exact completeEllipticESubstIntegrand_eq_trig_ae k
    _ = completeEllipticETrig k := by
          rfl

theorem completeEllipticETrig_integrand_antitone
    {k l θ : ℝ} (hk0 : 0 ≤ k) (hkl : k ≤ l) (hl1 : l ≤ 1) :
    Real.sqrt (1 - l ^ 2 * Real.sin θ ^ 2) ≤
      Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2) := by
  have hk1 : k ≤ 1 := le_trans hkl hl1
  have hl0 : 0 ≤ l := le_trans hk0 hkl
  have hk_sq_le_l_sq : k ^ 2 ≤ l ^ 2 := by
    nlinarith
  have hs0 : 0 ≤ Real.sin θ ^ 2 := by
    positivity
  have hs : |Real.sin θ| ≤ 1 := by
    simpa using Real.abs_sin_le_one θ
  have hs1 : Real.sin θ ^ 2 ≤ 1 := by
    have hsL : -1 ≤ Real.sin θ := (abs_le.mp hs).1
    have hsR : Real.sin θ ≤ 1 := (abs_le.mp hs).2
    nlinarith
  have hradl_nonneg : 0 ≤ 1 - l ^ 2 * Real.sin θ ^ 2 := by
    exact completeEllipticEParam_numerator_radicand_nonneg hl0 hl1 hs0 hs1
  have hradk_nonneg : 0 ≤ 1 - k ^ 2 * Real.sin θ ^ 2 := by
    exact completeEllipticEParam_numerator_radicand_nonneg hk0 hk1 hs0 hs1
  have hrad :
      1 - l ^ 2 * Real.sin θ ^ 2 ≤
        1 - k ^ 2 * Real.sin θ ^ 2 := by
    nlinarith
  have hsq :
      (Real.sqrt (1 - l ^ 2 * Real.sin θ ^ 2)) ^ 2 ≤
        (Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2)) ^ 2 := by
    rw [Real.sq_sqrt hradl_nonneg, Real.sq_sqrt hradk_nonneg]
    exact hrad
  have hleft_nonneg : 0 ≤ Real.sqrt (1 - l ^ 2 * Real.sin θ ^ 2) := by
    exact Real.sqrt_nonneg _
  have hright_nonneg : 0 ≤ Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2) := by
    exact Real.sqrt_nonneg _
  nlinarith

    {k l : ℝ} (hk0 : 0 ≤ k) (hkl : k ≤ l) (hl1 : l ≤ 1) :

              intro θ
              exact completeEllipticETrig_integrand_antitone hk0 hkl hl1)

/-
Elliptic-series definitions and coefficient algebra.

This block defines the coefficient sequence and supporting recurrence facts used
to convert the trigonometric integral expansion into `completeEllipticESeries`.
-/
def ellipticESeriesCoeff (n : ℕ) : ℝ :=
  ((((2 * n).factorial : ℝ) /
      (((2 : ℝ) ^ n * (n.factorial : ℝ)) ^ 2)) ^ 2) *
    (1 / (1 - 2 * (n : ℝ)))

def ellipticESeriesTerm (k : ℝ) (n : ℕ) : ℝ :=
  ellipticESeriesCoeff n * k ^ (2 * n)

def completeEllipticESeries (k : ℝ) : ℝ :=
  (Real.pi / 2) * ∑' n : ℕ, ellipticESeriesTerm k n

def seriesExpansionTarget (k : ℝ) : Prop :=
  completeEllipticETrig k = completeEllipticESeries k

  norm_num

  unfold ellipticESeriesCoeff
  norm_num

    ellipticESeriesTerm k 0 = 1 := by

  rw [ellipticESeriesCoeff_zero]
  norm_num

      sqrtOneSubCoeff n * wallisIntegralCandidate n * k ^ (2 * n) := by
  rw [ellipticESeriesTerm_eq_bridge]
  unfold wallisIntegralCandidate
  ring

theorem ellipticESeriesCoeff_zero_via_bridge :

def eTrigBinomialTerm (k : ℝ) (n : ℕ) (θ : ℝ) : ℝ :=
  sqrtOneSubCoeff n * k ^ (2 * n) * Real.sin θ ^ (2 * n)

def eTrigBinomialSeries (k θ : ℝ) : ℝ :=
  ∑' n : ℕ, eTrigBinomialTerm k n θ

def eTrigBinomialTarget (k : ℝ) : Prop :=
  Set.EqOn
    (fun θ : ℝ => Real.sqrt (1 - k ^ 2 * Real.sin θ ^ 2))
    (fun θ : ℝ => eTrigBinomialSeries k θ)
    (Set.uIcc (0 : ℝ) (Real.pi / 2))

def wallisTarget (n : ℕ) : Prop :=
  ∫ θ in (0 : ℝ)..(Real.pi / 2), Real.sin θ ^ (2 * n) =
    wallisIntegralCandidate n

def eTrigTermwiseTarget (k : ℝ) : Prop :=
  ∫ θ in (0 : ℝ)..(Real.pi / 2), eTrigBinomialSeries k θ =
    ∑' n : ℕ,
      ∫ θ in (0 : ℝ)..(Real.pi / 2), eTrigBinomialTerm k n θ

def eSeriesSummableTarget (k : ℝ) : Prop :=
  Summable (fun n : ℕ => ellipticESeriesTerm k n)

theorem eTrigBinomialTerm_zero (k θ : ℝ) :
    eTrigBinomialTerm k 0 θ = 1 := by
  unfold eTrigBinomialTerm
  rw [sqrtOneSubCoeff_zero]
  norm_num

theorem eTrigBinomialTerm_one (k θ : ℝ) :

theorem eTrigBinomialTerm_two (k θ : ℝ) :

theorem eTrigBinomialTerm_three (k θ : ℝ) :

theorem wallisTarget_zero :

theorem eSeriesSummableTarget_of_summable

theorem eSeries_eq_tsum_scaled_of_summable
    {k : ℝ} (hs : eSeriesSummableTarget k) :
    completeEllipticESeries k =
      ∑' n : ℕ, (Real.pi / 2) * ellipticESeriesTerm k n := by

  have hsum :
      HasSum
        (fun n : ℕ => (Real.pi / 2) * ellipticESeriesTerm k n)

  symm
  exact hsum.tsum_eq

theorem eTrig_eq_integral_binomial_of_target
    {k : ℝ} (hbinom : eTrigBinomialTarget k) :
    completeEllipticETrig k =
      ∫ θ in (0 : ℝ)..(Real.pi / 2), eTrigBinomialSeries k θ := by
  unfold completeEllipticETrig
  unfold eTrigBinomialTarget at hbinom
  apply intervalIntegral.integral_congr
  intro θ hθ
  exact hbinom hθ

theorem integral_eTrigBinomialTerm_eq_scaled_seriesTerm_of_wallis
    (k : ℝ) (n : ℕ) (hwallis : wallisTarget n) :
    ∫ θ in (0 : ℝ)..(Real.pi / 2), eTrigBinomialTerm k n θ =
      (Real.pi / 2) * ellipticESeriesTerm k n := by
  unfold wallisTarget at hwallis
  unfold eTrigBinomialTerm
  rw [intervalIntegral.integral_const_mul]
  rw [hwallis]
  calc
    (sqrtOneSubCoeff n * k ^ (2 * n)) * wallisIntegralCandidate n
        = sqrtOneSubCoeff n * wallisIntegralCandidate n * k ^ (2 * n) := by
          ring
    _ = (Real.pi / 2) * ellipticESeriesTerm k n := by
          exact (pi_half_mul_ellipticESeriesTerm_eq_bridge k n).symm

    {k : ℝ}
    (hbinom : eTrigBinomialTarget k)
    (htermwise : eTrigTermwiseTarget k)

      ∑' n : ℕ, (Real.pi / 2) * ellipticESeriesTerm k n := by

    completeEllipticETrig k

          ∫ θ in (0 : ℝ)..(Real.pi / 2), eTrigBinomialTerm k n θ := by
          exact htermwise
    _ = ∑' n : ℕ, (Real.pi / 2) * ellipticESeriesTerm k n := by
          apply tsum_congr
          intro n
          exact
            integral_eTrigBinomialTerm_eq_scaled_seriesTerm_of_wallis
              k n (hwallis n)

theorem two_mul_natCast_ne_one (n : ℕ) :
    (2 * (n : ℝ)) ≠ 1 := by
  intro h
  have hnat : 2 * n = 1 := by
    exact_mod_cast h
  have heven : Even (1 : ℕ) := by
    rw [← hnat]
    exact ⟨n, by ring⟩
  norm_num at heven

theorem one_sub_two_natCast_ne_zero (n : ℕ) :
    1 - 2 * (n : ℝ) ≠ 0 := by
  intro h
  have h' : 2 * (n : ℝ) = 1 := by
    linarith
  exact two_mul_natCast_ne_one n h'

theorem wallisEvenCoeff_succ (n : ℕ) :
    wallisEvenCoeff (n + 1) =
      ((2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 2)) * wallisEvenCoeff n := by
  unfold wallisEvenCoeff
  have hfac :
      ((2 * (n + 1)).factorial : ℝ) =
        (2 * (n : ℝ) + 2) * (2 * (n : ℝ) + 1) *
          ((2 * n).factorial : ℝ) := by
    have h1 : 2 * (n + 1) = (2 * n + 1) + 1 := by
      ring
    rw [h1, Nat.factorial_succ, Nat.cast_mul]
    have h2 : 2 * n + 1 = (2 * n) + 1 := by
      ring
    rw [h2, Nat.factorial_succ, Nat.cast_mul]
    push_cast
    ring
  have hden :
      (2 : ℝ) ^ (n + 1) * ((n + 1).factorial : ℝ) =
        (2 * (n : ℝ) + 2) * ((2 : ℝ) ^ n * (n.factorial : ℝ)) := by
    rw [pow_succ, Nat.factorial_succ, Nat.cast_mul]
    push_cast
    ring
  rw [hfac, hden]
  have hne1 : (2 * (n : ℝ) + 2) ≠ 0 := by
    positivity
  have hne2 : (2 : ℝ) ^ n * (n.factorial : ℝ) ≠ 0 := by
    positivity
  have hne3 :
      (2 * (n : ℝ) + 2) * ((2 : ℝ) ^ n * (n.factorial : ℝ)) ≠ 0 := by
    exact mul_ne_zero hne1 hne2
  field_simp [hne1, hne2, hne3]

/--
Standard double-factorial ratio, written as products.

This is the textbook factor

`1 * 3 * ... * (2n - 1) / (2 * 4 * ... * 2n)`.
-/
def standardDoubleFactorialRatio (n : ℕ) : ℝ :=
  (∏ i ∈ Finset.range n, (2 * (i : ℝ) + 1)) /
    ∏ i ∈ Finset.range n, (2 * (i : ℝ) + 2)

theorem standardDoubleFactorialRatio_zero :
    standardDoubleFactorialRatio 0 = 1 := by
  unfold standardDoubleFactorialRatio
  simp

theorem standardDoubleFactorialRatio_succ (n : ℕ) :
    standardDoubleFactorialRatio (n + 1) =
      ((2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 2)) *
        standardDoubleFactorialRatio n := by
  unfold standardDoubleFactorialRatio
  rw [Finset.prod_range_succ, Finset.prod_range_succ]
  have hden :
      (∏ i ∈ Finset.range n, (2 * (i : ℝ) + 2)) ≠ 0 := by
    positivity
  have hlast : 2 * (n : ℝ) + 2 ≠ 0 := by
    positivity
  field_simp [hden, hlast]

theorem wallisEvenCoeff_eq_standardDoubleFactorialRatio (n : ℕ) :
    wallisEvenCoeff n = standardDoubleFactorialRatio n := by
  induction n with
  | zero =>
      rw [wallisEvenCoeff_zero, standardDoubleFactorialRatio_zero]
  | succ n ih =>
      rw [wallisEvenCoeff_succ]
      rw [standardDoubleFactorialRatio_succ]
      rw [ih]

/--
Standard double-factorial form of the elliptic `E` series coefficient.

For `n`, the coefficient is

`((1 * 3 * ... * (2n - 1)) / (2 * 4 * ... * 2n)) ^ 2 *
  1 / (1 - 2n)`.
-/
theorem ellipticESeriesCoeff_eq_standardDoubleFactorial_form (n : ℕ) :
    ellipticESeriesCoeff n =
      standardDoubleFactorialRatio n ^ 2 *
        (1 / (1 - 2 * (n : ℝ))) := by
  rw [ellipticESeriesCoeff_eq_wallis_sq_mul]
  rw [wallisEvenCoeff_eq_standardDoubleFactorialRatio]

theorem wallisIntegralCandidate_succ (n : ℕ) :
    wallisIntegralCandidate (n + 1) =
      ((2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 2)) *
        wallisIntegralCandidate n := by
  unfold wallisIntegralCandidate
  rw [wallisEvenCoeff_succ]
  ring

theorem sqrtOneSubCoeff_succ (n : ℕ) :
    sqrtOneSubCoeff (n + 1) =
      ((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2)) * sqrtOneSubCoeff n := by
  unfold sqrtOneSubCoeff
  rw [wallisEvenCoeff_succ]
  have hs :
      1 - 2 * ((n + 1 : ℕ) : ℝ) = -(2 * (n : ℝ) + 1) := by
    push_cast
    ring
  rw [hs]
  have hne1 : 1 - 2 * (n : ℝ) ≠ 0 := by
    exact one_sub_two_natCast_ne_zero n
  have hne2 : 2 * (n : ℝ) + 2 ≠ 0 := by
    positivity
  have hne3 : 2 * (n : ℝ) + 1 ≠ 0 := by
    positivity
  field_simp [hne1, hne2, hne3]
  ring

def ellipticESeriesRatio (n : ℕ) : ℝ :=
  ((2 * (n : ℝ) - 1) * (2 * (n : ℝ) + 1)) / (2 * (n : ℝ) + 2) ^ 2

def ellipticESeriesTermRatio (k : ℝ) (n : ℕ) : ℝ :=
  ellipticESeriesRatio n * k ^ 2

theorem ellipticESeriesCoeff_succ (n : ℕ) :
    ellipticESeriesCoeff (n + 1) =
      ellipticESeriesRatio n * ellipticESeriesCoeff n := by
  unfold ellipticESeriesRatio
  rw [ellipticESeriesCoeff_eq_bridge, ellipticESeriesCoeff_eq_bridge]
  rw [sqrtOneSubCoeff_succ, wallisEvenCoeff_succ]
  have hne : 2 * (n : ℝ) + 2 ≠ 0 := by
    positivity
  field_simp [hne]

theorem ellipticESeriesTerm_succ (k : ℝ) (n : ℕ) :
    ellipticESeriesTerm k (n + 1) =
      ellipticESeriesTermRatio k n * ellipticESeriesTerm k n := by
  unfold ellipticESeriesTermRatio ellipticESeriesTerm
  rw [ellipticESeriesCoeff_succ]
  have hpow : 2 * (n + 1) = 2 + 2 * n := by
    ring
  rw [hpow, pow_add]
  ring

theorem abs_ellipticESeriesTerm_succ (k : ℝ) (n : ℕ) :
    |ellipticESeriesTerm k (n + 1)| =
      |ellipticESeriesTermRatio k n| * |ellipticESeriesTerm k n| := by
  rw [ellipticESeriesTerm_succ]
  rw [abs_mul]

theorem abs_ellipticESeriesRatio_le_one (n : ℕ) :
    |ellipticESeriesRatio n| ≤ 1 := by
  unfold ellipticESeriesRatio
  have hden : 0 < (2 * (n : ℝ) + 2) ^ 2 := by
    positivity
  have hden_ne : (2 * (n : ℝ) + 2) ^ 2 ≠ 0 := ne_of_gt hden
  apply abs_le.mpr
  constructor
  · have : -1 ≤
        ((2 * (n : ℝ) - 1) * (2 * (n : ℝ) + 1)) /
          (2 * (n : ℝ) + 2) ^ 2 := by
      field_simp [hden_ne]
      nlinarith
    simpa using this
  · have : ((2 * (n : ℝ) - 1) * (2 * (n : ℝ) + 1)) /
        (2 * (n : ℝ) + 2) ^ 2 ≤ 1 := by
      field_simp [hden_ne]
      nlinarith
    simpa using this

theorem eTrigBinomialTerm_eq_coeff_mul_pow
    (k θ : ℝ) (n : ℕ) :
    eTrigBinomialTerm k n θ =
      sqrtOneSubCoeff n * (k ^ 2 * sinSqSubst θ) ^ n := by
  calc
    eTrigBinomialTerm k n θ
        =
        sqrtOneSubCoeff n *
          (k ^ (2 * n) * Real.sin θ ^ (2 * n)) := by
            unfold eTrigBinomialTerm
            ring
    _ =
        sqrtOneSubCoeff n *
          ((k ^ 2) ^ n * (Real.sin θ ^ 2) ^ n) := by
            rw [pow_mul, pow_mul]
    _ =
        sqrtOneSubCoeff n *
          (k ^ 2 * Real.sin θ ^ 2) ^ n := by
            rw [← mul_pow]
    _ =
        sqrtOneSubCoeff n *
          (k ^ 2 * sinSqSubst θ) ^ n := by
            simp [sinSqSubst]

theorem abs_sqrtOneSubCoeffRatio_le_one (n : ℕ) :
    |((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2))| ≤ 1 := by
  have hnum :
      |2 * (n : ℝ) - 1| ≤ 2 * (n : ℝ) + 2 := by
    apply abs_le.mpr
    constructor <;> nlinarith
  have hden_nonneg : 0 ≤ 2 * (n : ℝ) + 2 := by
    positivity
  have hden_pos : 0 < 2 * (n : ℝ) + 2 := by
    positivity
  have hden_ne : 2 * (n : ℝ) + 2 ≠ 0 := ne_of_gt hden_pos
  have haux :
      |2 * (n : ℝ) - 1| / (2 * (n : ℝ) + 2) ≤
        (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 2) := by
    exact div_le_div_of_nonneg_right hnum hden_nonneg
  calc
    |((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2))|
        = |2 * (n : ℝ) - 1| / (2 * (n : ℝ) + 2) := by
            rw [abs_div, abs_of_nonneg hden_nonneg]
    _ ≤ (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 2) := haux
    _ = 1 := by
          field_simp [hden_ne]

theorem eTrigBinomialTerm_succ_eq_ratio_mul
    (k θ : ℝ) (n : ℕ) :
    eTrigBinomialTerm k (n + 1) θ =
      (((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2)) *
          (k ^ 2 * sinSqSubst θ)) *
        eTrigBinomialTerm k n θ := by
  unfold eTrigBinomialTerm
  rw [sqrtOneSubCoeff_succ]
  have hkpow : k ^ (2 * (n + 1)) = k ^ 2 * k ^ (2 * n) := by
    have hpow : 2 * (n + 1) = 2 + 2 * n := by
      ring
    rw [hpow, pow_add]
  have hspow :
      Real.sin θ ^ (2 * (n + 1)) =
        Real.sin θ ^ 2 * Real.sin θ ^ (2 * n) := by
    have hpow : 2 * (n + 1) = 2 + 2 * n := by
      ring
    rw [hpow, pow_add]
  rw [hkpow, hspow]
  unfold sinSqSubst
  ring

theorem abs_eTrigBinomialTerm_succ_le_sq_mul
    (k θ : ℝ) (n : ℕ) :
    |eTrigBinomialTerm k (n + 1) θ| ≤
      k ^ 2 * |eTrigBinomialTerm k n θ| := by
  rw [eTrigBinomialTerm_succ_eq_ratio_mul, abs_mul]
  have hratio :
      |((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2) *
          (k ^ 2 * sinSqSubst θ))| ≤
        k ^ 2 := by
    have hks : 0 ≤ k ^ 2 * sinSqSubst θ := by
      exact mul_nonneg (sq_nonneg k) (sinSqSubst_nonneg θ)
    rw [abs_mul, abs_of_nonneg hks]
    have hr : |((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2))| ≤ 1 := by
      exact abs_sqrtOneSubCoeffRatio_le_one n
    have hs1 : sinSqSubst θ ≤ 1 := by
      exact sinSqSubst_le_one θ
    have hk2 : 0 ≤ k ^ 2 := by
      positivity
    nlinarith
  exact mul_le_mul_of_nonneg_right hratio (abs_nonneg _)

theorem abs_eTrigBinomialTerm_le_abs_coeff_mul_abs_k_pow
    (k θ : ℝ) (n : ℕ) :
    |eTrigBinomialTerm k n θ| ≤
      |sqrtOneSubCoeff n| * |k| ^ (2 * n) := by
  unfold eTrigBinomialTerm
  rw [abs_mul, abs_mul, abs_pow, abs_pow]
  have hsin : |Real.sin θ| ≤ 1 := by
    simpa using Real.abs_sin_le_one θ
  have hpow : |Real.sin θ| ^ (2 * n) ≤ 1 := by
    exact pow_le_one₀ (abs_nonneg _) hsin
  have hfront_nonneg : 0 ≤ |sqrtOneSubCoeff n| * |k| ^ (2 * n) := by
    positivity
  calc
    |sqrtOneSubCoeff n| * |k| ^ (2 * n) * |Real.sin θ| ^ (2 * n)
        ≤ |sqrtOneSubCoeff n| * |k| ^ (2 * n) * 1 := by
          exact mul_le_mul_of_nonneg_left hpow hfront_nonneg
    _ = |sqrtOneSubCoeff n| * |k| ^ (2 * n) := by
          ring

/-
Scalar binomial square-root series support.

This block proves the analytic scalar identity
`sqrt (1 - x) = sqrtOneSubSeries x` for `|x| < 1`, then transports it into the
elliptic-integrand series expansion.
-/
def sqrtOneSubSeriesTerm (x : ℝ) (n : ℕ) : ℝ :=
  sqrtOneSubCoeff n * x ^ n

def sqrtOneSubSeries (x : ℝ) : ℝ :=
  ∑' n : ℕ, sqrtOneSubSeriesTerm x n

def sqrtOneSubSeriesTarget (x : ℝ) : Prop :=
  Real.sqrt (1 - x) = sqrtOneSubSeries x

theorem sqrtOneSubSeriesTerm_zero (x : ℝ) :
    sqrtOneSubSeriesTerm x 0 = 1 := by
  unfold sqrtOneSubSeriesTerm
  rw [sqrtOneSubCoeff_zero]
  norm_num

theorem sqrtOneSubSeriesTerm_one (x : ℝ) :
    sqrtOneSubSeriesTerm x 1 = (-(1 / 2 : ℝ)) * x := by
  unfold sqrtOneSubSeriesTerm
  rw [sqrtOneSubCoeff_one]
  ring

theorem sqrtOneSubSeriesTerm_succ_eq_ratio_mul
    (x : ℝ) (n : ℕ) :
    sqrtOneSubSeriesTerm x (n + 1) =
      (((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2)) * x) *
        sqrtOneSubSeriesTerm x n := by
  unfold sqrtOneSubSeriesTerm
  rw [sqrtOneSubCoeff_succ, pow_succ]
  ring

theorem abs_sqrtOneSubSeriesTerm_succ
    (x : ℝ) (n : ℕ) :
    |sqrtOneSubSeriesTerm x (n + 1)| =
      |((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2) * x)| *
        |sqrtOneSubSeriesTerm x n| := by
  rw [sqrtOneSubSeriesTerm_succ_eq_ratio_mul]
  rw [abs_mul]

theorem norm_sqrtOneSubSeriesTerm_succ_le_abs_mul
    (x : ℝ) (n : ℕ) :
    ‖sqrtOneSubSeriesTerm x (n + 1)‖ ≤
      |x| * ‖sqrtOneSubSeriesTerm x n‖ := by
  rw [Real.norm_eq_abs, Real.norm_eq_abs, abs_sqrtOneSubSeriesTerm_succ]
  have hr :
      |((2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2) * x)| ≤ |x| := by
    rw [abs_mul]
    calc
      |(2 * (n : ℝ) - 1) / (2 * (n : ℝ) + 2)| * |x|
          ≤ 1 * |x| := by
            exact mul_le_mul_of_nonneg_right
              (abs_sqrtOneSubCoeffRatio_le_one n)
              (abs_nonneg x)
      _ = |x| := by ring
  exact mul_le_mul_of_nonneg_right hr (abs_nonneg _)

theorem sqrtOneSubSeriesSummable_of_abs_lt_one
    {x : ℝ} (hx : |x| < 1) :
    Summable (fun n : ℕ => sqrtOneSubSeriesTerm x n) := by
  have hratio :
      ∀ᶠ n : ℕ in Filter.atTop,
        ‖sqrtOneSubSeriesTerm x (n + 1)‖ ≤
          |x| * ‖sqrtOneSubSeriesTerm x n‖ := by
    exact Filter.Eventually.of_forall
      (norm_sqrtOneSubSeriesTerm_succ_le_abs_mul x)
  exact summable_of_ratio_norm_eventually_le hx hratio

def binomHalf : ℕ → ℝ
  | 0 => 1
  | n + 1 => binomHalf n * (1 / 2 - (n : ℝ)) / ((n : ℝ) + 1)

theorem binomHalf_succ (n : ℕ) :
    binomHalf (n + 1) = binomHalf n * (1 / 2 - (n : ℝ)) / ((n : ℝ) + 1) := by
  rfl

theorem sqrtOneSubCoeff_eq_binomHalf_mul_neg_one_pow (n : ℕ) :
    sqrtOneSubCoeff n = binomHalf n * (-1 : ℝ) ^ n := by
  induction n with
  | zero =>
      unfold sqrtOneSubCoeff wallisEvenCoeff
      simp[binomHalf]
  | succ n ih =>
      rw [sqrtOneSubCoeff_succ n]
      rw [ih]
      rw [binomHalf_succ n]
      have h_pow : (-1 : ℝ) ^ (n + 1) = (-1 : ℝ) ^ n * (-1 : ℝ) := by
        rw [pow_add, pow_one]
      rw [h_pow]
      have hd1 : (n : ℝ) + 1 ≠ 0 := by positivity
      have hd2 : 2 * (n : ℝ) + 2 ≠ 0 := by positivity
      field_simp
      ring

theorem sqrtOneSubSeriesTerm_eq_binomHalf_term (x : ℝ) (n : ℕ) :
    sqrtOneSubSeriesTerm x n = binomHalf n * (-x) ^ n := by
  unfold sqrtOneSubSeriesTerm
  rw[sqrtOneSubCoeff_eq_binomHalf_mul_neg_one_pow n]
  have h_neg_x : (-x) ^ n = (-1 : ℝ) ^ n * x ^ n := by
    have h_eq : -x = -1 * x := by ring
    rw [h_eq, mul_pow]
  rw [h_neg_x]
  ring

theorem ring_choose_succ_right (a : ℝ) (n : ℕ) :
    Ring.choose a (n + 1) = Ring.choose a n * (a - n) / (n + 1) := by
  have h :=
    Ring.choose_smul_choose (r := a) (n := n + 1) (k := n) (Nat.le_succ n)
  have hcoeff : (((n + 1).choose n : ℕ) : ℝ) = n + 1 := by
    exact_mod_cast Nat.choose_succ_self_right n
  have hsub : n + 1 - n = 1 := by
    omega
  rw [nsmul_eq_mul, hcoeff, hsub] at h
  have hone : Ring.choose (a - n) 1 = a - n := by
    simp
  rw [hone] at h
  have hn1 : (n + 1 : ℝ) ≠ 0 := by
    positivity
  apply (eq_div_iff hn1).2
  simpa [mul_comm, mul_left_comm, mul_assoc] using h

theorem binomHalf_eq_choose (n : ℕ) :
    binomHalf n = Ring.choose ((1 : ℝ) / 2) n := by
  induction n with
  | zero =>
      simp [binomHalf]
  | succ n ih =>
      rw [binomHalf_succ, ih, ring_choose_succ_right ((1 : ℝ) / 2) n]

theorem sqrtOneSubSeriesTerm_eq_choose_term (x : ℝ) (n : ℕ) :
    sqrtOneSubSeriesTerm x n = Ring.choose ((1 : ℝ) / 2) n * (-x) ^ n := by
  rw [sqrtOneSubSeriesTerm_eq_binomHalf_term, binomHalf_eq_choose]

theorem sqrtOneSubSeries_cauchy_coeff (x : ℝ) (n : ℕ) :
    ∑ kl ∈ Finset.antidiagonal n,
      sqrtOneSubSeriesTerm x kl.1 * sqrtOneSubSeriesTerm x kl.2 =
        Ring.choose (1 : ℝ) n * (-x) ^ n := by
  calc
    ∑ kl ∈ Finset.antidiagonal n,
        sqrtOneSubSeriesTerm x kl.1 * sqrtOneSubSeriesTerm x kl.2
      =
      ∑ kl ∈ Finset.antidiagonal n,
        (Ring.choose ((1 : ℝ) / 2) kl.1 *
          Ring.choose ((1 : ℝ) / 2) kl.2) * (-x) ^ n := by
        apply Finset.sum_congr rfl
        intro kl hkl
        rcases Finset.mem_antidiagonal.mp hkl with hsum
        rw [sqrtOneSubSeriesTerm_eq_choose_term, sqrtOneSubSeriesTerm_eq_choose_term]
        calc
          Ring.choose ((1 : ℝ) / 2) kl.1 * (-x) ^ kl.1 *
              (Ring.choose ((1 : ℝ) / 2) kl.2 * (-x) ^ kl.2)
            =
              (Ring.choose ((1 : ℝ) / 2) kl.1 *
                Ring.choose ((1 : ℝ) / 2) kl.2) *
                  ((-x) ^ kl.1 * (-x) ^ kl.2) := by
                ring
          _ =
              (Ring.choose ((1 : ℝ) / 2) kl.1 *
                Ring.choose ((1 : ℝ) / 2) kl.2) *
                  (-x) ^ n := by
                rw [← pow_add, hsum]
    _ =
      (∑ kl ∈ Finset.antidiagonal n,
        Ring.choose ((1 : ℝ) / 2) kl.1 *
          Ring.choose ((1 : ℝ) / 2) kl.2) * (-x) ^ n := by
        rw [← Finset.sum_mul]
    _ = Ring.choose (1 : ℝ) n * (-x) ^ n := by
        have hadd :
            ∑ kl ∈ Finset.antidiagonal n,
              Ring.choose ((1 : ℝ) / 2) kl.1 *
                Ring.choose ((1 : ℝ) / 2) kl.2
              = Ring.choose (1 : ℝ) n := by
          calc
            ∑ kl ∈ Finset.antidiagonal n,
                Ring.choose ((1 : ℝ) / 2) kl.1 *
                  Ring.choose ((1 : ℝ) / 2) kl.2
              = Ring.choose (((1 : ℝ) / 2) + ((1 : ℝ) / 2)) n := by
                  symm
                  exact Ring.add_choose_eq
                    (r := ((1 : ℝ) / 2))
                    (s := ((1 : ℝ) / 2))
                    (k := n)
                    (h := Commute.all ((1 : ℝ) / 2) ((1 : ℝ) / 2))
            _ = Ring.choose (1 : ℝ) n := by
                  congr 1
                  ring
        rw [hadd]

theorem choose_one_mul_pow_eq_zero_of_two_le (x : ℝ) {n : ℕ} (hn : 2 ≤ n) :
    Ring.choose (1 : ℝ) n * (-x) ^ n = 0 := by
  have hlt : 1 < n := by
    linarith
  have hchoose_nat : Nat.choose 1 n = 0 := by
    exact Nat.choose_eq_zero_of_lt hlt
  have hchoose : Ring.choose (1 : ℝ) n = 0 := by
    rw [show (1 : ℝ) = ((1 : ℕ) : ℝ) by norm_num]
    rw [Ring.choose_natCast]
    exact_mod_cast hchoose_nat
  rw [hchoose]
  simp

theorem sqrtOneSubSeries_sq_eq_one_sub {x : ℝ} (hx : |x| < 1) :
    (sqrtOneSubSeries x) ^ 2 = 1 - x := by
  have hsabs :
      Summable (fun n : ℕ => ‖sqrtOneSubSeriesTerm x n‖) := by
    simpa [Real.norm_eq_abs] using
      sqrtOneSubSeries_abs_summable_of_abs_lt_one hx
  calc
    (sqrtOneSubSeries x) ^ 2
        = (∑' n : ℕ, sqrtOneSubSeriesTerm x n) *
            (∑' n : ℕ, sqrtOneSubSeriesTerm x n) := by
          simp [sqrtOneSubSeries, pow_two]
    _ =
      ∑' n : ℕ,
        ∑ kl ∈ Finset.antidiagonal n,
          sqrtOneSubSeriesTerm x kl.1 * sqrtOneSubSeriesTerm x kl.2 := by
          exact
            tsum_mul_tsum_eq_tsum_sum_antidiagonal_of_summable_norm hsabs hsabs
    _ = ∑' n : ℕ, Ring.choose (1 : ℝ) n * (-x) ^ n := by
          apply tsum_congr
          intro n
          exact sqrtOneSubSeries_cauchy_coeff x n
    _ = Finset.sum ({0, 1} : Finset ℕ)
          (fun n : ℕ => Ring.choose (1 : ℝ) n * (-x) ^ n) := by
          exact
            tsum_eq_sum
              (s := ({0, 1} : Finset ℕ))
              (f := fun n : ℕ => Ring.choose (1 : ℝ) n * (-x) ^ n)
              (by
                intro n hn
                have h2 : 2 ≤ n := by
                  rcases n with _ | _ | n
                  · simp at hn
                  · simp at hn
                  · omega
                exact choose_one_mul_pow_eq_zero_of_two_le x h2)
    _ = Ring.choose (1 : ℝ) 0 * (-x) ^ 0 +
          Ring.choose (1 : ℝ) 1 * (-x) ^ 1 := by
          simp
    _ = 1 - x := by
          have hchoose0 : Ring.choose (1 : ℝ) 0 = 1 := by
            simp
          have hchoose1 : Ring.choose (1 : ℝ) 1 = 1 := by
            rw [show (1 : ℝ) = ((1 : ℕ) : ℝ) by norm_num]
            rw [Ring.choose_natCast]
            norm_num
          rw [hchoose0, hchoose1]
          ring

theorem sqrtOneSubSeries_zero_val :
    sqrtOneSubSeries 0 = 1 := by
  unfold sqrtOneSubSeries
  have h : (fun n : ℕ => sqrtOneSubSeriesTerm 0 n) = (fun n : ℕ => if n = 0 then 1 else 0) := by
    ext n
    cases n with
    | zero => rw [sqrtOneSubSeriesTerm_zero]; simp
    | succ n =>
        unfold sqrtOneSubSeriesTerm
        simp [pow_succ]
  rw [h, tsum_ite_eq]

theorem sqrtOneSubSeriesTerm_continuous (n : ℕ) :
    Continuous (fun x : ℝ => sqrtOneSubSeriesTerm x n) := by
  unfold sqrtOneSubSeriesTerm
  exact continuous_const.mul (continuous_id.pow n)

theorem norm_sqrtOneSubSeriesTerm_le_on_Icc
    {r x : ℝ} (hr0 : 0 ≤ r) (hx : x ∈ Set.Icc (-r) r) (n : ℕ) :
    ‖sqrtOneSubSeriesTerm x n‖ ≤ |sqrtOneSubSeriesTerm r n| := by
  have hxr : |x| ≤ r := by
    exact abs_le.mpr ⟨hx.1, hx.2⟩
  unfold sqrtOneSubSeriesTerm
  rw [Real.norm_eq_abs, abs_mul, abs_mul, abs_pow, abs_pow, abs_of_nonneg hr0]
  apply mul_le_mul_of_nonneg_left
  · induction n with
    | zero =>
        simp
    | succ n ih =>
        rw [pow_succ, pow_succ]
        exact mul_le_mul ih hxr (abs_nonneg x) (pow_nonneg hr0 n)
  · exact abs_nonneg _

theorem sqrtOneSubSeries_continuousOn_Icc
    {r : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1) :
    ContinuousOn sqrtOneSubSeries (Set.Icc (-r) r) := by
  have hsumm : Summable (fun n : ℕ => |sqrtOneSubSeriesTerm r n|) := by
    have hrabs : |r| < 1 := by
      simpa [abs_of_nonneg hr0] using hr1
    exact sqrtOneSubSeries_abs_summable_of_abs_lt_one hrabs
  unfold sqrtOneSubSeries
  apply continuousOn_tsum
  · intro n
    exact (sqrtOneSubSeriesTerm_continuous n).continuousOn
  · exact hsumm
  · intro n x hx
    exact norm_sqrtOneSubSeriesTerm_le_on_Icc hr0 hx n

theorem sqrtOneSubSeries_nonneg_of_abs_lt_one {x : ℝ} (hx : |x| < 1) :
    0 ≤ sqrtOneSubSeries x := by
  by_contra hxneg
  have hxlt : sqrtOneSubSeries x < 0 := by
    exact lt_of_not_ge hxneg
  let r : ℝ := (|x| + 1) / 2
  have hr0 : 0 ≤ r := by
    dsimp [r]
    nlinarith [abs_nonneg x]
  have hxr : |x| < r := by
    dsimp [r]
    nlinarith
  have hr1 : r < 1 := by
    dsimp [r]
    nlinarith
  have hxmem : x ∈ Set.Icc (-r) r := by
    exact ⟨le_of_lt (abs_lt.mp hxr).1, le_of_lt (abs_lt.mp hxr).2⟩
  have h0mem : (0 : ℝ) ∈ Set.Icc (-r) r := by
    exact ⟨by nlinarith [hr0], by nlinarith [hr0]⟩
  have hseg :
      Set.uIcc x 0 ⊆ Set.Icc (-r) r := by
    intro y hy
    have hmin : -r ≤ min x 0 := by
      exact le_min hxmem.1 h0mem.1
    have hmax : max x 0 ≤ r := by
      exact max_le hxmem.2 h0mem.2
    exact ⟨le_trans hmin hy.1, le_trans hy.2 hmax⟩
  have hcont_Icc : ContinuousOn sqrtOneSubSeries (Set.Icc (-r) r) := by
    exact sqrtOneSubSeries_continuousOn_Icc hr0 hr1
  have hcont_seg : ContinuousOn sqrtOneSubSeries (Set.uIcc x 0) := by
    exact hcont_Icc.mono hseg
  have h0pos : 0 < sqrtOneSubSeries 0 := by
    rw [sqrtOneSubSeries_zero_val]
    norm_num
  have hzmem : (0 : ℝ) ∈ Set.uIcc (sqrtOneSubSeries x) (sqrtOneSubSeries 0) := by
    constructor
    · exact le_trans (min_le_left _ _) (le_of_lt hxlt)
    · exact le_trans (le_of_lt h0pos) (le_max_right _ _)
  rcases intermediate_value_uIcc hcont_seg hzmem with ⟨y, hyseg, hyzero⟩
  have hymem : y ∈ Set.Icc (-r) r := by
    exact hseg hyseg
  have hyabs_le : |y| ≤ r := by
    exact abs_le.mpr ⟨hymem.1, hymem.2⟩
  have hyabs : |y| < 1 := by
    exact lt_of_le_of_lt hyabs_le hr1
  have hsq : (sqrtOneSubSeries y) ^ 2 = 1 - y := by
    exact sqrtOneSubSeries_sq_eq_one_sub hyabs
  rw [hyzero] at hsq
  have hy1 : y = 1 := by
    nlinarith
  have hylt1 : y < 1 := by
    exact (abs_lt.mp hyabs).2
  linarith

/--
Scalar square-root binomial series.

For `|x| < 1`, proves that the constructed coefficient series evaluates to
`sqrt (1 - x)`. This is the analytic core behind the elliptic series expansion
for `|k| < 1`.
-/
theorem sqrtOneSubSeriesTarget_of_abs_lt_one (x : ℝ) (hx : |x| < 1) :
    sqrtOneSubSeriesTarget x := by
  unfold sqrtOneSubSeriesTarget
  have hxnonneg : 0 ≤ 1 - x := by
    have hx' := abs_lt.mp hx
    nlinarith
  have hsq :
      (sqrtOneSubSeries x) ^ 2 = (Real.sqrt (1 - x)) ^ 2 := by
    calc
      (sqrtOneSubSeries x) ^ 2 = 1 - x := by
        exact sqrtOneSubSeries_sq_eq_one_sub hx
      _ = (Real.sqrt (1 - x)) ^ 2 := by
        symm
        exact Real.sq_sqrt hxnonneg
  have hser_nonneg : 0 ≤ sqrtOneSubSeries x := by
    exact sqrtOneSubSeries_nonneg_of_abs_lt_one hx
  have hsqrt_nonneg : 0 ≤ Real.sqrt (1 - x) := by
    exact Real.sqrt_nonneg _
  have hEq : sqrtOneSubSeries x = Real.sqrt (1 - x) := by
    nlinarith [hsq, hser_nonneg, hsqrt_nonneg]
  exact hEq.symm

/-
Endpoint `k = 1` support.

This block proves the endpoint series value by combining closed-form coefficient
partial sums with Wallis squeeze bounds.
-/
theorem ellipticESeriesCoeff_succ_eq_partialStep (n : ℕ) :
    ellipticESeriesCoeff (n + 1) =
      (2 * (((n + 1 : ℕ) : ℝ)) + 1) * wallisEvenCoeff (n + 1) ^ 2
        - (2 * (n : ℝ) + 1) * wallisEvenCoeff n ^ 2 := by
  have hd1 : 2 * (n : ℝ) + 2 ≠ 0 := by positivity
  have hd2 : -(2 * (n : ℝ) + 1) ≠ 0 := by
    have : 0 ≤ (n : ℝ) := Nat.cast_nonneg n
    linarith
  have hs :
      1 - 2 * (((n + 1 : ℕ) : ℝ)) = -(2 * (n : ℝ) + 1) := by
    push_cast
    ring
  have hleft :
      ellipticESeriesCoeff (n + 1) =
        -((2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 2) ^ 2) *
          wallisEvenCoeff n ^ 2 := by
    rw[ellipticESeriesCoeff_eq_wallis_sq_mul, wallisEvenCoeff_succ, hs]
    field_simp [hd1, hd2]
  have hright :
      (2 * (((n + 1 : ℕ) : ℝ)) + 1) * wallisEvenCoeff (n + 1) ^ 2
        - (2 * (n : ℝ) + 1) * wallisEvenCoeff n ^ 2 =
      -((2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 2) ^ 2) *
        wallisEvenCoeff n ^ 2 := by
    rw[wallisEvenCoeff_succ]
    push_cast
    field_simp [hd1]
    ring
  exact hleft.trans hright.symm

theorem ellipticESeriesCoeff_partialSum_closed_form (n : ℕ) :
    (∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) =
      (2 * (n : ℝ) + 1) * wallisEvenCoeff n ^ 2 := by
  induction n with
  | zero =>
      rw[Finset.sum_range_one, ellipticESeriesCoeff_zero, wallisEvenCoeff_zero]
      norm_num
  | succ n ih =>
      rw[Finset.sum_range_succ, ih, ellipticESeriesCoeff_succ_eq_partialStep]
      push_cast
      ring

theorem completeEllipticESeries_one_partialSum_closed_form (n : ℕ) :
    (Real.pi / 2) * (∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) =
      (Real.pi / 2) * ((2 * (n : ℝ) + 1) * wallisEvenCoeff n ^ 2) := by
  rw [ellipticESeriesCoeff_partialSum_closed_form]

def oddWallisIntegralCandidate (n : ℕ) : ℝ :=
  1 / ((2 * (n : ℝ) + 1) * wallisEvenCoeff n)

def oddWallisTarget (n : ℕ) : Prop :=
  ∫ θ in (0 : ℝ)..(Real.pi / 2), Real.sin θ ^ (2 * n + 1) =
    oddWallisIntegralCandidate n

theorem wallisEvenCoeff_pos (n : ℕ) :
    0 < wallisEvenCoeff n := by
  unfold wallisEvenCoeff
  positivity

theorem oddWallisIntegralCandidate_succ (n : ℕ) :
    oddWallisIntegralCandidate (n + 1) =
      ((2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 3)) *
        oddWallisIntegralCandidate n := by
  unfold oddWallisIntegralCandidate
  rw [wallisEvenCoeff_succ]
  have hw : wallisEvenCoeff n ≠ 0 := by
    exact ne_of_gt (wallisEvenCoeff_pos n)
  have h1 : 2 * (n : ℝ) + 1 ≠ 0 := by
    positivity
  have h2 : 2 * (n : ℝ) + 2 ≠ 0 := by
    positivity
  have h3 : 2 * (n : ℝ) + 3 ≠ 0 := by
    positivity
  push_cast
  field_simp [hw, h1, h2, h3]
  ring

def oddWallisRecurrenceTarget : Prop :=
  ∀ n : ℕ,
    ∫ θ in (0 : ℝ)..(Real.pi / 2), Real.sin θ ^ (2 * (n + 1) + 1) =
      ((2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 3)) *
        ∫ θ in (0 : ℝ)..(Real.pi / 2), Real.sin θ ^ (2 * n + 1)

theorem oddWallisRecurrenceTarget_proved :
    oddWallisRecurrenceTarget := by
  intro n
  have hpow : 2 * (n + 1) + 1 = (2 * n + 1) + 2 := by
    omega
  rw [hpow]
  have hcoef :
      ((2 * (n : ℝ) + 1 + 1) / (2 * (n : ℝ) + 1 + 2)) =
        ((2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 3)) := by
    ring
  simpa [hcoef] using
    (integral_sin_pow
      (a := (0 : ℝ))
      (b := (Real.pi / 2))
      (2 * n + 1))

theorem oddWallisTarget_zero :
    oddWallisTarget 0 := by
  unfold oddWallisTarget oddWallisIntegralCandidate
  rw [wallisEvenCoeff_zero]
  norm_num

theorem oddWallisTarget_all :

  intro n
  induction n with
  | zero =>
      exact oddWallisTarget_zero
  | succ n ih =>
      unfold oddWallisTarget at ih ⊢
      rw [oddWallisRecurrenceTarget_proved n]
      rw [ih]
      rw [oddWallisIntegralCandidate_succ]

theorem oddWallisIntegral_eq_candidate (n : ℕ) :

theorem sin_pow_succ_le_self_of_mem_Icc
    {θ : ℝ} (hθ : θ ∈ Set.Icc (0 : ℝ) (Real.pi / 2)) (m : ℕ) :
    Real.sin θ ^ (m + 1) ≤ Real.sin θ ^ m := by
  have hsin0 : 0 ≤ Real.sin θ := by
    exact sin_nonneg_of_mem_Icc_zero_pi_div_two hθ
  have hsin1 : Real.sin θ ≤ 1 := by
    have hs : |Real.sin θ| ≤ 1 := by
      simpa using Real.abs_sin_le_one θ
    exact (abs_le.mp hs).2
  rw [pow_succ]
  exact mul_le_of_le_one_right (pow_nonneg hsin0 m) hsin1

def wallisScaledPartial (n : ℕ) : ℝ :=
  (Real.pi / 2) * ((2 * (n : ℝ) + 1) * wallisEvenCoeff n ^ 2)

theorem one_le_wallisScaledPartial (n : ℕ) :
    1 ≤ wallisScaledPartial n := by
  have h := oddWallisIntegral_le_wallisIntegralCandidate n
  unfold oddWallisIntegralCandidate wallisIntegralCandidate at h
  have hdenpos :
      0 < (2 * (n : ℝ) + 1) * wallisEvenCoeff n := by
    exact mul_pos (by positivity) (wallisEvenCoeff_pos n)
  have hmul := mul_le_mul_of_nonneg_right h (le_of_lt hdenpos)
  have hleft :
      (1 / ((2 * (n : ℝ) + 1) * wallisEvenCoeff n)) *
          ((2 * (n : ℝ) + 1) * wallisEvenCoeff n) = 1 := by
    exact one_div_mul_cancel (ne_of_gt hdenpos)
  have hright :
      (Real.pi / 2 * wallisEvenCoeff n) *
          ((2 * (n : ℝ) + 1) * wallisEvenCoeff n) =
        wallisScaledPartial n := by
    unfold wallisScaledPartial
    ring
  rwa [hleft, hright] at hmul

theorem wallisScaledPartial_le_ratio (n : ℕ) :
    wallisScaledPartial n ≤ (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 1) := by
  have h := wallisIntegralCandidate_succ_le_oddWallisIntegralCandidate n
  unfold wallisIntegralCandidate oddWallisIntegralCandidate at h
  rw [wallisEvenCoeff_succ] at h
  have hp1 : 0 < 2 * (n : ℝ) + 1 := by
    positivity
  have hp2 : 0 < 2 * (n : ℝ) + 2 := by
    positivity
  have hw : 0 < wallisEvenCoeff n := by
    exact wallisEvenCoeff_pos n
  have hmul_nonneg :
      0 ≤ (2 * (n : ℝ) + 2) *
          ((2 * (n : ℝ) + 1) * wallisEvenCoeff n) := by
    exact mul_nonneg
      (le_of_lt hp2)
      (mul_nonneg (le_of_lt hp1) (le_of_lt hw))
  have hmul := mul_le_mul_of_nonneg_right h hmul_nonneg
  have hleft :
      (Real.pi / 2 *
          (((2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 2)) *
            wallisEvenCoeff n)) *
        ((2 * (n : ℝ) + 2) *
          ((2 * (n : ℝ) + 1) * wallisEvenCoeff n)) =
      wallisScaledPartial n * (2 * (n : ℝ) + 1) := by
    unfold wallisScaledPartial
    field_simp [ne_of_gt hp2]
  have hright :
      (wallisEvenCoeff n)⁻¹ * (2 * (n : ℝ) + 1)⁻¹ *
        ((2 * (n : ℝ) + 2) *
          ((2 * (n : ℝ) + 1) * wallisEvenCoeff n)) =
      2 * (n : ℝ) + 2 := by
    field_simp [ne_of_gt hp1, ne_of_gt hw]
  have hmul' :
      wallisScaledPartial n * (2 * (n : ℝ) + 1) ≤
        2 * (n : ℝ) + 2 := by
    simpa [hleft, hright] using hmul
  have hdiv :
      wallisScaledPartial n * (2 * (n : ℝ) + 1) /
          (2 * (n : ℝ) + 1) ≤
        (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 1) := by
    exact div_le_div_of_nonneg_right hmul' (le_of_lt hp1)
  have hcancel :
      wallisScaledPartial n * (2 * (n : ℝ) + 1) /
          (2 * (n : ℝ) + 1) =
        wallisScaledPartial n := by
    field_simp [ne_of_gt hp1]
  simpa [hcancel] using hdiv

theorem wallisScaledPartial_squeeze (n : ℕ) :
    1 ≤ wallisScaledPartial n ∧
      wallisScaledPartial n ≤
        (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 1) := by
  constructor
  · exact one_le_wallisScaledPartial n
  · exact wallisScaledPartial_le_ratio n

theorem tendsto_inv_two_mul_nat_add_one :
    Filter.Tendsto (fun n : ℕ => (2 * (n : ℝ) + 1)⁻¹) Filter.atTop (nhds 0) := by
  have h_atTop : Filter.Tendsto (fun n : ℕ => 2 * (n : ℝ) + 1) Filter.atTop Filter.atTop := by
    apply Filter.Tendsto.atTop_add
    · apply Filter.Tendsto.const_mul_atTop (by norm_num)
      exact tendsto_natCast_atTop_atTop
    · exact tendsto_const_nhds
  exact tendsto_inv_atTop_zero.comp h_atTop

theorem tendsto_upper_bound_one :
    Filter.Tendsto (fun n : ℕ => (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 1)) Filter.atTop (nhds 1) := by
  have heq : (fun n : ℕ => (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 1)) =
      fun n : ℕ => 1 + (2 * (n : ℝ) + 1)⁻¹ := by
    ext n
    have hne : 2 * (n : ℝ) + 1 ≠ 0 := by positivity
    calc
      (2 * (n : ℝ) + 2) / (2 * (n : ℝ) + 1)
        = (2 * (n : ℝ) + 1 + 1) / (2 * (n : ℝ) + 1) := by ring
      _ = (2 * (n : ℝ) + 1) / (2 * (n : ℝ) + 1) + 1 / (2 * (n : ℝ) + 1) := by
          exact add_div (2 * (n : ℝ) + 1) 1 (2 * (n : ℝ) + 1)
      _ = 1 + (2 * (n : ℝ) + 1)⁻¹ := by
          rw [div_self hne, one_div]
  rw [heq]
  have h1 : nhds (1 : ℝ) = nhds (1 + 0) := by rw[add_zero]
  rw [h1]
  exact Filter.Tendsto.add tendsto_const_nhds tendsto_inv_two_mul_nat_add_one

theorem tendsto_partial_sums_coeff :
    Filter.Tendsto (fun n : ℕ => ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i)
      Filter.atTop (nhds (2 / Real.pi)) := by
  have heq : (fun n : ℕ => ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) =
      fun n : ℕ => (2 / Real.pi) * wallisScaledPartial n := by
    ext n
    have h1 := completeEllipticESeries_one_partialSum_closed_form n
    calc
      ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i
        = (2 / Real.pi) * ((Real.pi / 2) * ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) := by
          rw [← mul_assoc]
          have h_cancel : (2 / Real.pi) * (Real.pi / 2) = 1 := by
            have hpipos : 0 < Real.pi := Real.pi_pos
            have hpio : Real.pi ≠ 0 := ne_of_gt hpipos
            calc
              (2 / Real.pi) * (Real.pi / 2) = (2 * Real.pi) / (Real.pi * 2) := by ring
              _ = 1 := by
                rw [mul_comm]
                exact div_self (mul_ne_zero hpio (by norm_num))
          rw [h_cancel, one_mul]
      _ = (2 / Real.pi) * wallisScaledPartial n := by
          rw [h1]
          rfl
  rw [heq]
  have hlimit : nhds (2 / Real.pi) = nhds (2 / Real.pi * 1) := by rw [mul_one]
  rw [hlimit]
  exact Filter.Tendsto.const_mul (2 / Real.pi) tendsto_wallisScaledPartial

def eCoeffAbs (n : ℕ) : ℝ := |ellipticESeriesCoeff n|

theorem eCoeffAbs_zero : eCoeffAbs 0 = 1 := by
  unfold eCoeffAbs
  rw[ellipticESeriesCoeff_zero, abs_one]

theorem eCoeffAbs_succ (n : ℕ) : eCoeffAbs (n + 1) = -ellipticESeriesCoeff (n + 1) := by
  unfold eCoeffAbs
  have h_nonpos : ellipticESeriesCoeff (n + 1) ≤ 0 := by
    induction n with
    | zero =>
        rw[ellipticESeriesCoeff_one]
        norm_num
    | succ n ih =>
        rw[ellipticESeriesCoeff_succ]
        have hratio : 0 ≤ ellipticESeriesRatio (n + 1) := by
          unfold ellipticESeriesRatio
          have hnum : 0 ≤ (2 * ((n + 1 : ℕ) : ℝ) - 1) * (2 * ((n + 1 : ℕ) : ℝ) + 1) := by
            push_cast
            have h1 : 1 ≤ (n : ℝ) + 1 := by exact_mod_cast Nat.succ_le_succ (Nat.zero_le n)
            nlinarith
          have hden : 0 < (2 * ((n + 1 : ℕ) : ℝ) + 2) ^ 2 := by positivity
          exact div_nonneg hnum (le_of_lt hden)
        exact mul_nonpos_of_nonneg_of_nonpos hratio ih
  exact abs_of_nonpos h_nonpos

theorem eCoeffAbs_nonneg (n : ℕ) :
    0 ≤ eCoeffAbs n := by
  unfold eCoeffAbs
  exact abs_nonneg _

theorem sum_eCoeffAbs_succ_eq (n : ℕ) :
    ∑ i ∈ Finset.range (n + 1), eCoeffAbs i =
      2 - ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i := by
  induction n with
  | zero =>
      rw [Finset.sum_range_one, eCoeffAbs_zero]
      rw [Finset.sum_range_one, ellipticESeriesCoeff_zero]
      norm_num
  | succ n ih =>
      have hsum :
          (∑ i ∈ Finset.range (n + 1 + 1), ellipticESeriesCoeff i) =
            (∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) +
              ellipticESeriesCoeff (n + 1) := by
        rw [Finset.sum_range_succ]
      calc
        ∑ i ∈ Finset.range (n + 1 + 1), eCoeffAbs i
            =
          (∑ i ∈ Finset.range (n + 1), eCoeffAbs i) +
            eCoeffAbs (n + 1) := by
              rw [Finset.sum_range_succ]
        _ =
          (2 - ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) +
            eCoeffAbs (n + 1) := by
              rw [ih]
        _ =
          2 - ∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i -
            ellipticESeriesCoeff (n + 1) := by
              rw [eCoeffAbs_succ n]
              ring
        _ =
          2 -
            ((∑ i ∈ Finset.range (n + 1), ellipticESeriesCoeff i) +
              ellipticESeriesCoeff (n + 1)) := by
              ring
        _ =
          2 - ∑ i ∈ Finset.range (n + 1 + 1), ellipticESeriesCoeff i := by
              rw [hsum]

theorem sum_eCoeffAbs (n : ℕ) :
    ∑ i ∈ Finset.range n, eCoeffAbs i =
      if n = 0 then 0 else 2 - ∑ i ∈ Finset.range n, ellipticESeriesCoeff i := by
  cases n with
  | zero => simp
  | succ n =>
      have h_nz : n + 1 ≠ 0 := Nat.succ_ne_zero n
      rw [if_neg h_nz]
      exact sum_eCoeffAbs_succ_eq n

theorem sum_eCoeffAbs_le (n : ℕ) : ∑ i ∈ Finset.range n, eCoeffAbs i ≤ 2 := by
  rw [sum_eCoeffAbs]
  split_ifs with h
  · norm_num
  · have h_sum_pos : 0 ≤ ∑ i ∈ Finset.range n, ellipticESeriesCoeff i := by
      cases n with
      | zero => contradiction
      | succ m =>
          rw[ellipticESeriesCoeff_partialSum_closed_form]
          have hm : 0 ≤ 2 * (m : ℝ) + 1 := by positivity
          have hw : 0 ≤ wallisEvenCoeff m ^ 2 := by positivity
          exact mul_nonneg hm hw
    linarith

theorem summable_eCoeffAbs : Summable eCoeffAbs :=
  summable_of_sum_range_le eCoeffAbs_nonneg sum_eCoeffAbs_le

theorem summable_ellipticESeriesCoeff : Summable ellipticESeriesCoeff := by
  have h_abs : Summable (fun n => ‖ellipticESeriesCoeff n‖) := by
    simp_rw [Real.norm_eq_abs]
    exact summable_eCoeffAbs
  exact Summable.of_norm h_abs

theorem tendsto_partial_sums_coeff_unbiased :
    Filter.Tendsto (fun n : ℕ => ∑ i ∈ Finset.range n, ellipticESeriesCoeff i)
      Filter.atTop (nhds (2 / Real.pi)) := by
  rw[Metric.tendsto_atTop]

  rw [Metric.tendsto_atTop] at h_tendsto

theorem hasSum_ellipticESeriesCoeff :

/--
Endpoint series value at `k = 1`.

Uses the Wallis squeeze and coefficient partial sums to prove that the complete
elliptic series converges to `1` at the degenerate endpoint.
-/
theorem completeEllipticESeries_one_eq_one :
    completeEllipticESeries 1 = 1 := by
  unfold completeEllipticESeries
  have heq : (fun n : ℕ => ellipticESeriesTerm 1 n) = ellipticESeriesCoeff := by
    ext n
    unfold ellipticESeriesTerm
    have h1 : (1 : ℝ) ^ (2 * n) = 1 := by exact one_pow (2 * n)
    rw [h1, mul_one]
  rw [heq]
  have hsum := hasSum_ellipticESeriesCoeff.tsum_eq
  rw[hsum]
  have h_ne : Real.pi * 2 ≠ 0 := by positivity
  calc
    (Real.pi / 2) * (2 / Real.pi) = (Real.pi * 2) / (Real.pi * 2) := by ring
    _ = 1 := div_self h_ne

/--
Public all-cases perimeter theorem.

For nonnegative semiaxes `a` and `b`, with `a + b > 0`, the model perimeter is
given by the complete elliptic series evaluated at the eccentricity determined
by the major and minor semiaxes.

This theorem includes the strict ellipse case and the two degenerate endpoint
cases `b = 0` and `a = 0`.
-/
theorem ellipsePerimeterModel_eq_four_mul_completeEllipticESeries_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipsePerimeterModel a b =
      4 * max a b *
        completeEllipticESeries
          (eccentricity (max a b) (min a b)) := by
  by_cases ha_pos : 0 < a
  · by_cases hb_pos : 0 < b
    · exact
        ellipsePerimeterModel_eq_four_mul_completeEllipticESeries_of_pos
          ha_pos hb_pos
    · have hb_zero : b = 0 := by
        exact le_antisymm (le_of_not_gt hb_pos) hb
      subst b
      rw [max_eq_left (le_of_lt ha_pos), min_eq_right (le_of_lt ha_pos)]
      exact ellipsePerimeterModel_right_zero_eq_four_mul_series ha_pos
  · have ha_zero : a = 0 := by
      exact le_antisymm (le_of_not_gt ha_pos) ha
    subst a
    have hb_pos : 0 < b := by
      simpa using hsum
    rw [max_eq_right (le_of_lt hb_pos), min_eq_left (le_of_lt hb_pos)]
    exact ellipsePerimeterModel_left_zero_eq_four_mul_series hb_pos

/-
Monotonicity and public bounds.

These theorems package the geometric consequences of the final series theorem:
`completeEllipticESeries` is antitone on `0 ≤ k ≤ 1`, it lies between the
endpoint values `1` and `π / 2`, and the resulting perimeter lies between
`4 * max a b` and `2 * π * max a b`.
-/
theorem completeEllipticETrig_lower_bound_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    1 ≤ completeEllipticETrig k := by
  have h := completeEllipticETrig_antitone_on_unitInterval
    (k := k)
    (l := 1)
    hk.1
    hk.2
    le_rfl
  rw [completeEllipticETrig_one] at h
  exact h

theorem completeEllipticETrig_upper_bound_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    completeEllipticETrig k ≤ Real.pi / 2 := by
  have h := completeEllipticETrig_antitone_on_unitInterval
    (k := 0)
    (l := k)
    (by norm_num)
    hk.1
    hk.2
  rw [completeEllipticETrig_zero] at h
  exact h

theorem completeEllipticETrig_mem_Icc_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    completeEllipticETrig k ∈ Set.Icc (1 : ℝ) (Real.pi / 2) := by
  constructor
  · exact completeEllipticETrig_lower_bound_of_mem_Icc hk
  · exact completeEllipticETrig_upper_bound_of_mem_Icc hk

theorem completeEllipticESeries_eq_completeEllipticETrig_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    completeEllipticESeries k = completeEllipticETrig k := by
  by_cases hk_one : k = 1
  · subst k
    rw [completeEllipticESeries_one_eq_one, completeEllipticETrig_one]
  · have hk_lt_one : k < 1 := by
      exact lt_of_le_of_ne hk.2 hk_one
    have hk_abs : |k| < 1 := by
      rw [abs_of_nonneg hk.1]
      exact hk_lt_one
    exact (seriesExpansionTarget_final_clean hk_abs).symm

theorem completeEllipticESeries_lower_bound_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    1 ≤ completeEllipticESeries k := by
  rw [completeEllipticESeries_eq_completeEllipticETrig_of_mem_Icc hk]
  exact completeEllipticETrig_lower_bound_of_mem_Icc hk

theorem completeEllipticESeries_upper_bound_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    completeEllipticESeries k ≤ Real.pi / 2 := by
  rw [completeEllipticESeries_eq_completeEllipticETrig_of_mem_Icc hk]
  exact completeEllipticETrig_upper_bound_of_mem_Icc hk

theorem completeEllipticESeries_mem_Icc_of_mem_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) :
    completeEllipticESeries k ∈ Set.Icc (1 : ℝ) (Real.pi / 2) := by
  constructor
  · exact completeEllipticESeries_lower_bound_of_mem_Icc hk
  · exact completeEllipticESeries_upper_bound_of_mem_Icc hk

theorem completeEllipticESeries_antitone_on_unitInterval
    {k l : ℝ} (hk0 : 0 ≤ k) (hkl : k ≤ l) (hl1 : l ≤ 1) :
    completeEllipticESeries l ≤ completeEllipticESeries k := by
  have hk1 : k ≤ 1 := le_trans hkl hl1
  have hl0 : 0 ≤ l := le_trans hk0 hkl
  rw [completeEllipticESeries_eq_completeEllipticETrig_of_mem_Icc
        (show l ∈ Set.Icc (0 : ℝ) 1 from ⟨hl0, hl1⟩)]
  rw [completeEllipticESeries_eq_completeEllipticETrig_of_mem_Icc
        (show k ∈ Set.Icc (0 : ℝ) 1 from ⟨hk0, hk1⟩)]
  exact completeEllipticETrig_antitone_on_unitInterval hk0 hkl hl1

theorem max_pos_of_nonneg_of_pos_sum
    {a b : ℝ} (_ha : 0 ≤ a) (_hb : 0 ≤ b) (hsum : 0 < a + b) :
    0 < max a b := by
  by_cases hba : b ≤ a
  · rw [max_eq_left hba]
    nlinarith
  · have hab : a ≤ b := le_of_not_ge hba
    rw [max_eq_right hab]
    nlinarith

theorem min_le_max_of_nonneg
    {a b : ℝ} :
    min a b ≤ max a b := by
  by_cases hba : b ≤ a
  · rw [min_eq_right hba, max_eq_left hba]
    exact hba
  · have hab : a ≤ b := le_of_not_ge hba
    rw [min_eq_left hab, max_eq_right hab]
    exact hab

theorem eccentricity_max_min_mem_Icc_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    eccentricity (max a b) (min a b) ∈ Set.Icc (0 : ℝ) 1 := by
  have hmaxpos : 0 < max a b := by
    exact max_pos_of_nonneg_of_pos_sum ha hb hsum
  have hminnonneg : 0 ≤ min a b := by
    exact le_min ha hb
  have hminlemax : min a b ≤ max a b := by
    exact min_le_max_of_nonneg
  exact eccentricity_mem_Icc_of_le hmaxpos hminnonneg hminlemax

theorem ellipsePerimeter_lower_bound_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    4 * max a b ≤ ellipsePerimeter a b := by
  rw [ellipsePerimeter_eq_four_mul_completeEllipticESeries_of_nonneg
    ha hb hsum]
  have hk :
      eccentricity (max a b) (min a b) ∈ Set.Icc (0 : ℝ) 1 := by
    exact eccentricity_max_min_mem_Icc_of_nonneg ha hb hsum
  have hE :
      1 ≤ completeEllipticESeries
        (eccentricity (max a b) (min a b)) := by
    exact completeEllipticESeries_lower_bound_of_mem_Icc hk
  have hcoef : 0 ≤ 4 * max a b := by
    have hmax : 0 ≤ max a b := by
      exact le_of_lt (max_pos_of_nonneg_of_pos_sum ha hb hsum)
    nlinarith
  calc
    4 * max a b
        = (4 * max a b) * 1 := by
            ring
    _ ≤ (4 * max a b) *
          completeEllipticESeries
            (eccentricity (max a b) (min a b)) := by
            exact mul_le_mul_of_nonneg_left hE hcoef
    _ = 4 * max a b *
          completeEllipticESeries
            (eccentricity (max a b) (min a b)) := by
            ring

theorem ellipsePerimeter_upper_bound_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipsePerimeter a b ≤ 2 * Real.pi * max a b := by
  rw [ellipsePerimeter_eq_four_mul_completeEllipticESeries_of_nonneg
    ha hb hsum]
  have hk :
      eccentricity (max a b) (min a b) ∈ Set.Icc (0 : ℝ) 1 := by
    exact eccentricity_max_min_mem_Icc_of_nonneg ha hb hsum
  have hE :
      completeEllipticESeries
        (eccentricity (max a b) (min a b)) ≤ Real.pi / 2 := by
    exact completeEllipticESeries_upper_bound_of_mem_Icc hk
  have hcoef : 0 ≤ 4 * max a b := by
    have hmax : 0 ≤ max a b := by
      exact le_of_lt (max_pos_of_nonneg_of_pos_sum ha hb hsum)
    nlinarith
  calc
    4 * max a b *
          completeEllipticESeries
            (eccentricity (max a b) (min a b))
        = (4 * max a b) *
          completeEllipticESeries
            (eccentricity (max a b) (min a b)) := by
            ring
    _ ≤ (4 * max a b) * (Real.pi / 2) := by
            exact mul_le_mul_of_nonneg_left hE hcoef
    _ = 2 * Real.pi * max a b := by
            ring

theorem ellipsePerimeter_bounds_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    4 * max a b ≤ ellipsePerimeter a b ∧
      ellipsePerimeter a b ≤ 2 * Real.pi * max a b := by
  constructor
  · exact ellipsePerimeter_lower_bound_of_nonneg ha hb hsum
  · exact ellipsePerimeter_upper_bound_of_nonneg ha hb hsum

/--
Classical textbook series form.

Formats the result as

`P = 2 * π * A * ∑' n, C n * e ^ (2 * n)`,

where `A = max a b` is the semi-major axis and
`e = eccentricity (max a b) (min a b)`.
-/
theorem ellipsePerimeter_eq_two_pi_mul_tsum_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipsePerimeter a b =
      2 * Real.pi * max a b *
        ∑' n : ℕ,
          ellipticESeriesCoeff n *
            eccentricity (max a b) (min a b) ^ (2 * n) := by
  rw [ellipsePerimeter_eq_four_mul_tsum_of_nonneg ha hb hsum]
  ring

/-
Geometric ellipse arc-length bridge.

This block connects the analytic model to the classical geometric perimeter of
the ellipse.  The quadrant arc length is written in the standard normalized
major/minor-axis form, then proved equal to `A * completeEllipticETrig e`.
-/
def ellipseQuadrantArcLength (A B : ℝ) : ℝ :=
  ∫ θ in (0 : ℝ)..(Real.pi / 2),
    Real.sqrt (A ^ 2 * Real.cos θ ^ 2 + B ^ 2 * Real.sin θ ^ 2)

def ellipseGeometricPerimeter (a b : ℝ) : ℝ :=
  4 * ellipseQuadrantArcLength (max a b) (min a b)

theorem ellipseQuadrantArcLength_eq_major_mul_completeEllipticETrig_of_le
    {A B : ℝ} (hA : 0 < A) (hB : 0 ≤ B) (hBA : B ≤ A) :
    ellipseQuadrantArcLength A B =
      A * completeEllipticETrig (eccentricity A B) := by
  unfold ellipseQuadrantArcLength completeEllipticETrig
  rw [← intervalIntegral.integral_const_mul]
  apply intervalIntegral.integral_congr
  intro θ _hθ
  have hA0 : A ≠ 0 := ne_of_gt hA
  have he :
      eccentricity A B ^ 2 = eccentricitySq A B := by
    exact eccentricity_sq_eq_eccentricitySq_of_le hA hB hBA
  have hrad :
      A ^ 2 * Real.cos θ ^ 2 + B ^ 2 * Real.sin θ ^ 2 =
        A ^ 2 *
          (1 - eccentricity A B ^ 2 * Real.sin θ ^ 2) := by
    rw [he]
    unfold eccentricitySq
    field_simp [hA0]
    nlinarith [Real.sin_sq_add_cos_sq θ]
  calc
    Real.sqrt (A ^ 2 * Real.cos θ ^ 2 + B ^ 2 * Real.sin θ ^ 2)
        =
          Real.sqrt
            (A ^ 2 *
              (1 - eccentricity A B ^ 2 * Real.sin θ ^ 2)) := by
            rw [hrad]
    _ =
          Real.sqrt (A ^ 2) *
            Real.sqrt (1 - eccentricity A B ^ 2 * Real.sin θ ^ 2) := by
            rw [Real.sqrt_mul (sq_nonneg A)]
    _ =
          A * Real.sqrt
            (1 - eccentricity A B ^ 2 * Real.sin θ ^ 2) := by
            rw [Real.sqrt_sq_eq_abs, abs_of_nonneg hA.le]

theorem ellipseGeometricPerimeter_eq_ellipsePerimeter
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipseGeometricPerimeter a b = ellipsePerimeter a b := by
  unfold ellipseGeometricPerimeter ellipsePerimeter
  have hmaxpos : 0 < max a b := by
    exact max_pos_of_nonneg_of_pos_sum ha hb hsum
  have hminnonneg : 0 ≤ min a b := by
    exact le_min ha hb
  have hminlemax : min a b ≤ max a b := by
    exact min_le_max_of_nonneg
  rw [
    ellipseQuadrantArcLength_eq_major_mul_completeEllipticETrig_of_le
      hmaxpos hminnonneg hminlemax
  ]
  rw [ellipsePerimeterModel_eq_four_mul_completeEllipticETrig_of_nonneg
    ha hb hsum]
  ring

theorem ellipseGeometricPerimeter_eq_four_mul_completeEllipticESeries_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipseGeometricPerimeter a b =
      4 * max a b *
        completeEllipticESeries
          (eccentricity (max a b) (min a b)) := by
  rw [ellipseGeometricPerimeter_eq_ellipsePerimeter ha hb hsum]

theorem ellipseGeometricPerimeter_eq_four_mul_tsum_of_nonneg

/--

/-
Closed-endpoint continuity for the elliptic power series.

This packages the missing topological fact: the series representation is not
just glued at `k = 1`; the power series side is continuous on `[0, 1]`, so the
endpoint value is obtained as the within-limit from the closed unit interval.
-/
theorem ellipticESeriesTerm_continuous (n : ℕ) :
    Continuous (fun k : ℝ => ellipticESeriesTerm k n) := by
  unfold ellipticESeriesTerm
  exact continuous_const.mul (continuous_id.pow (2 * n))

theorem norm_ellipticESeriesTerm_le_eCoeffAbs_on_Icc
    {k : ℝ} (hk : k ∈ Set.Icc (0 : ℝ) 1) (n : ℕ) :
    ‖ellipticESeriesTerm k n‖ ≤ eCoeffAbs n := by
  have hkabs : |k| ≤ 1 := by
    exact abs_le.mpr ⟨by nlinarith [hk.1], hk.2⟩
  have hpow : |k| ^ (2 * n) ≤ 1 := by
    exact pow_le_one₀ (abs_nonneg k) hkabs
  unfold ellipticESeriesTerm eCoeffAbs
  rw [Real.norm_eq_abs, abs_mul, abs_pow]
  calc
    |ellipticESeriesCoeff n| * |k| ^ (2 * n)
        ≤ |ellipticESeriesCoeff n| * 1 := by
            exact mul_le_mul_of_nonneg_left hpow (abs_nonneg _)
    _ = |ellipticESeriesCoeff n| := by
            ring

theorem ellipticESeriesTerm_tsum_continuousOn_Icc_zero_one :
    ContinuousOn
      (fun k : ℝ => ∑' n : ℕ, ellipticESeriesTerm k n)
      (Set.Icc (0 : ℝ) 1) := by
  apply continuousOn_tsum
  · intro n
    exact (ellipticESeriesTerm_continuous n).continuousOn
  · exact summable_eCoeffAbs
  · intro n k hk
    exact norm_ellipticESeriesTerm_le_eCoeffAbs_on_Icc hk n

theorem completeEllipticESeries_continuousOn_Icc_zero_one :

/-
Full-loop parametric ellipse arc length.

This block introduces the classical full-loop arc-length integral for the
parametrized ellipse

  γ(θ) = (A * cos θ, B * sin θ)

whose speed is

  sqrt (A ^ 2 * sin θ ^ 2 + B ^ 2 * cos θ ^ 2).

The later symmetry theorem will identify this full-loop integral with
`4 * ellipseQuadrantArcLength A B`.
-/
def ellipseFullLoopArcLength (A B : ℝ) : ℝ :=
  ∫ θ in (0 : ℝ)..(2 * Real.pi),
    Real.sqrt (A ^ 2 * Real.sin θ ^ 2 + B ^ 2 * Real.cos θ ^ 2)

def ellipseParametricPerimeter (a b : ℝ) : ℝ :=
  ellipseFullLoopArcLength (max a b) (min a b)

theorem ellipseParametricPerimeter_unfold
    (a b : ℝ) :
    ellipseParametricPerimeter a b =
      ∫ θ in (0 : ℝ)..(2 * Real.pi),
        Real.sqrt
          ((max a b) ^ 2 * Real.sin θ ^ 2 +
            (min a b) ^ 2 * Real.cos θ ^ 2) := by
  rfl

theorem ellipseFullLoopArcLength_integrand_period_pi
    (A B θ : ℝ) :
    Real.sqrt
        (A ^ 2 * Real.sin (θ + Real.pi) ^ 2 +
          B ^ 2 * Real.cos (θ + Real.pi) ^ 2) =
      Real.sqrt
        (A ^ 2 * Real.sin θ ^ 2 +
          B ^ 2 * Real.cos θ ^ 2) := by
  rw [Real.sin_add_pi, Real.cos_add_pi]
  ring_nf

theorem ellipseFullLoopArcLength_integrand_reflect_pi
    (A B θ : ℝ) :
    Real.sqrt
        (A ^ 2 * Real.sin (Real.pi - θ) ^ 2 +
          B ^ 2 * Real.cos (Real.pi - θ) ^ 2) =
      Real.sqrt
        (A ^ 2 * Real.sin θ ^ 2 +
          B ^ 2 * Real.cos θ ^ 2) := by
  rw [Real.sin_pi_sub, Real.cos_pi_sub]
  ring_nf

theorem ellipseFullLoopArcLength_integrand_quadrant_swap
    (A B θ : ℝ) :
    Real.sqrt
        (A ^ 2 * Real.sin (Real.pi / 2 - θ) ^ 2 +
          B ^ 2 * Real.cos (Real.pi / 2 - θ) ^ 2) =
      Real.sqrt
        (A ^ 2 * Real.cos θ ^ 2 +
          B ^ 2 * Real.sin θ ^ 2) := by
  rw [Real.sin_pi_div_two_sub, Real.cos_pi_div_two_sub]

/-
Full-loop symmetry closure.

This proves that the classical full-loop parametric arc-length integral is
exactly four times the normalized quadrant integral. This closes the geometric
gap between the textbook parametrization and the quadrant elliptic-integral
model.
-/
def ellipseFullLoopSpeed (A B θ : ℝ) : ℝ :=
  Real.sqrt (A ^ 2 * Real.sin θ ^ 2 + B ^ 2 * Real.cos θ ^ 2)

def ellipseQuadrantSpeed (A B θ : ℝ) : ℝ :=
  Real.sqrt (A ^ 2 * Real.cos θ ^ 2 + B ^ 2 * Real.sin θ ^ 2)

theorem ellipseFullLoopSpeed_continuous (A B : ℝ) :
    Continuous (fun θ : ℝ => ellipseFullLoopSpeed A B θ) := by
  unfold ellipseFullLoopSpeed
  exact Real.continuous_sqrt.comp
    ((continuous_const.mul (Real.continuous_sin.pow 2)).add
      (continuous_const.mul (Real.continuous_cos.pow 2)))

theorem ellipseFullLoopSpeed_intervalIntegrable
    (A B a b : ℝ) :
    IntervalIntegrable
      (fun θ : ℝ => ellipseFullLoopSpeed A B θ)
      MeasureTheory.volume
      a
      b := by
  exact (ellipseFullLoopSpeed_continuous A B).intervalIntegrable _ _

theorem ellipseQuadrantSpeed_continuous (A B : ℝ) :
    Continuous (fun θ : ℝ => ellipseQuadrantSpeed A B θ) := by
  unfold ellipseQuadrantSpeed
  exact Real.continuous_sqrt.comp
    ((continuous_const.mul (Real.continuous_cos.pow 2)).add
      (continuous_const.mul (Real.continuous_sin.pow 2)))

theorem ellipseFullLoopSpeed_add_pi_div_two
    (A B θ : ℝ) :
    ellipseFullLoopSpeed A B (θ + Real.pi / 2) =
      ellipseQuadrantSpeed A B θ := by
  unfold ellipseFullLoopSpeed ellipseQuadrantSpeed
  rw [Real.sin_add_pi_div_two, Real.cos_add_pi_div_two]
  ring_nf

theorem ellipseFullLoopSpeed_reflect_pi_div_two
    (A B θ : ℝ) :
    ellipseFullLoopSpeed A B (Real.pi / 2 - θ) =
      ellipseQuadrantSpeed A B θ := by
  unfold ellipseFullLoopSpeed ellipseQuadrantSpeed
  rw [Real.sin_pi_div_two_sub, Real.cos_pi_div_two_sub]

theorem ellipseFullLoopSpeed_period_pi
    (A B θ : ℝ) :
    ellipseFullLoopSpeed A B (θ + Real.pi) =
      ellipseFullLoopSpeed A B θ := by
  unfold ellipseFullLoopSpeed
  rw [Real.sin_add_pi, Real.cos_add_pi]
  ring_nf

theorem ellipseFullLoop_first_quadrant_eq_quadrant
    (A B : ℝ) :
    (∫ θ in (0 : ℝ)..(Real.pi / 2),
      ellipseFullLoopSpeed A B θ) =
    ∫ θ in (0 : ℝ)..(Real.pi / 2),
      ellipseQuadrantSpeed A B θ := by
  have hsubst :=
    intervalIntegral.smul_integral_comp_mul_add
      (fun θ : ℝ => ellipseFullLoopSpeed A B θ)
      (-1 : ℝ)
      (Real.pi / 2)
      (a := (0 : ℝ))
      (b := Real.pi / 2)
  have hcomp :
      - (∫ θ in (0 : ℝ)..(Real.pi / 2),
          ellipseFullLoopSpeed A B (-θ + Real.pi / 2)) =
      ∫ θ in (-0 + Real.pi / 2)..(-(Real.pi / 2) + Real.pi / 2),
          ellipseFullLoopSpeed A B θ := by
    simpa only [smul_eq_mul, neg_one_mul] using hsubst
  have hsymm :
      (∫ θ in (-0 + Real.pi / 2)..(-(Real.pi / 2) + Real.pi / 2),
          ellipseFullLoopSpeed A B θ) =
      - (∫ θ in (0 : ℝ)..(Real.pi / 2),
          ellipseFullLoopSpeed A B θ) := by
    rw [
      show -0 + Real.pi / 2 = Real.pi / 2 by ring,
      show -(Real.pi / 2) + Real.pi / 2 = 0 by ring,
      intervalIntegral.integral_symm
    ]
  have hrev :
      (∫ θ in (0 : ℝ)..(Real.pi / 2),
          ellipseFullLoopSpeed A B θ) =
      ∫ θ in (0 : ℝ)..(Real.pi / 2),
        ellipseFullLoopSpeed A B (-θ + Real.pi / 2) := by
    exact neg_inj.mp (hcomp.trans hsymm).symm
  rw [hrev]
  apply intervalIntegral.integral_congr
  intro θ _hθ
  change
    ellipseFullLoopSpeed A B (-θ + Real.pi / 2) =
      ellipseQuadrantSpeed A B θ
  rw [show -θ + Real.pi / 2 = Real.pi / 2 - θ by ring]
  exact ellipseFullLoopSpeed_reflect_pi_div_two A B θ

theorem ellipseFullLoop_second_quadrant_eq_quadrant
    (A B : ℝ) :
    (∫ θ in (Real.pi / 2)..Real.pi,
      ellipseFullLoopSpeed A B θ) =
    ∫ θ in (0 : ℝ)..(Real.pi / 2),
      ellipseQuadrantSpeed A B θ := by
  have hshift :=
    intervalIntegral.integral_comp_add_right
      (fun θ : ℝ => ellipseFullLoopSpeed A B θ)
      (Real.pi / 2)
      (a := (0 : ℝ))
      (b := Real.pi / 2)
  have htarget :
      (∫ θ in (Real.pi / 2)..Real.pi,
        ellipseFullLoopSpeed A B θ) =
      ∫ θ in (0 : ℝ)..(Real.pi / 2),
        ellipseFullLoopSpeed A B (θ + Real.pi / 2) := by
    simpa only
      [zero_add, show Real.pi / 2 + Real.pi / 2 = Real.pi by ring]
      using hshift.symm
  rw [htarget]
  apply intervalIntegral.integral_congr
  intro θ _hθ
  exact ellipseFullLoopSpeed_add_pi_div_two A B θ

theorem ellipseFullLoop_half_eq_two_mul_quadrant
    (A B : ℝ) :
    (∫ θ in (0 : ℝ)..Real.pi,
      ellipseFullLoopSpeed A B θ) =
    2 * ellipseQuadrantArcLength A B := by
  have hsplit :
      (∫ θ in (0 : ℝ)..(Real.pi / 2),
          ellipseFullLoopSpeed A B θ) +
        (∫ θ in (Real.pi / 2)..Real.pi,
          ellipseFullLoopSpeed A B θ) =
      ∫ θ in (0 : ℝ)..Real.pi,
        ellipseFullLoopSpeed A B θ := by
    exact intervalIntegral.integral_add_adjacent_intervals
      (ellipseFullLoopSpeed_intervalIntegrable A B 0 (Real.pi / 2))
      (ellipseFullLoopSpeed_intervalIntegrable A B (Real.pi / 2) Real.pi)
  rw [← hsplit]
  rw [
    ellipseFullLoop_first_quadrant_eq_quadrant,
    ellipseFullLoop_second_quadrant_eq_quadrant
  ]
  unfold ellipseQuadrantArcLength ellipseQuadrantSpeed
  ring

theorem ellipseFullLoop_second_half_eq_first_half
    (A B : ℝ) :
    (∫ θ in Real.pi..(2 * Real.pi),
      ellipseFullLoopSpeed A B θ) =
    ∫ θ in (0 : ℝ)..Real.pi,
      ellipseFullLoopSpeed A B θ := by
  have hshift :=
    intervalIntegral.integral_comp_add_right
      (fun θ : ℝ => ellipseFullLoopSpeed A B θ)
      Real.pi
      (a := (0 : ℝ))
      (b := Real.pi)
  have htarget :
      (∫ θ in Real.pi..(2 * Real.pi),
        ellipseFullLoopSpeed A B θ) =
      ∫ θ in (0 : ℝ)..Real.pi,
        ellipseFullLoopSpeed A B (θ + Real.pi) := by
    simpa only
      [zero_add, show Real.pi + Real.pi = 2 * Real.pi by ring]
      using hshift.symm
  rw [htarget]
  apply intervalIntegral.integral_congr
  intro θ _hθ
  exact ellipseFullLoopSpeed_period_pi A B θ

theorem ellipseFullLoopArcLength_eq_four_mul_quadrant
    (A B : ℝ) :
    ellipseFullLoopArcLength A B =
      4 * ellipseQuadrantArcLength A B := by
  unfold ellipseFullLoopArcLength
  change
    (∫ θ in (0 : ℝ)..(2 * Real.pi),
      ellipseFullLoopSpeed A B θ) =
    4 * ellipseQuadrantArcLength A B
  have hsplit :
      (∫ θ in (0 : ℝ)..Real.pi,
          ellipseFullLoopSpeed A B θ) +
        (∫ θ in Real.pi..(2 * Real.pi),
          ellipseFullLoopSpeed A B θ) =
      ∫ θ in (0 : ℝ)..(2 * Real.pi),
        ellipseFullLoopSpeed A B θ := by
    exact intervalIntegral.integral_add_adjacent_intervals
      (ellipseFullLoopSpeed_intervalIntegrable A B 0 Real.pi)
      (ellipseFullLoopSpeed_intervalIntegrable A B Real.pi (2 * Real.pi))
  rw [← hsplit]
  rw [ellipseFullLoop_second_half_eq_first_half]
  rw [ellipseFullLoop_half_eq_two_mul_quadrant]
  ring

theorem ellipseParametricPerimeter_eq_ellipseGeometricPerimeter
    (a b : ℝ) :
    ellipseParametricPerimeter a b = ellipseGeometricPerimeter a b := by
  unfold ellipseParametricPerimeter ellipseGeometricPerimeter
  exact ellipseFullLoopArcLength_eq_four_mul_quadrant (max a b) (min a b)

theorem ellipseParametricPerimeter_eq_ellipsePerimeter
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipseParametricPerimeter a b = ellipsePerimeter a b := by
  rw [ellipseParametricPerimeter_eq_ellipseGeometricPerimeter]
  exact ellipseGeometricPerimeter_eq_ellipsePerimeter ha hb hsum

theorem ellipseParametricPerimeter_eq_two_pi_mul_tsum_of_nonneg
    {a b : ℝ} (ha : 0 ≤ a) (hb : 0 ≤ b) (hsum : 0 < a + b) :
    ellipseParametricPerimeter a b =
      2 * Real.pi * max a b *
        ∑' n : ℕ,
          ellipticESeriesCoeff n *
            eccentricity (max a b) (min a b) ^ (2 * n) := by
  rw [ellipseParametricPerimeter_eq_ellipsePerimeter ha hb hsum]
  exact ellipsePerimeter_eq_two_pi_mul_tsum_of_nonneg ha hb hsum
