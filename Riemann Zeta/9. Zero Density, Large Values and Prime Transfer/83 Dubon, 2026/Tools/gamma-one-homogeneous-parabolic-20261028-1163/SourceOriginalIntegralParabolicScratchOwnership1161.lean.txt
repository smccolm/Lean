import SourceCocycleConjugateCoboundary
import SourceParabolicFiniteCuspWitnesses
import SourceParabolicWitnessDenominators
import SourceParabolicUniformDenominator
import SourceOriginalBinaryCuspWitnesses
import SourceParabolicIntegralCoordinateSpan
import SourceOriginalIntegralParabolicCocycleSpan
import SourceOriginalIntegralParabolicH1Span
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceCocycleConjugateCoboundary, `SourceParabolicFiniteCuspWitnesses, `SourceParabolicWitnessDenominators, `SourceParabolicUniformDenominator, `SourceOriginalBinaryCuspWitnesses, `SourceParabolicIntegralCoordinateSpan, `SourceOriginalIntegralParabolicCocycleSpan, `SourceOriginalIntegralParabolicH1Span]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
