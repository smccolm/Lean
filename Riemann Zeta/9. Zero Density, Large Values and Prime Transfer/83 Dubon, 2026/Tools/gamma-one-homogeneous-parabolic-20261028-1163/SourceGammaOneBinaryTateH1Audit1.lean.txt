private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneFiniteOrderDischarge, `SourceGammaOneBinaryPeriodCharacters, `SourceGammaOneCoefficientHeckeCharacters, `SourceGammaOneBinaryParabolicH1, `SourceGammaOneBinaryTateH1, `SourceGammaOneBinaryTateHecke]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.finiteOrder_trace_sq_four, `Dubon2026.SourceFunctionField.GammaOne.finiteOrder_character_zero_of_parabolic, `Dubon2026.SourceFunctionField.GammaOne.finiteParabolicHoms_mem_iff_of_level, `Dubon2026.SourceFunctionField.GammaOne.originalGammaOne_level_ge_four, `Dubon2026.SourceFunctionField.GammaOne.originalGammaOne_finiteOrder_character_zero, `Dubon2026.SourceFunctionField.GammaOne.binaryPeriodCharacterMap_injective, `Dubon2026.SourceFunctionField.GammaOne.binaryPeriodCharacterMap_surjective, `Dubon2026.SourceFunctionField.GammaOne.binaryPeriodCharacterEquiv_apply, `Dubon2026.sourceGammaOneConstantHeckeH1_coefficient_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.binaryLatticeH1Equiv_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.binaryLatticeH1Equiv_hecke, `Dubon2026.SourceFunctionField.GammaOne.binaryParabolicH1_hecke_mem, `Dubon2026.SourceFunctionField.GammaOne.binaryTateCharacterRestriction_injective, `Dubon2026.SourceFunctionField.GammaOne.binaryTateCharacterRestriction_surjective, `Dubon2026.SourceFunctionField.GammaOne.binaryTateH1Equiv_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.binaryTateCharacterEquiv_prime, `Dubon2026.SourceFunctionField.GammaOne.binaryTateH1Equiv_prime, `Dubon2026.SourceFunctionField.GammaOne.binaryTateH1Equiv_word]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.binaryParabolicHoms, `Dubon2026.SourceFunctionField.GammaOne.binaryPeriodCharacterMap, `Dubon2026.SourceFunctionField.GammaOne.binaryPeriodCharacterEquiv, `Dubon2026.SourceFunctionField.GammaOne.binaryParabolicH1, `Dubon2026.SourceFunctionField.GammaOne.binaryLatticeH1Equiv, `Dubon2026.SourceFunctionField.GammaOne.binaryTateCharacterRestriction, `Dubon2026.SourceFunctionField.GammaOne.binaryTateCharacterEquiv, `Dubon2026.SourceFunctionField.GammaOne.binaryTateH1Equiv, `Dubon2026.SourceFunctionField.GammaOne.binaryTatePrimeOperators]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Binary Tate H1 postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Binary Tate H1 theorems discovered"
  logInfo m!"SOURCE GAMMA ONE BINARY TATE H1 AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
