import SourceParabolicCohomology
import SourceBinaryParabolicChange
import SourceGammaOneParabolicRestriction
import SourceParabolicCoefficientMap
import SourcePeriodParabolicClass
import SourceOriginalParabolicEigensystem
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceParabolicCohomology, `SourceBinaryParabolicChange, `SourceGammaOneParabolicRestriction, `SourceParabolicCoefficientMap, `SourcePeriodParabolicClass, `SourceOriginalParabolicEigensystem]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
