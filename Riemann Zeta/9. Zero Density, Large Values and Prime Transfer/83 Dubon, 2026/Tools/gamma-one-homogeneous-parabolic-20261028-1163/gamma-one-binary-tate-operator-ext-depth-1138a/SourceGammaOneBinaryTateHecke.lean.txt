/-! DUB17/18 -> original all-weight arithmetic realization -> original
finite Gamma1 homogeneous-coefficient Hecke operators. Define their
arithmetic Tate dual action using the literal Picard correspondence
and original homogeneousMatrixAction. The proved period inclusion and
native H1 coset formula identify these independently defined operators,
then the existing integral-word theorem gives every word identity.
This comparison is on the literal parabolic submodule; descent of the

variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w

private local instance : Algebra ℤ_[p] (ZMod N) :=
  (PadicInt.toZModPow (n + 1)).toAlgebra

  Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

/-- The actual arithmetic Picard Hecke dual with the original homogeneous coefficient action. -/
def binaryTatePrimeOperators :
    sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N)
      (TateModule p (Pic0 (AlgebraicClosure ℚ)
        (functionFieldOver (Gamma1 M) (AlgebraicClosure ℚ))) →ₗ[ℤ_[p]] V) :=
  fun ell =>
    letI : Fact ell.val.Prime := ⟨ell.property.1⟩
    let b := homogeneousMatrixAction w (Matrix.diagonal ![(ell.val : ZMod N), 1])
    let T := TateModule.tateMap p (heckePicard (AlgebraicClosure ℚ) M ell.val)
    { toFun := fun φ => (b.restrictScalars ℤ_[p]).comp (φ.comp T)
      map_add' := fun φ ψ => by
        ext x
        exact b.map_add _ _
      map_smul' := fun c φ => by
        ext x
        exact b.map_smul c _ }

theorem binaryTateCharacterEquiv_prime (ell : sourceBinaryHeckePrime (Q * N))
    (φ : TateModule p (Pic0 (AlgebraicClosure ℚ)
      (functionFieldOver (Gamma1 M) (AlgebraicClosure ℚ))) →ₗ[ℤ_[p]] V) :
    letI : Fact ell.val.Prime := ⟨ell.property.1⟩
    binaryTateCharacterEquiv Q p n w (binaryTatePrimeOperators Q p n w ell φ) =
      (((homogeneousMatrixAction w (Matrix.diagonal ![(ell.val : ZMod N), 1])).toAddMonoidHom.comp
        (binaryTateCharacterEquiv Q p n w φ)).comp (periodLatticeHecke M ell.val).toAddMonoidHom) := by
  letI : Fact ell.val.Prime := ⟨ell.property.1⟩
  dsimp only
  apply AddMonoidHom.ext
  intro l
  exact congrArg (homogeneousMatrixAction w (Matrix.diagonal ![(ell.val : ZMod N), 1]))
    (congrArg φ (arithmeticPeriodInclusion_hecke M p ell.val l))

theorem binaryTateH1Equiv_prime (hlarge : 3 < p)
    (ell : sourceBinaryHeckePrime (Q * N))
    (φ : TateModule p (Pic0 (AlgebraicClosure ℚ)
      (functionFieldOver (Gamma1 M) (AlgebraicClosure ℚ))) →ₗ[ℤ_[p]] V) :
    (binaryTateH1Equiv Q p n w hlarge (binaryTatePrimeOperators Q p n w ell φ)).val =
      sourceGammaOnePrimeOperators Q N w ell (binaryTateH1Equiv Q p n w hlarge φ).val := by
  letI : Fact ell.val.Prime := ⟨ell.property.1⟩
  change (binaryLatticeH1Equiv Q N w (originalGammaOne_level_ge_four Q p n hlarge)
      (binaryTateCharacterEquiv Q p n w (binaryTatePrimeOperators Q p n w ell φ))).val = _
  rw [binaryTateCharacterEquiv_prime]
  exact binaryLatticeH1Equiv_hecke Q N w (originalGammaOne_level_ge_four Q p n hlarge)
    ell.val (homogeneousMatrixAction w (Matrix.diagonal ![(ell.val : ZMod N), 1]))
      (binaryTateCharacterEquiv Q p n w φ)

theorem binaryTateH1Equiv_word (hlarge : 3 < p)
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N)))
    (φ : TateModule p (Pic0 (AlgebraicClosure ℚ)
      (functionFieldOver (Gamma1 M) (AlgebraicClosure ℚ))) →ₗ[ℤ_[p]] V) :
    (binaryTateH1Equiv Q p n w hlarge
        (FreeAlgebra.lift ℤ (binaryTatePrimeOperators Q p n w) a φ)).val =
      FreeAlgebra.lift ℤ (sourceGammaOnePrimeOperators Q N w) a
        (binaryTateH1Equiv Q p n w hlarge φ).val :=
  sourceHeckeWord_intertwine_addHom (binaryTatePrimeOperators Q p n w)
    (sourceGammaOnePrimeOperators Q N w)
    (((binaryParabolicH1 Q N w).subtype.comp
      (binaryTateH1Equiv Q p n w hlarge).toLinearMap).toAddMonoidHom)
    (binaryTateH1Equiv_prime Q p n w hlarge) a φ

end
