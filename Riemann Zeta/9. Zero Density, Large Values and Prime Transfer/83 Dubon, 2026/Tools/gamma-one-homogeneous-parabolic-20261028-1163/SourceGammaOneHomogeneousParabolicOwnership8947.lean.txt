import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOneFiniteOrderDischarge, `Dubon2026.SourceGammaOneBinaryPeriodCharacters, `Dubon2026.SourceGammaOneCoefficientHeckeCharacters, `Dubon2026.SourceGammaOneBinaryParabolicH1, `Dubon2026.SourceGammaOneBinaryTateH1, `Dubon2026.SourceGammaOneBinaryTateHecke, `Dubon2026.SourceParabolicCohomology, `Dubon2026.SourceBinaryParabolicChange, `Dubon2026.SourceGammaOneParabolicRestriction, `Dubon2026.SourceParabolicCoefficientMap, `Dubon2026.SourcePeriodParabolicClass, `Dubon2026.SourceOriginalParabolicEigensystem, `Dubon2026.SourceParabolicBinaryCoordinates, `Dubon2026.SourceParabolicBinaryRationalSpan, `Dubon2026.SourceParabolicBinaryH1RationalSpan, `Dubon2026.SourceCocycleConjugateCoboundary, `Dubon2026.SourceParabolicFiniteCuspWitnesses, `Dubon2026.SourceParabolicWitnessDenominators, `Dubon2026.SourceParabolicUniformDenominator, `Dubon2026.SourceOriginalBinaryCuspWitnesses, `Dubon2026.SourceParabolicIntegralCoordinateSpan, `Dubon2026.SourceOriginalIntegralParabolicCocycleSpan, `Dubon2026.SourceOriginalIntegralParabolicH1Span, `Dubon2026.SourcePrimitiveParabolicHeckeRelations]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
