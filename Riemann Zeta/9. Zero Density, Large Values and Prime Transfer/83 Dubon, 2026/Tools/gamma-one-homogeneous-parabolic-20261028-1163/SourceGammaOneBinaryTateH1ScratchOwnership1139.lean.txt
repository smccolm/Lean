import SourceGammaOneFiniteOrderDischarge
import SourceGammaOneBinaryPeriodCharacters
import SourceGammaOneCoefficientHeckeCharacters
import SourceGammaOneBinaryParabolicH1
import SourceGammaOneBinaryTateH1
import SourceGammaOneBinaryTateHecke
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneFiniteOrderDischarge, `SourceGammaOneBinaryPeriodCharacters, `SourceGammaOneCoefficientHeckeCharacters, `SourceGammaOneBinaryParabolicH1, `SourceGammaOneBinaryTateH1, `SourceGammaOneBinaryTateHecke]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
