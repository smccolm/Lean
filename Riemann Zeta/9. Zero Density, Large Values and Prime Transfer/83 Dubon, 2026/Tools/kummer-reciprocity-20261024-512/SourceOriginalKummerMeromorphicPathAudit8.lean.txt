private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverMeromorphicPrincipalDivisor, `SourceOriginalWeilCoverRegularDifferentialSubtraction, `SourceOriginalWeilCoverSimplePoleRegularPart, `SourceOriginalWeilCoverLocalRegularPrimitives, `SourceOriginalWeilCoverRegularPathLocalPrimitives, `SourceOriginalWeilCoverRegularPathPrimitives, `SourceOriginalWeilCoverPathIntegralSubtraction, `SourceOriginalWeilCoverRegularPathIntegralLaws]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_isPrincipal_of_meromorphic_orders, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_poles_sub_regular, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_residues_sub_regular, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_pole_regular_part, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analyticAt_readDifferential_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_readDifferential_local_primitive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chart_differentialCoeff_regular_at, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_change_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_path_local_primitive_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_isPrimitiveAlong_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_isPrimitiveAlong_of_simplePole_residue_zero, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_differentialCoeff_sub, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_sub_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isPrimitiveAlong_sub_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.pathIntegral_sub_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.pathIntegral_trans_of_primitive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.pathIntegral_trans_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.pathIntegral_symm_of_coeff_mem]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Meromorphic Path postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Meromorphic Path theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER MEROMORPHIC PATH AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
