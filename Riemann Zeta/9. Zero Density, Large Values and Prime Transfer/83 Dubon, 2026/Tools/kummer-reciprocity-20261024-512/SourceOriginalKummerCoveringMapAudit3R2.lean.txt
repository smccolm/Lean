import SourceAnalyticCoveringComparison
import SourceOriginalWeilRestrictionLocalHomeomorph
import SourceOriginalWeilRestrictionCovering
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceCurve.exists_nhds_injOn_of_analytic_chart_comparison
#print axioms Dubon2026.SourceCurve.isLocalHomeomorph_of_continuous_open_locally_injective
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isLocalHomeomorph_restriction
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isCoveringMap_restriction

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceAnalyticCoveringComparison, `SourceOriginalWeilRestrictionLocalHomeomorph, `SourceOriginalWeilRestrictionCovering]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.exists_nhds_injOn_of_analytic_chart_comparison, `Dubon2026.SourceCurve.isLocalHomeomorph_of_continuous_open_locally_injective, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isLocalHomeomorph_restriction, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isCoveringMap_restriction]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Covering Map postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Covering Map theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER COVERING MAP AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceAnalyticCoveringComparison
#lint in SourceOriginalWeilRestrictionLocalHomeomorph
