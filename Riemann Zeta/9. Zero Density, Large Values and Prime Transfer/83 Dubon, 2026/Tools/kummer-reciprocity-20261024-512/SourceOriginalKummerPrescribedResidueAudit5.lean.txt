private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCurveRationalResidueTrace, `SourceOriginalWeilCoverDifferentialResidues, `SourceOriginalWeilCoverSimplePoleBounds, `SourceOriginalWeilCoverTwoPoleResidues, `SourceOriginalWeilCoverPrescribedResidues]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.Place.trace_residue_eq_rationalEval, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialCoeff_apply, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialResidue_apply, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_pole_bound_iff, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialResidue_simplePole, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialResidue_eq_zero_of_regular, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialResidue_ne_zero_of_simplePole, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_pole_zero_bound, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_pole_add_bound, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_pole_smul_bound, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_simple_pole_sum_bound, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialResidue_finsum, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_two_pole_residue_sum, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_normalized_two_pole_differential, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_divisor_sum_support, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_differential_with_prescribed_residues, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_third_kind_differential]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverCanonicalLocalResidueData, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialCoeff, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDifferentialResidue]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Prescribed Residue postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Prescribed Residue theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER PRESCRIBED RESIDUE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
