import SourceOriginalWeilCoverLiftedCoordinateAnalytic
import SourceOriginalWeilCoverCellLift

open Lean

#print axioms Dubon2026.SourceFunctionField.SheetCover.exists_open_lift_of_starConvex
#print axioms Dubon2026.SourceFunctionField.SheetCover.exists_coordinate_of_open_lift
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analytic_lifted_coordinate
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_cell_lift

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCoverStarConvexLift, `SourceCoverLiftedCoordinate, `SourceOriginalWeilCoverLiftedCoordinateAnalytic, `SourceOriginalWeilCoverCellLift]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.SheetCover.exists_open_lift_of_starConvex, `Dubon2026.SourceFunctionField.SheetCover.exists_coordinate_of_open_lift, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analytic_lifted_coordinate, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_cell_lift]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Cell Lift postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Cell Lift theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER CELL LIFT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceCoverStarConvexLift
#lint in SourceCoverLiftedCoordinate
#lint in SourceOriginalWeilCoverLiftedCoordinateAnalytic
#lint in SourceOriginalWeilCoverCellLift
