private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverRegularChartReadingAt, `SourceOriginalWeilCoverLocalChartPathPrimitive, `SourceOriginalWeilCoverLocalChartPathIntegral, `SourceOriginalWeilCoverIntegralPeriodPathIndependence, `SourceOriginalWeilCoverPeriodExponential, `SourceOriginalWeilCoverPeriodExponentialConsumer, `SourceOriginalWeilCoverExponentialLocalTransport, `SourceOriginalWeilCoverExponentialChartDerivative, `SourceOriginalWeilCoverExponentialMeromorphicOrders, `SourceOriginalWeilCoverIntegralPeriodsPrincipal]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analyticAt_readDifferential_at_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isPrimitiveAlong_of_local_chart_primitive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_local_chart_path_integral, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_coeff_regular_off_divisor, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exp_pathIntegral_eq_of_integral_periods, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDivisorAvoidingPath_avoids, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverPeriodExponential_ne_zero, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverPeriodExponential_eq_path, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverPeriodExponential_transport, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_period_exponential, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_exponential_local_transport, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.hasDerivAt_cover_exponential_chart, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_meromorphic_orders_of_exponential_transport, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_meromorphic_orders_of_integral_periods, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_isPrincipal_of_integral_periods]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverDivisorAvoidingPath, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverPeriodExponential]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Integral Period postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Integral Period theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER INTEGRAL PERIOD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
