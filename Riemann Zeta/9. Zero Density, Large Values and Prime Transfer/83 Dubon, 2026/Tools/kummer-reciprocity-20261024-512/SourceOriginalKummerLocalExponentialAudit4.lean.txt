private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverLogarithmicReading, `SourceOriginalWeilCoverLocalExponentialModel, `SourceOriginalWeilCoverFinitePuncturePaths, `SourceOriginalWeilCoverDivisorAvoidingPaths]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_logarithmic_reading, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_logarithmic_primitive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_local_exponential_model, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_place_outside_finite, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_isPathConnected_compl_finite, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_divisor_base_point, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_path_avoiding_divisor]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Local Exponential postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Local Exponential theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER LOCAL EXPONENTIAL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
