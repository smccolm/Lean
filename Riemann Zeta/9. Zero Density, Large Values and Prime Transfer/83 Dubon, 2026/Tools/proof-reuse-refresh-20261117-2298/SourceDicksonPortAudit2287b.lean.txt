import SourceDicksonClassification2287b
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceDickson.dickson_classification

private def audit : CoreM Unit := do
  let env ← getEnv
  let modules : Array Name := #[`SourceDicksonObjects2269a, `SourceDicksonPGLBasic2271c, `SourceDicksonNatClassEquation2270d, `SourceDicksonPSLBasic2273c, `SourceDicksonPartitionHelpers2287a, `SourceDicksonCyclicPartition2287a, `SourceDicksonRecognitionA52287a, `SourceDicksonFieldReconstruction2287a, `SourceDicksonPartitionProof2287b, `SourceDicksonPSLRecognition2287a, `SourceDicksonTameClassification2287a, `SourceDicksonWildClassification2287b, `SourceDicksonClassification2287b]
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let mut nt := 0
  let mut nd := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Project postulate: {name}"
        let axioms ← Lean.collectAxioms name
        let forbidden := axioms.filter fun a => !allowed.contains a
        unless forbidden.isEmpty do throwError "Forbidden dependencies: {name}: {forbidden}"
        if info.isTheorem then nt := nt + 1 else nd := nd + 1
  logInfo m!"SOURCE DICKSON PORT AXIOM AUDIT PASS: {nt} theorems and {nd} other declarations, including private/generated declarations. This proves classification only, not the arithmetic large-image exclusions."

#eval audit
