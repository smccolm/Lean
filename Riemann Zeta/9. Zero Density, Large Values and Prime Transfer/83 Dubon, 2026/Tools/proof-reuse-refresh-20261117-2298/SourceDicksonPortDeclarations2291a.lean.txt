import SourceDicksonClassification2287b
import Mathlib.Tactic

open Lean
private def inspectDeclarations : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceDicksonObjects2269a, `SourceDicksonPGLBasic2271c, `SourceDicksonNatClassEquation2270d, `SourceDicksonPSLBasic2273c, `SourceDicksonPartitionHelpers2287a, `SourceDicksonCyclicPartition2287a, `SourceDicksonRecognitionA52287a, `SourceDicksonFieldReconstruction2287a, `SourceDicksonPartitionProof2287b, `SourceDicksonPSLRecognition2287a, `SourceDicksonTameClassification2287a, `SourceDicksonWildClassification2287b, `SourceDicksonClassification2287b]
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      let m := env.allImportedModuleNames[idx.toNat]!
      if mods.contains m then
        if info.isTheorem then logInfo m!"OWNER {m}: {name}"
        if !name.isInternal && !isPrivateName name then
          if !(← Batteries.Tactic.Lint.isAutoDecl name) then
            let kind := if info.isTheorem then "THEOREM" else "CONSTRUCTION"
            logInfo m!"PUBLIC {kind} {m}: {name}"

#eval inspectDeclarations
