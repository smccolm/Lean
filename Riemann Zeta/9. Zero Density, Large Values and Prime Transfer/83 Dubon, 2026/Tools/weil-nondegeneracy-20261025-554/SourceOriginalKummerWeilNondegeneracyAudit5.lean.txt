private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilComplexEvaluation, `SourceOriginalWeilComplexDatum, `SourceOriginalWeilComplexComparison, `SourceOriginalArithmeticWeilDetection, `SourceOriginalTateWeilNondegeneracy]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalConstantField.originalPlaceEval_liftPlace, `Dubon2026.SourceFunctionField.OriginalConstantField.originalDivisorEval_conorm, `Dubon2026.SourceFunctionField.OriginalConstantField.ord_fieldMap_of_multiple, `Dubon2026.SourceFunctionField.OriginalConstantField.conorm_disjoint, `Dubon2026.SourceFunctionField.OriginalConstantField.originalWeilComplexDatum_pairing, `Dubon2026.SourceFunctionField.OriginalConstantField.originalWeilComplexDatum_classLeft, `Dubon2026.SourceFunctionField.OriginalConstantField.originalWeilComplexDatum_classRight, `Dubon2026.SourceFunctionField.OriginalConstantField.originalWeilPairing_complex, `Dubon2026.SourceFunctionField.originalArithmeticWeilPairing_detects, `Dubon2026.SourceFunctionField.originalTateWeilPairing_detects, `Dubon2026.SourceFunctionField.originalTateWeilBilinear_injective]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalConstantField.originalWeilComplexDatum]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Weil Nondegeneracy postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Weil Nondegeneracy theorems discovered"
  logInfo m!"SOURCE ORIGINAL WEIL NONDEGENERACY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
