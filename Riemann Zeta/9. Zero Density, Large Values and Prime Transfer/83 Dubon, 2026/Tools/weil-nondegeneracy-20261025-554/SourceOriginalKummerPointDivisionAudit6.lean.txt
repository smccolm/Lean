private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourcePeriodDivision, `SourceCurvePicardDivisionFromPrincipal, `SourceCurvePicardDivisionFromAbel, `SourceOriginalWeilCoverClassDivision, `SourceOriginalWeilCoverPointDivision, `SourceOriginalComplexWeilDetection]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.PeriodDivision.integer_multiple_mem_of_scaled_difference, `Dubon2026.SourceCurve.Pic0.exists_nsmul_mk_of_principal_difference, `Dubon2026.SourceCurve.Pic0.exists_nsmul_mk_of_abel_data, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_degree_zero_class_has_division, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_point_difference_has_division, `Dubon2026.SourceFunctionField.originalComplexWeilPairing_detects]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Point Division postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Point Division theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER POINT DIVISION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
