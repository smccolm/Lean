private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverRawReciprocityData, `SourceFiniteIntegerGenerationReindex, `SourceOriginalWeilCoverGenerationReindex, `SourceFiniteThirdKindReindex, `SourceOriginalWeilCoverReciprocityReindex, `SourceOriginalWeilCoverReindexedReciprocity, `SourceOriginalWeilCoverAbelConverse]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.reindex_integer_generation, `Dubon2026.SourceFunctionField.CellGeometry.reindex_integer_generation_with_residues, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_regular_generation_reindex, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_meromorphic_generation_reindex, `Dubon2026.SourceFunctionField.CellGeometry.reindex_third_kind_reciprocity, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.reindex_cover_reciprocity, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_reindexed_reciprocity, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_isPrincipal_of_abelJacobi_mem]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverRawReciprocityData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Abel Converse postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Abel Converse theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER ABEL CONVERSE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
