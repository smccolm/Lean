  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceComplexZeroResidueCoefficient, `SourceRadialMeromorphicPrimitiveValues, `SourceRadialMeromorphicCellDifferences, `SourceRadialPoleAvoidingNeighborhoods, `SourceOriginalWeilCoverZeroResidueCoefficient, `SourceOriginalWeilCoverMeromorphicLocalPrimitives, `SourceRadialMeromorphicLocalAtlasData, `SourceOriginalWeilCoverMeromorphicLocalAtlas, `SourceRadialMeromorphicAtlasOverlap, `SourceRadialMeromorphicAtlasLocality, `SourceOriginalWeilCoverMeromorphicPathComparison, `SourceOriginalWeilCoverMeromorphicPathAtlas, `SourceOriginalWeilCoverMeromorphicLoopPeriodsV2, `SourceOriginalWeilCoverPathIntegralFiniteRegularity, `SourceFiniteCycleResidueLinearity, `SourceOriginalWeilCoverMeromorphicBasisExtensionV2, `SourceOriginalWeilCoverMeromorphicGenerationV2, `SourceOriginalWeilCoverFullCycleDataV2]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.residueSubtractedCoefficient_zero, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.zsmul_ρ_mem_Λres, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.hfun_sub_pval_mem, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.pval_sub_mem_of_linkedAt, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.pval_sub_mem, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.isOpen_Uset, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.mem_Uset_self, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.mem_Uset, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.ζ_ne_q, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_regCoeff_eq_coeff_of_zero, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_exists_logPrimitive, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.Φloc_self, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.Gθ_self, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_eventually_logPrimitive_sub, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.Gθ_sub_mem, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.eventually_Gθ_eq_Φloc, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_logPrimitive_along_path, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_meromorphic_path_sub_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_loop_period_mem_ΛJ, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_meromorphic_family_generation_of_jumps, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_sum_smul_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isPrimitiveAlong_sum_smul_of_coeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.pathIntegral_sum_smul_of_coeff_mem, `Dubon2026.SourceFunctionField.CellGeometry.sum_linear_combination_cycle_residue, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_meromorphic_cycle_generation_of_basis, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_meromorphic_cycle_generation_of_jump_data, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_full_cycle_data]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.pval, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.ΛJ, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.Uset, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.Φloc, `Dubon2026.SourceFunctionField.CellGeometry.RadialMeromorphicPeriods.Gθ, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverMeromorphicFamilyGeneration, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverMeromorphicCycleGeneration]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Meromorphic Generation postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Meromorphic Generation theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER MEROMORPHIC GENERATION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."
