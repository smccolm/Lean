private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourcePeakedAuxiliaryFunction, `SourceSiegelCuspPermutation, `SourcePeakedAuxiliaryCuspDecay, `SourcePeakedAuxiliaryHolomorphic, `SourcePeakedAuxiliaryModularForm]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.discr_slash, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.thetaFun_slash_apply, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.det_rel, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.mT_zero, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.mT_rowMul, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.idx_cast, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.uFun_smul, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.ord_eq, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.discr_isBigO, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.norm_C_eq_one, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.thetaFun_slash_isBigO, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.mdifferentiable_thetaFun, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.theta_apply, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.exists_modularForm_gamma1_coe_eq_prod_siegelFun_pow_mul_discriminant_pow]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.uFun, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.Ord, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.thetaFun, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.e00, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.e01, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.e10, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.e11, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.rowMul, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.mT, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.Form.theta]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Siegel expansion postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Siegel expansion theorems discovered"
  logInfo m!"SOURCE PEAKED MODULAR FORM AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
