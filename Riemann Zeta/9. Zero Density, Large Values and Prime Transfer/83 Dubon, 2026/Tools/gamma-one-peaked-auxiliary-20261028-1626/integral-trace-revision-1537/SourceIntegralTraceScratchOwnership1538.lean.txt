open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneEvenIntegralRatios, `SourceLevelOneCoefficientRing, `SourceLevelOneIntegralWeightFactor, `SourceTraceProductCoefficientRing, `SourceIntegralLevelOneTraceCoefficients, `SourceTraceExponentialCoefficientVanishing, `SourceIntegralTraceParameters, `SourceIntegralTracePowerCoefficients, `SourceIntegralTraceCoefficientExtraction, `SourceIntegralTraceModularForm, `SourceIntegralTraceCuspTransversal, `SourceIntegralTraceTranslates, `SourceIntegralTraceTranslationCoefficients, `SourceIntegralTracePeriodWidth, `SourceIntegralTraceAnalyticProducts, `SourceIntegralTraceSlashPeriods, `SourceIntegralTraceTranslationAssembly, `SourceIntegralTraceTransversalSplit, `SourceIntegralTraceDecomposition, `SourceIntegralTraceDegenerateCusp, `SourceEvenWeightPeakedTraceIntegrality]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
