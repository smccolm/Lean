open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceSiegelJacobiFiniteProducts, `SourceSiegelJacobiGaussProduct, `SourceSiegelJacobiFiniteIdentity, `SourceSiegelJacobiProductBounds, `SourceSiegelJacobiDominatedLimit, `SourceSiegelJacobiRangeLimit, `SourceSiegelJacobiProductIdentity, `SourceSiegelFunction, `SourceSiegelProductCore, `SourceSiegelThetaIdentification, `SourceSiegelHolomorphic, `SourceSiegelEtaTransform, `SourceSiegelThetaTransform, `SourceSiegelInversion, `SourceSiegelSL2Transformation, `SourceSiegelLevelShiftFactors, `SourceSiegelLevelShifts, `SourceSiegelResidueReduction, `SourceSiegelProductTransformation]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
