import SourceSiegelStableCoefficients
import SourceSiegelProductLocalUniform
import SourceSiegelProductTaylorExpansion
import SourceSiegelPolynomialFactors
import SourceSiegelIntegralCoefficients
import SourceSiegelQParameter
import SourceSiegelResidueProductIndices
import SourceSiegelResidueProductStreams
import SourceSiegelLeadingUnit
import SourceSiegelNormalizedProduct
import SourceSiegelCoefficientSeries
import SourceSiegelIntegralExpansion
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceSiegelStableCoefficients, `SourceSiegelProductLocalUniform, `SourceSiegelProductTaylorExpansion, `SourceSiegelPolynomialFactors, `SourceSiegelIntegralCoefficients, `SourceSiegelQParameter, `SourceSiegelResidueProductIndices, `SourceSiegelResidueProductStreams, `SourceSiegelLeadingUnit, `SourceSiegelNormalizedProduct, `SourceSiegelCoefficientSeries, `SourceSiegelIntegralExpansion]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
