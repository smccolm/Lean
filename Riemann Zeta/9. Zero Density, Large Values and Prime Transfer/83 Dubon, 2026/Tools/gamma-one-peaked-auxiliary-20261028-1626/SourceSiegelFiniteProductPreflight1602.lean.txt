private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceSiegelFiniteSeriesProducts1596b, `SourceSiegelProductCoefficients1597b, `SourceSiegelProductLeadingCoefficient1599b, `SourceSiegelProductPhases1600a, `SourceSiegelFiniteProductExpansion1601b]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.Expands.one, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.Expands.mul, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.Expands.pow, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.Expands.prod, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.coeff_prod_mem, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.expands_series, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.integral_series, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.expands_productSeries, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.integral_productSeries, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.series_zero, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.productSeries_zero, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.not_dvd_val, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.productSeries_zero_eq, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.productSeries_zero_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.integral_level_inverse_series_zero, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.integral_level_inverse_productSeries_zero, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.phase_pow, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.expo_pow, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.exp_orderSum_eq, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.normalized_product_eq, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.qParam_pow_eq, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.hasSum_productSeries, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.tendsto_productSeries, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.exists_integral_hasSum_prod_siegelFun_pow]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.Expands, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.rho, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.expo, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.normalized, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.series, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.exponent, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.productSeries, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.phase, `Dubon2026.SourceFunctionField.GammaOne.PeakedAuxiliary.FiniteExpansion.orderSum]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Siegel expansion postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Siegel expansion theorems discovered"
  logInfo m!"SOURCE SIEGEL FINITE PRODUCT PREFLIGHT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
