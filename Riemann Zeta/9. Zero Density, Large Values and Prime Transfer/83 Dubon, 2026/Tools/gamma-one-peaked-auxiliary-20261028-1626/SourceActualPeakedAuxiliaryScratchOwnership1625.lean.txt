import SourcePeakedDiscriminantSeries
import SourcePeakedSeriesOperations
import SourcePeakedShiftedCoefficients
import SourcePeakedShiftedSeries
import SourcePeakedWidthOneIdentification
import SourcePeakedIntegralQExpansion
import SourcePeakedTranslatedIntegralExpansion
import SourcePeakedScaledOrders
import SourcePeakedOrderGapDecay
import SourceActualPeakedAuxiliary
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourcePeakedDiscriminantSeries, `SourcePeakedSeriesOperations, `SourcePeakedShiftedCoefficients, `SourcePeakedShiftedSeries, `SourcePeakedWidthOneIdentification, `SourcePeakedIntegralQExpansion, `SourcePeakedTranslatedIntegralExpansion, `SourcePeakedScaledOrders, `SourcePeakedOrderGapDecay, `SourceActualPeakedAuxiliary]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
