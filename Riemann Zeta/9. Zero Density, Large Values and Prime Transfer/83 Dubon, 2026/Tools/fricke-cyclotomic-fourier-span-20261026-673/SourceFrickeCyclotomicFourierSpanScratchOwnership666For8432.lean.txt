import SourceConstantExtensionIntegralAlgebra
import SourceAlgebraicConstantRoots
import SourceAlgebraicConstantLift
import SourceAlgebraicConstantInterpolation
import SourceAlgebraicConstantIntegralSpan
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceConstantExtensionIntegralAlgebra, `SourceAlgebraicConstantRoots, `SourceAlgebraicConstantLift, `SourceAlgebraicConstantInterpolation, `SourceAlgebraicConstantIntegralSpan]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
