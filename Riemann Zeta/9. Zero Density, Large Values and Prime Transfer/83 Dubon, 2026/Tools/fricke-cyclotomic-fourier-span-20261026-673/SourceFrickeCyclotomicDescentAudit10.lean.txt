private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCuspLaurentFunctionIdentity, `SourceQExpansionLaurentWidth, `SourcePoleBoundedCuspLaurent, `SourceFrickeCyclotomicLaurent, `SourceCuspLaurentPolynomial, `SourceCuspLaurentFractionIdentity, `SourceFrickeCyclotomicFractionGerm, `SourceFrickeLaurentIntegralDescent, `SourceFrickeFractionPeriodicity, `SourceFrickeCyclotomicMonic]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.mdifferentiable_eq_of_eventuallyEq_atImInfty, `Dubon2026.SourceFunctionField.CuspTaylor.represents_qExpansion, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.eq_upperHalfPlane, `Dubon2026.SourceFunctionField.CuspTaylor.represents_cusp_const, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_add, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_mul, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_inv, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_div, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_pow, `Dubon2026.SourceFunctionField.qExpansion_eq_expand, `Dubon2026.SourceFunctionField.qExpansion_laurent_width, `Dubon2026.SourceFunctionField.represents_jqNModC, `Dubon2026.SourceFricke.periodic_discriminant_width, `Dubon2026.SourceFricke.represents_discriminant_width, `Dubon2026.SourceFricke.exists_cusp_represents_of_poleBounded, `Dubon2026.SourceFricke.coeffMap_discriminantLaurentK, `Dubon2026.SourceFricke.coeffMap_jqNModC_kN, `Dubon2026.SourceFricke.frickeF_eq_frickeH_div, `Dubon2026.SourceFricke.represents_frickeLaurentK, `Dubon2026.SourceFricke.periodic_frickeF, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_sum, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_aeval, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.isIntegral_of_monic_relation, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.eventually_ne_zero, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.ne_zero_of_function_ne_zero, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_of_mul_eq, `Dubon2026.SourceFricke.periodic_jf_width, `Dubon2026.SourceFricke.cyclotomicLevelRing_holomorphic, `Dubon2026.SourceFricke.cyclotomicLevelRing_periodic, `Dubon2026.SourceFricke.exists_cyclotomic_cusp_represents, `Dubon2026.SourceFricke.exists_cyclotomic_fraction_represents, `Dubon2026.SourceFricke.exists_integral_cyclotomic_laurent_of_monic, `Dubon2026.SourceFricke.exists_polynomial_monic_relation_of_isIntegral, `Dubon2026.SourceFricke.coeffMap_aeval_jqNModC_kN, `Dubon2026.SourceFricke.apply_T_of_periodic, `Dubon2026.SourceFricke.periodic_comp_ofComplex_of_T, `Dubon2026.SourceFricke.periodic_cyclotomic_fraction, `Dubon2026.SourceFricke.mdifferentiable_polynomial_jf, `Dubon2026.SourceFricke.exists_cyclotomic_monic_relation]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.discriminantLaurentK, `Dubon2026.SourceFricke.frickeLaurentK, `Dubon2026.SourceFricke.cyclotomicLevelRing]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Fricke Cyclotomic Descent postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Fricke Cyclotomic Descent theorems discovered"
  logInfo m!"SOURCE FRICKE CYCLOTOMIC DESCENT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
