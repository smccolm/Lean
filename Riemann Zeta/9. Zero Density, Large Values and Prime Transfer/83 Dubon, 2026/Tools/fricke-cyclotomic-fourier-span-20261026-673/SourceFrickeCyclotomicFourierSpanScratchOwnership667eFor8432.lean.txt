import SourceTranscendentalConstantAlgebraV2
import SourceTranscendentalConstantEvaluationV2
import SourceTranscendentalConstantDenominatorsV2
import SourceTranscendentalConstantFieldRangeV2
import SourceTranscendentalConstantScaledRelationV2
import SourceTranscendentalConstantNormalityV2
import SourceTranscendentalConstantIntegralSpanV2
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceTranscendentalConstantAlgebraV2, `SourceTranscendentalConstantEvaluationV2, `SourceTranscendentalConstantDenominatorsV2, `SourceTranscendentalConstantFieldRangeV2, `SourceTranscendentalConstantScaledRelationV2, `SourceTranscendentalConstantNormalityV2, `SourceTranscendentalConstantIntegralSpanV2]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
