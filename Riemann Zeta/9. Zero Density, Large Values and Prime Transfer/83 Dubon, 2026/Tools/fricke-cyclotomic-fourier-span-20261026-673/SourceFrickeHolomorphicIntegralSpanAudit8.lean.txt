private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMonicQuotientRelation, `SourceHolomorphicMonicQuotient, `SourceIntegralCoordinateMonic, `SourceFrickeCyclotomicLaurentLifts, `SourceFrickeClearedLaurentMonic, `SourceFrickeIntegralHolomorphicRepresentative, `SourceCuspLaurentSpan, `SourceFrickeHolomorphicIntegralSpan]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.cleared_monic_relation_eq_scaled, `Dubon2026.SourceFunctionField.monic_div_relation_of_cleared, `Dubon2026.SourceFunctionField.cleared_monic_relation_of_div, `Dubon2026.SourceFunctionField.mdiff_eventually_ne_zero_ofComplex, `Dubon2026.SourceFunctionField.quotient_locally_bounded_of_monic, `Dubon2026.SourceFunctionField.exists_mdifferentiable_quotient_of_monic, `Dubon2026.SourceFunctionField.exists_coordinate_monic_relation, `Dubon2026.SourceFricke.exists_cyclotomic_function_of_laurent_algebra, `Dubon2026.SourceFricke.exists_cyclotomic_function_fraction_of_laurent_field, `Dubon2026.SourceFricke.cyclotomic_polynomial_jf_mem, `Dubon2026.SourceFricke.represents_cyclotomic_polynomial_jf, `Dubon2026.SourceFricke.cleared_cyclotomic_relation_of_laurent, `Dubon2026.SourceFricke.exists_holomorphic_cyclotomic_fraction_of_integral_laurent, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_smul, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.mem_span_of_laurent_span, `Dubon2026.SourceFricke.holomorphicCyclotomicFrickeFractions_periodic, `Dubon2026.SourceFricke.periodic_of_principal_invariant, `Dubon2026.SourceFricke.mem_span_holomorphic_cyclotomic_fractions_of_principal_invariant]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.holomorphicCyclotomicFrickeFractions]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Fricke Holomorphic Integral Span postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Fricke Holomorphic Integral Span theorems discovered"
  logInfo m!"SOURCE FRICKE HOLOMORPHIC INTEGRAL SPAN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
