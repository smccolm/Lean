private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMeromorphicCuspPoleClearing, `SourceFrickePoleClearedFourierSpan]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.exists_nonnegative_add_nat_mul, `Dubon2026.SourceFricke.meromorphicOrderAt_discriminant_cusp_pos, `Dubon2026.SourceFricke.periodic_mul_discriminant, `Dubon2026.SourceFricke.exists_bounded_mul_discriminant_of_meromorphic, `Dubon2026.SourceFricke.exists_pole_cleared_cyclotomic_qExpansion, `Dubon2026.SourceFricke.mem_span_pole_cleared_cyclotomic_of_principal_invariant]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.cyclotomicPoleBoundedFunctions]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Fricke Pole Cleared Fourier Span postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Fricke Pole Cleared Fourier Span theorems discovered"
  logInfo m!"SOURCE FRICKE POLE CLEARED FOURIER SPAN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
