private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceLaurentConstantTranscendence, `SourceLaurentGeneratedConstants, `SourceConstantExtensionChain, `SourceConstantExtensionMonotonicity, `SourceFiniteConstantIntegralCapture, `SourceLaurentIntegralConstantSpan]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.ConstantDescent.constant_transcendental_of_coeff_mem, `Dubon2026.SourceFricke.ConstantDescent.constant_mem_coeffMap_fieldRange, `Dubon2026.SourceFricke.ConstantDescent.generated_mem_coeffMap_fieldRange, `Dubon2026.SourceFricke.ConstantDescent.generated_coeff_mem, `Dubon2026.SourceFricke.ConstantDescent.constant_transcendental_generated, `Dubon2026.SourceFricke.ConstantDescent.coeffMap_fieldRange_mono, `Dubon2026.SourceFricke.ConstantDescent.constantChain_le_cons, `Dubon2026.SourceFricke.ConstantDescent.le_constantChain, `Dubon2026.SourceFricke.ConstantDescent.mem_constantChain_of_mem, `Dubon2026.SourceFricke.ConstantDescent.constantChain_le, `Dubon2026.SourceFricke.ConstantDescent.constantChain_le_append_left, `Dubon2026.SourceFricke.ConstantDescent.constantChain_le_append_right, `Dubon2026.SourceFricke.ConstantDescent.adjoin_constants_mono, `Dubon2026.SourceFricke.ConstantDescent.generated_constants_mono, `Dubon2026.SourceFricke.ConstantDescent.integral_constants_mono, `Dubon2026.SourceFricke.ConstantDescent.generated_mem_iff_closure, `Dubon2026.SourceFricke.ConstantDescent.exists_list_mem_algebra, `Dubon2026.SourceFricke.ConstantDescent.exists_list_mem_field, `Dubon2026.SourceFricke.ConstantDescent.exists_list_integral_coordinates, `Dubon2026.SourceFricke.ConstantDescent.exists_list_mem_integral_generated, `Dubon2026.SourceFricke.ConstantDescent.mem_span_laurent_of_algebra_span, `Dubon2026.SourceFricke.ConstantDescent.integral_constantChain_step, `Dubon2026.SourceFricke.ConstantDescent.integral_constantChain_subset_span, `Dubon2026.SourceFricke.ConstantDescent.mem_span_integral_generated_of_complex]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.ConstantDescent.constantChain]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Laurent Integral Constant Span postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Laurent Integral Constant Span theorems discovered"
  logInfo m!"SOURCE LAURENT INTEGRAL CONSTANT SPAN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
