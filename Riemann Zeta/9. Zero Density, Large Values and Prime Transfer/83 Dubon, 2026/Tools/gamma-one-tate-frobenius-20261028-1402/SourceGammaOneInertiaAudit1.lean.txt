private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneArithmeticCharts, `SourceGammaOneSpecializationCharts, `SourceGammaOneDecompositionCharts, `SourceGammaOneConstantReductionCharts, `SourceGammaOneSpecializationEquivariance, `SourceGammaOneInertiaPlaceReduction, `SourceGammaOneInertiaPicard, `SourceGammaOneTateInertia]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut_mem_adjoin, `Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut_isIntegral, `Dubon2026.SourceFunctionField.GammaOne.nonunit_arithmeticPlace_iff, `Dubon2026.SourceFunctionField.GammaOne.chartDomain_arithmeticPlace, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_principal_reduction_map_with_charts_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.nonunit_decompositionResiduePlace_iff, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResidueAut_j, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResiduePlace_j_mem, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.chartReducesTo_decomposition, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_constant_reduction_with_charts_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_reduction_decomposition_equivariant_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_reduction_invariant_on_inertia_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.reduction_arithmeticDivisor_invariant, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.reduction_arithmeticPicard_invariant, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_arithmeticPicard_eq_self_of_inertia_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_arithmeticTateGaloisRep_eq_one_of_inertia_of_degree_genus_eq]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE INERTIA AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
