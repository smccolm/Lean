/-! DUB-17/18 -> original Gamma1 arithmetic association -> actual Picard
Frobenius relation. Equivariant specialization and invertible-torsion
injectivity transport the exact norm-derived relation to the original field.
The literal coefficient Frobenius relation is H*F = F*F*D + p, with the
Picard diamond D labelled p. Normalization to a target Galois representation
is a subsequent obligation; no untwisted quadratic is claimed here. -/
namespace Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
noncomputable section
open SourceCurve IntermediateField CongruenceSubgroup ConstantField
attribute [local instance] specializationResidueAlgebra
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (M p : ℕ) [NeZero M] [Fact p.Prime]
  [CharP (IsLocalRing.ResidueField A) p]
local notation "Γ" => Gamma1 M
local notation "FF" => laurentBaseChange Qbar (x1FunctionField M)
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Γ

theorem exists_gammaOne_picard_frobenius_relation_of_degree_genus_eq
    (hcop : Nat.Coprime M p)
    (hdeg : Module.finrank Qbar⟮jCoordinateOver Γ Qbar⟯ (functionFieldOver Γ Qbar) =
      Module.finrank kk⟮residueJCoordinate A Γ⟯ Fb)
    (hgen : genusFF kk Fb = genusFF Qbar FF)
    (d : A.decompositionSubgroup ℚ) (hd : ∀ a : kk, d • a = a ^ p) :
    ∃ g : Gamma0 M, (g.val 1 1 : ZMod M) = (p : ZMod M) ∧
      ∀ (m : ℕ), (m : kk) ≠ 0 → ∀ (x : Pic0 Qbar FF), x ∈ Pic0.torsion m →
        heckePicard Qbar M p (gamma1ArithmeticPicardRep Qbar M d.val x) =
          gamma1ArithmeticPicardRep Qbar M d.val
            (gamma1ArithmeticPicardRep Qbar M d.val (arithmeticDiamondPicard M g x)) + p • x := by
  run_tac Lean.logInfo "obtain ⟨g, hlabel, hhecke⟩ :="
  obtain ⟨g, hlabel, hhecke⟩ :=
    exists_gammaOne_hecke_reduction_torsion_picard_of_degree_eq A M p hcop hdeg d hd
  run_tac Lean.logInfo "obtain ⟨R, hR, hinj, he⟩ :="
  obtain ⟨R, hR, hinj, he⟩ :=
    exists_gammaOne_reduction_picard_equivariant_of_degree_genus_eq A M hdeg hgen
  run_tac Lean.logInfo "refine ⟨g⁻¹, norm_diamond_inverse_label M p g hlabel, ?_⟩"
  refine ⟨g⁻¹, norm_diamond_inverse_label M p g hlabel, ?_⟩
  run_tac Lean.logInfo "intro m hm x hx"
  intro m hm x hx
  run_tac Lean.logInfo "have hm0 : m ≠ 0 := fun hz => hm (by rw [hz, Nat.cast_zero])"
  have hm0 : m ≠ 0 := fun hz => hm (by rw [hz, Nat.cast_zero])
  run_tac Lean.logInfo "have hFx := gamma1ArithmeticPicardRep_torsion M d.val m x hx"
  have hFx := gamma1ArithmeticPicardRep_torsion M d.val m x hx
  run_tac Lean.logInfo "have hDx := arithmeticDiamondPicard_torsion M g⁻¹ m x hx"
  have hDx := arithmeticDiamondPicard_torsion M g⁻¹ m x hx
  run_tac Lean.logInfo "have hFDx := gamma1ArithmeticPicardRep_torsion M d.val m _ hDx"
  have hFDx := gamma1ArithmeticPicardRep_torsion M d.val m _ hDx
  run_tac Lean.logInfo "have hFFDx := gamma1ArithmeticPicardRep_torsion M d.val m _ hFDx"
  have hFFDx := gamma1ArithmeticPicardRep_torsion M d.val m _ hFDx
  run_tac Lean.logInfo "have hleft : heckePicard Qbar M p (gamma1ArithmeticPicardRep Qbar M d.val x"
  have hleft : heckePicard Qbar M p (gamma1ArithmeticPicardRep Qbar M d.val x) ∈ Pic0.torsion m := by
    apply Pic0.mem_torsion_iff.mpr
    rw [← map_nsmul, Pic0.mem_torsion_iff.mp hFx, map_zero]
  run_tac Lean.logInfo "apply hinj m hm hleft (add_mem hFFDx ((Pic0.torsion m).nsmul_mem hx p))"
  apply hinj m hm hleft (add_mem hFFDx ((Pic0.torsion m).nsmul_mem hx p))
  run_tac Lean.logInfo "let F := gamma1ArithmeticPicardRep Qbar M d.val"
  let F := gamma1ArithmeticPicardRep Qbar M d.val
  run_tac Lean.logInfo "let D := arithmeticDiamondPicard M g⁻¹"
  let D := arithmeticDiamondPicard M g⁻¹
  run_tac Lean.logInfo "have hT : ModularGroup.T ∈ Γ := by simp [Gamma1_mem, ModularGroup.T]"
  have hT : ModularGroup.T ∈ Γ := by simp [Gamma1_mem, ModularGroup.T]
  run_tac Lean.logInfo "let C := decompositionResiduePicard A Γ hT d"
  let C := decompositionResiduePicard A Γ hT d
  run_tac Lean.logInfo "let Ci := decompositionResiduePicard A Γ hT d⁻¹"
  let Ci := decompositionResiduePicard A Γ hT d⁻¹
  run_tac Lean.logInfo "have hfirst : C (R.pic0Map (D (F x))) = R.pic0Map (F (F (D x))) :="
  have hfirst : C (R.pic0Map (D (F x))) = R.pic0Map (F (F (D x))) :=
    (congrArg (fun z => C (R.pic0Map z))
      (arithmeticDiamondPicard_galois M g⁻¹ d.val x).symm).trans (he d (F (D x))).symm
  run_tac Lean.logInfo "have hsecond : Ci (R.pic0Map (F x)) = R.pic0Map x :="
  have hsecond : Ci (R.pic0Map (F x)) = R.pic0Map x :=
    (congrArg Ci (he d x)).trans
      (decompositionResiduePicard_inverse A Γ hT d (R.pic0Map x))
  run_tac Lean.logInfo "have hmap : R.pic0Map (F (F (D x)) + p • x) ="
  have hmap : R.pic0Map (F (F (D x)) + p • x) =
      R.pic0Map (F (F (D x))) + p • R.pic0Map x :=
    (R.pic0Map.map_add _ _).trans
      (congrArg (fun z => R.pic0Map (F (F (D x))) + z) (R.pic0Map.map_nsmul p x))
  run_tac Lean.logInfo "exact (hhecke R hR _ m hm0 (Pic0.mem_torsion_iff.mp hFx)).trans"
  exact (hhecke R hR _ m hm0 (Pic0.mem_torsion_iff.mp hFx)).trans
    ((congrArg₂ (· + ·) hfirst (congrArg (fun z => p • z) hsecond)).trans hmap.symm)

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
