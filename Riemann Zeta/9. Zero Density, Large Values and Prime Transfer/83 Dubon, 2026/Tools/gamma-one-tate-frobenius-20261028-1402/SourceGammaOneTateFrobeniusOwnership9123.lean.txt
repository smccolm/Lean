open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOneArithmeticCharts, `Dubon2026.SourceGammaOneSpecializationCharts, `Dubon2026.SourceGammaOneDecompositionCharts, `Dubon2026.SourceGammaOneConstantReductionCharts, `Dubon2026.SourceGammaOneSpecializationEquivariance, `Dubon2026.SourceGammaOneInertiaPlaceReduction, `Dubon2026.SourceGammaOneInertiaPicard, `Dubon2026.SourceGammaOneTateInertia, `Dubon2026.SourceGammaOneHeckePrincipalNorm, `Dubon2026.SourceGammaOneResidueDivisors, `Dubon2026.SourceGammaOneHeckeResidueNorm, `Dubon2026.SourceGammaOneIntegralDivisorReduction, `Dubon2026.SourceGammaOneHeckeIntegralAgreement, `Dubon2026.SourceGammaOneHeckeTorsionDivisors, `Dubon2026.SourceGammaOneResiduePicard, `Dubon2026.SourceGammaOneReductionPicardEquivariance, `Dubon2026.SourceGammaOneNormDiamondOrientation, `Dubon2026.SourceGammaOneHeckePicardDescent, `Dubon2026.SourceGammaOneHeckeTorsionPicard, `Dubon2026.SourceTorsionReductionDiamond, `Dubon2026.SourceGammaOnePicardFrobeniusRelation, `Dubon2026.SourceTateTorsionQuadratic, `Dubon2026.SourceGammaOneTateFrobeniusRelation, `Dubon2026.SourceTateDualArithmeticQuadratic, `Dubon2026.SourceGammaOneArithmeticFrobenius, `Dubon2026.SourceGammaOneGeometricArithmeticFrobenius]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
