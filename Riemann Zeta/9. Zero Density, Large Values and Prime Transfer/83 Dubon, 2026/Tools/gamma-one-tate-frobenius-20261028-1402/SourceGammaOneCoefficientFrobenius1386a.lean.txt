variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "T" => TateModule p (Pic0 Qbar (functionFieldOver (Gamma1 M) Qbar))

private local instance : Algebra ℤ_[p] (ZMod N) :=
  (PadicInt.toZModPow (n + 1)).toAlgebra

  Module.restrictScalars ℤ_[p] (ZMod N) V

  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

/-- Complementary homogeneous coefficient action paired with the actual
coefficient Galois action on the original Picard Tate module. -/
def binaryTateCoefficientFrobeniusOperator (ell : sourceBinaryHeckePrime (Q * N))
    (σ : Qbar ≃ₐ[ℚ] Qbar) : Module.End (ZMod N) (T →ₗ[ℤ_[p]] V) :=
  let u := homogeneousMatrixAction w (Matrix.diagonal ![1, (ell.val : ZMod N)])
  { toFun := fun φ => (u.restrictScalars ℤ_[p]).comp
      (φ.comp (arithmeticTateGaloisRep M p σ))
    map_add' := fun φ ψ => by
      apply LinearMap.ext
      intro x
      exact u.map_add _ _
    map_smul' := fun c φ => by
      apply LinearMap.ext
      intro x
      exact u.map_smul c _ }

theorem binaryTateCoefficientFrobeniusOperator_quadratic_of_picard
    (ell : sourceBinaryHeckePrime (Q * N)) (g : Gamma0 M)
    (hlabel : (g.val 1 1 : ZMod N) = (ell.val : ZMod N))
    (σ : Qbar ≃ₐ[ℚ] Qbar)
    (hquad : letI : Fact ell.val.Prime := ⟨ell.property.1⟩
      arithmeticHeckeTate M p ell.val * arithmeticTateGaloisRep M p σ =
        arithmeticTateGaloisRep M p σ * arithmeticTateGaloisRep M p σ *
          arithmeticDiamondTate M p g + ell.val • 1)
    (φ : T →ₗ[ℤ_[p]] V)
    (hfixed : ((sourceGammaOneBinaryDiamondCoefficient Q N w g).restrictScalars ℤ_[p]).comp
      (φ.comp (arithmeticDiamondTate M p g)) = φ) :
    binaryTateCoefficientFrobeniusOperator Q p n w ell σ (binaryTatePrimeOperators Q p n w ell φ) =
      binaryTateCoefficientFrobeniusOperator Q p n w ell σ
        (binaryTateCoefficientFrobeniusOperator Q p n w ell σ φ) +
      (ell.val : ZMod N) ^ (w + 1) • φ := by
  letI : Fact ell.val.Prime := ⟨ell.property.1⟩
  exact sourceTateDual_diamond_quadratic
    (homogeneousMatrixAction w (Matrix.diagonal ![(ell.val : ZMod N), 1]))
    (homogeneousMatrixAction w (Matrix.diagonal ![1, (ell.val : ZMod N)]))
    (sourceGammaOneBinaryDiamondCoefficient Q N w g)
    (arithmeticHeckeTate M p ell.val) (arithmeticTateGaloisRep M p σ)
    (arithmeticDiamondTate M p g) ell.val w hquad
    (arithmeticTateGaloisRep_diamond M p g σ)
    (sourceGammaOneComplementaryHeckeDiamond Q N w g ell.val hlabel)
    (homogeneousComplementaryHecke_product w (ell.val : ZMod N)) φ hfixed

end
