/-! DUB-17/18 -> original Gamma1 Tate Frobenius -> exact Hecke
specialization on finite Picard torsion. Descend the proved integer-divisor
identity through the native quotient, retaining the inverse matrix lift in
the existing Picard diamond convention. The norm lift has label p-inverse;
the Picard lift has label p by the separately proved inverse identity. -/
namespace Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
noncomputable section
open SourceCurve IntermediateField CongruenceSubgroup ConstantField
attribute [local instance] specializationResidueAlgebra
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (M p : ℕ) [NeZero M] [Fact p.Prime]
  [CharP (IsLocalRing.ResidueField A) p]
local notation "Γ" => Gamma1 M
local notation "FF" => functionFieldOver Γ Qbar
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Γ

theorem exists_gammaOne_hecke_reduction_torsion_picard_of_degree_eq
    (hcop : Nat.Coprime M p)
    (hdeg : Module.finrank Qbar⟮jCoordinateOver Γ Qbar⟯ FF =
      Module.finrank kk⟮residueJCoordinate A Γ⟯ Fb)
    (d : A.decompositionSubgroup ℚ) (hd : ∀ a : kk, d • a = a ^ p) :
    ∃ g : Gamma0 M, (p : ZMod M) * (g.val 1 1 : ZMod M) = 1 ∧
      ∀ (R : ConstantReduction A FF Fb), R.toRegularProlongation = regularProlongation A Γ →
        ∀ (x : Pic0 Qbar FF) (m : ℕ), m ≠ 0 → m • x = 0 →
          R.pic0Map (heckePicard Qbar M p x) =
            decompositionResiduePicard A Γ (by simp [Gamma1_mem, ModularGroup.T]) d
              (R.pic0Map (arithmeticDiamondPicard M g⁻¹ x)) +
            p • decompositionResiduePicard A Γ (by simp [Gamma1_mem, ModularGroup.T]) d⁻¹
              (R.pic0Map x) := by
  obtain ⟨g, hlabel, hdiv⟩ :=
    exists_gammaOne_hecke_reduction_torsion_divisor_of_degree_eq A M p hcop hdeg d hd
  refine ⟨g, hlabel, fun R hR x m hm hx => ?_⟩
  obtain ⟨D, rfl⟩ := Pic0.mk_surjective x
  have hT : ModularGroup.T ∈ Γ := by simp [Gamma1_mem, ModularGroup.T]
  let E := fieldDegZeroEquiv (arithmeticDiamondAut M g)
    (functionFieldOver_residue_bijective Γ hT)
    (functionFieldOver_residue_bijective Γ hT) D
  have hdiamond : arithmeticDiamondPicard M g⁻¹ (Pic0.mk D) = Pic0.mk E :=
    arithmeticDiamondPicard_inverse_lift_mk M g D
  rw [hdiamond]
  change Pic0.mk (R.degZeroMap ⟨heckeDivisor Qbar M p D.val,
    heckeDivisor_mem_degZero Qbar M p D.property⟩) =
    Pic0.mk (decompositionResidueDegZero A Γ hT d (R.degZeroMap E)) +
      p • Pic0.mk (decompositionResidueDegZero A Γ hT d⁻¹ (R.degZeroMap D))
  rw [← map_nsmul, ← map_add]
  apply congrArg Pic0.mk
  apply Subtype.ext
  change R.divMap (heckeDivisor Qbar M p D.val) =
    decompositionResidueDivisorEquiv A Γ d
      (R.divMap (fieldDivisorEquiv (arithmeticDiamondAut M g) D.val)) +
    p • decompositionResidueDivisorEquiv A Γ d⁻¹ (R.divMap D.val)
  rw [← decompositionResidueDivisorEquiv_symm]
  exact hdiv R hR D m hm hx

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
