private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialCellBoundaryCoordinates, `SourceOriginalWeilCoverPathPrimitiveDerivative, `SourceOriginalWeilCoverBoundaryCoefficient, `SourceRadialAffinePrimitiveIntegral, `SourceOriginalWeilCoverArcPathIntegral, `SourceOriginalWeilCoverEdgePathIntegral, `SourceOriginalWeilCoverCoordinateCycleLoops, `SourceOriginalWeilCoverCoordinateCyclePrimitives]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialCell.bdry_coordinates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_hasDerivAt_path_primitive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_continuousOn_boundary_coefficient, `Dubon2026.SourceFunctionField.CellGeometry.OriginalEdgePaths.arc_integral_eq_primitive_sub, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_pathIntegral_arcPath, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_pathIntegral_edgePath, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_coordinate_cycle_loops, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_coordinate_cycle_primitives]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverBoundaryIntegrand, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverEdgeIntegral, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverCoordinateCycleLoops]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Coordinate Loops postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Coordinate Loops theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER COORDINATE LOOPS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
