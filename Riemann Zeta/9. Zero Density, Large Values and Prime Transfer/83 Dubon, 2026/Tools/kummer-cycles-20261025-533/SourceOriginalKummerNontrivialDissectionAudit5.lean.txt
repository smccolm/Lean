private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverVertexNontrivial, `SourceOriginalLiftedMarkedEulerNontrivialAssembly, `SourceOriginalWeilCoverNontrivialEulerData, `SourceOriginalWeilCoverNontrivialEulerDissection, `SourceOriginalWeilCoverNontrivialTreeCotree]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_lifted_vertex_count_ge_two, `Dubon2026.SourceFunctionField.CellGeometry.OriginalGlobalCellCover.exists_lifted_marked_euler_dissection_nontrivial, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_nontrivial_euler_dissection_of_centered_base, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_nontrivial_euler_cell_dissection, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_nontrivial_tree_cotree]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Nontrivial Dissection postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Nontrivial Dissection theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER NONTRIVIAL DISSECTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
