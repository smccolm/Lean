private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverOneFormAt, `SourceOriginalWeilCoverRegularNeighborhood, `SourceOriginalWeilCoverOneFormLaw, `SourceOriginalWeilCoverCellCoefficient, `SourceOriginalWeilCoverCellRegularCoefficient, `SourceOriginalWeilCoverCellCoefficientPullback, `SourceOriginalWeilCoverCellCoefficientTransition]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_one_form_law_at, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_eventually_differentialCoeff_mem, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_one_form_law, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_coeff_eq_near, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_coeff_comp_mul_deriv, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_coeff_transition]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverCoeffIn]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Cell Calculus postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Cell Calculus theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER CELL CALCULUS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
