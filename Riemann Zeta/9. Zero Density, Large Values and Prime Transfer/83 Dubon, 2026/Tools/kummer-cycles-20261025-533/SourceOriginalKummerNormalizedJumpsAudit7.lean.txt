private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverCoordinateTransition, `SourceRadialPairedArcOverlap, `SourceOriginalWeilCoverCellPrimitiveJumps, `SourceOriginalWeilCoverDissectionPrimitiveJumps, `SourceRadialDualTreeNormalization, `SourceOriginalWeilCoverNormalizedCellPrimitives, `SourceOriginalWeilCoverNormalizedCyclePrimitives]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_analyticAt_coordinate_transition, `Dubon2026.SourceFunctionField.CellGeometry.RadialCellDissection.paired_arc_overlap, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_primitive_jump_on_arc, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_exists_regular_primitives_and_jumps, `Dubon2026.SourceFunctionField.CellGeometry.RadialCellDissection.exists_normalising_constants, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_exists_normalized_primitives_and_jumps, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_normalized_cycle_primitives]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverNormalizedCellPrimitives]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Normalized Jumps postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Normalized Jumps theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER NORMALIZED JUMPS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
