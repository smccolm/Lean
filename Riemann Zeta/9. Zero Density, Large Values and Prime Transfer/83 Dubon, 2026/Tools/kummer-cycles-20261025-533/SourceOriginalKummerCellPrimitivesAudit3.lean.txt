import SourceOriginalWeilCoverCellPrimitives
import SourceOriginalWeilCoverCellPrimitiveFamily
import SourceOriginalWeilCoverRegularCellPrimitives
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_cell_exists_regular_primitives
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_cell_primitives_of_coordinates
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_regular_cell_primitives
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverCellPrimitives

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverCellPrimitives, `SourceOriginalWeilCoverCellPrimitiveFamily, `SourceOriginalWeilCoverRegularCellPrimitives]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_cell_exists_regular_primitives, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_cell_primitives_of_coordinates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_regular_cell_primitives]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverCellPrimitives]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Cell Primitives postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Cell Primitives theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER CELL PRIMITIVES AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceOriginalWeilCoverCellPrimitives
#lint in SourceOriginalWeilCoverCellPrimitiveFamily
#lint in SourceOriginalWeilCoverRegularCellPrimitives
