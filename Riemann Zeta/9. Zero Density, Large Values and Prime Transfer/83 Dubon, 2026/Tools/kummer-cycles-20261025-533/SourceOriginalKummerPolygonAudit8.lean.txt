private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialPolygonCorners, `SourceRadialPolygonState, `SourceRadialPolygonGluing, `SourceRadialPolygonWordCycles, `SourceRadialDissectionPolygonWord, `SourceRadialFundamentalCycleSystem, `SourceOriginalWeilCoverPolygonWord, `SourceOriginalWeilCoverCycleSystem]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.eV_not, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.sV_mem_ends, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.Cs_side, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.side_injective, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.corner_arc, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.crossAt_mono, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.corner_mono, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.crossAt_pair, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.corner_junction, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.mem_arcs, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.arcs_nodup, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.cyc_arcs, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.state_init, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.state_step, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.state_final, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.filter_indicator, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.letter_bd, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.w3_of_w1, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.exists_polygon_word, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionTrees.exists_fundamental_cycle_system, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_polygon_word, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_cycle_system]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.Cs, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.sV, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.eV, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.crossAt, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.Corner, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.arcs, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.State, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.State.mk, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPolygon.PolygonWordConclusion, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionTrees.FundamentalCycleSystem]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Polygon postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Polygon theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER POLYGON AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
