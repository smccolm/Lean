import SourceMarkedChartStabilizerInclusion

/-! DUB-17/18 -> reverse genus -> finite elliptic fibers -> original-field
marked-point chart. Consume the actual constructed point and original
modular-function embedding to prove that its field stabilizer fixes the
marked point up to sign. All coordinate premises are discharged by the
constructed unit-multiple formulas, rather than separately assumed. The
opposite inclusion and the required orbit cardinality remain open. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open WeierstrassCurve CongruenceSubgroup GammaOne GammaOne.ToricCoordinates
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    [CharP (IsLocalRing.ResidueField A) ℓ]
local notation "kk" => IsLocalRing.ResidueField A
local notation "F" => qExpFunctionFieldC kk (Gamma1 N)
private local instance : Algebra kk F := (F).toSubalgebra.algebra

theorem exists_original_marked_chart_stabilizer_inclusion
    (hℓN : ¬ ℓ ∣ N) (hN : 2 ≤ N)
    (x : F) (hx : (x : LaurentSeries kk) = jqModC kk)
    (k Ω : Type*) [Field k] [Field Ω] [DecidableEq Ω]
    [Algebra kk k] [Algebra kk Ω] [Algebra k Ω] [IsScalarTower kk k Ω] [IsAlgClosure k Ω]
    (E : WeierstrassCurve k) [E.IsElliptic] (hE : Transcendental kk E.j)
    (hgen : IntermediateField.adjoin kk ({E.j} : Set k) = ⊤) :
    ∃ (ζ : kk) (P : (E.baseChange Ω).toAffine.Point) (ψ : F →ₐ[kk] Ω),
      IsPrimitiveRoot ζ N ∧ addOrderOf P = N ∧ ψ x = algebraMap k Ω E.j ∧
      (∀ t : (ZMod N)ˣ, ∃ (a b : Ω) (h : (E.baseChange Ω).toAffine.Nonsingular a b),
        (t : ZMod N).val • P = Affine.Point.some a b h ∧
        ψ (w₁ N kk ζ t) = (E.baseChange Ω).c₄ * (12 * a + (E.baseChange Ω).b₂) /
          (E.baseChange Ω).c₆ ∧
        ψ (w₂ N kk ζ t) = (6 * a ^ 2 + (E.baseChange Ω).b₂ * a + (E.baseChange Ω).b₄) /
          (E.baseChange Ω).c₄) ∧
      ∀ σ : Ω ≃ₐ[k] Ω, (σ.toAlgHom.restrictScalars kk).comp ψ = ψ →
        Affine.Point.map (W' := E) σ.toAlgHom P = P ∨
          Affine.Point.map (W' := E) σ.toAlgHom P = -P := by
  obtain ⟨ζ, P, ψ, hζ, hPord, hψ, hcoord⟩ :=
    exists_original_marked_chart_construction A N ℓ hℓN hN x hx k Ω E hE hgen
  refine ⟨ζ, P, ψ, hζ, hPord, hψ, hcoord, fun σ hσ => ?_⟩
  letI : Fact (1 < N) := ⟨by omega⟩
  obtain ⟨a, b, h, hP, hw1, hw2⟩ := hcoord 1
  simp only [Units.val_one, ZMod.val_one, one_nsmul] at hP
  rw [hP]
  exact point_fixed_up_to_sign_of_embedding_fixed E hE ψ (w₁ N kk ζ 1) (w₂ N kk ζ 1)
    a b h hw1 hw2 σ hσ

end
end Dubon2026.SourceFunctionField.MarkedChart
