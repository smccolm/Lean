/-! DUB-17/18 -> reverse genus -> finite elliptic fibers -> natural marked
chart. Over the algebraic closure of the original generic j field, construct
the actual original-field chart with Galois and exact diamond covariance,
sign invariance, and separation up to sign. Every orbit and stabilizer premise
is discharged. The signed lifts suffice for the exact required chart formula;
an unsigned lift is not an input. Descent to an arbitrary full-torsion field
and finite elliptic-fiber bounds remain downstream obligations. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open WeierstrassCurve CongruenceSubgroup GammaOne GammaOne.ToricCoordinates IntegralDiamond
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    [CharP (IsLocalRing.ResidueField A) ℓ]
local notation "kk" => IsLocalRing.ResidueField A
local notation "F" => qExpFunctionFieldC kk (Gamma1 N)
private local instance : Algebra kk F := (F).toSubalgebra.algebra

theorem exists_original_natural_marked_chart_generic
    (hℓN : ¬ ℓ ∣ N) (hN : 2 ≤ N)
    (x : F) (hx : (x : LaurentSeries kk) = jqModC kk)
    (k Ω : Type*) [Field k] [Field Ω] [DecidableEq Ω]
    [Algebra kk k] [Algebra kk Ω] [Algebra k Ω] [IsScalarTower kk k Ω] [IsAlgClosure k Ω]
    (E : WeierstrassCurve k) [E.IsElliptic] (hE : Transcendental kk E.j)
    (hgen : IntermediateField.adjoin kk ({E.j} : Set k) = ⊤) :
    ∃ Ψ : {P : (E.baseChange Ω).toAffine.Point // addOrderOf P = N} →
        {ψ : F →ₐ[kk] Ω // ψ x = algebraMap k Ω E.j},
      (∀ (σ : Ω ≃ₐ[k] Ω) P P', P'.val = Affine.Point.map (W' := E) σ.toAlgHom P.val →
        (Ψ P').val = (σ.toAlgHom.restrictScalars kk).comp (Ψ P).val) ∧
      (∀ (γ : Gamma0 N) P P',
        P'.val = (((gamma0Unit N γ)⁻¹ : (ZMod N)ˣ) : ZMod N).val • P.val →
        (Ψ P').val = (Ψ P).val.comp (originalResidueDiamondAction A N ℓ hℓN γ).toAlgHom) ∧
      (∀ P P', P'.val = -P.val → Ψ P' = Ψ P) ∧
      (∀ P P', Ψ P' = Ψ P → P'.val = P.val ∨ P'.val = -P.val) := by
  obtain ⟨P, ψ, hP, hψ, hstab, hdiamond⟩ :=
    exists_original_marked_chart_stabilizer_equality A N ℓ hℓN hN x hx k Ω E hE hgen
  letI := valuation_residue_isAlgClosed A
  have hNK : (N : kk) ≠ 0 := fun h => hℓN ((CharP.cast_eq_zero_iff kk ℓ N).mp h)
  have htrans : ∀ Q : (E.baseChange Ω).toAffine.Point, addOrderOf Q = N →
      ∃ σ : Ω ≃ₐ[k] Ω, Affine.Point.map (W' := E) σ.toAlgHom P = Q ∨
        Affine.Point.map (W' := E) σ.toAlgHom P = -Q := by
    intro Q hQ
    apply exists_point_conjugacy_up_to_sign_of_unit_moves N hNK E hE hgen P hP ?_ Q hQ
    intro u
    obtain ⟨γ, hγ⟩ := gamma0Unit_surjective N u⁻¹
    obtain ⟨σ, _, hσ⟩ := hdiamond γ
    exact ⟨σ, by simpa only [hγ, inv_inv] using hσ⟩
  obtain ⟨Ψ, hΨ, hnat, hsep⟩ :=
    exists_natural_sign_chart_of_stabilizers E N x P hP ψ hψ hstab htrans
  refine ⟨Ψ, fun σ Q Q' h => hnat Q Q' σ h, ?_, ?_, ?_⟩
  · intro γ Q Q' hQ'
    obtain ⟨δ, hδψ, hδ⟩ := hdiamond γ
    exact sign_chart_diamond_of_signed_lift E N P ψ (fun Q => (Ψ Q).val) htrans hΨ
      (originalResidueDiamondAction A N ℓ hℓN γ)
      (((gamma0Unit N γ)⁻¹ : (ZMod N)ˣ) : ZMod N).val δ hδψ hδ Q Q' hQ'
  · intro Q Q' h
    exact (hsep Q Q').mpr (Or.inr h)
  · intro Q Q' h
    exact (hsep Q Q').mp h

end
end Dubon2026.SourceFunctionField.MarkedChart
