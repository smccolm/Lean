import SourceTateOriginalJCoordinate
import SourceMarkedChartCoordinateEmbedding
import SourceMarkedChartVariableInvariants
import SourceMarkedChartEqualJTransfer
import SourceMarkedChartInvariantSeparation
import SourceMarkedChartFieldDescent
import SourceMarkedChartTorsionDescent
import SourceMarkedChartEmbeddingBound
import SourceMarkedChartResidueDegree
import SourceMarkedChartDescendedPoint
import SourceMarkedChartOriginalConstruction
import SourceMarkedChartStabilizerInclusion
import SourceMarkedChartOriginalStabilizer
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceTateOriginalJCoordinate, `SourceMarkedChartCoordinateEmbedding, `SourceMarkedChartVariableInvariants, `SourceMarkedChartEqualJTransfer, `SourceMarkedChartInvariantSeparation, `SourceMarkedChartFieldDescent, `SourceMarkedChartTorsionDescent, `SourceMarkedChartEmbeddingBound, `SourceMarkedChartResidueDegree, `SourceMarkedChartDescendedPoint, `SourceMarkedChartOriginalConstruction, `SourceMarkedChartStabilizerInclusion, `SourceMarkedChartOriginalStabilizer]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
