import Mathlib.AlgebraicGeometry.EllipticCurve.Affine.Point

/-! DUB-17/18 -> reverse genus -> finite elliptic fibers -> original marked
chart -> reverse stabilizer inclusion. These are the two literal Galois
actions needed for the index comparison: on actual exact-order points up
to sign, and on the original field's normalized embeddings. No abstract
replacement of either carrier or assumed action compatibility is used. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open WeierstrassCurve MarkedTorsion

section Points
variable {k Ω : Type*} [Field k] [Field Ω] [DecidableEq Ω] [Algebra k Ω]
    (E : WeierstrassCurve k) (N : ℕ)

/-- The actual field automorphisms act on exact-order points modulo sign. -/
abbrev signClassGaloisAction : MulAction (Ω ≃ₐ[k] Ω) (SignClasses (E.baseChange Ω).toAffine.Point N) where
  smul σ s := mapSignClasses (Affine.Point.map (W' := E) σ.toAlgHom)
    (Affine.Point.map_injective σ.toAlgHom) s
  one_smul s := by
    apply Subtype.ext
    change Affine.Point.map (W' := E) (1 : Ω ≃ₐ[k] Ω).toAlgHom '' s.val = s.val
    have hm : (Affine.Point.map (W' := E) (1 : Ω ≃ₐ[k] Ω).toAlgHom :
        (E.baseChange Ω).toAffine.Point → (E.baseChange Ω).toAffine.Point) = id := by
      funext P
      cases P <;> rfl
    rw [hm, Set.image_id]
  mul_smul σ τ s := by
    apply Subtype.ext
    change Affine.Point.map (W' := E) (σ * τ).toAlgHom '' s.val =
      Affine.Point.map (W' := E) σ.toAlgHom '' (Affine.Point.map (W' := E) τ.toAlgHom '' s.val)
    rw [Set.image_image]
    congr 1
    funext P
    exact (Affine.Point.map_map τ.toAlgHom σ.toAlgHom P).symm

theorem signClass_smul_eq_iff
    (P Q : (E.baseChange Ω).toAffine.Point) (hP : addOrderOf P = N) (hQ : addOrderOf Q = N)
    (σ : Ω ≃ₐ[k] Ω) :
    letI := signClassGaloisAction (Ω := Ω) E N
    σ • (⟨{P, -P}, P, hP, rfl⟩ : SignClasses (E.baseChange Ω).toAffine.Point N) =
      ⟨{Q, -Q}, Q, hQ, rfl⟩ ↔
      Affine.Point.map (W' := E) σ.toAlgHom P = Q ∨
        Affine.Point.map (W' := E) σ.toAlgHom P = -Q := by
  letI := signClassGaloisAction (Ω := Ω) E N
  have he : (σ • (⟨{P, -P}, P, hP, rfl⟩ : SignClasses (E.baseChange Ω).toAffine.Point N) =
      ⟨{Q, -Q}, Q, hQ, rfl⟩) ↔
      (Affine.Point.map (W' := E) σ.toAlgHom '' ({P, -P} : Set (E.baseChange Ω).toAffine.Point)) =
        {Q, -Q} := Subtype.ext_iff
  rw [he, Set.image_pair, map_neg, signPair_eq_iff]

theorem signClass_stabilizer_iff
    (P : (E.baseChange Ω).toAffine.Point) (hP : addOrderOf P = N) (σ : Ω ≃ₐ[k] Ω) :
    letI := signClassGaloisAction (Ω := Ω) E N
    σ ∈ MulAction.stabilizer (Ω ≃ₐ[k] Ω)
      (⟨{P, -P}, P, hP, rfl⟩ : SignClasses (E.baseChange Ω).toAffine.Point N) ↔
      Affine.Point.map (W' := E) σ.toAlgHom P = P ∨
        Affine.Point.map (W' := E) σ.toAlgHom P = -P := by
  letI := signClassGaloisAction (Ω := Ω) E N
  rw [MulAction.mem_stabilizer_iff, signClass_smul_eq_iff]

end Points

section Embeddings
variable {K k F Ω : Type*} [Field K] [Field k] [Field F] [Field Ω]
    [Algebra K k] [Algebra K F] [Algebra K Ω] [Algebra k Ω] [IsScalarTower K k Ω]
    (x : F) (t : k)

/-- Field automorphisms act on embeddings with the prescribed original j value. -/
abbrev normalizedEmbeddingAction : MulAction (Ω ≃ₐ[k] Ω)
    {ψ : F →ₐ[K] Ω // ψ x = algebraMap k Ω t} where
  smul σ ψ := ⟨(σ.toAlgHom.restrictScalars K).comp ψ.val, by
    change σ (ψ.val x) = algebraMap k Ω t
    rw [ψ.property, σ.commutes]⟩
  one_smul ψ := Subtype.ext (AlgHom.ext fun _ => rfl)
  mul_smul _ _ ψ := Subtype.ext (AlgHom.ext fun _ => rfl)

local instance : MulAction (Ω ≃ₐ[k] Ω) {ψ : F →ₐ[K] Ω // ψ x = algebraMap k Ω t} :=
  normalizedEmbeddingAction x t

theorem normalizedEmbedding_stabilizer_iff
    (ψ : F →ₐ[K] Ω) (hψ : ψ x = algebraMap k Ω t) (σ : Ω ≃ₐ[k] Ω) :
    σ ∈ MulAction.stabilizer (Ω ≃ₐ[k] Ω)
      (⟨ψ, hψ⟩ : {φ : F →ₐ[K] Ω // φ x = algebraMap k Ω t}) ↔
      (σ.toAlgHom.restrictScalars K).comp ψ = ψ := by
  rw [MulAction.mem_stabilizer_iff]
  exact Subtype.ext_iff

end Embeddings
end
end Dubon2026.SourceFunctionField.MarkedChart
