/-! DUB-17/18 -> reverse genus -> finite elliptic fibers -> marked-point
stabilizers. The already proved irreducible integral modular polynomial
gives transitivity on its roots over any field generated by a generic j.
Together with the native cyclic-quotient bijection this will move cyclic
subgroups; the diamond construction supplies their generators up to sign.
No new modular-polynomial irreducibility theorem is assumed. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open IntegralJCoordinate Polynomial

theorem modularPolynomial_irreducible_coordinate
    {K k : Type*} [Field K] [IsAlgClosed K] [Field k] [Algebra K k]
    (N : ℕ) [NeZero N] (hN : (N : K) ≠ 0) (t : k) (ht : Transcendental K t)
    (hgen : IntermediateField.adjoin K ({t} : Set k) = ⊤) :
    Irreducible (relationAtSeed (modularPolynomial N) t) := by
  let e : RatFunc K ≃ₐ[K] k := (RatFunc.algEquivOfTranscendental t ht).trans
    ((IntermediateField.equivOfEq hgen).trans IntermediateField.topEquiv)
  have he : e RatFunc.X = t := RatFunc.algEquivOfTranscendental_X t ht
  have hi := (modularPolynomial_irreducible_ratFunc N K hN).map
    (Polynomial.mapEquiv e.toRingEquiv)
  change Irreducible ((relationAtSeed (modularPolynomial N) (RatFunc.X : RatFunc K)).map
    e.toRingHom) at hi
  rw [relationAtSeed_map, he] at hi
  exact hi

theorem exists_modular_root_conjugacy
    {K k Ω : Type*} [Field K] [IsAlgClosed K] [Field k] [Field Ω]
    [Algebra K k] [Algebra k Ω] [Normal k Ω]
    (N : ℕ) [NeZero N] (hN : (N : K) ≠ 0) (t : k) (ht : Transcendental K t)
    (hgen : IntermediateField.adjoin K ({t} : Set k) = ⊤) (x y : Ω)
    (hx : (relationAtSeed (modularPolynomial N) (algebraMap k Ω t)).IsRoot x)
    (hy : (relationAtSeed (modularPolynomial N) (algebraMap k Ω t)).IsRoot y) :
    ∃ σ : Ω ≃ₐ[k] Ω, σ x = y := by
  let P := relationAtSeed (modularPolynomial N) t
  have hirr : Irreducible P := modularPolynomial_irreducible_coordinate N hN t ht hgen
  have hm : P.Monic := (modularPolynomial_monic N).map _
  have hmin : ∀ z : Ω,
      (relationAtSeed (modularPolynomial N) (algebraMap k Ω t)).IsRoot z →
      minpoly k z = P := by
    intro z hz
    rw [← relationAtSeed_map, Polynomial.IsRoot, Polynomial.eval_map, ← Polynomial.aeval_def] at hz
    exact (minpoly.eq_of_irreducible_of_monic hirr hz hm).symm
  exact (Normal.minpoly_eq_iff_mem_orbit (F := k) (E := Ω)).mp
    ((hmin y hy).trans (hmin x hx).symm)

end
end Dubon2026.SourceFunctionField.MarkedChart
