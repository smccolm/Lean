import SourceMarkedOrbitMap
import SourceMarkedNaturalSignChart
import SourceMarkedSignChartDiamond
import SourceMarkedOriginalNaturalChart
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceFunctionField.MarkedChart.exists_injective_equivariant_map_of_stabilizer_eq
#print axioms Dubon2026.SourceFunctionField.MarkedChart.exists_natural_sign_chart_of_stabilizers
#print axioms Dubon2026.SourceFunctionField.MarkedChart.sign_chart_diamond_of_signed_lift
#print axioms Dubon2026.SourceFunctionField.MarkedChart.exists_original_natural_marked_chart_generic

private def runMarkedNaturalChartAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMarkedOrbitMap, `SourceMarkedNaturalSignChart, `SourceMarkedSignChartDiamond, `SourceMarkedOriginalNaturalChart]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.MarkedChart.exists_injective_equivariant_map_of_stabilizer_eq, `Dubon2026.SourceFunctionField.MarkedChart.exists_natural_sign_chart_of_stabilizers, `Dubon2026.SourceFunctionField.MarkedChart.sign_chart_diamond_of_signed_lift, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_natural_marked_chart_generic]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked natural chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked natural chart theorems discovered"
  logInfo m!"SOURCE MARKED NATURAL CHART AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runMarkedNaturalChartAudit

#lint in SourceMarkedOrbitMap
#lint in SourceMarkedNaturalSignChart
#lint in SourceMarkedSignChartDiamond
#lint in SourceMarkedOriginalNaturalChart
