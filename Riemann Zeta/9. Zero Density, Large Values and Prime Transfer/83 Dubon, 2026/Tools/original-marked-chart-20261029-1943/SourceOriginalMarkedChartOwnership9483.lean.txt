private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceTateOriginalJCoordinate, `Dubon2026.SourceMarkedChartCoordinateEmbedding, `Dubon2026.SourceMarkedChartVariableInvariants, `Dubon2026.SourceMarkedChartEqualJTransfer, `Dubon2026.SourceMarkedChartInvariantSeparation, `Dubon2026.SourceMarkedChartFieldDescent, `Dubon2026.SourceMarkedChartTorsionDescent, `Dubon2026.SourceMarkedChartEmbeddingBound, `Dubon2026.SourceMarkedChartResidueDegree, `Dubon2026.SourceMarkedChartDescendedPoint, `Dubon2026.SourceMarkedChartOriginalConstruction, `Dubon2026.SourceMarkedChartStabilizerInclusion, `Dubon2026.SourceMarkedChartOriginalStabilizer, `Dubon2026.SourceMarkedTorsionSignClasses, `Dubon2026.SourceMarkedTorsionSignedIndex, `Dubon2026.SourceMarkedTorsionActualCount, `Dubon2026.SourceMarkedChartEmbeddingConjugacy, `Dubon2026.SourceMarkedChartPointTransport, `Dubon2026.SourceMarkedChartDiamondNormalization, `Dubon2026.SourceMarkedChartOriginalDiamond, `Dubon2026.SourceMarkedCyclicRootTransitivity, `Dubon2026.SourceMarkedCyclicGeneratorConjugacy, `Dubon2026.SourceMarkedTorsionGeneratorUnit, `Dubon2026.SourceMarkedPointSignTransitivity, `Dubon2026.SourceMarkedChartOriginalSignOrbit, `Dubon2026.SourceMarkedChartStabilizerActions, `Dubon2026.SourceMarkedStabilizerIndexComparison, `Dubon2026.SourceMarkedChartStabilizerEquality, `Dubon2026.SourceMarkedChartOriginalStabilizerEquality, `Dubon2026.SourceMarkedOrbitMap, `Dubon2026.SourceMarkedNaturalSignChart, `Dubon2026.SourceMarkedSignChartDiamond, `Dubon2026.SourceMarkedOriginalNaturalChart]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
