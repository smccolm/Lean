/-! DUB-17/18 -> reverse genus -> finite elliptic fibers -> original marked
chart -> stabilizer comparison. Native extension of embeddings in a normal
field makes every pair of j-normalized embeddings conjugate. This permits
the already proved diamond coordinate formulas to supply generator moves,
before any separate Tate transvection construction is expanded. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section

theorem coordinate_algHom_ext {K k L : Type*} [Field K] [Field k] [Field L]
    [Algebra K k] [Algebra K L] (t : k)
    (hgen : IntermediateField.adjoin K ({t} : Set k) = ⊤)
    (f g : k →ₐ[K] L) (h : f t = g t) : f = g := by
  let B := IntermediateField.adjoin K ({t} : Set k)
  have he : f.comp B.val = g.comp B.val := by
    apply IntermediateField.adjoin_algHom_ext
    intro y hy
    rw [Set.mem_singleton_iff] at hy
    subst y
    exact h
  ext y
  have hy : y ∈ B := by rw [show B = ⊤ from hgen]; exact IntermediateField.mem_top
  exact AlgHom.congr_fun he ⟨y, hy⟩

theorem exists_normalized_embedding_conjugacy
    {K k F Ω : Type*} [Field K] [Field k] [Field F] [Field Ω]
    [Algebra K k] [Algebra K F] [Algebra K Ω] [Algebra k Ω]
    [IsScalarTower K k Ω] [Normal k Ω]
    (t : k) (ht : Transcendental K t)
    (hgen : IntermediateField.adjoin K ({t} : Set k) = ⊤)
    (x : F) (ψ ψ' : F →ₐ[K] Ω)
    (hψ : ψ x = algebraMap k Ω t) (hψ' : ψ' x = algebraMap k Ω t) :
    ∃ σ : Ω ≃ₐ[k] Ω, (σ.toAlgHom.restrictScalars K).comp ψ = ψ' := by
  have htΩ : Transcendental K (algebraMap k Ω t) :=
    (transcendental_algebraMap_iff (algebraMap k Ω).injective).mpr ht
  have hx : Transcendental K x := by
    intro hx
    apply htΩ
    rw [← hψ]
    exact hx.algHom ψ
  obtain ⟨ι, hι⟩ := exists_coordinate_embedding t ht hgen x hx
  have he : ∀ φ : F →ₐ[K] Ω, φ x = algebraMap k Ω t →
      φ.comp ι = IsScalarTower.toAlgHom K k Ω := by
    intro φ hφ
    apply coordinate_algHom_ext t hgen
    change φ (ι t) = algebraMap k Ω t
    rw [hι, hφ]
  letI : Algebra k F := ι.toRingHom.toAlgebra
  letI : Algebra F Ω := ψ.toRingHom.toAlgebra
  haveI : IsScalarTower k F Ω := IsScalarTower.of_algebraMap_eq fun z =>
    (AlgHom.congr_fun (he ψ hψ) z).symm
  let φ : F →ₐ[k] Ω := { ψ'.toRingHom with
    commutes' := fun z => AlgHom.congr_fun (he ψ' hψ') z }
  let τ : Ω →ₐ[k] Ω := φ.liftNormal Ω
  let σ : Ω ≃ₐ[k] Ω := AlgEquiv.ofBijective τ (AlgHom.normal_bijective k Ω Ω τ)
  refine ⟨σ, AlgHom.ext fun y => ?_⟩
  change φ.liftNormal Ω (algebraMap F Ω y) = φ y
  exact φ.liftNormal_commutes Ω y

end
end Dubon2026.SourceFunctionField.MarkedChart
