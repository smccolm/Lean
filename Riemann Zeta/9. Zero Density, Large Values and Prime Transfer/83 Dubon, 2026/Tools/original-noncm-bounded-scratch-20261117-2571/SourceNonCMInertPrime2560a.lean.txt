/-! DUB-17/18 -> eventual original residual large image -> dihedral exclusion
-> fixed-character obstruction. Reuse the source definition of non-CM and its
proved inert-prime reformulation to produce a nonzero original coefficient.
The residual dihedral character and its uniform conductor bound remain open. -/
namespace Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
noncomputable section

/-- Every nontrivial primitive quadratic character has a good inert prime with nonzero actual original Fourier coefficient. -/
theorem nonCM_exists_original_inert_prime_coefficient
    (Q : ℕ) [NeZero Q] {k : ℤ} (f : NonCMPrimitiveCuspForm Q k)
    (D : ℕ+) (χ : DirichletCharacter ℂ D) (hχp : χ.IsPrimitive)
    (hχ1 : χ ≠ 1) (hχq : χ.IsQuadratic) :
    ∃ q : ℕ, q.Prime ∧ q.Coprime (Q * D) ∧ characterCoefficients χ q = -1 ∧
      sourcePrimitiveFieldCoefficient f.toPrimitiveCuspForm q ≠ 0 := by
  by_contra h
  apply f.nonCM
  refine ⟨D, χ, hχp, hχ1, hχq, ?_⟩
  apply (coefficientSelfTwist_iff_inert_zero χ hχq _).mpr
  intro q hq hgood hinert
  have hz : sourcePrimitiveFieldCoefficient f.toPrimitiveCuspForm q = 0 := by
    by_contra hn
    exact h ⟨q, hq, hgood, hinert, hn⟩
  have he := congrArg (fun x : sourcePrimitiveCoefficientField f.toPrimitiveCuspForm => (x : ℂ)) hz
  simpa only [sourcePrimitiveFieldCoefficient_coe, ZeroMemClass.coe_zero] using he

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
