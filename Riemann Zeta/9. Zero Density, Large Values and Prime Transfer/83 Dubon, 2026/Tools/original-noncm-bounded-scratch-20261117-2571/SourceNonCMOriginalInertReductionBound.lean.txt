/-! DUB-17/18 -> original dihedral image exclusion -> fixed-character residual
obstruction. Non-CM supplies one actual inert coefficient. Its integer relation
excludes zero reduction in every sufficiently large characteristic and every
original local coefficient embedding. A uniform bound on the dihedral character
conductor and its original Galois/Dirichlet comparison remain open. -/
namespace Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
open Dubon2026.SourceDickson
noncomputable section
open Polynomial
open scoped ValuativeRel

/-- For one source quadratic character, an actual non-CM form has a good inert prime with nonzero original residue coefficient in every sufficiently large characteristic. -/
theorem nonCM_exists_original_inert_reduction_bound
    (Q : ℕ) [NeZero Q] {k : ℤ} (f : NonCMPrimitiveCuspForm Q k) (hk : 2 ≤ k)
    (D : ℕ+) (χ : DirichletCharacter ℂ D) (hχp : χ.IsPrimitive)
    (hχ1 : χ ≠ 1) (hχq : χ.IsQuadratic) :
    ∃ B : ℕ, ∀ (p : ℕ) [Fact p.Prime], B < p →
      ∀ {E : Type} [Field E] [CharZero E] [ValuativeRel E] [CharP 𝓀[E] p]
        (τ : sourcePrimitiveCoefficientField f.toPrimitiveCuspForm →ₐ[ℚ] E),
        ∃ q : ℕ, q.Prime ∧ q.Coprime (Q * D * p) ∧ characterCoefficients χ q = -1 ∧
          IsLocalRing.residue 𝒪[E]
            ⟨τ (sourcePrimitiveFieldCoefficient f.toPrimitiveCuspForm q),
              sourcePrimitiveCoefficient_valuation_le_one f.toPrimitiveCuspForm hk τ q⟩ ≠ 0 := by
  obtain ⟨q, H, hq, hgood, hinert, hH0, hH⟩ :=
    nonCM_exists_original_inert_integer_obstruction Q f hk D χ hχp hχ1 hχq
  refine ⟨max q (H.coeff 0).natAbs, ?_⟩
  intro p _ hp E _ _ _ _ τ
  have hqp : q < p := lt_of_le_of_lt (le_max_left _ _) hp
  have hpH : (H.coeff 0).natAbs < p := lt_of_le_of_lt (le_max_right _ _) hp
  have hcop : q.Coprime p := by
    apply hq.coprime_iff_not_dvd.mpr
    intro hd
    have he := (Nat.prime_dvd_prime_iff_eq hq (Fact.out : p.Prime)).mp hd
    omega
  refine ⟨q, hq, Nat.coprime_mul_iff_right.mpr ⟨hgood, hcop⟩, hinert, ?_⟩
  intro hz
  exact not_int_dvd_of_natAbs_lt hH0 hpH
    (original_coefficient_zero_reduction_dvd_relation_constant Q p f.toPrimitiveCuspForm hk q H hH τ hz)

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
