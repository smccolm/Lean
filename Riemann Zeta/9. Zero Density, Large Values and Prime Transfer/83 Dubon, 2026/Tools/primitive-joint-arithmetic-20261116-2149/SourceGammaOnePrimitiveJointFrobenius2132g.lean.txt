/-- Each original generator has exactly its geometric/boundary pair
in the literal joint operator image. -/
theorem binaryPrimitiveJointHeckeDiamondWord_ι
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J) (s : S) :
    binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ s) =
      (Sum.elim Hq (fun _ : Gamma0 M => 1) s,
        binaryPrimitiveBoundaryHeckeDiamondOperators Q N w s) := by
  change (FreeAlgebra.lift ℤ (Sum.elim Hq (fun _ : Gamma0 M => 1)) (FreeAlgebra.ι ℤ s),
    FreeAlgebra.lift ℤ (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w) (FreeAlgebra.ι ℤ s)) = _
  exact Prod.ext (FreeAlgebra.lift_ι_apply (Sum.elim Hq (fun _ : Gamma0 M => 1)) s)
    (FreeAlgebra.lift_ι_apply (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w) s)

/-- The actual joint representation obeys the exact original Hecke
quadratic at every original good-prime local valuation, with the
geometric hypotheses proved and the diamond coefficient explicit. -/
theorem binaryPrimitiveJointGaloisRep_local_frobenius
    (g : Gamma0 M) (hlabel : (g.val 1 1 : ZMod M) = ell.val)
    (d : (A).decompositionSubgroup ℚ) (hd : ∀ a : kk, d • a = a ^ ell.val) :
    binaryPrimitiveJointGaloisRep Q p n w d.val *
        binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ (Sum.inl ell)) =
      binaryPrimitiveJointGaloisRep Q p n w d.val * binaryPrimitiveJointGaloisRep Q p n w d.val +
        (ell.val ^ (w + 1)) •
          binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ (Sum.inr g)) := by
  letI := GaussReduction.residue_charP_of_prime_mem_nonunits A ell.val ell.property.1
    (sourceRationalLocalValuation_prime ell.val ell.property.1 e)
  have haway := ell.property.1.coprime_iff_not_dvd.mp (binaryPrimitiveLevelPrime Q N ell).property.2
  have hgeo := gammaOne_local_degree_and_genus_eq M ell.val ell.property.1 haway e
  have hp : (p : kk) ≠ 0 := by
    rw [Ne, CharP.cast_eq_zero_iff _ ell.val]
    intro h
    exact (ell.property.1.coprime_iff_not_dvd.mp ell.property.2)
      (dvd_mul_of_dvd_right (h.trans (dvd_pow_self p (Nat.succ_ne_zero n))) Q)
  rw [binaryPrimitiveJointHeckeDiamondWord_ι, binaryPrimitiveJointHeckeDiamondWord_ι]
  apply Prod.ext
  · simpa only [← Nat.cast_pow, Nat.cast_smul_eq_nsmul] using
      gammaOne_diamond_quotient_hecke_frobenius_of_intertwining Q p n w A ell
        hgeo.1 hgeo.2 d hd hp (Hq ell) (hH ell)
  · simpa only [← Nat.cast_pow, Nat.cast_smul_eq_nsmul] using
      binaryPrimitiveBoundaryGaloisRep_frobenius Q N w A ell g hlabel d hd

end
