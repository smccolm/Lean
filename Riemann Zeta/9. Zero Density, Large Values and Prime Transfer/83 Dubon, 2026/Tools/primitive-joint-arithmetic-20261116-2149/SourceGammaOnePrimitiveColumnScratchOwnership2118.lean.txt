open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOnePrimitiveCuspBoundary, `SourceGammaOnePrimitiveWidth, `SourceGammaOnePrimitiveWidthColumns, `SourceGammaOnePrimitiveCuspValues, `SourceGammaOnePrimitiveCuspConjugation, `SourceGammaOnePrimitiveDiamond, `SourceGammaOnePrimitiveUpperColumns, `SourceGammaOnePrimitiveFixedCosets, `SourceGammaOnePrimitiveFixedOrbit, `SourceGammaOnePrimitivePrimeOrbits, `SourceGammaOnePrimeHeckeCardinality, `SourceGammaOnePrimitiveTwoOrbits, `SourceGammaOnePrimitiveOrbitRepresentatives, `SourceGammaOnePrimitiveDividedTerm, `SourceGammaOnePrimitiveMultipliedTerm, `SourceGammaOnePrimitiveTermColumns, `SourceGammaOneFiniteBinaryBoundary, `SourceGammaOnePrimitiveBoundaryOrbit, `SourceGammaOnePrimitiveHeckeTrace, `SourceGammaOnePrimitiveColumnBoundary, `SourceGammaOnePrimitiveColumnHecke, `SourceGammaOnePrimitiveColumnDiamond]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
