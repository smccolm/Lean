private theorem pair_commute_of_eq {X Y : Type} [Ring X] [Ring Y]
    (a b : X × Y) (ax bx : X) (ay by' : Y)
    (ha : a = (ax, ay)) (hb : b = (bx, by'))

  subst a
  subst b
  exact Prod.ext hx.eq hy.eq

/-- Each original joint diamond commutes with every joint generator;
the geometric diamond is one and the actual boundary operators commute. -/
theorem binaryPrimitiveJointHeckeDiamondWord_diamond_commute
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (g : Gamma0 M) (s : S) :
    Commute (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ (Sum.inr g)))
      (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ s)) := by
  exact pair_commute_of_eq _ _ 1 (Sum.elim Hq (fun _ : Gamma0 M => 1) s)
    (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w (Sum.inr g))
    (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w s)
    (binaryPrimitiveJointHeckeDiamondWord_ι Q p n w Hq (Sum.inr g))
    (binaryPrimitiveJointHeckeDiamondWord_ι Q p n w Hq s)
    (Commute.one_left _) (binaryPrimitiveBoundaryHeckeDiamondOperators_commute Q N w (Sum.inr g) s)

/-- The actual joint prime/diamond generators commute, with original local
Frobenius existence and all geometric hypotheses proved. -/
theorem binaryPrimitiveJointHeckeDiamondWord_generators_commute
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    (a b : S) :
    Commute (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ a))
      (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ b)) := by
  cases a with
  | inl ell =>
    obtain ⟨e⟩ := sourceRationalLocalClosure_nonempty_embedding ell.val ell.property.1
    obtain ⟨g, hg, σ, hσ, hpoly⟩ :=
      binaryPrimitiveJointGaloisRep_exists_local_frobenius Q p n w Hq hH ell e
    exact sourceHecke_commute_of_unit_quadratic _ _ _ _ (ell.val ^ (w + 1))
      ((Group.isUnit σ.val).map (binaryPrimitiveJointGaloisRep Q p n w))
      (binaryPrimitiveJointGaloisRep_word Q p n w Hq hH σ.val (FreeAlgebra.ι ℤ b))
      (binaryPrimitiveJointHeckeDiamondWord_diamond_commute Q p n w Hq g b) hpoly
  | inr g => exact binaryPrimitiveJointHeckeDiamondWord_diamond_commute Q p n w Hq g b

/-- The literal joint ring-hom image carrying the original adic character
is commutative, without assuming commutativity or changing its carrier. -/
theorem binaryPrimitiveJointHeckeDiamondWord_range_isMulCommutative
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell)) :
    IsMulCommutative (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq).range :=
  sourceIntegralWordRange_isMulCommutative _
    (binaryPrimitiveJointHeckeDiamondWord_generators_commute Q p n w Hq hH)
