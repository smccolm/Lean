  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.primitiveCuspWidth, `Dubon2026.SourceFunctionField.GammaOne.primitiveCuspGenerator, `Dubon2026.SourceFunctionField.GammaOne.primitiveCuspTranslation, `Dubon2026.SourceFunctionField.GammaOne.primitiveCuspValue, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveCuspBoundary, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveColumnBoundary, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveLevelPrime, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveColumnHecke, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveColumnDiamond]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Primitive column postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Primitive column theorems discovered"
  logInfo m!"SOURCE GAMMAONE PRIMITIVE COLUMN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."
