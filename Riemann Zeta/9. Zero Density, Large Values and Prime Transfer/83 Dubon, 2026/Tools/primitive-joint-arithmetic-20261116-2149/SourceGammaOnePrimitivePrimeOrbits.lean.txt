/-! DUB-17/18 -> original all-weight association -> arithmetic realization
of the exact joint operator character -> primitive boundary Frobenius.
Every actual primitive cusp translation has prime-th power in the
principal prime-level subgroup. Consequently its original Hecke-coset
orbits have length one or that prime. This retains torsion coefficients
by using the primitive loop, rather than a level-multiple loop. The
remaining fixed-orbit count and two-term trace identity are separate. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open CongruenceSubgroup
open scoped MatrixGroups

/-- The prime-th power of the actual primitive cusp loop lies in the
principal subgroup of that prime level. -/
theorem primitiveCuspGenerator_pow_mem_principal (M ell : ℕ)
    (q : SL(2, ℤ) ⧸ Gamma1 M) :
    (primitiveCuspGenerator M q).val ^ ell ∈ Gamma ell := by
  have hT : ModularGroup.T ^ (primitiveCuspWidth M q * ell) ∈ Gamma ell := by
    have hd : (ell : ℤ) ∣ (primitiveCuspWidth M q * ell : ℕ) := by
      exact_mod_cast dvd_mul_left ell (primitiveCuspWidth M q)
    simpa only [zpow_natCast] using
      ModularGroup_T_pow_mem_Gamma (ell : ℤ) (primitiveCuspWidth M q * ell : ℕ) hd
  have he : (primitiveCuspGenerator M q).val =
      MulAut.conj q.out⁻¹ (ModularGroup.T ^ primitiveCuspWidth M q) := by
    change q.out⁻¹ * ModularGroup.T ^ primitiveCuspWidth M q * q.out =
      q.out⁻¹ * ModularGroup.T ^ primitiveCuspWidth M q * (q.out⁻¹)⁻¹
    rw [inv_inv]
  rw [he, ← map_pow, ← pow_mul]
  exact (Gamma_normal ell).conj_mem _ hT q.out⁻¹

/-- The prime-th power fixes every original Gamma1 Hecke coset. -/
theorem primitiveCuspHecke_pow_fixes (M ell : ℕ)
    (q : SL(2, ℤ) ⧸ Gamma1 M)
    (x : Gamma1 M ⧸ sourceGammaOneHeckeUpper M ell) :
    primitiveCuspGenerator M q ^ ell • x = x := by
  induction x using QuotientGroup.induction_on with
  | H r =>
    rw [MulAction.Quotient.smul_coe, smul_eq_mul, QuotientGroup.eq,
      ← inv_mem_iff, mul_inv_rev, inv_inv]
    have hc := (Gamma_normal ell).conj_mem _
      (primitiveCuspGenerator_pow_mem_principal M ell q) r.val⁻¹
    obtain ⟨_, h01, _, _⟩ := Gamma_mem.mp hc
    change (((r⁻¹ * (primitiveCuspGenerator M q ^ ell * r)).val 0 1 : ℤ) : ZMod ell) = 0
    simpa only [Subgroup.coe_mul, Subgroup.coe_inv, Subgroup.coe_pow, inv_inv,
      mul_assoc] using h01

/-- The original primitive cusp action has only fixed or prime-length
orbits in the actual good-prime Hecke coset space. -/
theorem primitiveCuspHecke_period_one_or_prime (M ell : ℕ) (hell : ell.Prime)
    (q : SL(2, ℤ) ⧸ Gamma1 M)
    (x : Gamma1 M ⧸ sourceGammaOneHeckeUpper M ell) :
    Function.minimalPeriod (primitiveCuspGenerator M q • ·) x = 1 ∨
      Function.minimalPeriod (primitiveCuspGenerator M q • ·) x = ell := by
  apply (Nat.dvd_prime hell).mp
  exact MulAction.pow_smul_eq_iff_minimalPeriod_dvd.mp
    (primitiveCuspHecke_pow_fixes M ell q x)

end
end Dubon2026.SourceFunctionField.GammaOne
