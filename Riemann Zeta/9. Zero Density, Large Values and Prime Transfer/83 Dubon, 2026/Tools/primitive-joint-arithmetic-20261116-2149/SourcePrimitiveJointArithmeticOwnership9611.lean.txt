private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOnePrimitiveCuspBoundary, `Dubon2026.SourceGammaOnePrimitiveWidth, `Dubon2026.SourceGammaOnePrimitiveWidthColumns, `Dubon2026.SourceGammaOnePrimitiveCuspValues, `Dubon2026.SourceGammaOnePrimitiveCuspConjugation, `Dubon2026.SourceGammaOnePrimitiveDiamond, `Dubon2026.SourceGammaOnePrimitiveUpperColumns, `Dubon2026.SourceGammaOnePrimitiveFixedCosets, `Dubon2026.SourceGammaOnePrimitiveFixedOrbit, `Dubon2026.SourceGammaOnePrimitivePrimeOrbits, `Dubon2026.SourceGammaOnePrimeHeckeCardinality, `Dubon2026.SourceGammaOnePrimitiveTwoOrbits, `Dubon2026.SourceGammaOnePrimitiveOrbitRepresentatives, `Dubon2026.SourceGammaOnePrimitiveDividedTerm, `Dubon2026.SourceGammaOnePrimitiveMultipliedTerm, `Dubon2026.SourceGammaOnePrimitiveTermColumns, `Dubon2026.SourceGammaOneFiniteBinaryBoundary, `Dubon2026.SourceGammaOnePrimitiveBoundaryOrbit, `Dubon2026.SourceGammaOnePrimitiveHeckeTrace, `Dubon2026.SourceGammaOnePrimitiveColumnBoundary, `Dubon2026.SourceGammaOnePrimitiveColumnHecke, `Dubon2026.SourceGammaOnePrimitiveColumnDiamond, `Dubon2026.SourceGammaOnePrimitiveHeckeDiamondWords, `Dubon2026.SourceGammaOnePrimitiveJointSquare, `Dubon2026.SourceGammaOnePrimitiveUnitActions, `Dubon2026.SourceGammaOnePrimitiveUnitHecke, `Dubon2026.SourceGammaOnePrimitiveBoundaryQuadratic, `Dubon2026.SourceGammaOnePrimitiveBoundaryGalois, `Dubon2026.SourceGammaOnePrimitiveBoundaryDescent, `Dubon2026.SourceGammaOnePrimitiveJointRing, `Dubon2026.SourceGammaOnePrimitiveJointCharacter, `Dubon2026.SourceGammaOnePrimitiveJointCharacterValues, `Dubon2026.SourceGammaOnePrimitiveBoundaryGaloisWords, `Dubon2026.SourceGammaOnePrimitiveJointGalois, `Dubon2026.SourceGammaOnePrimitiveJointDescent, `Dubon2026.SourceGammaOneGeometricPrimitiveJointCharacter, `Dubon2026.SourceGammaOnePrimitiveJointFrobenius, `Dubon2026.SourceGammaOnePrimitiveBoundaryOperatorCommute, `Dubon2026.SourceUnitQuadraticCommutation, `Dubon2026.SourceGammaOnePrimitiveJointFrobeniusWitness, `Dubon2026.SourceGammaOnePrimitiveJointArithmeticData, `Dubon2026.SourceGammaOnePrimitiveJointArithmeticSystem, `Dubon2026.SourceGammaOnePrimitiveJointCommutativity]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
