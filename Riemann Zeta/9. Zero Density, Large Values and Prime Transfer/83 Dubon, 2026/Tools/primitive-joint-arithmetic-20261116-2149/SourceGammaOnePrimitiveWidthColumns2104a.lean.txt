/-! DUB-17/18 -> original all-weight association -> exact joint arithmetic
realization -> primitive boundary Frobenius. The two good-prime cusp
column transforms preserve the actual primitive return time because
their coordinate factors are units at the original level. This supplies
the width equality needed for the exact one and prime multiplicities;
no cancellation by a possibly nonunit cusp width is used. -/

/-- Divisibility by the primitive width is precisely membership of the
literal conjugated translation in the original Gamma1 subgroup. -/
theorem primitiveCuspWidth_dvd_iff_translation (M n : ℕ) (δ : SL(2, ℤ)) :
    primitiveCuspWidth M (δ⁻¹ : SL(2, ℤ) ⧸ Gamma1 M) ∣ n ↔
      δ * ModularGroup.T ^ n * δ⁻¹ ∈ Gamma1 M := by
  change Function.minimalPeriod (ModularGroup.T • ·)
    (δ⁻¹ : SL(2, ℤ) ⧸ Gamma1 M) ∣ n ↔ _
  rw [← MulAction.pow_smul_eq_iff_minimalPeriod_dvd]
  change ((ModularGroup.T ^ n * δ⁻¹ : SL(2, ℤ)) : SL(2, ℤ) ⧸ Gamma1 M) =
    (δ⁻¹ : SL(2, ℤ) ⧸ Gamma1 M) ↔ _
  rw [QuotientGroup.eq, ← inv_mem_iff, mul_inv_rev, inv_inv]
  simp only [mul_assoc]

private theorem translation_mem_iff (M n : ℕ) (δ : SL(2, ℤ)) :
    δ * ModularGroup.T ^ n * δ⁻¹ ∈ Gamma1 M ↔
      (n : ZMod M) * (δ 0 0 : ℤ) * (δ 1 0 : ℤ) = 0 ∧
      (n : ZMod M) * (δ 1 0 : ℤ) * (δ 1 0 : ℤ) = 0 := by
  rw [← zpow_natCast, conj_T_zpow_eq_unipMat, Gamma1_mem]
  simp only [unipMat_apply_00, unipMat_apply_11, unipMat_apply_10]
  push_cast
  constructor
  · rintro ⟨ha, _, hc⟩
    constructor
    · linear_combination -ha
    · exact neg_eq_zero.mp hc
  · rintro ⟨ha, hc⟩
    simp only [ha, hc, sub_zero, add_zero, neg_zero, and_self]

/-- Independent unit rescaling of the actual two cusp coordinates
preserves the primitive Gamma1 width, in particular for both Hecke transforms. -/
theorem primitiveCuspWidth_eq_of_unit_columns (M : ℕ) (δ σ : SL(2, ℤ))
    (u v : (ZMod M)ˣ)
    (h0 : ((σ 0 0 : ℤ) : ZMod M) = (u : ZMod M) * (δ 0 0 : ℤ))
    (h1 : ((σ 1 0 : ℤ) : ZMod M) = (v : ZMod M) * (δ 1 0 : ℤ)) :
    primitiveCuspWidth M (δ⁻¹ : SL(2, ℤ) ⧸ Gamma1 M) =
      primitiveCuspWidth M (σ⁻¹ : SL(2, ℤ) ⧸ Gamma1 M) := by
  have hmem (n : ℕ) : σ * ModularGroup.T ^ n * σ⁻¹ ∈ Gamma1 M ↔
      δ * ModularGroup.T ^ n * δ⁻¹ ∈ Gamma1 M := by
    rw [translation_mem_iff, translation_mem_iff, h0, h1]
    have e0 : (n : ZMod M) * ((u : ZMod M) * (δ 0 0 : ℤ)) *
        ((v : ZMod M) * (δ 1 0 : ℤ)) =
        ((u * v : (ZMod M)ˣ) : ZMod M) *
          ((n : ZMod M) * (δ 0 0 : ℤ) * (δ 1 0 : ℤ)) := by
      rw [Units.val_mul]
      ring
    have e1 : (n : ZMod M) * ((v : ZMod M) * (δ 1 0 : ℤ)) *
        ((v : ZMod M) * (δ 1 0 : ℤ)) =
        ((v * v : (ZMod M)ˣ) : ZMod M) *
          ((n : ZMod M) * (δ 1 0 : ℤ) * (δ 1 0 : ℤ)) := by
      rw [Units.val_mul]
      ring
    rw [e0, e1, Units.mul_right_eq_zero, Units.mul_right_eq_zero]
  apply Nat.dvd_antisymm
  · apply (primitiveCuspWidth_dvd_iff_translation M _ δ).mpr
    exact (hmem _).mp ((primitiveCuspWidth_dvd_iff_translation M _ σ).mp dvd_rfl)
  · apply (primitiveCuspWidth_dvd_iff_translation M _ σ).mpr
    exact (hmem _).mpr ((primitiveCuspWidth_dvd_iff_translation M _ δ).mp dvd_rfl)

end
end Dubon2026.SourceFunctionField.GammaOne
