variable (Q N w : ℕ) [NeZero Q] [NeZero N]
local notation "M" => Q * N ^ 2
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "C" => Rep.trivial (ZMod N) (Gamma1 M) V

/-- The genuine finite primitive boundary operators for all original
prime and normalizer generators. -/
def binaryPrimitiveBoundaryHeckeDiamondOperators : S → Module.End (ZMod N) (P M → V) :=
  Sum.elim (binaryPrimitiveColumnHecke Q N w) (binaryPrimitiveColumnDiamond Q N w)

/-- The actual primitive boundary intertwines every original
Hecke/diamond word, using the two separately proved native comparisons. -/
theorem binaryPrimitiveBoundaryHeckeDiamond_word (a : FreeAlgebra ℤ S) (x : H1 C) :
    binaryPrimitiveColumnBoundary Q N w
      (FreeAlgebra.lift ℤ (sourceGammaOneHeckeDiamondOperators Q N w) a x) =
      FreeAlgebra.lift ℤ (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w) a
        (binaryPrimitiveColumnBoundary Q N w x) := by
  apply sourceHeckeWord_intertwine_addHom _ _
    (binaryPrimitiveColumnBoundary Q N w).toAddMonoidHom
  intro s y
  cases s with
  | inl ell => exact binaryPrimitiveColumnBoundary_hecke Q N w ell y
  | inr g => exact binaryPrimitiveColumnBoundary_diamond Q N w g y

/-- A word vanishing on the explicit finite boundary sends every
original cohomology class into the independently defined parabolic H1. -/
theorem binaryPrimitiveBoundaryHeckeDiamond_word_zero_parabolic
    (h2 : IsUnit (2 : ZMod N)) (a : FreeAlgebra ℤ S)
    (ha : FreeAlgebra.lift ℤ (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w) a = 0)
    (x : H1 C) :
    FreeAlgebra.lift ℤ (sourceGammaOneHeckeDiamondOperators Q N w) a x ∈
      binaryParabolicH1 Q N w := by
  apply (binaryPrimitiveColumnBoundary_eq_zero_iff Q N w h2 _).mp
  rw [binaryPrimitiveBoundaryHeckeDiamond_word, ha, LinearMap.zero_apply]

end
