omit [NeZero Q] [NeZero N] in
/-- The first finite-unit action commutes with each actual prime
boundary operator, using the proved two-term decomposition. -/
theorem binaryPrimitiveColumnFirst_hecke_commute (u : (ZMod M)ˣ)
    (ell : sourceBinaryHeckePrime (Q * N)) :
    Commute (binaryPrimitiveColumnFirst Q N w u) (binaryPrimitiveColumnHecke Q N w ell) := by
  rw [binaryPrimitiveColumnHecke_eq_sum]
  exact ((Commute.all u _).map (binaryPrimitiveColumnFirst Q N w)).add_right
    (binaryPrimitiveColumnFirst_second_commute Q N w u _)

omit [NeZero Q] in
/-- The finite-unit boundary action commutes with the full original
diamond, including its homogeneous coefficient endomorphism. -/
theorem binaryPrimitiveColumnFirst_diamond_commute (u : (ZMod M)ˣ) (g : Gamma0 M) :
    Commute (binaryPrimitiveColumnFirst Q N w u) (binaryPrimitiveColumnDiamond Q N w g) := by
  have hc : Commute (binaryPrimitiveCoefficientUnits Q N w u)
      (sourceGammaOneBinaryDiamondCoefficient Q N w g) := by
    rw [binaryPrimitiveCoefficientUnits_eq, sourceGammaOneBinaryDiamondCoefficient_eq_diagonal]
    have hd := (Commute.all
      ![(binaryPrimitiveUnitReduction Q N u : ZMod N), 1]
      ![(g.val 1 1 : ZMod N), (g.val 0 0 : ZMod N)]).map
        (Matrix.diagonalRingHom (Fin 2) (ZMod N))
    exact hd.map (homogeneousMatrixRepresentation w)
  apply LinearMap.ext
  intro f
  funext c
  change binaryPrimitiveCoefficientUnits Q N w u
      (sourceGammaOneBinaryDiamondCoefficient Q N w g
        (f (scale (gamma0Unit M g)⁻¹ (gamma0Unit M g) (scale 1 u c)))) =
    sourceGammaOneBinaryDiamondCoefficient Q N w g
      (binaryPrimitiveCoefficientUnits Q N w u
        (f (scale 1 u (scale (gamma0Unit M g)⁻¹ (gamma0Unit M g) c))))
  rw [scale_comm]
  exact LinearMap.congr_fun hc.eq _

/-- The actual primitive boundary Galois action commutes with every
original prime/diamond word on the same finite coefficient module. -/
theorem binaryPrimitiveBoundaryGaloisRep_word
    (σ : Qbar ≃ₐ[ℚ] Qbar)
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N) ⊕ Gamma0 M)) (f : F) :
    binaryPrimitiveBoundaryGaloisRep Q N w σ
        (FreeAlgebra.lift ℤ (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w) a f) =
      FreeAlgebra.lift ℤ (binaryPrimitiveBoundaryHeckeDiamondOperators Q N w) a
        (binaryPrimitiveBoundaryGaloisRep Q N w σ f) := by
  apply sourceHeckeWord_intertwine_addHom _ _
    (binaryPrimitiveBoundaryGaloisRep Q N w σ).toAddMonoidHom
  intro s y
  cases s with
  | inl ell =>
    exact LinearMap.congr_fun
      (binaryPrimitiveColumnFirst_hecke_commute Q N w
        (sourceAbsoluteCyclotomicCharacter M σ) ell).eq y
  | inr g =>
    exact LinearMap.congr_fun
      (binaryPrimitiveColumnFirst_diamond_commute Q N w
        (sourceAbsoluteCyclotomicCharacter M σ) g).eq y

end
