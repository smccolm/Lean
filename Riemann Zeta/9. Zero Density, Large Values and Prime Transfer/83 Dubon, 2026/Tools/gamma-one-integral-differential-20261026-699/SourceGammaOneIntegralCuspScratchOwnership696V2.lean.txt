private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceWeierstrassIntegralDenominator, `SourceBoundedIntegralSeriesProduct, `SourceIntegralEisensteinSix, `SourceBoundedIntegralSeriesConstant, `SourceFrickeNumeratorIntegralBound, `SourceBoundedRationalSeriesImage, `SourceLaurentSpanDescent, `SourceFrickeLaurentSpanDescent, `SourceRationalCuspWeightLaurent, `SourceRationalCuspWeightIntegrality, `SourceIntegralRationalLaurentSeries, `SourceBoundedIntegralLaurentAlgebra, `SourceBoundedRationalLaurentImage, `SourceCyclotomicLaurentTrace, `SourceBoundedRationalLaurentIntegral, `SourceBoundedCyclotomicLaurentFraction, `SourceBoundedCyclotomicLaurentGenerators, `SourceBoundedFrickeLaurentField, `SourceBoundedRationalJAlgebra, `SourceBoundedRationalCuspImage, `SourceIntegralLaurentWeightFactors, `SourceRationalCuspWeightCancellation, `SourceGammaOneRationalBoundedDenominators]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
