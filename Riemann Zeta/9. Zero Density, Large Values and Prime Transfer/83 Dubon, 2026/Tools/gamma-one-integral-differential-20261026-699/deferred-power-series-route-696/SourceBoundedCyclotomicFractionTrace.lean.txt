/-! DUB-17/18 -> actual Gamma1 differential realization -> original
integral-ratio field -> rational bounded-series fractions. Normalize a
nonzero denominator coefficient, then trace numerator and denominator.
The denominator trace has a coefficient equal to the positive extension
degree. Both traces keep one uniform integral bound, and their quotient
identity is the original rational-series identity. -/
namespace Dubon2026.SourceFricke
noncomputable section
open SourceFunctionField

theorem hasBoundedIntegralCoefficients_C_kN (N : ℕ) [NeZero N] (c : kN N) :
    HasBoundedIntegralCoefficients (PowerSeries.C c) := by
  letI := finiteDimensional_kN N
  letI : NumberField (kN N) := {}
  obtain ⟨d, hd, hi⟩ := numberField_exists_positive_integral_multiple (fun _ : Unit => c)
  apply hasBoundedIntegralCoefficients_C_of_multiple c d (ne_of_gt hd)
  simpa only [zsmul_eq_mul] using hi ()

theorem exists_rational_bounded_series_fraction (N : ℕ) [NeZero N]
    (y : PowerSeries ℚ) (q r : PowerSeries (kN N)) (hq0 : q ≠ 0)
    (hq : HasBoundedIntegralCoefficients q) (hr : HasBoundedIntegralCoefficients r)
    (hyq : y.map (algebraMap ℚ (kN N)) * q = r) :
    ∃ h u : PowerSeries ℚ, h ≠ 0 ∧ HasBoundedIntegralCoefficients h ∧
      HasBoundedIntegralCoefficients u ∧ h * y = u := by
  classical
  letI := finiteDimensional_kN N
  obtain ⟨n, hn⟩ : ∃ n : ℕ, q.coeff n ≠ 0 := by
    by_contra! hh
    apply hq0
    apply PowerSeries.ext
    intro n
    simpa only [map_zero] using hh n
  let x := (q.coeff n)⁻¹
  have hx := hasBoundedIntegralCoefficients_C_kN N x
  refine ⟨cyclotomicTraceSeries N (PowerSeries.C x * q),
    cyclotomicTraceSeries N (PowerSeries.C x * r), ?_,
    hasBoundedIntegralCoefficients_cyclotomicTrace (hx.mul hq),
    hasBoundedIntegralCoefficients_cyclotomicTrace (hx.mul hr), ?_⟩
  · intro hz
    have he := congrArg (fun f : PowerSeries ℚ => f.coeff n) hz
    simp only [cyclotomicTraceSeries_coeff, map_zero] at he
    rw [PowerSeries.coeff_C_mul, show x = (q.coeff n)⁻¹ from rfl, inv_mul_cancel₀ hn] at he
    have ht : Algebra.trace ℚ (kN N) (1 : kN N) = (Module.finrank ℚ (kN N) : ℚ) := by
      simpa only [map_one, nsmul_eq_mul, mul_one] using
        Algebra.trace_algebraMap (R := ℚ) (S := kN N) (1 : ℚ)
    rw [ht] at he
    exact (Nat.cast_ne_zero.mpr (ne_of_gt (Module.finrank_pos : 0 < Module.finrank ℚ (kN N)))) he
  · have hrel : y.map (algebraMap ℚ (kN N)) * (PowerSeries.C x * q) = PowerSeries.C x * r := by
      rw [← hyq]
      ring
    have he := cyclotomicTraceSeries_rational_mul N y (PowerSeries.C x * q)
    rw [hrel] at he
    simpa only [mul_comm] using he.symm

end
end Dubon2026.SourceFricke
