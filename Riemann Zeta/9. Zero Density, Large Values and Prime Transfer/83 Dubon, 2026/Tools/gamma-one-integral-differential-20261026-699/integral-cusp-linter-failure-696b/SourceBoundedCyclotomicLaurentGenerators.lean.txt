/-! DUB-17/18 -> actual Gamma1 differential realization -> original
integral-ratio field -> bounded Laurent fractions. The literal j series is
an integer Laurent image. The original Fricke numerator has the proved
uniform coefficient bound, and its original discriminant denominator is
an integer Laurent image and nonzero. These are the exact generators used
by the source cyclotomic function field. -/
namespace Dubon2026.SourceFricke
noncomputable section
open SourceFunctionField

theorem hasBoundedIntegralLaurentCoefficients_jqNModC
    (K : Type*) [Field K] [CharZero K] (N : ℕ) [NeZero N] :
    HasBoundedIntegralLaurentCoefficients (jqNModC K N) := by
  have he : coeffMap (Int.castRingHom K) (jqNModC ℤ N) = jqNModC K N := by
    rw [jqNModC, coeffMap_qExpand, jqNModC]
    congr 1
    exact map_jqModC (Int.castRingHom K)
  rw [← he]
  exact hasBoundedIntegralLaurentCoefficients_integer_map _

theorem hasBoundedIntegralLaurentCoefficients_discriminantLaurentK (N : ℕ) [NeZero N] :
    HasBoundedIntegralLaurentCoefficients (discriminantLaurentK N) := by
  have h := hasBoundedIntegralLaurentCoefficients_integer_map (K := kN N)
    (qExpand ℤ N ((PowerSeries.X * dedekindEtaUnit : PowerSeries ℤ) : LaurentSeries ℤ))
  simpa only [discriminantLaurentK, coeffMap_qExpand, coeffMap_ofPowerSeries] using h

theorem discriminantLaurentK_ne_zero (N : ℕ) [NeZero N] : discriminantLaurentK N ≠ 0 := by
  have hη : dedekindEtaUnit ≠ 0 := by
    intro hz
    have he := constantCoeff_dedekindEtaUnit
    rw [hz, map_zero] at he
    exact zero_ne_one he
  intro hz
  apply mul_ne_zero PowerSeries.X_ne_zero hη
  apply PowerSeries.map_injective (Int.castRingHom (kN N)) Int.cast_injective
  apply (HahnSeries.ofPowerSeries_injective (Γ := ℤ) (R := kN N))
  apply (qExpand (kN N) N).injective
  simpa only [discriminantLaurentK, map_zero] using hz

theorem exists_bounded_frickeLaurentK_fraction (N : ℕ) [NeZero N] (i : FrickeIdx N) :
    ∃ q r : LaurentSeries (kN N), q ≠ 0 ∧ HasBoundedIntegralLaurentCoefficients q ∧
      HasBoundedIntegralLaurentCoefficients r ∧ frickeLaurentK N i * q = r := by
  refine ⟨discriminantLaurentK N,
    (PowerSeries.C (-1 / 2592 : kN N) *
      frickeHQSeriesK (frickeIdx_hyps i.1 i.2).1 (i.1 1).val : PowerSeries (kN N)),
    discriminantLaurentK_ne_zero N,
    hasBoundedIntegralLaurentCoefficients_discriminantLaurentK N,
    (hasBoundedIntegralCoefficients_frickeNumerator i).laurent, ?_⟩
  rw [frickeLaurentK, div_mul_cancel₀ _ (discriminantLaurentK_ne_zero N),
    map_mul, HahnSeries.ofPowerSeries_C]
  simp only [map_div₀, map_neg, map_one, map_ofNat]

end
end Dubon2026.SourceFricke
