/-! DUB-17/18 -> actual Gamma1 differential realization -> integral cusp
generators. The existing analytic j-derivative formula supplies a nonzero
complex scalar. Its constant coefficient is an integer, because the literal
j, discriminant and Euler derivative have integer coefficients and the
E4-squared E6 factor has constant coefficient one. This descends the identity
to integers without computing an unnecessary exact normalization scalar. -/
namespace Dubon2026.SourceFricke
noncomputable section
open UpperHalfPlane SourceFunctionField SourceDifferential ModularForm
open scoped ModularForm

theorem qExpansion_E6_eq_map_eisenstein6 :
    qExpansion 1 (E₆ : ℍ → ℂ) = eisenstein6.map (Int.castRingHom ℂ) := by
  rw [← eisensteinQSeries_map (by norm_num : 3 ≤ 6) (by decide : Even 6),
    eisensteinQSeries_six_eq_integer_map]
  ext n
  simp only [PowerSeries.coeff_map, Int.coe_castRingHom, map_intCast]

theorem exists_integer_qEuler_jqModC_formula :
    ∃ c : ℤ, c ≠ 0 ∧
      qEuler ℤ (jqModC ℤ) * ((PowerSeries.X * dedekindEtaUnit : PowerSeries ℤ) : LaurentSeries ℤ) =
        HahnSeries.C c * ((eisenstein4 : LaurentSeries ℤ) ^ 2 * (eisenstein6 : LaurentSeries ℤ)) := by
  let ι : LaurentSeries ℤ →+* LaurentSeries ℂ := coeffMap (Int.castRingHom ℂ)
  let D : LaurentSeries ℤ := ((PowerSeries.X * dedekindEtaUnit : PowerSeries ℤ) : LaurentSeries ℤ)
  let E : LaurentSeries ℤ := (eisenstein4 : LaurentSeries ℤ) ^ 2 * (eisenstein6 : LaurentSeries ℤ)
  let G : LaurentSeries ℤ := qEuler ℤ (jqModC ℤ) * D
  have hι : Function.Injective ι := by
    intro a b he
    apply HahnSeries.ext
    funext n
    apply Int.cast_injective (α := ℂ)
    exact congrArg (fun s : LaurentSeries ℂ => s.coeff n) he
  have hD : ι D = (qExpansion 1 ModularForm.discriminant : LaurentSeries ℂ) := by
    rw [qExpansion_discriminant_eq_map_X_mul_dedekindEtaUnit]
    exact coeffMap_ofPowerSeries _ _
  have hE : ι E = (qExpansion 1 (E₄ : ℍ → ℂ) : LaurentSeries ℂ) ^ 2 *
      (qExpansion 1 (E₆ : ℍ → ℂ) : LaurentSeries ℂ) := by
    rw [show ι E = ι (eisenstein4 : LaurentSeries ℤ) ^ 2 * ι (eisenstein6 : LaurentSeries ℤ)
      from by simp only [E, map_mul, map_pow]]
    rw [qExpansion_E4_eq_map_eisenstein4, qExpansion_E6_eq_map_eisenstein6]
    rfl
  have hQ : ι (qEuler ℤ (jqModC ℤ)) = qEuler ℂ (jqModC ℂ) := by
    rw [← map_jqModC (Int.castRingHom ℂ)]
    apply HahnSeries.ext
    funext n
    simp only [ι, coeffMap_coeff, qEuler_coeff, Int.coe_castRingHom, Int.cast_mul]
  obtain ⟨c, hc, he⟩ := ModularQDerivative.exists_qEuler_jqModC_formula
  have hG : ι G = algebraMap ℂ (LaurentSeries ℂ) c * ι E := by
    rw [G, map_mul, hQ, hD, hE]
    exact he
  have hE0 : E.coeff 0 = 1 := by
    change (((eisenstein4 ^ 2 * eisenstein6 : PowerSeries ℤ) : LaurentSeries ℤ)).coeff 0 = 1
    rw [show (0 : ℤ) = ((0 : ℕ) : ℤ) from rfl, HahnSeries.ofPowerSeries_apply_coeff,
      PowerSeries.coeff_zero_eq_constantCoeff, map_mul, map_pow,
      constantCoeff_eisenstein4, constantCoeff_eisenstein6, one_pow, one_mul]
  have hgc : (G.coeff 0 : ℂ) = c := by
    have h := congrArg (fun s : LaurentSeries ℂ => s.coeff 0) hG
    dsimp only at h
    rw [algebraMap_laurentSeries_apply_alt, HahnSeries.coeff_single_zero_mul] at h
    simpa only [ι, coeffMap_coeff, Int.coe_castRingHom, hE0, Int.cast_one, mul_one] using h
  refine ⟨G.coeff 0, ?_, ?_⟩
  · intro hz
    apply hc
    rw [← hgc, hz, Int.cast_zero]
  · apply hι
    change ι G = ι (HahnSeries.C (G.coeff 0) * E)
    rw [hG, map_mul]
    congr 1
    rw [algebraMap_laurentSeries_apply_alt, ← hgc]
    exact (coeffMap_single (Int.castRingHom ℂ) 0 (G.coeff 0)).symm

end
end Dubon2026.SourceFricke
