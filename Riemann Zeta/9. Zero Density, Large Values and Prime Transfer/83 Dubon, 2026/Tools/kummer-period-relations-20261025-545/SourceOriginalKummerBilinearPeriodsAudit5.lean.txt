private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialWeightedCycleSums, `SourceOriginalWeilCoverCycleResidueBilinear, `SourceOriginalWeilCoverBilinearPeriodPolygonData, `SourceOriginalWeilCoverBilinearPeriodPolygon, `SourceOriginalWeilCoverBilinearCycleData]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.sum_smul_eq_sum_cycle, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.sum_mul_jump_eq_cycle_bilinear, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_loop_bilinear_eq_neg_edge_jumps, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_cycle_bilinear_eq_neg_residue, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_riemann_cycle_bilinear, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_bilinear_period_polygon_of_primitives, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_bilinear_cycle_data]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.cycleScalarIntegral, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverBilinearCyclePeriodPolygonConclusion, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverBilinearCycleData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Bilinear Periods postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Bilinear Periods theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER BILINEAR PERIODS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
