private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverNormalizedAbelReadings, `SourceFiniteSkewBilinearSwap, `SourceOriginalWeilCoverCellResidueReciprocity, `SourceOriginalWeilCoverThirdKindPolygonData, `SourceOriginalWeilCoverThirdKindPolygon, `SourceOriginalWeilCoverThirdKindCycleData]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_abel_vector_eq_reading_add_periods, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_weighted_readings_eq_abel_add_periods, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.finite_skew_bilinear_swap, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_residue_reciprocity_of_weighted_cells, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_third_kind_reciprocity_of_polygon, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_third_kind_cycle_data]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverThirdKindCyclePeriodPolygonConclusion, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverThirdKindCycleData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Third Kind postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Third Kind theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER THIRD KIND AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
