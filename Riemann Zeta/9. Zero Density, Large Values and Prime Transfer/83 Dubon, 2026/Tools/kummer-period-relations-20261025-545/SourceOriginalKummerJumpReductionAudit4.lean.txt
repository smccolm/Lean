private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialArcIndexInverse, `SourceRadialPrimitiveJumpCirculation, `SourceRadialJumpCycleReduction, `SourceOriginalWeilCoverReducedCellJumps]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialCellDissection.arcOf_side, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.jump_kirchhoff, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.vector_decomposition_of_scalar_lift, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.jumps_kirchhoff, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.closure_jumps_le, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_reduced_cell_jumps]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverReducedCellJumps]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Jump Reduction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Jump Reduction theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER JUMP REDUCTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
