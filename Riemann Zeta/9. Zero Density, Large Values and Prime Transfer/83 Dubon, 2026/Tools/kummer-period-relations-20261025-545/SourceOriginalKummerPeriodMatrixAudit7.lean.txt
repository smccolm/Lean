private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialCycleIntegralSums, `SourceRadialWordPeriodMatrix, `SourceRadialJumpPeriodClosure, `SourceRadialCyclePeriodPolygonData, `SourceRadialCyclePeriodMatrixAssembly, `SourceOriginalWeilCoverPeriodMatrix, `SourceOriginalWeilCoverCyclePeriodData]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.integer_decomposition_of_lift, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.sum_chain_smul, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.wordMatrix_of_not_mem, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.jump_formula_of_word, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.closure_generated_jumps_le_periods, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.period_matrix_of_primitive_word, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_period_matrix_of_primitive_word, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_cycle_period_data]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.zv, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.wordMatrix, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.JumpFormula, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.CyclePeriodPolygonConclusion, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverCyclePeriodData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Period Matrix postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Period Matrix theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER PERIOD MATRIX AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
