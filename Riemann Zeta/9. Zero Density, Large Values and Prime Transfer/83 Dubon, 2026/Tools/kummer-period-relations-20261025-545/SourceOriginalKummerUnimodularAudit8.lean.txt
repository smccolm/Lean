private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialComplementaryWordMatrix, `SourcePeriodGeneratorsRealSpan, `SourceOriginalWeilCoverComplementaryPeriodCoordinates, `SourceOriginalWeilCoverComplementaryPeriodSpan, `SourceOriginalWeilCoverComplementaryWordMatrix, `SourceRadialUnimodularPeriodPolygon, `SourceOriginalWeilCoverUnimodularPeriodPolygon, `SourceOriginalWeilCoverUnimodularCycleData]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.complementaryWordMatrix_transpose, `Dubon2026.SourceFunctionField.PeriodRealDual.integer_coordinates_of_finset_generators, `Dubon2026.SourceFunctionField.PeriodRealDual.span_top_of_integer_coordinates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_complementary_period_coordinates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_complementary_periods_span_real, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_complementaryWordMatrix_isUnit, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_unimodular_period_polygon_of_basis, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_unimodular_cycle_data]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.complementaryWordMatrix, `Dubon2026.SourceFunctionField.CellGeometry.RadialDissectionPeriods.UnimodularCyclePeriodPolygonConclusion, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverUnimodularCycleData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Unimodular postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Unimodular theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER UNIMODULAR AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
