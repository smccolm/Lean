private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCurveConvergentDiscGerms, `SourceCurveConvergentDiscOrders, `SourceCurveConvergentDiscUniformizer, `SourceCurveConvergentDiscPlaces, `SourceCurveConvergentPlaceFamily, `SourceOriginalWeilCoverPlaceFamily]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.AnalyticDisc.realize_const, `Dubon2026.SourceCurve.AnalyticDisc.realize_add, `Dubon2026.SourceCurve.AnalyticDisc.realize_mul, `Dubon2026.SourceCurve.AnalyticDisc.realize_sub_const, `Dubon2026.SourceCurve.AnalyticDisc.order_ne_top, `Dubon2026.SourceCurve.AnalyticDisc.order_zero, `Dubon2026.SourceCurve.AnalyticDisc.order_eq_top_iff, `Dubon2026.SourceCurve.AnalyticDisc.order_const, `Dubon2026.SourceCurve.AnalyticDisc.order_mul, `Dubon2026.SourceCurve.AnalyticDisc.order_add, `Dubon2026.SourceCurve.AnalyticDisc.normal_germ, `Dubon2026.SourceCurve.AnalyticDisc.normal_form, `Dubon2026.SourceCurve.AnalyticDisc.normal_uniformizer, `Dubon2026.SourceCurve.AnalyticDisc.order_sub_uniformizer_one, `Dubon2026.SourceCurve.AnalyticDisc.exists_normalized_place, `Dubon2026.SourceCurve.AnalyticDisc.exists_place_family, `Dubon2026.SourceCurve.AnalyticDisc.exists_place_family_of_common_disc, `Dubon2026.SourceFunctionField.OriginalChart.exists_originalWeil_cover_place_family, `Dubon2026.SourceFunctionField.OriginalChart.coverFamilyIntegral]
  let constructions : Array Name := #[`Dubon2026.SourceCurve.AnalyticDisc.normal]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Place Family postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Place Family theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER PLACE FAMILY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
