private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCurveRationalChartValues, `SourceCurveAnalyticChartOrders, `SourceCurveAnalyticChartComparison, `SourceCurveAnalyticChartDifferenceOrders, `SourceCurveAnalyticChartBranchEquality, `SourceOriginalWeilCoverChartPackage, `SourceOriginalWeilCoverCharts, `SourceOriginalWeilCoverChartTransitions]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.Place.rationalEval_zero, `Dubon2026.SourceCurve.Place.rationalEval_inv, `Dubon2026.SourceCurve.Place.rationalEval_decomp, `Dubon2026.SourceCurve.AnalyticChart.Orders.analyticAt, `Dubon2026.SourceCurve.AnalyticChart.Orders.eventually_mem, `Dubon2026.SourceCurve.AnalyticChart.Orders.meromorphicOrderAt_sub_const, `Dubon2026.SourceCurve.AnalyticChart.Orders.uniformizer_sub_order, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.sub_ne_zero, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.uniformizer_mem, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.transition_self, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.analyticAt_transition, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.tendsto_transition, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.order_transition, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.eventually_mem', `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.eventually_mem, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.analyticAt_difference, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.decomp, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.difference_step, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.le_difference_order, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.difference_order_top_of_mem, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.eventuallyEq_of_mem, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.eventuallyEq, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.difference_order_top, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.deriv_transition_ne_zero, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.eventually_good, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.meromorphicAt_difference, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.ord_eq_of_difference_order_top, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.eventually_place_eq, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_parameter_package, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.place_spec, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.place_eval_uniformizer, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeChart_source, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeChart_target, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.mem_placeChart_source, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeChart_trans_source, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isOpen_placeChart_trans_source, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analyticOnNhd_placeChart_transition]
  let constructions : Array Name := #[`Dubon2026.SourceCurve.AnalyticChart.Orders, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.transition, `Dubon2026.SourceCurve.AnalyticChart.ComparisonData.difference, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeRadius, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeFamily, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeChart]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Compatibility postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Compatibility theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER COMPATIBILITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
