private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceLipschitzTwo, `SourceLipschitzTwoRealBoundary, `SourceWeierstrassRowSeries, `SourceWeierstrassLatticeRows, `SourceWeierstrassTorsionExpansion, `SourceWeierstrassTorsionTail, `SourceWeierstrassTorsionHolomorphy, `SourceFrickeHolomorphy]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.summable_one_div_add_int_sq, `Dubon2026.SourceFricke.tsum_one_div_add_int_sq_neg, `Dubon2026.SourceFricke.tsum_one_div_sub_int_sq, `Dubon2026.SourceFricke.lipschitz_two_of_im_pos, `Dubon2026.SourceFricke.lipschitz_two_of_im_neg, `Dubon2026.SourceFricke.summable_nat_mul_cexp_pow, `Dubon2026.SourceFricke.cexp_two_pi_I_mul_ne_one, `Dubon2026.SourceFricke.lipschitz_two_closed_of_im_pos, `Dubon2026.SourceFricke.lipschitz_two_closed_of_im_neg, `Dubon2026.SourceFricke.lipschitz_two_closed, `Dubon2026.SourceFricke.summable_one_div_sub_int_sq, `Dubon2026.SourceFricke.tsum_int_one_div_sq, `Dubon2026.SourceFricke.two_pi_I_sq, `Dubon2026.SourceFricke.neg_pi_sq_div_three_div, `Dubon2026.SourceFricke.tsum_row_split, `Dubon2026.SourceFricke.tsum_row_zero, `Dubon2026.SourceFricke.lipschitz_two_of_im_neg', `Dubon2026.SourceFricke.cexp_two_pi_I_mul_add, `Dubon2026.SourceFricke.cexp_two_pi_I_mul_nat_mul, `Dubon2026.SourceFricke.cexp_two_pi_I_mul_neg, `Dubon2026.SourceFricke.lipschitz_row_pair, `Dubon2026.SourceFricke.lipschitz_row_zero, `Dubon2026.SourceFricke.lipschitz_rows_qExpansion, `Dubon2026.SourceFricke.hasSum_weierstrassP_prod, `Dubon2026.SourceFricke.summable_weierstrassP_prod, `Dubon2026.SourceFricke.weierstrassP_eq_tsum_tsum, `Dubon2026.SourceFricke.summable_weierstrassP_row, `Dubon2026.SourceFricke.wpRow_periodPairOfTau_split, `Dubon2026.SourceFricke.weierstrassP_periodPairOfTau_eq_row_sum, `Dubon2026.SourceFricke.weierstrassP_periodPairOfTau_qExpansion, `Dubon2026.SourceFricke.zetaN_ne_zero, `Dubon2026.SourceFricke.qN_ne_zero, `Dubon2026.SourceFricke.qN_pow_self, `Dubon2026.SourceFricke.cexp_two_pi_I_mul_torsionPt, `Dubon2026.SourceFricke.norm_qN, `Dubon2026.SourceFricke.torsionPt_hyps, `Dubon2026.SourceFricke.weierstrassP_torsionPt_qExpansion, `Dubon2026.SourceFricke.norm_zetaN, `Dubon2026.SourceFricke.norm_qN_le_one, `Dubon2026.SourceFricke.norm_qN_lt_one, `Dubon2026.SourceFricke.norm_qN_le_of_le_im, `Dubon2026.SourceFricke.norm_wpTail_le, `Dubon2026.SourceFricke.summable_wpTail_majorant, `Dubon2026.SourceFricke.summable_wpTail, `Dubon2026.SourceFricke.weierstrassP_torsionPt_eq_tsum_prod, `Dubon2026.SourceFricke.wpTorsion_eq_wpTorsionSeries, `Dubon2026.SourceFricke.one_sub_zetaN_pow_mul_qN_pow_ne_zero, `Dubon2026.SourceFricke.differentiable_qN, `Dubon2026.SourceFricke.differentiable_wpTail, `Dubon2026.SourceFricke.differentiableOn_wpTorsionSeries, `Dubon2026.SourceFricke.mdifferentiable_wpTorsion, `Dubon2026.SourceFricke.frickeIdx_hyps, `Dubon2026.SourceFricke.mdifferentiable_wpNormZ, `Dubon2026.SourceFricke.mdifferentiable_frickeF]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.wpSummand, `Dubon2026.SourceFricke.zetaN, `Dubon2026.SourceFricke.qN, `Dubon2026.SourceFricke.wpTail, `Dubon2026.SourceFricke.wpTorsion, `Dubon2026.SourceFricke.wpTorsionSeries]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE WEIERSTRASS EXPANSION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
