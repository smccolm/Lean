open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceLaurentIntegralDescent, `Dubon2026.SourceCurveIntegralExpressionOrders, `Dubon2026.SourceCurveIntegralDifferentialOrders, `Dubon2026.SourceGammaOneDifferentialIntegralDescent, `Dubon2026.SourceCurveRegularIntegralCriterion, `Dubon2026.SourceGammaOneDifferentialRegularity, `Dubon2026.SourceFrickePeriodHomothetyV2, `Dubon2026.SourceFrickeModularLatticeV2, `Dubon2026.SourceFrickeTorsionTransformationV2, `Dubon2026.SourceFrickeFunctionTransformationV2, `Dubon2026.SourceLipschitzTwo, `Dubon2026.SourceLipschitzTwoRealBoundary, `Dubon2026.SourceWeierstrassRowSeries, `Dubon2026.SourceWeierstrassLatticeRows, `Dubon2026.SourceWeierstrassTorsionExpansion, `Dubon2026.SourceWeierstrassTorsionTail, `Dubon2026.SourceWeierstrassTorsionHolomorphy, `Dubon2026.SourceFrickeHolomorphy, `Dubon2026.SourceFrickeCyclotomicField, `Dubon2026.SourceWeierstrassNormalizedTorsion, `Dubon2026.SourceWeierstrassTorsionCuspBound, `Dubon2026.SourceWeierstrassTorsionFourierCoefficients, `Dubon2026.SourceWeierstrassTorsionFourierSum, `Dubon2026.SourceWeierstrassTorsionFourierExpansion, `Dubon2026.SourceFrickeCuspInputs]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
