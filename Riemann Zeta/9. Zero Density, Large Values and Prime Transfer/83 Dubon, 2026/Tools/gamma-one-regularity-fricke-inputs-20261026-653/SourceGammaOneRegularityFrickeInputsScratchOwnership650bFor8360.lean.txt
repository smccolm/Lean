import SourceLipschitzTwo
import SourceLipschitzTwoRealBoundary
import SourceWeierstrassRowSeries
import SourceWeierstrassLatticeRows
import SourceWeierstrassTorsionExpansion
import SourceWeierstrassTorsionTail
import SourceWeierstrassTorsionHolomorphy
import SourceFrickeHolomorphy
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceLipschitzTwo, `SourceLipschitzTwoRealBoundary, `SourceWeierstrassRowSeries, `SourceWeierstrassLatticeRows, `SourceWeierstrassTorsionExpansion, `SourceWeierstrassTorsionTail, `SourceWeierstrassTorsionHolomorphy, `SourceFrickeHolomorphy]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
