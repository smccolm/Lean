import SourceFrickeCyclotomicField
import SourceWeierstrassNormalizedTorsion
import SourceWeierstrassTorsionCuspBound
import SourceWeierstrassTorsionFourierCoefficients
import SourceWeierstrassTorsionFourierSum
import SourceWeierstrassTorsionFourierExpansion
import SourceFrickeCuspInputs
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceFrickeCyclotomicField, `SourceWeierstrassNormalizedTorsion, `SourceWeierstrassTorsionCuspBound, `SourceWeierstrassTorsionFourierCoefficients, `SourceWeierstrassTorsionFourierSum, `SourceWeierstrassTorsionFourierExpansion, `SourceFrickeCuspInputs]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
