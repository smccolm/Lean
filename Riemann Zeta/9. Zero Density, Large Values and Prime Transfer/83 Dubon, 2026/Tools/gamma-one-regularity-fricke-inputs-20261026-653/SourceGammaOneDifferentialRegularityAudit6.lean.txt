private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceLaurentIntegralDescent, `SourceCurveIntegralExpressionOrders, `SourceCurveIntegralDifferentialOrders, `SourceGammaOneDifferentialIntegralDescent, `SourceCurveRegularIntegralCriterion, `SourceGammaOneDifferentialRegularity]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.isIntegral_adjoin_of_isIntegral_coeffMap, `Dubon2026.SourceCurve.Place.ord_natPow, `Dubon2026.SourceCurve.Place.ord_nonneg_of_integral_coordinate, `Dubon2026.SourceCurve.Place.cusp_coefficient_order_of_integral_expression, `Dubon2026.SourceCurve.Place.ordDifferential_nonneg_of_finite_expression, `Dubon2026.SourceFunctionField.GammaOne.coeffMap_qEuler_natural, `Dubon2026.SourceFunctionField.GammaOne.integral_subfield_of_integral_coeffMap, `Dubon2026.SourceFunctionField.GammaOne.differential_integral_expressions_of_matching_expansion, `Dubon2026.SourceCurve.regular_smul_D_of_two_integral_expressions, `Dubon2026.SourceFunctionField.GammaOne.regular_of_diffQExp_matches_cusp, `Dubon2026.SourceFunctionField.GammaOne.gamma1_regular_of_diffQExp_matches_cusp]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIFFERENTIAL REGULARITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
