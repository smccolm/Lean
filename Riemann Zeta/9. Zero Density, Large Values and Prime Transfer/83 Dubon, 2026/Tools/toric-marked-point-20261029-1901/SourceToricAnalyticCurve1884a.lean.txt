import Dubon2026.SourceEllipticComplexUniformization
import Dubon2026.SourceEllipticPeriodModularCurve
import Dubon2026.SourceGammaOneDivisionForms

/-! DUB-17/18 -> original Gamma1 genus preservation -> finite elliptic
fibers -> exact-order toric point. Reuse the native complex period-model
homomorphism instead of developing a general nonarchimedean Tate
uniformization. This module identifies the required analytic Tate equation
by an explicit variable change; the formal-series specialization is still
required. -/
namespace Dubon2026.SourceFunctionField.ToricAnalytic
noncomputable section
open WeierstrassCurve UpperHalfPlane ModularForm SourceEllipticPeriod
open SourceEllipticVariableChange

/-- The fixed nonzero complex normalization of the torsion Weierstrass function. -/
def lambda : ℂ := 2 * (Real.pi : ℂ) * Complex.I

theorem lambda_ne_zero : lambda ≠ 0 := by
  exact mul_ne_zero (mul_ne_zero (by norm_num) (by exact_mod_cast Real.pi_ne_zero))
    Complex.I_ne_zero

/-- The literal change from the analytic Tate equation to the short period equation. -/
def tateChange : VariableChange ℂ :=
  ⟨Units.mk0 lambda lambda_ne_zero, lambda ^ 2 / 12, lambda / 2, 0⟩

/-- The complex Tate equation obtained from the actual period curve by the fixed change. -/
def tateCurve (τ : ℍ) : WeierstrassCurve ℂ :=
  tateChange • periodCurve (PeriodLattice.ofTau τ)

theorem tateCurve_a₁ (τ : ℍ) : (tateCurve τ).a₁ = 1 := by
  simp only [tateCurve, variableChange_a₁, periodCurve, tateChange,
    Units.val_inv_eq_inv_val, Units.val_mk0]
  field_simp [lambda_ne_zero]

theorem tateCurve_a₂ (τ : ℍ) : (tateCurve τ).a₂ = 0 := by
  simp only [tateCurve, variableChange_a₂, periodCurve, tateChange,
    Units.val_inv_eq_inv_val, Units.val_mk0]
  ring

theorem tateCurve_a₃ (τ : ℍ) : (tateCurve τ).a₃ = 0 := by
  simp [tateCurve, variableChange_a₃, periodCurve, tateChange]

theorem tateCurve_a₄ (τ : ℍ) : (tateCurve τ).a₄ = (1 - E₄ τ) / 48 := by
  simp only [tateCurve, variableChange_a₄, periodCurve, tateChange,
    Units.val_inv_eq_inv_val, Units.val_mk0, gTwo_ofTau]
  have hπ : (Real.pi : ℂ) ≠ 0 := by exact_mod_cast Real.pi_ne_zero
  dsimp [lambda]
  field_simp
  simp only [Complex.I_sq]
  ring

theorem tateCurve_a₆ (τ : ℍ) :
    (tateCurve τ).a₆ = (2 * E₆ τ + 1 - 3 * E₄ τ) / 1728 := by
  simp only [tateCurve, variableChange_a₆, periodCurve, tateChange,
    Units.val_inv_eq_inv_val, Units.val_mk0, gTwo_ofTau, gThree_ofTau]
  have hπ : (Real.pi : ℂ) ≠ 0 := by exact_mod_cast Real.pi_ne_zero
  dsimp [lambda]
  field_simp
  simp only [Complex.I_sq]
  ring

end
end Dubon2026.SourceFunctionField.ToricAnalytic
