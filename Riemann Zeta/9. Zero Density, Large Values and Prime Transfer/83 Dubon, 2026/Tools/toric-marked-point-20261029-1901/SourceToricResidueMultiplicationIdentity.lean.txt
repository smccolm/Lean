/-! DUB-17/18 -> original Gamma1 genus preservation -> finite elliptic

noncomputable section
open WeierstrassCurve Polynomial UpperHalfPlane SourceFricke GammaOne.ToricFourier
local notation "Zbar" => integralClosure ℤ ℂ

theorem cuspTateCurve_mul_eval (N : ℕ) [NeZero N] (c : ZMod N) (hc : c ≠ 0)
    (m : ℕ) (hmc : (m : ZMod N) * c ≠ 0) :
    cuspToricX N ((m : ZMod N) * c) hmc *
      (cuspTateCurve.ΨSq m).eval (cuspToricX N c hc) =
        (cuspTateCurve.Φ m).eval (cuspToricX N c hc) := by
  apply Subtype.ext
  funext τ
  have h := tateCurve_divisionForm_mul_eval τ N c hc m hmc
  rw [← cuspTateCurve_evaluate τ, ← cuspToricX_evaluate τ N c hc,
    ← cuspToricX_evaluate τ N _ hmc,
    WeierstrassCurve.map_ΨSq, WeierstrassCurve.map_Φ,
    Polynomial.eval_map_apply, Polynomial.eval_map_apply] at h
  exact h

theorem tatePowerSeries_complex_mul_eval (N : ℕ) [NeZero N] (c : ZMod N) (hc : c ≠ 0)
    (m : ℕ) (hmc : (m : ZMod N) * c ≠ 0) :
    xS (zetaN N ^ ((m : ZMod N) * c).val) *
      ((tatePowerSeries.map (PowerSeries.map (Int.castRingHom ℂ))).ΨSq m).eval
        (xS (zetaN N ^ c.val)) =
      ((tatePowerSeries.map (PowerSeries.map (Int.castRingHom ℂ))).Φ m).eval
        (xS (zetaN N ^ c.val)) := by
  have h := congrArg cuspExpansionHom (cuspTateCurve_mul_eval N c hc m hmc)
  rw [map_mul, ← Polynomial.eval_map_apply, ← Polynomial.eval_map_apply,
    ← WeierstrassCurve.map_ΨSq, ← WeierstrassCurve.map_Φ,
    cuspExpansionTateCurve, cuspExpansionToricX N c hc,
    cuspExpansionToricX N _ hmc] at h
  exact h

theorem localizedTateCurve_mul_eval (N : ℕ) [NeZero N] (c : ZMod N) (hc : c ≠ 0)
    (m : ℕ) (hmc : (m : ZMod N) * c ≠ 0) :
    localizedToricX N ((m : ZMod N) * c) *
      ((localizedTateCurve N).ΨSq m).eval (localizedToricX N c) =
        ((localizedTateCurve N).Φ m).eval (localizedToricX N c) := by
  apply PowerSeries.map_injective (coefficientComplex N) (coefficientComplex_injective N)
  rw [map_mul, ← Polynomial.eval_map_apply, ← Polynomial.eval_map_apply,
    ← WeierstrassCurve.map_ΨSq, ← WeierstrassCurve.map_Φ,
    map_localizedTateCurve,
    map_localizedToricX N c hc (coefficientComplex N)
      (isPrimitiveRoot_zetaN (NeZero.ne N)) (coefficientComplex_algebraMap N (integralZeta N)),
    map_localizedToricX N _ hmc (coefficientComplex N)
      (isPrimitiveRoot_zetaN (NeZero.ne N)) (coefficientComplex_algebraMap N (integralZeta N))]
  exact tatePowerSeries_complex_mul_eval N c hc m hmc

theorem tatePowerSeries_residue_mul_eval (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (K : Type*) [Field K] [IsAlgClosed K] [CharP K ℓ]
    {ζ : K} (hζ : IsPrimitiveRoot ζ N) (c : ZMod N) (hc : c ≠ 0)
    (m : ℕ) (hmc : (m : ZMod N) * c ≠ 0) :
    xS (ζ ^ ((m : ZMod N) * c).val) *
      ((tatePowerSeries.map (PowerSeries.map (Int.castRingHom K))).ΨSq m).eval
        (xS (ζ ^ c.val)) =
      ((tatePowerSeries.map (PowerSeries.map (Int.castRingHom K))).Φ m).eval
        (xS (ζ ^ c.val)) := by
  obtain ⟨φ, hφ⟩ := IntegralSlash.exists_algebraicIntegerReduction_primitiveRoot
    N ℓ hℓN K ζ hζ
  have hroot : φ (integralZeta N) = ζ := hφ (integralZeta N) rfl
  let f := coefficientResidue N φ (GammaOne.ToricFourier.natCast_ne_zero hζ)
  have hf : f (algebraMap Zbar (ToricCoefficientRing N) (integralZeta N)) = ζ :=
    (coefficientResidue_algebraMap N φ _ _).trans hroot
  have h := congrArg (PowerSeries.map f) (localizedTateCurve_mul_eval N c hc m hmc)
  rw [map_mul, ← Polynomial.eval_map_apply, ← Polynomial.eval_map_apply,
    ← WeierstrassCurve.map_ΨSq, ← WeierstrassCurve.map_Φ,
    map_localizedTateCurve, map_localizedToricX N c hc f hζ hf,
    map_localizedToricX N _ hmc f hζ hf] at h
  exact h

end
