/-! DUB-17/18 -> original Gamma1 genus preservation -> finite elliptic
fibers -> original-field toric invariants. Identify the actual Tate
invariants with the repository's original integer E4 and E6 series.
The c6 sign is negative. These identities supply the weight-zero ratios
needed by the marked-point chart, including characteristics two and three. -/
namespace Dubon2026.SourceFunctionField
noncomputable section
open PowerSeries WeierstrassCurve

theorem c4_tatePowerSeries : tatePowerSeries.c₄ = eisenstein4 := by
  have he : eisenstein4 = 1 - 48 * tateA4 := by
    apply PowerSeries.map_injective (Int.castRingHom ℚ) (Int.cast_injective (R := ℚ))
    simpa only [map_sub, map_one, map_mul, map_ofNat] using
      GammaOne.ToricFourier.eisenstein4_map_eq ℚ
  rw [he]
  simp only [WeierstrassCurve.c₄, WeierstrassCurve.b₂, WeierstrassCurve.b₄,
    tatePowerSeries_a₁, tatePowerSeries_a₂, tatePowerSeries_a₃, tatePowerSeries_a₄]
  ring

theorem c6_tatePowerSeries : tatePowerSeries.c₆ = -SourceFricke.eisenstein6 := by
  have he : tatePowerSeries.c₆ =
      -1 + PowerSeries.C (72 : ℤ) * tateA4 - PowerSeries.C (864 : ℤ) * tateA6 := by
    simp only [WeierstrassCurve.c₆, WeierstrassCurve.b₂, WeierstrassCurve.b₄,
      WeierstrassCurve.b₆, tatePowerSeries_a₁, tatePowerSeries_a₂, tatePowerSeries_a₃,
      tatePowerSeries_a₄, tatePowerSeries_a₆, map_ofNat]
    ring
  rw [he]
  ext n
  simp only [map_sub, map_add, map_neg, PowerSeries.coeff_one, PowerSeries.coeff_C_mul,
    coeff_tateA4, coeff_tateA6, SourceFricke.eisenstein6, PowerSeries.coeff_mk]
  by_cases hn : n = 0
  · subst n
    simp
  · simp only [hn, if_false, neg_zero, zero_add, ArithmeticFunction.sigma_apply,
      Nat.cast_sum, Nat.cast_pow, ← Finset.mul_sum]
    have hs := Finset.sum_congr rfl (fun d (_ : d ∈ n.divisors) => twelve_mul_tateB d)
    simp only [Finset.sum_add_distrib, ← Finset.mul_sum] at hs
    linear_combination 72 * hs

theorem c4_tateLaurent (K : Type*) [CommRing K] :
    (tateLaurent K).c₄ = laurentOfInt K eisenstein4 := by
  rw [tateLaurent, WeierstrassCurve.map_c₄, c4_tatePowerSeries]

theorem c6_tateLaurent (K : Type*) [CommRing K] :
    (tateLaurent K).c₆ = -laurentOfInt K SourceFricke.eisenstein6 := by
  rw [tateLaurent, WeierstrassCurve.map_c₆, c6_tatePowerSeries, map_neg]

end
end Dubon2026.SourceFunctionField
