import SourceTateIntegralCoefficients
import SourceTateIntegralCurve
import SourceTateEisensteinInvariants
import SourceGammaOneToricOriginalRatios
import SourceGammaOneToricOriginalCoordinates
import SourceGammaOneToricOriginalCoordinateValues
import SourceIntegralDiamondScalarRatioAction
import SourceGammaOneToricRatioDiamond
import SourceGammaOneToricCoordinateDiamond
import SourceGammaOneOriginalToricCoordinates
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceTateIntegralCoefficients, `SourceTateIntegralCurve, `SourceTateEisensteinInvariants, `SourceGammaOneToricOriginalRatios, `SourceGammaOneToricOriginalCoordinates, `SourceGammaOneToricOriginalCoordinateValues, `SourceIntegralDiamondScalarRatioAction, `SourceGammaOneToricRatioDiamond, `SourceGammaOneToricCoordinateDiamond, `SourceGammaOneOriginalToricCoordinates]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
