/-! DUB-17/18 -> original Gamma1 genus preservation -> finite elliptic
fibers -> marked-point chart. Connect the actual exact-order point and its
unit multiples to the two functions already constructed inside the original
Gamma1 field. Their coordinate values are proved after the natural Laurent
field embedding; the original-field diamond law remains the previously
proved actual action, not a new action certificate. -/
namespace Dubon2026.SourceFunctionField.ToricAnalytic
noncomputable section
open WeierstrassCurve HahnSeries GammaOne.ToricFourier SourceEllipticExceptional
open GammaOne GammaOne.ToricCoordinates

theorem exists_originalField_markedToricCoordinates
    (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime] (hℓN : ¬ ℓ ∣ N) (hN : 2 ≤ N)
    (K L : Type*) [Field K] [IsAlgClosed K] [CharP K ℓ] [Field L] [IsAlgClosed L]
    [DecidableEq L] [Algebra (LaurentSeries K) L] {ζ : K} (hζ : IsPrimitiveRoot ζ N) :
    ∃ P : ((tateLaurent K).baseChange L).toAffine.Point,
      addOrderOf P = N ∧ ∀ t : (ZMod N)ˣ,
        ∃ (x y : L) (h : ((tateLaurent K).baseChange L).toAffine.Nonsingular x y),
          (t : ZMod N).val • P = Affine.Point.some x y h ∧
          algebraMap (LaurentSeries K) L ((w₁ N K ζ t : qExpFunctionFieldC K
            (CongruenceSubgroup.Gamma1 N)) : LaurentSeries K) =
              ((tateLaurent K).baseChange L).c₄ * (1 + 12 * x) /
                ((tateLaurent K).baseChange L).c₆ ∧
          algebraMap (LaurentSeries K) L ((w₂ N K ζ t : qExpFunctionFieldC K
            (CongruenceSubgroup.Gamma1 N)) : LaurentSeries K) =
              (x + 6 * x ^ 2 + 2 * ((tateLaurent K).baseChange L).a₄) /
                ((tateLaurent K).baseChange L).c₄ := by
  obtain ⟨P, hord, hcoord⟩ := exists_exactOrder_toric_point N ℓ hℓN hN K L hζ
  refine ⟨P, hord, fun t => ?_⟩
  cases hp : (t : ZMod N).val • P with
  | zero =>
      have he := hcoord t
      rw [hp] at he
      cases he
  | some x y h =>
      have hx : x = seriesToField K L (xS (ζ ^ (t : ZMod N).val)) := by
        have he := hcoord t
        rw [hp, xco_some] at he
        exact Option.some.inj he
      refine ⟨x, y, h, rfl, ?_, ?_⟩
      · rw [hx]
        have he := congrArg (algebraMap (LaurentSeries K) L) (coe_w₁ N hζ hN t)
        simpa only [map_div, map_mul, map_add, map_one, map_ofNat,
          WeierstrassCurve.baseChange, WeierstrassCurve.map_c₄, WeierstrassCurve.map_c₆,
          seriesToField, RingHom.comp_apply] using he
      · rw [hx]
        have he := congrArg (algebraMap (LaurentSeries K) L) (coe_w₂ N hζ hN t)
        simpa only [map_div, map_mul, map_add, map_pow, map_ofNat,
          WeierstrassCurve.baseChange, WeierstrassCurve.map_c₄, WeierstrassCurve.map_a₄,
          seriesToField, RingHom.comp_apply] using he

end
end Dubon2026.SourceFunctionField.ToricAnalytic
