theorem reducedFieldAction_range_finite (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) : Finite (reducedFieldAction N ℓ hℓN).range := by
  letI : (plusMinus (Gamma1 N)).FiniteIndex := Subgroup.finiteIndex_of_le (le_plusMinus _)
  letI : (reducedFieldAction N ℓ hℓN).ker.FiniteIndex := by
    rw [reducedFieldAction_kernel]
    infer_instance
  apply Nat.finite_of_card_ne_zero
  rw [← Subgroup.index_ker]
  exact Subgroup.FiniteIndex.index_ne_zero

theorem reduced_fixedField_degree_eq_signed_relativeIndex
    (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime] (hℓN : ¬ ℓ ∣ N) :
    Module.finrank (IntermediateField.fixedField (reducedFieldAction N ℓ hℓN).range)
      (qExpFunctionFieldC (ZMod ℓ) (Gamma1 N)) =
        (plusMinus (Gamma1 N)).relIndex (Gamma0 N) := by
  letI := reducedFieldAction_range_finite N ℓ hℓN
  letI := Fintype.ofFinite (reducedFieldAction N ℓ hℓN).range
  rw [FixedPoints.finrank_eq_card, ← Nat.card_eq_fintype_card, ← Subgroup.index_ker,
    reducedFieldAction_kernel]
  rfl

end
