theorem signed_index_le_residue_ratio_degree (hℓN : ¬ ℓ ∣ N) :
    (plusMinus (Gamma1 N)).index ≤
      relfinrank kk⟮jqModC kk⟯ (qExpFunctionFieldC kk (Gamma1 N)) := by
  have hT : ModularGroup.T ∈ Gamma1 N := by simp [Gamma1_mem, ModularGroup.T]
  obtain ⟨hp, hle⟩ := ReducedDegreeTransport.primeField_degree_pos_and_le_residue A (Gamma1 N) ℓ hT
  exact (signed_index_le_reduced_degree_of_pos N ℓ hℓN hp).trans hle

    Module.finrank L⟮jCoordinateOver (Gamma1 N) L⟯ (functionFieldOver (Gamma1 N) L) =
        Module.finrank kk⟮residueJCoordinate A (Gamma1 N)⟯ (residueImage A (Gamma1 N)) ∧
      residueImage A (Gamma1 N) = qExpFunctionFieldC kk (Gamma1 N) :=
  gammaOne_degree_and_residueField_of_integral_ratio_index A N
    (signed_index_le_residue_ratio_degree A N ℓ hℓN)

end
