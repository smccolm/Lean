import SourceIntegralDiamondFieldComposition

/-! DUB-17/18 -> original Gamma1 degree/genus -> reduced field action.
An actually invariant even form has its original Fourier reduction after
the diamond action. Native even-weight sign invariance gives the same
result for the signed Gamma1 subgroup. This supplies the identity map
and the required signed-kernel inclusion of the field action. -/
namespace Dubon2026.SourceFunctionField.IntegralDiamond
noncomputable section
open UpperHalfPlane CongruenceSubgroup GammaOne
open scoped MatrixGroups ModularForm

theorem reducedDiamond_of_invariant (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) {k : ℤ} (hk : Even k) (δ : Gamma0 N)
    (f : ModularForm (Gamma1 N) k) {p : PowerSeries ℤ} (hp : IsIntegralQExp f p)
    (hi : (⇑f : ℍ → ℂ) ∣[k] δ.val = f) :
    reducedDiamond N ℓ hk δ f hp = red ℓ p := by
  have hw : IsIntegralQExp (scaledDiamond N δ 0 f) p := by
    rw [coe_scaledDiamond, pow_zero, one_smul, hi]
    exact hp
  rw [reducedDiamond_eq N ℓ hℓN hk δ f hp hw, normalizedRed, pow_zero, div_one]

theorem even_slash_eq_of_signed_gammaOne (N : ℕ) {k : ℤ} (hk : Even k)
    (f : ModularForm (Gamma1 N) k) (δ : Gamma0 N)
    (hδ : δ.val ∈ plusMinus (Gamma1 N)) : (⇑f : ℍ → ℂ) ∣[k] δ.val = f := by
  rcases (mem_plusMinus_iff _ _).mp hδ with h | h
  · rw [ModularForm.SL_slash]
    exact SlashInvariantFormClass.slash_action_eq f _ (Subgroup.mem_map_of_mem _ h)
  · rw [← IntegralTrace.slash_neg_even hk, ModularForm.SL_slash]
    exact SlashInvariantFormClass.slash_action_eq f _ (Subgroup.mem_map_of_mem _ h)

theorem reducedFieldPull_of_signed_gammaOne (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (δ : Gamma0 N) (hδ : δ.val ∈ plusMinus (Gamma1 N))
    (x : qExpFunctionFieldC (ZMod ℓ) (Gamma1 N)) :
    reducedFieldPull N ℓ δ x = (x : LaurentSeries (ZMod ℓ)) := by
  obtain ⟨k, hk, f, g, pf, pg, hf, hg, hg0, hx⟩ := exists_even_originalReducedField_ratio x
  rw [reducedFieldPull_eq N ℓ hℓN δ x hk f g hf hg hg0 hx,
    reducedDiamond_of_invariant N ℓ hℓN hk δ f hf (even_slash_eq_of_signed_gammaOne N hk f δ hδ),
    reducedDiamond_of_invariant N ℓ hℓN hk δ g hg (even_slash_eq_of_signed_gammaOne N hk g δ hδ), hx]

theorem reducedFieldPullHom_one_matrix (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (x : qExpFunctionFieldC (ZMod ℓ) (Gamma1 N)) :
    reducedFieldPullHom N ℓ hℓN 1 x = x := by
  apply Subtype.ext
  exact reducedFieldPull_of_signed_gammaOne N ℓ hℓN 1 (one_mem _) x

end
end Dubon2026.SourceFunctionField.IntegralDiamond
