/-! DUB-17/18 -> original Gamma1 degree/genus -> reduced field action.
The actual weight-zero constant form is slash invariant. Its clearing
exponent is zero, proving preservation of one without a new witness
assumption. Together with exact zero preservation this supplies the
constant laws required for the original-field homomorphism. -/
namespace Dubon2026.SourceFunctionField.IntegralDiamond
noncomputable section
open UpperHalfPlane CongruenceSubgroup GammaOne
open scoped MatrixGroups ModularForm

theorem reducedDiamond_one (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (δ : Gamma0 N) :
    reducedDiamond N ℓ ((show Even (0 : ℤ) from ⟨0, rfl⟩) : Even (0 : ℤ)) δ 1 (integralQExp_one N) = 1 := by
  have hw : IsIntegralQExp (scaledDiamond N δ 0 (1 : ModularForm (Gamma1 N) 0)) 1 := by
    rw [coe_scaledDiamond, pow_zero, one_smul, ModularForm.one_coe_eq_one,
      ModularForm.is_invariant_one]
    simpa only [ModularForm.one_coe_eq_one] using integralQExp_one N
  rw [reducedDiamond_eq N ℓ hℓN (show Even (0 : ℤ) from ⟨0, rfl⟩) δ 1 (integralQExp_one N) hw,
    normalizedRed, map_one, pow_zero, div_one]

theorem reducedFieldPull_one (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (δ : Gamma0 N) : reducedFieldPull N ℓ δ 1 = 1 := by
  rw [reducedFieldPull_eq N ℓ hℓN δ 1 (show Even (0 : ℤ) from ⟨0, rfl⟩) 1 1 (integralQExp_one N)
    (integralQExp_one N) (by rw [map_one]; exact one_ne_zero)
    (by rw [map_one, div_one]; rfl), reducedDiamond_one N ℓ hℓN, div_self]
  exact one_ne_zero

theorem reducedFieldPull_zero (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (δ : Gamma0 N) : reducedFieldPull N ℓ δ 0 = 0 := by
  rw [reducedFieldPull_eq N ℓ hℓN δ 0 (show Even (0 : ℤ) from ⟨0, rfl⟩) 0 1 (integralQExp_zero N 0)
    (integralQExp_one N) (by rw [map_one]; exact one_ne_zero)
    (by rw [map_zero, zero_div]; rfl), reducedDiamond_zero N ℓ hℓN, zero_div]

end
end Dubon2026.SourceFunctionField.IntegralDiamond
