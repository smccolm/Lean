/-! DUB-17/18 -> original Gamma1 degree/genus -> reduced field action.
The image remains in the original field: the two actual integral diamond
forms have the same weight, and removing their clearing powers changes
their ratio only by a scalar in the original constant field. Native field
closure avoids matching the two clearing exponents with new witnesses. -/
namespace Dubon2026.SourceFunctionField.IntegralDiamond
noncomputable section
open UpperHalfPlane CongruenceSubgroup GammaOne
open scoped MatrixGroups ModularForm

theorem reducedDiamond_ratio_mem (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) {k : ℤ} (hk : Even k) (δ : Gamma0 N)
    (f g : ModularForm (Gamma1 N) k) {pf pg : PowerSeries ℤ}
    (hf : IsIntegralQExp f pf) (hg : IsIntegralQExp g pg) (hg0 : red ℓ pg ≠ 0) :
    reducedDiamond N ℓ hk δ f hf / reducedDiamond N ℓ hk δ g hg ∈
      qExpFunctionFieldC (ZMod ℓ) (Gamma1 N) := by
  let u := diamondWitness N hk δ f hf
  let v := diamondWitness N hk δ g hg
  have hv : red ℓ v.val.2 ≠ 0 := by
    intro hz
    apply hg0
    apply (normalizedRed_zero_iff_original N ℓ hℓN hk δ g hg v.property).mp
    exact (normalizedRed_eq_zero_iff N ℓ hℓN v.val.1 v.val.2).mpr hz
  have hratio : red ℓ u.val.2 / red ℓ v.val.2 ∈
      qExpFunctionFieldC (ZMod ℓ) (Gamma1 N) := by
    simpa only [red_apply] using div_mem_qExpFunctionFieldC
      (scaledDiamond N δ u.val.1 f) (scaledDiamond N δ v.val.1 g)
      u.property v.property (by simpa only [red_apply] using hv)
  rw [reducedDiamond_eq N ℓ hℓN hk δ f hf u.property,
    reducedDiamond_eq N ℓ hℓN hk δ g hg v.property,
    normalizedRed, normalizedRed, div_div_div_comm]
  exact div_mem hratio (div_mem (pow_mem (natCast_mem _ N) _)
    (pow_mem (natCast_mem _ N) _))

theorem reducedFieldPull_mem (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    (hℓN : ¬ ℓ ∣ N) (δ : Gamma0 N)
    (x : qExpFunctionFieldC (ZMod ℓ) (Gamma1 N)) :
    reducedFieldPull N ℓ δ x ∈ qExpFunctionFieldC (ZMod ℓ) (Gamma1 N) := by
  let v := originalEvenRepresentative N ℓ x
  exact reducedDiamond_ratio_mem N ℓ hℓN v.evenWeight δ v.numerator v.denominator
    v.numeratorIntegral v.denominatorIntegral v.denominatorNonzero

end
end Dubon2026.SourceFunctionField.IntegralDiamond
