private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralDiamondFullBaseInclusion1692d, `SourceIntegralDiamondGammaZeroFixed1695a, `SourceIntegralDiamondEisensteinAction1696b, `SourceIntegralDiamondSignedMembership1697a, `SourceIntegralDiamondExactKernel1698a, `SourceIntegralDiamondFixedFieldDegree1699b]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.IntegralDiamond.divisor_j_mem_gammaZeroField, `Dubon2026.SourceFunctionField.IntegralDiamond.fullBase_le_gammaZeroField, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldPull_of_gammaZero_ratio, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldEquiv_fixes_gammaZeroField, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldAction_fixes_fullBase, `Dubon2026.SourceFunctionField.IntegralDiamond.fullBase_le_originalReducedField, `Dubon2026.SourceFunctionField.IntegralDiamond.red_eisenstein4_ne_zero, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldPull_eisenstein_ratio, `Dubon2026.SourceFunctionField.IntegralDiamond.gammaZero_diagonal_product, `Dubon2026.SourceFunctionField.IntegralDiamond.signed_gammaOne_of_upperLeft, `Dubon2026.SourceFunctionField.IntegralDiamond.gammaZero_upperLeft_ne_zero, `Dubon2026.SourceFunctionField.IntegralDiamond.signed_gammaOne_small_level, `Dubon2026.SourceFunctionField.IntegralDiamond.signed_gammaOne_of_reducedFieldEquiv_eq_one, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldAction_eq_one_iff, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldAction_kernel, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldAction_range_finite, `Dubon2026.SourceFunctionField.IntegralDiamond.reduced_fixedField_degree_eq_signed_relativeIndex]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Reduced diamond action postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Reduced diamond action theorems discovered"
  logInfo m!"SOURCE REDUCED DIAMOND KERNEL PREFLIGHT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
