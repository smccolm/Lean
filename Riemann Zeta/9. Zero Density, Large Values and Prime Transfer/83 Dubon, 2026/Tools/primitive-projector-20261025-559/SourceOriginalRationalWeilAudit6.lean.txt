private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFractionBilinearInjective, `SourceTateRationalInjection, `SourceOriginalRationalWeilNondegeneracy, `SourceOriginalTateWeilSkew, `SourceBilinearBaseChangeLaws, `SourceOriginalRationalWeilLaws]
  let theorems : Array Name := #[`Dubon2026.sourceFraction_bilinear_baseChange_injective, `Dubon2026.TateModule.rational_one_tmul_injective, `Dubon2026.SourceFunctionField.originalRationalWeilBilinear_injective, `Dubon2026.SourceFunctionField.originalTateWeilPairing_skew, `Dubon2026.SourceFunctionField.originalTateWeilBilinear_self, `Dubon2026.sourceBilinear_baseChange_skew, `Dubon2026.sourceBilinear_baseChange_adjoint, `Dubon2026.sourceBilinear_baseChange_multiplier, `Dubon2026.SourceFunctionField.originalRationalWeilBilinear_self, `Dubon2026.SourceFunctionField.originalRationalWeilBilinear_hecke, `Dubon2026.SourceFunctionField.originalRationalWeilBilinear_cyclotomic]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.originalRationalWeilBilinear]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Rational Weil postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Rational Weil theorems discovered"
  logInfo m!"SOURCE ORIGINAL RATIONAL WEIL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
