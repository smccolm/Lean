private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFiniteOperatorCoefficientAnnihilator, `SourcePrimitiveGoodCoefficientAnnihilator, `SourceMappedKernelProjection, `SourcePrimitiveGoodCoefficientProjection, `SourcePrimitiveGoodProjectionHecke]
  let theorems : Array Name := #[`Dubon2026.finite_operator_coefficient_annihilator, `Dubon2026.primitiveGoodAnnihilator_coefficient_polynomial, `Dubon2026.exists_mapped_kernel_projector_polynomial, `Dubon2026.exists_primitive_good_coefficient_projection, `Dubon2026.primitive_projection_hecke_scalar, `Dubon2026.exists_primitive_good_coefficient_hecke_projection]
  let constructions : Array Name := #[`Dubon2026.primitiveGoodCoefficientProjector]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Coefficient Projection postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Coefficient Projection theorems discovered"
  logInfo m!"SOURCE ORIGINAL PRIMITIVE COEFFICIENT PROJECTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
