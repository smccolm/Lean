private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegerCoefficientExtension, `SourceIntegerCoefficientExtensionInjective, `SourceOriginalHeckeCoefficientFaithfulness, `SourceIntegerCoefficientExtensionLaws, `SourcePrimitiveTensorOperators, `SourcePrimitiveTensorProjection, `SourcePrimitiveTensorEigencharacter, `SourcePrimitiveTensorProjectionNonzero]
  let theorems : Array Name := #[`Dubon2026.integerCoefficientExtension_tmul, `Dubon2026.integerCoefficientExtension_injective, `Dubon2026.originalHecke_coefficient_extension_injective, `Dubon2026.integerCoefficientExtension_algebraMap, `Dubon2026.integerCoefficientExtension_smul, `Dubon2026.integerCoefficientExtension_aeval, `Dubon2026.integerCoefficientExtension_shifted_squares, `Dubon2026.integerCoefficientExtension_mul_eq_zero, `Dubon2026.integerCoefficientExtension_idempotent, `Dubon2026.integerCoefficientExtension_mul_eq_smul, `Dubon2026.primitiveCoefficientCuspEvaluation_injective, `Dubon2026.primitiveCoefficientCuspEvaluation_hecke, `Dubon2026.primitiveCoefficientCuspEvaluation_annihilator, `Dubon2026.primitiveCoefficientCuspEvaluation_projector, `Dubon2026.exists_primitive_tensor_projection, `Dubon2026.primitiveTensorEigencharacter_hecke, `Dubon2026.primitiveTensorEigencharacter_annihilator, `Dubon2026.primitiveTensorEigencharacter_projector, `Dubon2026.primitiveGoodTensorProjector_ne_zero, `Dubon2026.exists_nonzero_primitive_tensor_projection]
  let constructions : Array Name := #[`Dubon2026.integerCoefficientExtension, `Dubon2026.primitiveIntegralHeckeIndex, `Dubon2026.primitiveCoefficientCuspEvaluation, `Dubon2026.primitiveGoodTensorAnnihilator, `Dubon2026.primitiveGoodTensorProjector, `Dubon2026.primitiveTensorEigencharacter]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tensor Projection postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tensor Projection theorems discovered"
  logInfo m!"SOURCE ORIGINAL PRIMITIVE TENSOR PROJECTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
