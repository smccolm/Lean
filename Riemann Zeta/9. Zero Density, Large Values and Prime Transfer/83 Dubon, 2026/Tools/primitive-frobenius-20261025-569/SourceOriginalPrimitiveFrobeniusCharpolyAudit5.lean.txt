private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceValuationFrobeniusRoots, `SourceValuationCyclotomicFrobenius, `SourceRankTwoQuadraticTrace, `SourceCyclotomicFrobeniusCharpoly, `SourceOriginalPrimitiveFrobeniusCharpolyRealization]
  let theorems : Array Name := #[`Dubon2026.valuation_root_mem, `Dubon2026.valuation_frobenius_root, `Dubon2026.valuation_cyclotomic_frobenius, `Dubon2026.rank_two_cayley_hamilton, `Dubon2026.rank_two_trace_of_determinant_quadratic, `Dubon2026.rank_two_charpoly_of_determinant_quadratic, `Dubon2026.cyclotomic_frobenius_charpoly, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_continuous_unramified_frobenius_charpoly_realization]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Frobenius Charpoly postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Frobenius Charpoly theorems discovered"
  logInfo m!"SOURCE ORIGINAL PRIMITIVE FROBENIUS CHARPOLY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
