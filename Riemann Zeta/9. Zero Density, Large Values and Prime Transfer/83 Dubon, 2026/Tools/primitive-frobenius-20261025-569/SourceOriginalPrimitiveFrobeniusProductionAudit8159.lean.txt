  let theorems : Array Name := #[`Dubon2026.spanning_liftBaseChange_surjective, `Dubon2026.projector_range_specialization_surjective, `Dubon2026.projector_range_specialization_bijective, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitiveTate_prescribed_specialization_of_basis, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_prescribed_continuous_galois_realization, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_prescribed_residual_frobenius_realization]
  let constructions : Array Name := #[`Dubon2026.projectorRangeSpecializationEquiv, `Dubon2026.imageSpecializationPairing]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue functional postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue functional theorems discovered"
  logInfo m!"SOURCE ORIGINAL PRIMITIVE FROBENIUS AUDIT PASS: {count} theorems including private/generated declarations. The same original primitive representation has cyclotomic determinant, exact Frobenius trace/determinant/characteristic polynomial, and a prescribed-embedding residual realization. Source reduction compatibility, arithmetic quotient and norm-local comparisons, automorphy and all-weight continuation remain open; 16/20 gates."
