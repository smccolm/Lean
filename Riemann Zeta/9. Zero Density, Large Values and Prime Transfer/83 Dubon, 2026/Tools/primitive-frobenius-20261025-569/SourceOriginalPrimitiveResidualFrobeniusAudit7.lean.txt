private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMatrixIntertwinerCharpoly, `SourceIntegralMatrixCharpoly, `SourceRankTwoCharpolyCoefficients, `SourceOriginalSpecializationFrobeniusCharpoly, `SourceOriginalResidualFrobeniusCharpoly, `SourceOriginalResidualInertia, `SourceOriginalPrimitiveResidualFrobeniusRealization]
  let theorems : Array Name := #[`Dubon2026.matrix_intertwiner_charpoly, `Dubon2026.matrix_intertwiner_one, `Dubon2026.integral_matrix_charpoly_of_intertwiner, `Dubon2026.residue_matrix_charpoly_of_intertwiner, `Dubon2026.residue_matrix_one_of_intertwiner, `Dubon2026.rank_two_trace_det_of_charpoly, `Dubon2026.SourceFunctionField.OriginalChart.original_specialization_frobenius_charpoly, `Dubon2026.SourceFunctionField.OriginalChart.original_residual_frobenius_charpoly, `Dubon2026.SourceFunctionField.OriginalChart.original_residual_inertia_identity, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_continuous_residual_frobenius_realization]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Residual Frobenius postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Residual Frobenius theorems discovered"
  logInfo m!"SOURCE ORIGINAL PRIMITIVE RESIDUAL FROBENIUS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
