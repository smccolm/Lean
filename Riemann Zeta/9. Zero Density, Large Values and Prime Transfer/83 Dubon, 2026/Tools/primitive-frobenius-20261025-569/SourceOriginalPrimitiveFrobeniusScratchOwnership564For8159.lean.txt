import SourceSpanningProjectorImage
import SourceSpecializationBaseChangeIntertwiner
import SourcePolynomialSpecialization
import SourceOriginalSpecializationProjector
import SourceOriginalSpecializationPrimitiveImage
import SourcePolynomialCommutant
import SourcePrimitiveOriginalTateGalois
import SourceImageSpecializationPairing
import SourceAlternatingRankTwoDeterminant
import SourceImageSpecializationDeterminant
import SourceOriginalSpecializationCyclotomicDeterminant
import SourceOriginalPrimitiveCyclotomicRealization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceSpanningProjectorImage, `SourceSpecializationBaseChangeIntertwiner, `SourcePolynomialSpecialization, `SourceOriginalSpecializationProjector, `SourceOriginalSpecializationPrimitiveImage, `SourcePolynomialCommutant, `SourcePrimitiveOriginalTateGalois, `SourceImageSpecializationPairing, `SourceAlternatingRankTwoDeterminant, `SourceImageSpecializationDeterminant, `SourceOriginalSpecializationCyclotomicDeterminant, `SourceOriginalPrimitiveCyclotomicRealization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
