open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceSpanningProjectorImage, `Dubon2026.SourceSpecializationBaseChangeIntertwiner, `Dubon2026.SourcePolynomialSpecialization, `Dubon2026.SourceOriginalSpecializationProjector, `Dubon2026.SourceOriginalSpecializationPrimitiveImage, `Dubon2026.SourcePolynomialCommutant, `Dubon2026.SourcePrimitiveOriginalTateGalois, `Dubon2026.SourceImageSpecializationPairing, `Dubon2026.SourceAlternatingRankTwoDeterminant, `Dubon2026.SourceImageSpecializationDeterminant, `Dubon2026.SourceOriginalSpecializationCyclotomicDeterminant, `Dubon2026.SourceOriginalPrimitiveCyclotomicRealization, `Dubon2026.SourceValuationFrobeniusRoots, `Dubon2026.SourceValuationCyclotomicFrobenius, `Dubon2026.SourceRankTwoQuadraticTrace, `Dubon2026.SourceCyclotomicFrobeniusCharpoly, `Dubon2026.SourceOriginalPrimitiveFrobeniusCharpolyRealization, `Dubon2026.SourceMatrixIntertwinerCharpoly, `Dubon2026.SourceIntegralMatrixCharpoly, `Dubon2026.SourceRankTwoCharpolyCoefficients, `Dubon2026.SourceOriginalSpecializationFrobeniusCharpoly, `Dubon2026.SourceOriginalResidualFrobeniusCharpoly, `Dubon2026.SourceOriginalResidualInertia, `Dubon2026.SourceOriginalPrimitiveResidualFrobeniusRealization, `Dubon2026.SourceOriginalPrescribedTateSpecialization, `Dubon2026.SourceOriginalPrescribedResidualFrobenius]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
