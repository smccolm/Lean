import SourceValuationFrobeniusRoots
import SourceValuationCyclotomicFrobenius
import SourceRankTwoQuadraticTrace
import SourceCyclotomicFrobeniusCharpoly
import SourceOriginalPrimitiveFrobeniusCharpolyRealization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceValuationFrobeniusRoots, `SourceValuationCyclotomicFrobenius, `SourceRankTwoQuadraticTrace, `SourceCyclotomicFrobeniusCharpoly, `SourceOriginalPrimitiveFrobeniusCharpolyRealization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
