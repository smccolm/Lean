/-! DUB-17/18 -> original Gamma1 global charts -> exact ramification.
Actual native modular forms separate every pair of distinct Gamma1 orbits.
Use the proved integral-closure argument for distinct Gamma0 orbits and
the proved division-form argument inside one Gamma0 orbit. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane CongruenceSubgroup
open scoped MatrixGroups

theorem exists_separating_modular_forms (N : ℕ) [NeZero N] (τ τ' : ℍ)
    (hτ : ∀ γ ∈ Gamma1 N, γ • τ ≠ τ') :
    ∃ (k : ℤ) (g h : ModularForm (Gamma1 N : Subgroup (GL (Fin 2) ℝ)) k),
      g τ * h τ' ≠ g τ' * h τ := by
  classical
  by_cases h0 : ∃ γ : Gamma0 N, (γ : SL(2, ℤ)) • τ = τ'
  · obtain ⟨γ, he⟩ := h0
    exact exists_separating_forms_of_gamma0_orbit N τ τ' hτ γ γ.2 he
  · have hne : ∀ γ : Gamma0 N, (γ : SL(2, ℤ)) • τ ≠ τ' := by
      simpa only [not_exists] using h0
    obtain ⟨k, g, h, hsep⟩ := exists_original_separating_modular_forms N τ τ' hne
    have hle : (Gamma1 N : Subgroup (GL (Fin 2) ℝ)) ≤
        (Gamma0 N : Subgroup (GL (Fin 2) ℝ)) := Subgroup.map_mono (Gamma1_in_Gamma0 N)
    exact ⟨k, ModularNormPolynomial.restrict hle g, ModularNormPolynomial.restrict hle h, hsep⟩

end
end Dubon2026.SourceFunctionField.GammaOne
