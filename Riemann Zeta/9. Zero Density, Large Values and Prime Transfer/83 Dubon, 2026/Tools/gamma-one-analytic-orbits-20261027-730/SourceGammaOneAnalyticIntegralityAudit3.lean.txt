private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneHolomorphicFunctions, `SourceGammaOneQuotientCuspSeries, `SourceGammaOneAnalyticIntegrality]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_invariant, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_realization_of_mem_all_analytic_places, `Dubon2026.SourceFunctionField.GammaOne.quotient_cusp_represents, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_cusp_meromorphic, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_cusp_pole_bound, `Dubon2026.SourceFunctionField.GammaOne.isIntegral_of_mem_all_analytic_places]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Analytic Integrality postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Analytic Integrality theorems discovered"
  logInfo m!"SOURCE GAMMA ONE ANALYTIC INTEGRALITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
