import Dubon2026.SourcePeriodDivisionRotations
import Mathlib.Tactic

open Lean

private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.DivisionRotations.exists_rotation_of_division_power]
  let localModules : Array Name := #[`Dubon2026.SourcePeriodDivisionRotations]
  let required : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.DivisionRotations.scale_lattice_eq_of_invariant_powers, `Dubon2026.SourceFunctionField.GammaOne.DivisionRotations.exists_rotation_of_division_power, `Dubon2026.SourceFunctionField.JOrbitSeparation.g₂_scale, `Dubon2026.SourceFunctionField.JOrbitSeparation.g₃_scale, `Dubon2026.SourceFunctionField.LatticeInvariants.lattice_eq_of_invariants]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE PERIOD DIVISION ROTATIONS DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in lattice-preserving rotations derived from equal division-value powers. Only declaration values are traversed. A complex square root and its root-of-unity equation give unit norm; the existing invariant scaling and lattice uniqueness prove preservation of the actual period lattice. This must still be consumed by actual division-form orbit separation. Original chart/period, Hecke/diamond/Picard/H1 arithmetic comparison, all-weight characteristic-zero association and literal higher symmetric-power continuation remain; 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
