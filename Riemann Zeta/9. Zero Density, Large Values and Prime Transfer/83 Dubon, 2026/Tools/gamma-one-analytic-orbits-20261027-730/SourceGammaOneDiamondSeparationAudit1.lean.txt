import SourceGammaOneDiamondSeparation
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceFunctionField.GammaOne.exists_gamma1_smul_eq_of_cross_products
#print axioms Dubon2026.SourceFunctionField.GammaOne.exists_separating_forms_of_gamma0_orbit

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneDiamondSeparation]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_gamma1_smul_eq_of_cross_products, `Dubon2026.SourceFunctionField.GammaOne.exists_separating_forms_of_gamma0_orbit]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Diamond Separation postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Diamond Separation theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIAMOND SEPARATION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceGammaOneDiamondSeparation
