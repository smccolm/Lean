private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_separating_forms_of_gamma0_orbit]
  let localModules : Array Name := #[`SourceGammaOneDiamondSeparation, `SourceGammaOneDivisionForms, `SourceGammaOneDivisionLatticeOrbit, `SourceGammaOneDivisionPowerOrbit, `SourcePeriodDivisionRotations]

  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue

          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"

Only declaration values are traversed. Original chart/period, Hecke/diamond/Picard/H1 arithmetic comparison, all-weight characteristic-zero association and literal higher symmetric-power continuation remain; 16/20; DUB-17--20 OPEN."
