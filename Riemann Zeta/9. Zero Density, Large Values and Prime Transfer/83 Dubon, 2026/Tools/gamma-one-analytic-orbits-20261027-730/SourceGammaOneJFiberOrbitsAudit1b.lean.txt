private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneJFiberOrbits]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.j_rep, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.card_stabilizer_rep, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.finite_orbitQuot, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.toFiber_injective, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.toFiber_surjective, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.sum_orders]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.OrbitQuot, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.rep, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.Fiber, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.toFiber, `Dubon2026.SourceFunctionField.GammaOne.AnalyticFiberOrbits.equiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One J Fiber Orbits postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One J Fiber Orbits theorems discovered"
  logInfo m!"SOURCE GAMMA ONE J FIBER ORBITS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
