private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceModularSidePairingAction, `SourceModularSidePairingIntegrals, `SourceModularBoundarySidePairing, `SourceModularGreenDomain, `SourceModularGreenLines, `SourceModularGreenSliceGeometry, `SourceModularGreenSlicing, `SourceModularGreenArc, `SourceModularGreenFormula]
  let theorems : Array Name := #[`Dubon2026.SourceModularSidePairing.pairElt_mem, `Dubon2026.SourceModularSidePairing.inv_mul_eq_pairElt, `Dubon2026.SourceModularSidePairing.sum_comp_smul, `Dubon2026.SourceModularSidePairing.slash_tile_pair, `Dubon2026.SourceModularSidePairing.slash_T_apply, `Dubon2026.SourceModularSidePairing.slash_S_apply, `Dubon2026.SourceModularSidePairing.T_smul_coe, `Dubon2026.SourceModularSidePairing.S_smul_coe, `Dubon2026.SourceModularSidePairing.T_smul_leftRay, `Dubon2026.SourceModularSidePairing.tilePull_T_pt, `Dubon2026.SourceModularSidePairing.exp_arc_im_pos, `Dubon2026.SourceModularSidePairing.S_smul_arc, `Dubon2026.SourceModularSidePairing.tilePull_S_pt, `Dubon2026.SourceModularSidePairing.tilePull_eq_div, `Dubon2026.SourceModularSidePairing.tilePull_add_mul, `Dubon2026.SourceModularSidePairing.slash_eq_of_twist, `Dubon2026.SourceModularSidePairing.sum_integral_rightRay, `Dubon2026.SourceModularSidePairing.sum_integral_arc, `Dubon2026.SourceModularSidePairing.boundary_side_pairing, `Dubon2026.SourceModularGreen.three_quarters_le, `Dubon2026.SourceModularGreen.sqrt3_sq, `Dubon2026.SourceModularGreen.sqrt3_pos, `Dubon2026.SourceModularGreen.sqrt3_lt_one, `Dubon2026.SourceModularGreen.sqrt3_le_h, `Dubon2026.SourceModularGreen.h_pos, `Dubon2026.SourceModularGreen.h_le_one, `Dubon2026.SourceModularGreen.mem_D_iff, `Dubon2026.SourceModularGreen.mk_mem_D, `Dubon2026.SourceModularGreen.preimage_D, `Dubon2026.SourceModularGreen.isClosed_T, `Dubon2026.SourceModularGreen.measurableSet_T, `Dubon2026.SourceModularGreen.T_subset_prod, `Dubon2026.SourceModularGreen.setIntegral_D, `Dubon2026.SourceModularGreen.integrableOn_T_of_bound, `Dubon2026.SourceModularGreen.setIntegral_T_eq_x_y, `Dubon2026.SourceModularGreen.setIntegral_T_eq_y_x, `Dubon2026.SourceModularGreen.mk_eq, `Dubon2026.SourceModularGreen.hasDerivAt_vert, `Dubon2026.SourceModularGreen.hasDerivAt_horiz, `Dubon2026.SourceModularGreen.continuous_mk_left, `Dubon2026.SourceModularGreen.continuous_mk_right, `Dubon2026.SourceModularGreen.tendsto_const_mul_exp, `Dubon2026.SourceModularGreen.integral_vert, `Dubon2026.SourceModularGreen.integral_horiz, `Dubon2026.SourceModularGreen.r_nonneg, `Dubon2026.SourceModularGreen.r_one, `Dubon2026.SourceModularGreen.r_pos, `Dubon2026.SourceModularGreen.r_le_half, `Dubon2026.SourceModularGreen.mem_D_neg_r, `Dubon2026.SourceModularGreen.continuous_r, `Dubon2026.SourceModularGreen.continuous_mk2, `Dubon2026.SourceModularGreen.continuousOn_corr, `Dubon2026.SourceModularGreen.integrableOn_vert, `Dubon2026.SourceModularGreen.sliceIntegral, `Dubon2026.SourceModularGreen.integral_T_horiz, `Dubon2026.SourceModularGreen.integrableOn_T_apply, `Dubon2026.SourceModularGreen.integral_T_vert, `Dubon2026.SourceModularGreen.exp_mul_I_eq, `Dubon2026.SourceModularGreen.continuous_γ, `Dubon2026.SourceModularGreen.cos_two_pi_div_three, `Dubon2026.SourceModularGreen.sin_two_pi_div_three, `Dubon2026.SourceModularGreen.pi_div_three_le, `Dubon2026.SourceModularGreen.abs_cos_le, `Dubon2026.SourceModularGreen.sin_nonneg_arc, `Dubon2026.SourceModularGreen.h_cos, `Dubon2026.SourceModularGreen.r_sin, `Dubon2026.SourceModularGreen.γ_mem_D, `Dubon2026.SourceModularGreen.continuousOn_Φ, `Dubon2026.SourceModularGreen.continuousOn_arc, `Dubon2026.SourceModularGreen.arc_sin, `Dubon2026.SourceModularGreen.arc_cos, `Dubon2026.SourceModularGreen.arc_integrand, `Dubon2026.SourceModularGreen.arc_split, `Dubon2026.SourceModularGreen.integral_eq_boundary]
  let constructions : Array Name := #[`Dubon2026.SourceModularSidePairing.pairElt, `Dubon2026.SourceModularSidePairing.tilePull, `Dubon2026.SourceModularGreen.D, `Dubon2026.SourceModularGreen.h, `Dubon2026.SourceModularGreen.T, `Dubon2026.SourceModularGreen.r, `Dubon2026.SourceModularGreen.corr, `Dubon2026.SourceModularGreen.γ]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Modular Green Boundary postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Modular Green Boundary theorems discovered"
  logInfo m!"SOURCE MODULAR GREEN BOUNDARY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
