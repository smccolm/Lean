import SourceModularSidePairingAction
import SourceModularSidePairingIntegrals
import SourceModularBoundarySidePairing
import SourceModularGreenDomain
import SourceModularGreenLines
import SourceModularGreenSliceGeometry
import SourceModularGreenSlicing
import SourceModularGreenArc
import SourceModularGreenFormula
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceModularSidePairingAction, `SourceModularSidePairingIntegrals, `SourceModularBoundarySidePairing, `SourceModularGreenDomain, `SourceModularGreenLines, `SourceModularGreenSliceGeometry, `SourceModularGreenSlicing, `SourceModularGreenArc, `SourceModularGreenFormula]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
