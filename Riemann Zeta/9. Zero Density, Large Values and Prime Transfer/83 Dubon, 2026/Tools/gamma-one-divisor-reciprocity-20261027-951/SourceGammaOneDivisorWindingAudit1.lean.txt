private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceSmoothComparisonQuotient, `SourceSmoothComparisonCusps, `SourceSmoothComparisonWindingLog, `SourceGammaOneDivisorWindingComparison]
  let theorems : Array Name := #[`Dubon2026.SourceSmoothComparison.quot_local, `Dubon2026.SourceSmoothComparison.contDiffAt_quot, `Dubon2026.SourceSmoothComparison.quot_ne_zero, `Dubon2026.SourceSmoothComparison.tendsto_div_quot, `Dubon2026.SourceSmoothComparison.eventually_eq_quot_mul, `Dubon2026.SourceSmoothComparison.quot_smul, `Dubon2026.SourceSmoothComparison.quot_eq_limUnder, `Dubon2026.SourceSmoothComparison.exists_height_n_eq_zero, `Dubon2026.SourceSmoothComparison.eventually_atImInfty, `Dubon2026.SourceSmoothComparison.quot_eq_of_high, `Dubon2026.SourceSmoothComparison.tendsto_quot_smul, `Dubon2026.SourceSmoothComparison.tendsto_fderiv_quot_smul, `Dubon2026.SourceSmoothComparison.quot_cusp_data, `Dubon2026.SourceSmoothComparison.exists_filled_winding_log, `Dubon2026.SourceFunctionField.GammaOne.exists_original_divisor_winding_comparison]
  let constructions : Array Name := #[`Dubon2026.SourceSmoothComparison.HasLocalModels, `Dubon2026.SourceSmoothComparison.quot]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Divisor Winding postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Divisor Winding theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIVISOR WINDING AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
