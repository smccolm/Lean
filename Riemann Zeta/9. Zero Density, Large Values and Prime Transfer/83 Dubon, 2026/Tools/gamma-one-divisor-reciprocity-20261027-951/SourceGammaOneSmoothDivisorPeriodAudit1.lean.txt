private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceSmoothOrbitIntegralGeometry, `SourceSmoothOrbitIntegralChangeVariables, `SourceSmoothCompactTranslateIntegral, `SourceSmoothDivisorPeriodIntegral, `SourceGammaOneSmoothDivisorPeriodIntegral]
  let theorems : Array Name := #[`Dubon2026.SourceSmoothOrbitIntegral.sm_image, `Dubon2026.SourceSmoothOrbitIntegral.continuousAt_ofComplex', `Dubon2026.SourceSmoothOrbitIntegral.hasFDerivAt_sm_matrix, `Dubon2026.SourceSmoothOrbitIntegral.continuousOn_sm, `Dubon2026.SourceSmoothOrbitIntegral.denom_ofComplex, `Dubon2026.SourceSmoothOrbitIntegral.denom_ne_zero', `Dubon2026.SourceSmoothOrbitIntegral.continuousOn_dsm, `Dubon2026.SourceSmoothOrbitIntegral.eq_zero_of_notMem, `Dubon2026.SourceSmoothOrbitIntegral.continuousOn_goc, `Dubon2026.SourceSmoothOrbitIntegral.det_identity, `Dubon2026.SourceSmoothOrbitIntegral.integral_term, `Dubon2026.SourceSmoothOrbitIntegral.integrable_term, `Dubon2026.SourceSmoothOrbitIntegral.integrable_term', `Dubon2026.SourceSmoothOrbitIntegral.compact_translate_eventually_eq, `Dubon2026.SourceSmoothOrbitIntegral.compact_translate_limit, `Dubon2026.SourceSmoothDivisor.g_smul, `Dubon2026.SourceSmoothDivisor.exists_comparison_period_limit, `Dubon2026.SourceFunctionField.GammaOne.exists_original_divisor_smooth_period_limit]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Smooth Divisor Period postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Smooth Divisor Period theorems discovered"
  logInfo m!"SOURCE GAMMA ONE SMOOTH DIVISOR PERIOD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
