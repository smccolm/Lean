private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceSmoothLocalModelRegularization, `Dubon2026.SourceSmoothLocalModelProducts, `Dubon2026.SourceModularSmoothProjectiveRepresentatives, `Dubon2026.SourceSmoothModularGeometry, `Dubon2026.SourceSmoothProjectiveOrbitSum, `Dubon2026.SourceSmoothDivisorOrbitData, `Dubon2026.SourceSmoothModularLogDerivative, `Dubon2026.SourceSmoothOrbitIntegralChangeVariables, `Dubon2026.SourceSmoothCompactTranslateIntegral, `Dubon2026.SourceSmoothDivisorPeriodIntegral, `Dubon2026.SourceGammaOneSmoothDivisorPeriodIntegral, `Dubon2026.SourceSmoothComparisonQuotient, `Dubon2026.SourceSmoothComparisonCusps, `Dubon2026.SourceSmoothComparisonWindingLog, `Dubon2026.SourceGammaOneDivisorWindingComparison, `Dubon2026.SourceModularSidePairingAction, `Dubon2026.SourceModularSidePairingIntegrals, `Dubon2026.SourceModularBoundarySidePairing, `Dubon2026.SourceModularGreenDomain, `Dubon2026.SourceModularGreenLines, `Dubon2026.SourceModularGreenSliceGeometry, `Dubon2026.SourceModularGreenSlicing, `Dubon2026.SourceModularGreenArc, `Dubon2026.SourceModularGreenMeasure, `Dubon2026.SourceModularGreenDensity, `Dubon2026.SourceModularGreenSidePairing, `Dubon2026.SourceWindingLogCuspBounds, `Dubon2026.SourceModularSmoothedGreenLimit, `Dubon2026.SourceWindingBoundaryIntegral, `Dubon2026.SourceGammaOneDivisorWindingBoundary, `Dubon2026.SourceFinitePermutationCoboundary, `Dubon2026.SourceParabolicCosetPotentials, `Dubon2026.SourceUntwistPeterssonLimit, `Dubon2026.SourceGammaOnePeterssonReciprocity]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
