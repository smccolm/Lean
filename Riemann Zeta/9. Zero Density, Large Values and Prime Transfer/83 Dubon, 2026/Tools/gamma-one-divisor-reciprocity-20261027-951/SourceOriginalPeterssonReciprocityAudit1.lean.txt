private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceSmoothComparisonDbar, `SourceSmoothedWindingIntegralSplit, `SourceGammaOneSmoothedReciprocity, `SourceUntwistPeterssonLimit, `SourceGammaOnePeterssonReciprocity]
  let theorems : Array Name := #[`Dubon2026.SourceSmoothComparison.order_eq_zero_of_ne_zero, `Dubon2026.SourceSmoothComparison.differentiableAt_of_zero_order, `Dubon2026.SourceSmoothComparison.dbar_quotient_split, `Dubon2026.SourceSmoothComparison.weighted_dbar_quotient_split_ae, `Dubon2026.SourceSmoothComparison.integrable_smoothed_dbar, `Dubon2026.SourceSmoothComparison.smoothed_integral_quotient_split, `Dubon2026.SourceFunctionField.GammaOne.exists_original_divisor_smoothed_reciprocity, `Dubon2026.SourcePeriodUntwist.Φ_dbar_ae, `Dubon2026.SourcePeriodUntwist.petersson_plane_density, `Dubon2026.SourcePeriodUntwist.tendsto_Φ_smoothed_petersson, `Dubon2026.SourceFunctionField.GammaOne.exists_original_divisor_petersson_reciprocity]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Petersson Reciprocity postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Petersson Reciprocity theorems discovered"
  logInfo m!"SOURCE ORIGINAL PETERSSON RECIPROCITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
