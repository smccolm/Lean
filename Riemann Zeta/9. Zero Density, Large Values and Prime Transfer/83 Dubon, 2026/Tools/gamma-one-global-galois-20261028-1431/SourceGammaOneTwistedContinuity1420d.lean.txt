private theorem continuous_discrete_of_eventually_one {G A : Type*} [Group G]
    [TopologicalSpace G] [IsTopologicalGroup G] [Monoid A] (ρ : G →* A)
    (h : ∀ᶠ σ in 𝓝 (1 : G), ρ σ = 1) : @Continuous G A _ ⊥ ρ := by
  letI : TopologicalSpace A := ⊥
  letI : DiscreteTopology A := ⟨rfl⟩
  apply continuous_of_tendsto_nhds_one
  apply tendsto_const_nhds.congr'
  exact h.mono (fun _ he => he.symm)

private theorem one_congruence {R W G : Type*} [CommRing R] [AddCommGroup W]
    [Module R W] [Monoid G] (ρ : G →* Module.End R W) (U : Submodule R W) :
    ∀ x : W, ρ 1 x - x ∈ U := by
  intro x
  rw [map_one]
  change x - x ∈ U
  rw [sub_self]
  exact U.zero_mem

theorem binaryArithmeticTateGaloisRep_continuous :
    @Continuous _ _ _ ⊥ (binaryArithmeticTateGaloisRep Q p n w).toFun := by
  let U : Set G := {σ | ∀ x : T, arithmeticTateGaloisRep M p σ x - x ∈
    (maximalIdeal ℤ_[p] ^ (n + 1)) • (⊤ : Submodule ℤ_[p] T)}
  have ho : IsOpen U := arithmeticTateGalois_uniform_congruence_isOpen M p (n + 1)
  have hm : (1 : G) ∈ U := one_congruence (arithmeticTateGaloisRep M p)
    ((maximalIdeal ℤ_[p] ^ (n + 1)) • (⊤ : Submodule ℤ_[p] T))
  have hi : IsOpen ((fun σ : G => σ⁻¹) ⁻¹' U) := ho.preimage continuous_inv
  have hχ : IsOpen {σ : G | sourceAbsoluteCyclotomicCharacter N σ = 1} :=
    (sourceAbsoluteCyclotomicCharacter N).continuous.isOpen_preimage {a | a = 1} (isOpen_discrete _)
  apply continuous_discrete_of_eventually_one (binaryArithmeticTateGaloisRep Q p n w)
  filter_upwards [hi.mem_nhds (by simpa only [Set.mem_preimage, inv_one] using hm),
    hχ.mem_nhds (map_one (sourceAbsoluteCyclotomicCharacter N))] with σ hσ hc
  have hs := binaryArithmeticTateGaloisRep_eq_one_of_congruence Q p n w σ
  exact hs hc hσ

end
