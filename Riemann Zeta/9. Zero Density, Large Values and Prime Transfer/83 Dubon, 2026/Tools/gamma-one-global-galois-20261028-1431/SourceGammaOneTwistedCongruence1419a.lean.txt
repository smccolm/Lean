/-! DUB-17/18 -> original all-weight arithmetic association -> continuity
of the genuine finite homogeneous-valued Tate action. A trivial actual
cyclotomic exponent and the proved finite-coordinate Tate congruence
force the actual dual action to be the identity. The next consumer obtains
these conditions on an open neighborhood from the existing continuous
cyclotomic character and the proved uniform Tate congruences. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open SourceCurve CongruenceSubgroup IsLocalRing
variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "J" => Pic0 Qbar (functionFieldOver (Gamma1 M) Qbar)
local notation "T" => TateModule p J

private local instance : Algebra ℤ_[p] (ZMod N) := (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V := Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

theorem binaryArithmeticTateGaloisRep_eq_one_of_congruence (σ : Qbar ≃ₐ[ℚ] Qbar)
    (hχ : sourceAbsoluteCyclotomicCharacter N σ = 1)
    (hT : ∀ x : T, arithmeticTateGaloisRep M p σ⁻¹ x - x ∈
      (maximalIdeal ℤ_[p] ^ (n + 1)) • (⊤ : Submodule ℤ_[p] T)) :
    binaryArithmeticTateGaloisRep Q p n w σ = 1 := by
  have hV : binaryCyclotomicCoefficientRep Q p n w σ = 1 :=
    (congrArg (sourceCyclotomicHomogeneousAction (ZMod N) w) hχ).trans
      (sourceCyclotomicHomogeneousAction (ZMod N) w).map_one
  apply LinearMap.ext
  intro φ
  apply LinearMap.ext
  intro x
  have hp : TateModule.proj p J (n + 1) (arithmeticTateGaloisRep M p σ⁻¹ x) =
      TateModule.proj p J (n + 1) x := by
    have hz := (TateModule.proj_eq_zero_iff (n + 1) _).mpr (hT x)
    exact sub_eq_zero.mp ((map_sub (TateModule.proj p J (n + 1)) _ _).symm.trans hz)
  have he := binaryArithmeticTateDual_eq_of_proj_eq Q p n w φ hp
  change binaryCyclotomicCoefficientRep Q p n w σ (φ (arithmeticTateGaloisRep M p σ⁻¹ x)) = φ x
  exact (congrArg (fun b : Module.End (ZMod N) V =>
    b (φ (arithmeticTateGaloisRep M p σ⁻¹ x))) hV).trans he

end
end Dubon2026.SourceFunctionField.GammaOne
