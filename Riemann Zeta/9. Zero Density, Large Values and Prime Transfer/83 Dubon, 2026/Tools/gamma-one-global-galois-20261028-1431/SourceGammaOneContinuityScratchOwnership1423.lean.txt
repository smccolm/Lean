import SourceGammaOneFunctionFieldDefinition
import SourceGammaOneFinitePlaceOrbit
import SourceGammaOnePlaceFieldDefinition
import SourceGammaOnePicardFieldDefinition
import SourceGammaOneTateOpenCongruence
import SourceGammaOneFiniteTateDual
import SourceGammaOneTwistedCongruence
import SourceGammaOneTwistedContinuity
import SourceGammaOneTwistedFiniteFactor
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneFunctionFieldDefinition, `SourceGammaOneFinitePlaceOrbit, `SourceGammaOnePlaceFieldDefinition, `SourceGammaOnePicardFieldDefinition, `SourceGammaOneTateOpenCongruence, `SourceGammaOneFiniteTateDual, `SourceGammaOneTwistedCongruence, `SourceGammaOneTwistedContinuity, `SourceGammaOneTwistedFiniteFactor]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
