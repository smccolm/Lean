open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceTwistedCoefficientDualAction, `Dubon2026.SourceCyclotomicHomogeneousAction, `Dubon2026.SourceAbsoluteCyclotomicCharacter, `Dubon2026.SourceGammaOneTwistedGalois, `Dubon2026.SourceCyclotomicHomogeneousHecke, `Dubon2026.SourceGammaOneTwistedGaloisHecke, `Dubon2026.SourceAbsoluteCyclotomicLocal, `Dubon2026.SourceGammaOneTwistedGaloisFrobenius, `Dubon2026.SourceTwistedCoefficientDualTrivial, `Dubon2026.SourceGammaOneTwistedGaloisInertia, `Dubon2026.SourceGammaOneParabolicTateLift, `Dubon2026.SourceGammaOneParabolicGaloisFrobenius, `Dubon2026.SourceGammaOneFunctionFieldDefinition, `Dubon2026.SourceGammaOneFinitePlaceOrbit, `Dubon2026.SourceGammaOnePlaceFieldDefinition, `Dubon2026.SourceGammaOnePicardFieldDefinition, `Dubon2026.SourceGammaOneTateOpenCongruence, `Dubon2026.SourceGammaOneFiniteTateDual, `Dubon2026.SourceGammaOneTwistedCongruence, `Dubon2026.SourceGammaOneTwistedContinuity, `Dubon2026.SourceGammaOneTwistedFiniteFactor, `Dubon2026.SourceGammaOneTwistedNormInertia, `Dubon2026.SourceGammaOneTwistedArithmeticDescent, `Dubon2026.SourceCyclotomicHomogeneousDiamond, `Dubon2026.SourceGammaOneTwistedGaloisDiamond, `Dubon2026.SourceGammaOneParabolicGaloisOrbit]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
