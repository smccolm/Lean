private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneFunctionFieldDefinition, `SourceGammaOneFinitePlaceOrbit, `SourceGammaOnePlaceFieldDefinition, `SourceGammaOnePicardFieldDefinition, `SourceGammaOneTateOpenCongruence, `SourceGammaOneFiniteTateDual, `SourceGammaOneTwistedCongruence, `SourceGammaOneTwistedContinuity, `SourceGammaOneTwistedFiniteFactor]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.gammaOne_function_fieldOfDefinition, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_place_galois_orbit_finite, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_place_fieldOfDefinition, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_picard_fieldOfDefinition, `Dubon2026.SourceFunctionField.GammaOne.gammaOnePicard_isOpen_stabilizer, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGalois_congruence_isOpen, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGalois_uniform_congruence_isOpen, `Dubon2026.SourceFunctionField.GammaOne.binaryArithmeticTateDual_finite, `Dubon2026.SourceFunctionField.GammaOne.binaryArithmeticTateDual_eq_of_proj_eq, `Dubon2026.SourceFunctionField.GammaOne.binaryArithmeticTateGaloisRep_eq_one_of_congruence, `Dubon2026.SourceFunctionField.GammaOne.binaryArithmeticTateGaloisRep_eventually_one, `Dubon2026.SourceFunctionField.GammaOne.binaryArithmeticTateGaloisRep_continuous, `Dubon2026.SourceFunctionField.GammaOne.binaryArithmeticTateGaloisRep_faithful_finite_factor]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.gammaOnePicardArithmeticAction]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CONTINUITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
