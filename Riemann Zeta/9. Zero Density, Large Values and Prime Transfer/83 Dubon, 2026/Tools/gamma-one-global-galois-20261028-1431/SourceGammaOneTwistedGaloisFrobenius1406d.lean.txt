private theorem transfer_quadratic {R W : Type*} [CommRing R] [AddCommGroup W] [Module R W]
    (F G H : Module.End R W) (a : R) (x : W) (he : F = G)
    (hq : G (H x) = G (G x) + a • x) : F (H x) = F (F x) + a • x := by
  cases he
  exact hq

variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "Γ" => Gamma1 M
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "T" => TateModule p (Pic0 Qbar (functionFieldOver Γ Qbar))
variable (A : ValuationSubring Qbar)
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Γ
attribute [local instance] specializationResidueAlgebra

private local instance : Algebra ℤ_[p] (ZMod N) := (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V := Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

theorem exists_gammaOne_galois_frobenius_quadratic_of_degree_genus_eq
    (ell : sourceBinaryHeckePrime (Q * N)) [CharP kk ell.val]
    (hdeg : Module.finrank Qbar⟮jCoordinateOver Γ Qbar⟯ (functionFieldOver Γ Qbar) =
      Module.finrank kk⟮residueJCoordinate A Γ⟯ Fb)
    (hgen : genusFF kk Fb = genusFF Qbar (functionFieldOver Γ Qbar))
    (d : A.decompositionSubgroup ℚ) (hd : ∀ a : kk, d • a = a ^ ell.val)
    (hp : (p : kk) ≠ 0) :
    ∃ g : Gamma0 M, (g.val 1 1 : ZMod N) = (ell.val : ZMod N) ∧
      ∀ (φ : T →ₗ[ℤ_[p]] V),
        ((sourceGammaOneBinaryDiamondCoefficient Q N w g).restrictScalars ℤ_[p]).comp
          (φ.comp (arithmeticDiamondTate M p g)) = φ →
        binaryArithmeticTateGaloisRep Q p n w d.val (binaryTatePrimeOperators Q p n w ell φ) =
          binaryArithmeticTateGaloisRep Q p n w d.val
            (binaryArithmeticTateGaloisRep Q p n w d.val φ) +
            (ell.val : ZMod N) ^ (w + 1) • φ := by
  have hs := exists_gammaOne_arithmetic_frobenius_quadratic_of_degree_genus_eq Q p n w A ell
  have h1 := hs hdeg
  have h2 := h1 hgen
  have h3 := h2 d hd hp
  obtain ⟨g, hg, hq⟩ := h3
  have hχ := sourceAbsoluteCyclotomicCharacter_frobenius A N ell.val
    (by simpa only [Nat.cast_pow] using pow_ne_zero (n + 1) hp) d hd
  have he := binaryArithmeticTateGaloisRep_frobenius_operator Q p n w ell d.val hχ
  refine ⟨g, hg, fun φ hφ => ?_⟩
  exact transfer_quadratic (binaryArithmeticTateGaloisRep Q p n w d.val)
    (binaryTateArithmeticFrobeniusOperator Q p n w ell d.val)
    (binaryTatePrimeOperators Q p n w ell) ((ell.val : ZMod N) ^ (w + 1)) φ he (hq φ hφ)

end
