The continuous actual cyclotomic character and the proved uniform Tate

quotient representation is supplied as an input. -/

variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "G" => Qbar ≃ₐ[ℚ] Qbar
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "J" => Pic0 Qbar (functionFieldOver (Gamma1 M) Qbar)
local notation "T" => TateModule p J

private local instance : Algebra ℤ_[p] (ZMod N) := (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V := Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

theorem binaryArithmeticTateGaloisRep_continuous :
    letI : TopologicalSpace (Module.End (ZMod N) (T →ₗ[ℤ_[p]] V)) := ⊥
    Continuous (binaryArithmeticTateGaloisRep Q p n w) := by
  letI : TopologicalSpace (Module.End (ZMod N) (T →ₗ[ℤ_[p]] V)) := ⊥
  letI : DiscreteTopology (Module.End (ZMod N) (T →ₗ[ℤ_[p]] V)) := ⟨rfl⟩
  let U : Set G := {σ | ∀ x : T, arithmeticTateGaloisRep M p σ x - x ∈
    (maximalIdeal ℤ_[p] ^ (n + 1)) • (⊤ : Submodule ℤ_[p] T)}
  have ho : IsOpen U := arithmeticTateGalois_uniform_congruence_isOpen M p (n + 1)
  have hm : (1 : G) ∈ U := by
    intro x
    rw [map_one]
    change x - x ∈ _
    rw [sub_self]
    exact Submodule.zero_mem _
  have hi : IsOpen ((fun σ : G => σ⁻¹) ⁻¹' U) := ho.preimage continuous_inv
  have hχ : IsOpen {σ : G | sourceAbsoluteCyclotomicCharacter N σ = 1} :=
    (sourceAbsoluteCyclotomicCharacter N).continuous.isOpen_preimage {a | a = 1} (isOpen_discrete _)
  apply continuous_of_tendsto_nhds_one
  apply tendsto_const_nhds.congr'
  filter_upwards [hi.mem_nhds (by simpa only [Set.mem_preimage, inv_one] using hm),
    hχ.mem_nhds (map_one (sourceAbsoluteCyclotomicCharacter N))] with σ hσ hc
  have hs := binaryArithmeticTateGaloisRep_eq_one_of_congruence Q p n w σ
  exact (hs hc hσ).symm

end
