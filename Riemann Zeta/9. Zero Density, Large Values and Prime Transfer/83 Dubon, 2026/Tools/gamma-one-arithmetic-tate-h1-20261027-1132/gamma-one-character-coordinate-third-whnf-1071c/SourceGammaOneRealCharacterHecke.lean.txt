/-- The actual native H1 Hecke action in the native trivial-coefficient character coordinates. -/
def constantCharacterHecke : Module.End R (Additive (Gamma1 M) →+ R) :=
  (H1IsoOfIsTrivial (Rep.trivial R (Gamma1 M) R)).hom.hom.comp
    ((sourceGammaOneConstantHeckeH1 R R M d LinearMap.id).comp
      (H1IsoOfIsTrivial (Rep.trivial R (Gamma1 M) R)).inv.hom)

theorem constantCharacterHecke_apply (φ : Additive (Gamma1 M) →+ R) (γ : Gamma1 M) :
    constantCharacterHecke R M d φ (Additive.ofMul γ) =
      ∑ q : Gamma1 M ⧸ sourceGammaOneHeckeUpper M d,
        φ (Additive.ofMul (sourceGammaOneHeckeConjugation M d
          (cosetTransition (sourceGammaOneHeckeUpper M d) γ q))) := by
  dsimp only [constantCharacterHecke, LinearMap.comp_apply]
  refine (sourceGammaOneConstantHeckeH1_hom_apply R M d
    ((H1IsoOfIsTrivial (Rep.trivial R (Gamma1 M) R)).inv φ) γ).trans ?_
  apply Finset.sum_congr rfl
  intro q _
  exact congrArg (fun ψ : Additive (Gamma1 M) →+ R => ψ
    (Additive.ofMul (sourceGammaOneHeckeConjugation M d
      (cosetTransition (sourceGammaOneHeckeUpper M d) γ q))))
    (Iso.inv_hom_id_apply (H1IsoOfIsTrivial (Rep.trivial R (Gamma1 M) R)) φ)

variable (M d : ℕ) [NeZero M] [NeZero d]

theorem realCharacterHecke_period (f : CuspForm ((Gamma1 M).map (mapGL ℝ)) 2) :

  intro γ
  have hγ : γ = Additive.ofMul γ.toMul := rfl
  rw [hγ, constantCharacterHecke_apply]
  simp only [sourceGamma1RealPeriodHom_apply]
  rw [periodValue_cuspHeckeTrace]
  exact (map_sum Complex.reAddGroupHom
    (fun q : Gamma1 M ⧸ sourceGammaOneHeckeUpper M d =>
      sourceWeightTwoPeriodValue (Gamma1 M) f
        (sourceGammaOneHeckeConjugation M d
          (cosetTransition (sourceGammaOneHeckeUpper M d) γ.toMul q))) Finset.univ).symm

end
