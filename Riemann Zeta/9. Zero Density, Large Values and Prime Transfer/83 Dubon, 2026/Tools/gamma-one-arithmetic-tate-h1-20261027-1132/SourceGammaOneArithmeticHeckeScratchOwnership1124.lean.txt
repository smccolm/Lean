import SourceGammaOneHeckeConstantField
import SourceGammaOneHeckeSpanSeparation
import SourceGammaOneHeckeValuationExtension
import SourceGammaOneHeckeConstantPlaces
import SourceGammaOneHeckeConorm
import SourceGammaOneHeckeComplexPlaces
import SourceGammaOneHeckeComplexDivisors
import SourceGammaOneHeckeComplexPicard
import SourceGammaOneArithmeticTateHecke
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneHeckeConstantField, `SourceGammaOneHeckeSpanSeparation, `SourceGammaOneHeckeValuationExtension, `SourceGammaOneHeckeConstantPlaces, `SourceGammaOneHeckeConorm, `SourceGammaOneHeckeComplexPlaces, `SourceGammaOneHeckeComplexDivisors, `SourceGammaOneHeckeComplexPicard, `SourceGammaOneArithmeticTateHecke]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
