/-! DUB17/18 -> actual Gamma1 arithmetic realization -> arithmetic
Tate/Hecke and original period comparison. The actual geometric Picard
naturality discharges the commuting-map premise of Tate naturality.
The original arithmetic prime Hecke operator therefore acts on the
p-adic tensor of the same integral period lattice by the native dual
cusp Hecke map. Neither prime is required to avoid the level. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open SourceCurve CongruenceSubgroup
variable (M p : ℕ) [NeZero M] [Fact p.Prime]
local notation "Qbar" => AlgebraicClosure ℚ
local notation "FQ" => functionFieldOver (Gamma1 M) Qbar

theorem tateComplexEquiv_hecke (d : ℕ) [NeZero d] (x : TateModule p (Pic0 Qbar FQ)) :
    ConstantField.tateComplexEquiv M p (TateModule.tateMap p (heckePicard Qbar M d) x) =
      TateModule.tateMap p (heckePicard ℂ M d) (ConstantField.tateComplexEquiv M p x) :=
  ConstantField.tateComplexEquiv_naturality M p (heckePicard Qbar M d) (heckePicard ℂ M d)
    (ConstantField.complexPicardMap_heckePicard M d) x

theorem arithmeticPicardTatePeriodTensorEquiv_hecke (q : ℕ) [Fact q.Prime]
    (x : TateModule p (Pic0 Qbar FQ)) :
    arithmeticPicardTatePeriodTensorEquiv M p (TateModule.tateMap p (heckePicard Qbar M q) x) =
      (periodLatticeHecke M q).baseChange ℤ_[p]
        (arithmeticPicardTatePeriodTensorEquiv M p x) := by
  have hT : ModularGroup.T ∈ Gamma1 M := by simp [Gamma1_mem, ModularGroup.T]
  change complexPicardTatePeriodTensorEquiv M p hT
      (ConstantField.tateComplexEquiv M p (TateModule.tateMap p (heckePicard Qbar M q) x)) =
    (periodLatticeHecke M q).baseChange ℤ_[p]
      (complexPicardTatePeriodTensorEquiv M p hT (ConstantField.tateComplexEquiv M p x))
  rw [tateComplexEquiv_hecke]
  exact complexPicardTatePeriodTensorEquiv_hecke M p hT q (ConstantField.tateComplexEquiv M p x)

end
end Dubon2026.SourceFunctionField.GammaOne
