/-! DUB17/18 -> original all-weight arithmetic realization -> full/parabolic

variable (M : ℕ) [NeZero M]

private local instance (d : ℕ) [NeZero d] :
    Fintype (Gamma1 M ⧸ sourceGammaOneHeckeUpper M d) := by
  letI := sourceGammaOneHeckeUpper_finiteIndex M d
  exact Fintype.ofFinite _

/-- The original real cusp periods as classes in native real Gamma1 H1. -/
def realPeriodH1 :
    CuspForm ((Gamma1 M).map (mapGL ℝ)) 2 →ₗ[ℝ] H1 (Rep.trivial ℝ (Gamma1 M) ℝ) :=
  (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).inv.hom.comp
    (sourceGamma1RealPeriodHom M)

theorem realPeriodH1_hom_apply
    (f : CuspForm ((Gamma1 M).map (mapGL ℝ)) 2) (γ : Gamma1 M) :
    (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom
      (realPeriodH1 M f) (Additive.ofMul γ) =
        (sourceWeightTwoPeriodValue (Gamma1 M) f γ).re := by
  change (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom
    ((H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).inv
      (sourceGamma1RealPeriodHom M f)) (Additive.ofMul γ) = _
  rw [Iso.inv_hom_id_apply]
  rfl

theorem realPeriodH1_cuspHeckeTrace (d : ℕ) [NeZero d]
    (f : CuspForm ((Gamma1 M).map (mapGL ℝ)) 2) :
    realPeriodH1 M (cuspHeckeTrace M d f) =
      sourceGammaOneConstantHeckeH1 ℝ ℝ M d LinearMap.id (realPeriodH1 M f) := by
  apply (ConcreteCategory.bijective_of_isIso
    (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom).injective
  apply AddMonoidHom.ext
  intro γ
  have hγ : γ = Additive.ofMul γ.toMul := rfl
  rw [hγ, realPeriodH1_hom_apply, sourceGammaOneConstantHeckeH1_hom_apply]
  refine (congrArg Complex.re (periodValue_cuspHeckeTrace M d f γ.toMul)).trans ?_
  refine (map_sum Complex.reAddGroupHom _ Finset.univ).trans ?_
  apply Finset.sum_congr rfl
  intro q _
  exact (realPeriodH1_hom_apply M f _).symm

end
