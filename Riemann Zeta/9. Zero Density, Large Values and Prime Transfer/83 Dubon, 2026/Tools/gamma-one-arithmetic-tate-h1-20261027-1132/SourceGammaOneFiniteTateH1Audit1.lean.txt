private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneIntegralCharacterFactor, `SourceGammaOneFiniteParabolicCharacters, `SourceGammaOneFiniteParabolicH1, `SourceGammaOneFiniteParabolicHecke, `SourceGammaOneArithmeticPeriodInclusion, `SourceGammaOneArithmeticTateFiniteH1, `SourceGammaOneArithmeticTateFiniteHecke]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.latticePeriodHom_surjective, `Dubon2026.SourceFunctionField.GammaOne.latticePeriodHom_parabolic, `Dubon2026.SourceFunctionField.GammaOne.latticePeriodHom_finite_order, `Dubon2026.SourceFunctionField.GammaOne.latticePeriodHom_ker_le_integer_character, `Dubon2026.SourceFunctionField.GammaOne.exists_integer_lattice_factor, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeCharacterMap_injective, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeCharacterMap_surjective, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeCharacterEquiv_apply, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeH1Equiv_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.periodLatticeHecke_latticePeriod, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeH1Equiv_hecke, `Dubon2026.SourceFunctionField.GammaOne.finiteParabolicH1_hecke_mem, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeH1Equiv_hecke_restrict, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPeriodInclusion_isBaseChange, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPeriodInclusion_hecke, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateCharacterRestriction_injective, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateCharacterRestriction_surjective, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateFiniteH1Equiv_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateCharacterEquiv_hecke, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateFiniteH1Equiv_hecke, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateFiniteH1Equiv_hecke_restrict]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.latticePeriodHom, `Dubon2026.SourceFunctionField.GammaOne.finiteParabolicHoms, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeCharacterMap, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeCharacterEquiv, `Dubon2026.SourceFunctionField.GammaOne.finiteParabolicH1, `Dubon2026.SourceFunctionField.GammaOne.finiteLatticeH1Equiv, `Dubon2026.SourceFunctionField.GammaOne.finiteParabolicHecke, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPeriodInclusion, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateCharacterRestriction, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateCharacterEquiv, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateFiniteH1Equiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Finite Tate H1 postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Finite Tate H1 theorems discovered"
  logInfo m!"SOURCE GAMMA ONE FINITE TATE H1 AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
