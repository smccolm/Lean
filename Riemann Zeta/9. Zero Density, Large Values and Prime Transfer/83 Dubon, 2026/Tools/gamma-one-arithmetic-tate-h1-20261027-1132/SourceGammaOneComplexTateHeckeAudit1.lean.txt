private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneFinitePeriodLattice, `SourceGammaOneComplexTatePeriods, `SourceGammaOnePeriodTateHecke, `SourceGammaOneComplexTateHecke]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.integralLattice_eq_dualPeriodHom_range, `Dubon2026.SourceFunctionField.GammaOne.integralLattice_finite, `Dubon2026.SourceFunctionField.GammaOne.integralLattice_free, `Dubon2026.SourceFunctionField.GammaOne.complexPicardTatePeriodTensorEquiv_symm_tmul, `Dubon2026.SourceFunctionField.GammaOne.periodLatticeHecke_val, `Dubon2026.SourceFunctionField.GammaOne.periodQuotientHecke_eq_quotEnd, `Dubon2026.SourceFunctionField.GammaOne.periodTateTensor_hecke, `Dubon2026.SourceFunctionField.GammaOne.complexPicardTatePeriodQuotient_hecke, `Dubon2026.SourceFunctionField.GammaOne.complexPicardTatePeriodTensorEquiv_hecke]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.dualPeriodHom, `Dubon2026.SourceFunctionField.GammaOne.integralPeriodBasis, `Dubon2026.SourceFunctionField.GammaOne.complexPicardTatePeriodTensorEquiv, `Dubon2026.SourceFunctionField.GammaOne.periodLatticeHecke]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Complex Tate Hecke postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Complex Tate Hecke theorems discovered"
  logInfo m!"SOURCE GAMMA ONE COMPLEX TATE HECKE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
