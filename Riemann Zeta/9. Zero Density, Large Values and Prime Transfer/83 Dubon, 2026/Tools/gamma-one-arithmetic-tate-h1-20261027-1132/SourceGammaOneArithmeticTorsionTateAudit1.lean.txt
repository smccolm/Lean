  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneComplexConstantAutomorphisms, `SourceGammaOneComplexTranslations, `SourceGammaOneOffImageValues, `SourceGammaOneTranslationOrbits, `SourceGammaOneReducedRiemannRochSpace, `SourceGammaOnePrincipalDivisors, `SourceGammaOnePrincipalDivisorIdentities, `SourceGammaOneFiniteTorsionAction, `SourceGammaOneReducedDivisorInvariance, `SourceGammaOneFixedReducedDivisor, `SourceGammaOneInvariantSupport, `SourceGammaOneTorsionDivisorDescent, `SourceGammaOneTorsionComparison, `SourceGammaOneTateComparison, `SourceGammaOneArithmeticTatePeriods]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.ConstantField.complexAut_fixes_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.complexAut_fixes_liftPlace, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.exists_complex_translation, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.off_image_exists_nonarithmetic_j_value, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.j_value_arithmetic_pos, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.translation_orbit_injective, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.divisor_pow_fixed, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.exists_reduced_space, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.finite_principal_support, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.principalDivisor_apply, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.degree_principalDivisor, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.principal_mul, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.principal_inv, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.principal_smul, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.principal_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.mem_lSpace_add_principal_iff, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.mem_lSpace_arithmetic_iff, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.complex_torsion_action_factorial, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.complex_torsion_divisor_factorial_principal, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.reduced_divisor_fixed, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.complexAut_fixes_lift_single, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.exists_fixed_reduced_divisor, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.invariant_support_lifts, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.exists_conorm_of_support_lifts, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.exists_torsion_divisor_descent, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.picard_principal_eq_zero, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.exists_torsion_preimage, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.torsion_cardinality_eq_complex, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.arithmetic_torsion_cardinality_period_rank, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.finite_order_mem_complexPicardMap_range, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.tateComplexEquiv_apply, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.tateComplexEquiv_naturality, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPicardTatePeriodTensorEquiv_symm_tmul]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.ConstantField.principalDivisor, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.tateComplexEquiv, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPicardTatePeriodTensorEquiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Hecke Geometric Fibers postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Hecke Geometric Fibers theorems discovered"
  logInfo m!"SOURCE GAMMA ONE ARITHMETIC TORSION TATE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
