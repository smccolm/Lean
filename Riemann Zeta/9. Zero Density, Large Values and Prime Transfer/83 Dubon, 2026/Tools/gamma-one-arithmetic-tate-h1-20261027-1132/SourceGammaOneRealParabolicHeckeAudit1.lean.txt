private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRealPeriodH1Hecke, `SourceGammaOneRealParabolicH1, `SourceGammaOneRealParabolicHecke]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_cuspHeckeTrace, `Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_mem_parabolic, `Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_range, `Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_injective, `Dubon2026.SourceFunctionField.GammaOne.realParabolicH1_hecke_mem, `Dubon2026.SourceFunctionField.GammaOne.realParabolicPeriodEquiv_cuspHeckeTrace]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.realPeriodH1, `Dubon2026.SourceFunctionField.GammaOne.realParabolicH1, `Dubon2026.SourceFunctionField.GammaOne.realParabolicHecke, `Dubon2026.SourceFunctionField.GammaOne.realParabolicPeriodEquiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Real Parabolic Hecke postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Real Parabolic Hecke theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REAL PARABOLIC HECKE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
