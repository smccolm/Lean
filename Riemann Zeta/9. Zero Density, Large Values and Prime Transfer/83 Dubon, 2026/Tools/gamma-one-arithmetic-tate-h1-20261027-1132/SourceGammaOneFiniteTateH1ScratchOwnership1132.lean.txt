import SourceGammaOneIntegralCharacterFactor
import SourceGammaOneFiniteParabolicCharacters
import SourceGammaOneFiniteParabolicH1
import SourceGammaOneFiniteParabolicHecke
import SourceGammaOneArithmeticPeriodInclusion
import SourceGammaOneArithmeticTateFiniteH1
import SourceGammaOneArithmeticTateFiniteHecke
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneIntegralCharacterFactor, `SourceGammaOneFiniteParabolicCharacters, `SourceGammaOneFiniteParabolicH1, `SourceGammaOneFiniteParabolicHecke, `SourceGammaOneArithmeticPeriodInclusion, `SourceGammaOneArithmeticTateFiniteH1, `SourceGammaOneArithmeticTateFiniteHecke]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
