/-! DUB17/18 -> actual Gamma1 arithmetic Tate/Hecke comparison ->
divisor naturality. This is the finite-support conorm on the actual
Hecke intersection curve. Only its on-image, off-image and single-divisor
formulas are needed for the correspondence comparison. No separate
Picard or torsion theory of the intersection curve is developed. -/
namespace Dubon2026.SourceFunctionField.GammaOne.HeckeConstantField
noncomputable section
open SourceCurve CongruenceSubgroup
local notation "Qbar" => AlgebraicClosure ℚ
variable (M d : ℕ) [NeZero M] [NeZero d]
local notation "FQ" => functionFieldOver (Gamma1 M ⊓ Gamma0 (d * M)) Qbar
local notation "FC" => functionFieldOver (Gamma1 M ⊓ Gamma0 (d * M)) ℂ

/-- Pushforward along the actual extension of arithmetic places to complex places. -/
def conorm : Divisor Qbar FQ →+ Divisor ℂ FC :=
  Finsupp.mapDomain.addMonoidHom (liftPlace M d)

theorem conorm_apply_liftPlace (D : Divisor Qbar FQ) (v : Place Qbar FQ) :
    conorm M d D (liftPlace M d v) = D v :=
  Finsupp.mapDomain_apply (liftPlace_injective M d) D v

theorem conorm_apply_of_forall_ne (D : Divisor Qbar FQ) {w : Place ℂ FC}
    (h : ∀ v, w ≠ liftPlace M d v) : conorm M d D w = 0 :=
  Finsupp.mapDomain_notin_range D w (by rintro ⟨v, rfl⟩; exact h v rfl)

theorem conorm_single (v : Place Qbar FQ) (n : ℤ) :
    conorm M d (Finsupp.single v n) = Finsupp.single (liftPlace M d v) n :=
  Finsupp.mapDomain_single

end
end Dubon2026.SourceFunctionField.GammaOne.HeckeConstantField
