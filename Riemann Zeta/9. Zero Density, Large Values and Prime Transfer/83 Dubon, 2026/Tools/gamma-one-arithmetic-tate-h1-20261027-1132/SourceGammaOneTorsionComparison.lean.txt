import SourceGammaOneTorsionDivisorDescent

/-! DUB17/18 -> actual Gamma1 arithmetic realization -> actual Picard
torsion comparison. Every nonzero-order complex torsion class now has
an arithmetic torsion preimage under the original coefficient map.
Injectivity gives bijectivity and the exact count in terms of the
original integral period rank. No rank-versus-genus assertion is used. -/
namespace Dubon2026.SourceFunctionField.GammaOne.ConstantField
noncomputable section
open SourceCurve CongruenceSubgroup
local notation "Qbar" => AlgebraicClosure ℚ
variable (N : ℕ) [NeZero N]
local notation "FQ" => functionFieldOver (Gamma1 N) Qbar
local notation "FC" => functionFieldOver (Gamma1 N) ℂ

theorem picard_principal_eq_zero (f : FC) (hf : f ≠ 0) :
    Pic0.mk (⟨principalDivisor N ℂ f, degree_principalDivisor N ℂ f⟩ :
      Divisor.degZero (K := ℂ) (F := FC)) = 0 :=
  (Pic0.mk_eq_zero_iff _).mpr ⟨f, hf, fun _ => rfl⟩

theorem exists_torsion_preimage (n : ℕ) (hn : n ≠ 0)
    (x : Pic0 ℂ FC) (hx : x ∈ Pic0.torsion n) :
    ∃ y : Pic0 Qbar FQ, y ∈ Pic0.torsion n ∧ complexPicardMap N y = x := by
  obtain ⟨D, rfl⟩ := Pic0.mk_surjective x
  obtain ⟨E, f, hf0, hE⟩ := exists_torsion_divisor_descent N n hn D hx
  have he0 : Divisor.degree E = 0 := by
    rw [← degree_conorm N E, hE, map_add, degree_principalDivisor, add_zero]
    exact D.property
  let E0 : Divisor.degZero (K := Qbar) (F := FQ) := ⟨E, he0⟩
  have hy : complexPicardMap N (Pic0.mk E0) = Pic0.mk D := by
    have heq : conormDegZero N E0 = D +
        ⟨principalDivisor N ℂ f, degree_principalDivisor N ℂ f⟩ := Subtype.ext hE
    rw [complexPicardMap_mk, heq, map_add, picard_principal_eq_zero N f hf0, add_zero]
  refine ⟨Pic0.mk E0, ?_, hy⟩
  rw [Pic0.mem_torsion_iff] at hx ⊢
  apply complexPicardMap_injective N
  rw [map_nsmul, hy, hx, map_zero]

theorem torsion_cardinality_eq_complex (n : ℕ) (hn : n ≠ 0) :
    Nat.card (Pic0.torsion (K := Qbar) (F := FQ) n) =
      Nat.card (Pic0.torsion (K := ℂ) (F := FC) n) := by
  let f : Pic0.torsion (K := Qbar) (F := FQ) n → Pic0.torsion (K := ℂ) (F := FC) n :=
    fun y => ⟨complexPicardMap N y.val, complexPicardMap_mem_torsion N y.property⟩
  have hf : Function.Bijective f := by
    constructor
    · intro x y h
      exact Subtype.ext (complexPicardMap_injective N (congrArg Subtype.val h))
    · intro x
      obtain ⟨y, hy, he⟩ := exists_torsion_preimage N n hn x.val x.property
      exact ⟨⟨y, hy⟩, Subtype.ext he⟩
  exact Nat.card_congr (Equiv.ofBijective f hf)

theorem arithmetic_torsion_cardinality_period_rank (n : ℕ) (hn : n ≠ 0) :
    Nat.card (Pic0.torsion (K := Qbar) (F := FQ) n) = n ^ Module.finrank ℤ (sourceWeightTwoIntegralLattice (Gamma1 N)) := by
  rw [torsion_cardinality_eq_complex N n hn]
  exact complex_torsion_cardinality N n hn

end
end Dubon2026.SourceFunctionField.GammaOne.ConstantField
