import SourceGammaOneRealPeriodH1Hecke

/-! DUB17/18 -> original all-weight arithmetic realization -> full/parabolic
Gamma1 H1 comparison. The actual native real H1 classes whose original
characters vanish on every trace-square-four element are exactly the
range of the real cusp-period map. Existing real-period surjectivity
discharges the reverse inclusion. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open CongruenceSubgroup Matrix.SpecialLinearGroup groupCohomology CategoryTheory
open scoped MatrixGroups
variable (M : ℕ) [NeZero M]

/-- The actual parabolic submodule of native real Gamma1 H1. -/
def realParabolicH1 : Submodule ℝ (H1 (Rep.trivial ℝ (Gamma1 M) ℝ)) :=
  (sourceGamma1RealParabolicHoms M).comap
    (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom.hom

theorem realPeriodH1_mem_parabolic
    (f : CuspForm ((Gamma1 M).map (mapGL ℝ)) 2) :
    realPeriodH1 M f ∈ realParabolicH1 M := by
  intro γ hγ
  change (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom
    (realPeriodH1 M f) (Additive.ofMul γ) = 0
  rw [realPeriodH1_hom_apply, sourceGamma1_period_eq_zero_of_trace M f γ hγ,
    Complex.zero_re]

theorem realPeriodH1_range :
    LinearMap.range (realPeriodH1 M) = realParabolicH1 M := by
  ext x
  constructor
  · rintro ⟨f, rfl⟩
    exact realPeriodH1_mem_parabolic M f
  · intro hx
    let φ : sourceGamma1RealParabolicHoms M :=
      ⟨(H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom x, hx⟩
    obtain ⟨f, hf⟩ := sourceGamma1RealParabolicPeriodHom_surjective M φ
    refine ⟨f, ?_⟩
    apply (ConcreteCategory.bijective_of_isIso
      (H1IsoOfIsTrivial (Rep.trivial ℝ (Gamma1 M) ℝ)).hom).injective
    apply AddMonoidHom.ext
    intro γ
    have hγ : γ = Additive.ofMul γ.toMul := rfl
    rw [hγ, realPeriodH1_hom_apply]
    exact congrArg (fun ψ : sourceGamma1RealParabolicHoms M =>
      ψ.val (Additive.ofMul γ.toMul)) hf

theorem realPeriodH1_injective : Function.Injective (realPeriodH1 M) := by
  intro f g h
  apply sourceGamma1RealPeriodHom_injective M
  apply AddMonoidHom.ext
  intro γ
  have hγ : γ = Additive.ofMul γ.toMul := rfl
  rw [hγ, sourceGamma1RealPeriodHom_apply, sourceGamma1RealPeriodHom_apply]
  rw [← realPeriodH1_hom_apply M f γ.toMul, ← realPeriodH1_hom_apply M g γ.toMul, h]

end
end Dubon2026.SourceFunctionField.GammaOne
