/-- The original primitive cusp form constructs its actual continuous rank-two
representation over a genuine finite local coefficient field, with exact
original norm-Frobenius trace and determinant and actual complex oddness. -/
theorem exists_gammaOne_original_continuous_rankTwo
    (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k) (hlarge : 3 < p) :
    letI := sourcePrimitiveCoefficientField_numberField f hk
    letI := rationalPrimeLocalField_charZero p (Fact.out : p.Prime)
    let F := rationalPrimeLocalField p (Fact.out : p.Prime)
    let L := commonCoefficientField (sourcePrimitiveCoefficientField f) F
    ∃ (u : UniformSpace L) (_ : ValuativeRel L),
      letI : TopologicalSpace L := u.toTopologicalSpace
      IsUniformAddGroup L ∧ IsNonarchimedeanLocalField L ∧
      (ValuativeRel.valuation F).HasExtension (ValuativeRel.valuation L) ∧
      FiniteDimensional F L ∧ CharP 𝓀[L] p ∧
      ∃ τ : sourcePrimitiveCoefficientField f →ₐ[ℚ] L,
      ∃ ρ : (Qbar ≃ₐ[ℚ] Qbar) →* Matrix.GeneralLinearGroup (Fin 2) L,
        Continuous ρ ∧
        (∀ (q : ℕ) (hq : q.Prime), ¬q ∣ Q * p →
          ∀ (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
            (φ : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq)),
            (∀ z : IsLocalRing.ResidueField (algebraicLocalIntegerRing
              (rationalPrimeLocalClosure q hq)), φ • z = z ^ q) →
            Matrix.trace (ρ (embeddedNormalGaloisRestriction e φ)).val =
              τ (sourcePrimitiveFieldCoefficient f q) ∧
            (ρ (embeddedNormalGaloisRestriction e φ)).val.det = (q : L) ^ Int.toNat (k - 1)) ∧
        ∀ e : Qbar →ₐ[ℚ] ℂ,
          (ρ (sourceAbsoluteComplexConjugation e)).val.det = -1 ∧
          Matrix.trace (ρ (sourceAbsoluteComplexConjugation e)).val = 0 := by
  letI := sourcePrimitiveCoefficientField_numberField f hk
  letI := rationalPrimeLocalField_charZero p (Fact.out : p.Prime)
  let F := rationalPrimeLocalField p (Fact.out : p.Prime)
  let L := commonCoefficientField (sourcePrimitiveCoefficientField f) F
  have hs := exists_gammaOne_original_spanning_odd_trace_algebra Q p f hk hlarge
  dsimp only at hs
  obtain ⟨u, v, hu, hv, hext, hfin, hchar, τ, t, D, ht, hD, hOne, hCyc, hProd,
    hFrob, hOdd, A, rA, aA, T, σ, hTOne, hTσ, hTCyc, hNondeg, hPol, hCorner, hspan⟩ := hs
  letI : UniformSpace L := u
  letI : TopologicalSpace L := u.toTopologicalSpace
  letI : ValuativeRel L := v
  letI := hu
  letI := hv
  letI := rA
  letI := aA
  let α : (Qbar ≃ₐ[ℚ] Qbar) →* A := (Units.coeHom A).comp σ
  have htα : Continuous (fun g => T (α g)) := ht.congr (fun g => (hTσ g).symm)
  let ec : Qbar →ₐ[ℚ] ℂ := IsAlgClosed.lift
  let c := sourceAbsoluteComplexConjugation ec
  have hc : α c * α c = 1 := by
    rw [← map_mul, sourceAbsoluteComplexConjugation_sq, map_one]
  have htc : T (α c) = 0 := (hTσ c).trans (hOdd ec).2
  obtain ⟨ρ, hρ, htrace⟩ := exists_continuous_matrix_of_odd_trace_algebra α T hspan htα
    hTOne hTCyc hNondeg hPol (α c) hc htc
  have htr : ∀ g, Matrix.trace (ρ g).val = t g := fun g => (htrace g).trans (hTσ g)
  have hdet := matrixTwoRepresentation_det_of_trace ρ t D hOne hProd htr
  refine ⟨u, v, hu, hv, hext, hfin, hchar, τ, ρ, hρ, ?_, ?_⟩
  · intro q hq ha e φ hφ
    exact ⟨(htr _).trans (hFrob q hq ha e φ hφ).1,
      (hdet _).trans (hFrob q hq ha e φ hφ).2⟩
  · intro e
    exact ⟨(hdet _).trans (hOdd e).1, (htr _).trans (hOdd e).2⟩

end
