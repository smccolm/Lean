/-- The original norm inertia outside Q*p also kills the actual finite
joint determinant term, using the existing cyclotomic inertia theorem. -/
theorem binaryPrimitiveJointDeterminant_normInertia
    (q : ℕ) (hq : q.Prime) (haway : ¬q ∣ Q * p)
    (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
    (g : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq))
    (hg : g ∈ rationalPrimeNormInertia q hq) :
    binaryPrimitiveJointDeterminant Q p n w (embeddedNormalGaloisRestriction e g) = 1 := by
  have hM : ¬q ∣ M := by
    intro h
    rcases hq.dvd_mul.mp h with hQ | hp
    · exact haway (dvd_mul_of_dvd_left hQ p)
    · exact haway (dvd_mul_of_dvd_right (hq.dvd_of_dvd_pow (hq.dvd_of_dvd_pow hp)) Q)
  change binaryPrimitiveJointWeightedDiamond Q p n w
    (sourceAbsoluteCyclotomicCharacter M (embeddedNormalGaloisRestriction e g)) = 1
  rw [sourceAbsoluteCyclotomicCharacter_normInertia M q hq hM e g hg, map_one]

/-- Every actual specialized trace translate is invariant under original
norm inertia outside Q*p. Both action and determinant invariance are proved. -/
theorem binaryPrimitiveJointSpecializedTrace_normInertia_translate
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    {C : Type} [Ring C] (χ : (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq).range →+* C)
    (q : ℕ) (hq : q.Prime) (haway : ¬q ∣ Q * p)
    (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
    (g : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq))
    (hg : g ∈ rationalPrimeNormInertia q hq) (σ : Qbar ≃ₐ[ℚ] Qbar) :
    binaryPrimitiveJointSpecializedTrace Q p n w Hq hH χ
        (embeddedNormalGaloisRestriction e g * σ) =
      binaryPrimitiveJointSpecializedTrace Q p n w Hq hH χ σ := by
  apply congrArg χ
  apply Subtype.ext
  exact sourceUnitTrace_kernel_translate (binaryPrimitiveJointGaloisRep Q p n w)
    (binaryPrimitiveJointDeterminant Q p n w) (embeddedNormalGaloisRestriction e g)
    (binaryPrimitiveJointGaloisRep_normInertia Q p n w q hq haway e g hg)
    (binaryPrimitiveJointDeterminant_normInertia Q p n w q hq haway e g hg) σ

end
