private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceAdicPrimePowerScalar, `Dubon2026.SourceCyclotomicNormFrobenius, `Dubon2026.SourceAbsoluteComplexConjugation, `Dubon2026.SourceAdicCyclotomicDeterminant, `Dubon2026.SourceTraceLeftIdealBasis, `Dubon2026.SourceTraceLeftActionTrace, `Dubon2026.SourceSpanningTraceContinuity, `Dubon2026.SourceTraceLeftActionContinuity, `Dubon2026.SourceScalarCornerCharacter, `Dubon2026.SourceTraceDiagonalRepresentation, `Dubon2026.SourceOddTraceIdempotent, `Dubon2026.SourceContinuousTraceRealization, `Dubon2026.SourceMatrixTraceDeterminant, `Dubon2026.SourceOriginalTraceMatrixConsumer, `Dubon2026.SourceGammaOneLocalRankTwo, `Dubon2026.SourceGammaOneContinuousRankTwo, `Dubon2026.SourceCyclotomicNormInertia, `Dubon2026.SourceUnitTraceKernel, `Dubon2026.SourceGammaOneJointTraceInertia, `Dubon2026.SourceGammaOneOriginalAdicInertia, `Dubon2026.SourceGammaOneLocalTraceInertia, `Dubon2026.SourceTraceDiagonalKernel, `Dubon2026.SourceContinuousTraceKernel, `Dubon2026.SourceTraceActionKernel, `Dubon2026.SourceOriginalTraceMatrixKernel, `Dubon2026.SourceGammaOneLocalUnramifiedRankTwo, `Dubon2026.SourceGammaOneUnramifiedRankTwo, `Dubon2026.SourceGammaOneJointArithmeticKernel, `Dubon2026.SourceGammaOneOriginalAdicArithmetic, `Dubon2026.SourceGammaOneLocalTraceArithmetic, `Dubon2026.SourceAbsoluteArithmeticFactor, `Dubon2026.SourceGammaOneLocalArithmeticRankTwo, `Dubon2026.SourceGammaOneArithmeticRankTwo, `Dubon2026.CompactOrbitSpanFinite, `Dubon2026.SourceIntegralCoordinateLattice, `Dubon2026.SourceCompactOrbitLattice, `Dubon2026.SourceOriginalResidualLocalInterface, `Dubon2026.SourceGammaOneFixedLocalResidual, `Dubon2026.SourceGammaOneFixedDeterminantResidual]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
