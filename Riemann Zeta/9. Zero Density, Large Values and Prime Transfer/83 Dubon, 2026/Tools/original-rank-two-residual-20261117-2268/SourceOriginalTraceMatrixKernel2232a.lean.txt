noncomputable section
variable {G K : Type} [Group G] [TopologicalSpace G] [IsTopologicalGroup G]
  [Field K] [CharZero K] [TopologicalSpace K] [IsTopologicalRing K]

/-- The original continuous trace, odd group element and constructed
spanning trace algebra give actual continuous matrices with both original
trace and determinant. This is consumed by the original cusp-form theorem. -/
theorem exists_continuous_matrix_of_original_trace_kernel
    (t : G → K) (D : G →* K) (ht : Continuous t) (hOne : t 1 = 2)
    (hProd : ∀ g h, t g * t h = t (g * h) + t (g * h⁻¹) * D h)
    (c : G) (hc : c * c = 1) (htc : t c = 0)
    (hAlgebra : ∃ (A : Type) (_ : Ring A) (_ : Algebra K A) (T : A →ₗ[K] K)
      (σ : G →* Aˣ), T 1 = 2 ∧ (∀ g, T (σ g : A) = t g) ∧
      (∀ x y, T (x * y) = T (y * x)) ∧
      (∀ x, (∀ y, T (x * y) = 0) → x = 0) ∧
      (∀ x y, x * y + y * x - T x • y - T y • x +
        (T x * T y - T (x * y)) • (1 : A) = 0) ∧
      (∀ e : A, e * e = e → T e = 1 → ∀ x, e * x * e = T (e * x) • e) ∧
      Submodule.span K (Set.range (fun g => (σ g : A))) = ⊤) :
    ∃ ρ : G →* Matrix.GeneralLinearGroup (Fin 2) K,
      Continuous ρ ∧ (∀ g, Matrix.trace (ρ g).val = t g ∧ (ρ g).val.det = D g) ∧
      ∀ n, (∀ g, t (n * g) = t g) → ρ n = 1 := by
  obtain ⟨A, rA, aA, T, σ, hTOne, hTσ, hTCyc, hNondeg, hPol, _, hspan⟩ := hAlgebra
  letI := rA
  letI := aA
  let α : G →* A := (Units.coeHom A).comp σ
  have htα : Continuous (fun g => T (α g)) := ht.congr (fun g => (hTσ g).symm)
  have hcα : α c * α c = 1 := by rw [← map_mul, hc, map_one]
  obtain ⟨ρ, hρ, htrace, hker⟩ := exists_continuous_matrix_of_odd_trace_algebra_kernel α T hspan htα
    hTOne hTCyc hNondeg hPol (α c) hcα ((hTσ c).trans htc)
  have htr : ∀ g, Matrix.trace (ρ g).val = t g := fun g => (htrace g).trans (hTσ g)
  refine ⟨ρ, hρ, (fun g => ⟨htr g, matrixTwoRepresentation_det_of_trace ρ t D hOne hProd htr g⟩), ?_⟩
  intro n hn
  apply hker n
  exact sourceTraceAction_eq_one_of_translate α T hspan hNondeg n
    (fun g => (hTσ (n * g)).trans ((hn g).trans (hTσ g).symm))

end
end Dubon2026
