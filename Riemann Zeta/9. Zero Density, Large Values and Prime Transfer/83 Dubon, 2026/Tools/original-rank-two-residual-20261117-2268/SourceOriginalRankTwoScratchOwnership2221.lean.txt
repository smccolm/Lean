open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceAdicPrimePowerScalar, `SourceCyclotomicNormFrobenius, `SourceAbsoluteComplexConjugation, `SourceAdicCyclotomicDeterminant, `SourceOriginalAdicDeterminant, `SourceAdicQuotientContinuity, `SourceOriginalLocalDeterminant, `SourcePrimitiveLocalOddness, `SourceGroupAlgebraQuotientSpan, `SourceSpanningTraceAlgebra, `SourceTraceOffDiagonal, `SourceTraceOppositePair, `SourceTraceLeftIdealCoordinates, `SourceTraceLeftIdealBasis, `SourceTraceLeftActionTrace, `SourceSpanningTraceContinuity, `SourceTraceLeftActionContinuity, `SourceScalarCornerCharacter, `SourceTraceDiagonalRepresentation, `SourceOddTraceIdempotent, `SourceContinuousTraceRealization, `SourceMatrixTraceDeterminant, `SourceOriginalTraceMatrixConsumer, `SourceGammaOneLocalRankTwo, `SourceGammaOneContinuousRankTwo]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
