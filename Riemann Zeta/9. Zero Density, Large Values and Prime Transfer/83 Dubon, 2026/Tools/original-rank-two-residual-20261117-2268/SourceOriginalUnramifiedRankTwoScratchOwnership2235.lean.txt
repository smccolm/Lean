import SourceCyclotomicNormInertia
import SourceUnitTraceKernel
import SourceGammaOneJointTraceInertia
import SourceGammaOneOriginalAdicInertia
import SourceGammaOneLocalTraceInertia
import SourceTraceDiagonalKernel
import SourceContinuousTraceKernel
import SourceTraceActionKernel
import SourceOriginalTraceMatrixKernel
import SourceGammaOneLocalUnramifiedRankTwo
import SourceGammaOneUnramifiedRankTwo
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceCyclotomicNormInertia, `SourceUnitTraceKernel, `SourceGammaOneJointTraceInertia, `SourceGammaOneOriginalAdicInertia, `SourceGammaOneLocalTraceInertia, `SourceTraceDiagonalKernel, `SourceContinuousTraceKernel, `SourceTraceActionKernel, `SourceOriginalTraceMatrixKernel, `SourceGammaOneLocalUnramifiedRankTwo, `SourceGammaOneUnramifiedRankTwo]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
