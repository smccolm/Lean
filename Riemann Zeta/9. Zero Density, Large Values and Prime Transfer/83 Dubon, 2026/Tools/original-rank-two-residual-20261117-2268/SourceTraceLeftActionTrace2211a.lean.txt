import SourceTraceLeftIdealBasis2203b
import Mathlib.LinearAlgebra.Matrix.Trace

/-! DUB-17/18 -> actual rank-two realization -> exact original matrix trace.
The native matrix of the original left action in the proved two-vector
basis has exactly the original algebra trace. This retains the original
Fourier values once applied to the original Galois action. -/
namespace Dubon2026
noncomputable section
open Module
variable {K A : Type*} [Field K] [Ring A] [Algebra K A]

/-- The native matrix of the original left action has the original trace,
using the actual complementary products of the constructed opposite pair. -/
theorem traceIdempotentPair_leftAction_trace (T : A →ₗ[K] K)
    (hCyc : ∀ x y, T (x * y) = T (y * x))
    (hPol : ∀ x y, x * y + y * x - T x • y - T y • x +
      (T x * T y - T (x * y)) • (1 : A) = 0)
    (e b c : A) (he : e * e = e) (hte : T e = 1)
    (heb : e * b = b) (hec : e * c = 0) (hce : c * e = c)
    (hbe : b * e = 0) (hbc : b * c = e) (hcb : c * b = 1 - e) (a : A) :
    let B := traceIdempotentPairBasis T hCyc hPol e b c he hte heb hec hce hbe hbc hcb
    Matrix.trace (LinearMap.toMatrix B B (algebraIdempotentLeftAction e a)) = T a := by
  let B := traceIdempotentPairBasis T hCyc hPol e b c he hte heb hec hce hbe hbc hcb
  change Matrix.trace (LinearMap.toMatrix B B (algebraIdempotentLeftAction e a)) = T a
  have h0 : (B 0).val = e := by
    simp [B, traceIdempotentPairBasis, Basis.coe_ofEquivFun, traceIdempotentPairEquiv]
  have h1 : (B 1).val = c := by
    simp [B, traceIdempotentPairBasis, Basis.coe_ofEquivFun, traceIdempotentPairEquiv]
  have hr0 (z : algebraIdempotentLeftIdeal (S := K) e) : B.repr z 0 = T z.val := rfl
  have hr1 (z : algebraIdempotentLeftIdeal (S := K) e) : B.repr z 1 = T (b * z.val) := rfl
  rw [Matrix.trace_fin_two, LinearMap.toMatrix_apply, LinearMap.toMatrix_apply,
    hr0, hr1, algebraIdempotentLeftAction_val, algebraIdempotentLeftAction_val, h0, h1]
  rw [hCyc b (a * c), mul_assoc a c b, hcb, mul_sub, mul_one, map_sub]
  abel

end
end Dubon2026
