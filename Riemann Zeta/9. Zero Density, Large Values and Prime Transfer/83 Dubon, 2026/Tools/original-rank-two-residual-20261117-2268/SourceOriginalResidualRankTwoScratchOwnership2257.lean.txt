import CompactOrbitSpanFinite
import SourceIntegralCoordinateLattice
import SourceCompactOrbitLattice
import SourceMatrixIntertwiningInvariants
import SourceCompactResidualRealization
import SourceGammaOneLocalResidualRankTwo
import SourceGammaOneResidualRankTwo
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`CompactOrbitSpanFinite, `SourceIntegralCoordinateLattice, `SourceCompactOrbitLattice, `SourceMatrixIntertwiningInvariants, `SourceCompactResidualRealization, `SourceGammaOneLocalResidualRankTwo, `SourceGammaOneResidualRankTwo]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
