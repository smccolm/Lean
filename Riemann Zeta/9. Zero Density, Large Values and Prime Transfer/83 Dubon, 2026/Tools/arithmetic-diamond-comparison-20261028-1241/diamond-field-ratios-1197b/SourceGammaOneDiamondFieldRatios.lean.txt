/-! DUB-17/18 -> original all-weight association -> arithmetic Gamma1
diamond/Picard comparison -> diamond on the original complex function field.
Reuse the original equal-weight ratio theorem and native modular translation.
The translated ratio stays in the same field and gives the exact action at

variable (M : ℕ) (g : Gamma0 M)

/-- The native slash diamond on actual modular forms of the original level. -/
def modularDiamond {k : ℤ} (f : ModularForm (Gamma1 M) k) :
    ModularForm (Gamma1 M) k :=
  ModularNormPolynomial.restrict (diamondGL_le_conj M g)
    (ModularForm.translate f (mapGL ℝ g.val))

theorem modularDiamond_coe {k : ℤ} (f : ModularForm (Gamma1 M) k) :
    ⇑(modularDiamond M g f) = ⇑f ∣[k] mapGL ℝ g.val := rfl

theorem modularDiamond_ne_zero {k : ℤ} (f : ModularForm (Gamma1 M) k)
    (hf : f ≠ 0) : modularDiamond M g f ≠ 0 := by
  intro he
  have hh := congrArg (fun F : ModularForm (Gamma1 M) k => (F : ℍ → ℂ)) he
  change ⇑(modularDiamond M g f) = (0 : ℍ → ℂ) at hh
  rw [modularDiamond_coe, SlashAction.slash_eq_zero_iff] at hh
  exact hf (DFunLike.coe_injective hh)

theorem modularDiamond_translate_coe {k : ℤ} (f : ModularForm (Gamma1 M) k)
    (δ : SL(2, ℤ)) :
    (ModularForm.translate (modularDiamond M g f) (δ : GL (Fin 2) ℝ) : ℍ → ℂ) =
      (ModularForm.translate f ((g.val * δ : SL(2, ℤ)) : GL (Fin 2) ℝ) : ℍ → ℂ) := by
  change (⇑f ∣[k] mapGL ℝ g.val) ∣[k] mapGL ℝ δ =
    ⇑f ∣[k] mapGL ℝ (g.val * δ)
  rw [map_mul, SlashAction.slash_mul]

variable [NeZero M]

theorem exists_diamond_field_cusp_realization (x : functionFieldOver (Gamma1 M) ℂ) :
    ∃ y : functionFieldOver (Gamma1 M) ℂ, ∀ δ : SL(2, ℤ),
      cuspRealize (Gamma1 M) M δ (y : LaurentSeries ℂ) =ᶠ[𝓝[≠] (0 : ℂ)]
        cuspRealize (Gamma1 M) M (g.val * δ) (x : LaurentSeries ℂ) := by
  have hT : ModularGroup.T ∈ Gamma1 M := by simp [Gamma1_mem, ModularGroup.T]
  have hΓ : Gamma M ≤ Gamma1 M := by
    rw [Gamma_eq_Gamma1_inf_Gamma0]
    exact inf_le_left
  obtain ⟨k, f, h, hh, hx⟩ := functionFieldOver_modular_ratio hT x
  let F := modularDiamond M g f
  let H := modularDiamond M g h
  have hH : H ≠ 0 := modularDiamond_ne_zero M g h hh
  let y : functionFieldOver (Gamma1 M) ℂ :=
    ⟨qL F / qL H, modular_ratio_mem M F H hH⟩
  refine ⟨y, fun δ => ?_⟩
  have hy : (y : LaurentSeries ℂ) * qL H = qL F :=
    div_mul_cancel₀ _ (qL_ne_zero hT H hH)
  filter_upwards [cuspRealize_eventuallyEq_of_ratio M hΓ hT F H y hH hy δ,
    cuspRealize_eventuallyEq_of_ratio M hΓ hT f h x hh hx (g.val * δ)] with q h1 h2
  rw [h1, h2]
  change cuspFunction M (ModularForm.translate F (δ : GL (Fin 2) ℝ)) q /
      cuspFunction M (ModularForm.translate H (δ : GL (Fin 2) ℝ)) q = _
  rw [show (ModularForm.translate F (δ : GL (Fin 2) ℝ) : ℍ → ℂ) =
      (ModularForm.translate f ((g.val * δ : SL(2, ℤ)) : GL (Fin 2) ℝ) : ℍ → ℂ)
      from modularDiamond_translate_coe M g f δ,
    show (ModularForm.translate H (δ : GL (Fin 2) ℝ) : ℍ → ℂ) =
      (ModularForm.translate h ((g.val * δ : SL(2, ℤ)) : GL (Fin 2) ℝ) : ℍ → ℂ)
      from modularDiamond_translate_coe M g h δ]
  rfl

end
