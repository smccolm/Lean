private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneDiamondCoefficient, `SourceGammaOneDiamondRestriction, `SourceGammaOneDiamondCoefficientMatrix, `SourceGammaOneBinaryNormalizerTate, `SourceGammaOneBinaryDiamondLabels]
  let theorems : Array Name := #[`Dubon2026.sourceGammaOneDiamondLift_gammaOne, `Dubon2026.sourceGammaOneDiamondLift_conjugation, `Dubon2026.sourceGammaOneBinaryDiamondCoefficient_conjugation, `Dubon2026.sourceGammaOneBinaryRestrictionCocycles_diamond, `Dubon2026.sourceGammaOneBinaryRestrictionH1_diamond, `Dubon2026.sourceGammaOneDiamondLift_mod, `Dubon2026.sourceGammaOneBinaryDiamondCoefficient_eq_diagonal, `Dubon2026.SourceFunctionField.GammaOne.binaryTateH1Equiv_normalizerDiamond, `Dubon2026.SourceFunctionField.GammaOne.binaryTate_normalizerDiamond_galois, `Dubon2026.sourceGammaOneBinaryDiamondCoefficient_lift_independent, `Dubon2026.sourceGammaOneBinaryDiamondH1_lift_independent]
  let constructions : Array Name := #[`Dubon2026.sourceGammaOneDiamondLift, `Dubon2026.sourceGammaOneBinaryDiamondCoefficient]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Diamond Period postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Diamond Period theorems discovered"
  logInfo m!"SOURCE DIAMOND NORMALIZER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
