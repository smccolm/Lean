import SourceGammaOneDiamondCoefficient
import Dubon2026.HomogeneousDiagonalAction

/-! DUB-17/18 -> original all-weight association -> original finite
Gamma1 arithmetic realization -> homogeneous diamond action.
At the exact coefficient modulus, the original raised normalizer is
diagonal. Its inverse therefore acts by the original homogeneous
substitution with the diagonal entries reversed. The divisibility
uses the actual level Q*N^2, with no weight or prime exclusions.
-/
namespace Dubon2026
noncomputable section
open CongruenceSubgroup Matrix.SpecialLinearGroup
open scoped MatrixGroups
variable (Q N w : ℕ) [NeZero N]

theorem sourceGammaOneDiamondLift_mod (g : Gamma0 (Q * N ^ 2)) :
    (mapGL (ZMod N) (sourceGammaOneDiamondLift Q N g).val).val =
      Matrix.diagonal ![(g.val 0 0 : ZMod N), (g.val 1 1 : ZMod N)] := by
  obtain ⟨t, ht⟩ := (ZMod.intCast_zmod_eq_zero_iff_dvd (g.val 1 0) (Q * N ^ 2)).mp
    (Gamma0_mem.mp g.property)
  have he : g.val 1 0 = (N : ℤ) * (N * (Q * t)) := by
    rw [ht]
    push_cast
    ring
  have hc : ((g.val 1 0 / (N : ℤ) : ℤ) : ZMod N) = 0 := by
    rw [he, Int.mul_ediv_cancel_left _ (Int.natCast_ne_zero.mpr (NeZero.ne N))]
    push_cast
    simp
  have hm : (sourceGammaOneDiamondLift Q N g).val.val =
      !![g.val 0 0, (N : ℤ) * g.val 0 1; g.val 1 0 / (N : ℤ), g.val 1 1] := rfl
  rw [mapGL_coe_matrix]
  change (sourceGammaOneDiamondLift Q N g).val.val.map (algebraMap ℤ (ZMod N)) = _
  rw [hm]
  ext i j
  fin_cases i <;> fin_cases j <;> simp [hc]

theorem sourceGammaOneBinaryDiamondCoefficient_eq_diagonal (g : Gamma0 (Q * N ^ 2)) :
    sourceGammaOneBinaryDiamondCoefficient Q N w g =
      homogeneousMatrixAction w
        (Matrix.diagonal ![(g.val 1 1 : ZMod N), (g.val 0 0 : ZMod N)]) := by
  change homogeneousMatrixAction w
    (mapGL (ZMod N) ((sourceGammaOneDiamondLift Q N g)⁻¹).val).val = _
  rw [← map_inv, sourceGammaOneDiamondLift_mod]
  congr 2
  simp [Matrix.SpecialLinearGroup.coe_inv, Matrix.adjugate_fin_two]

end
end Dubon2026
