/-! DUB-17/18 -> original all-weight association -> arithmetic Gamma1
diamond/Picard comparison. Define both genuine Picard diamonds from the
proved original arithmetic and complex field automorphisms. Native closed
residue fields discharge degree preservation, and original order transport
preserves principal divisors. Periods are not used in either definition.
Comparison with the proved period operator and Galois compatibility remain.
-/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open CongruenceSubgroup SourceCurve
local notation "Qbar" => AlgebraicClosure ℚ
variable (M : ℕ) [NeZero M] (g : Gamma0 M)
local notation "FQ" => functionFieldOver (Gamma1 M) Qbar
local notation "FC" => functionFieldOver (Gamma1 M) ℂ

/-- The arithmetic Picard diamond induced by the original arithmetic function field. -/
def arithmeticDiamondPicard : Pic0 Qbar FQ ≃+ Pic0 Qbar FQ :=
  fieldPicardEquiv (ConstantField.arithmeticDiamondAut M g).symm
    (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T]))
    (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T]))

/-- The complex Picard diamond induced by the original complex function field. -/
def complexDiamondPicard : Pic0 ℂ FC ≃+ Pic0 ℂ FC :=
  fieldPicardEquiv (diamondFieldAut M g).symm
    (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T]))
    (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T]))

theorem arithmeticDiamondPicard_mk (D : Divisor.degZero (K := Qbar) (F := FQ)) :
    arithmeticDiamondPicard M g (Pic0.mk D) =
      Pic0.mk (fieldDegZeroEquiv (ConstantField.arithmeticDiamondAut M g).symm
        (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T]))
        (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T])) D) := rfl

theorem complexDiamondPicard_mk (D : Divisor.degZero (K := ℂ) (F := FC)) :
    complexDiamondPicard M g (Pic0.mk D) =
      Pic0.mk (fieldDegZeroEquiv (diamondFieldAut M g).symm
        (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T]))
        (functionFieldOver_residue_bijective (Gamma1 M) (by simp [Gamma1_mem, ModularGroup.T])) D) := rfl

theorem arithmeticDiamondPicard_torsion (n : ℕ) (x : Pic0 Qbar FQ)
    (hx : x ∈ Pic0.torsion n) : arithmeticDiamondPicard M g x ∈ Pic0.torsion n :=
  fieldPicardEquiv_torsion _ _ _ n x hx

theorem complexDiamondPicard_torsion (n : ℕ) (x : Pic0 ℂ FC)
    (hx : x ∈ Pic0.torsion n) : complexDiamondPicard M g x ∈ Pic0.torsion n :=
  fieldPicardEquiv_torsion _ _ _ n x hx

theorem arithmeticDiamondHom_lift_independent (h : Gamma0 M)
    (hlabel : Gamma0Map M g = Gamma0Map M h) (x : FQ) :
    ConstantField.arithmeticDiamondHom M g x = ConstantField.arithmeticDiamondHom M h x := by
  apply (ConstantField.fieldMap M).injective
  rw [ConstantField.fieldMap_arithmeticDiamondHom, ConstantField.fieldMap_arithmeticDiamondHom]
  exact diamondFieldHom_lift_independent M g h hlabel _

end
end Dubon2026.SourceFunctionField.GammaOne
