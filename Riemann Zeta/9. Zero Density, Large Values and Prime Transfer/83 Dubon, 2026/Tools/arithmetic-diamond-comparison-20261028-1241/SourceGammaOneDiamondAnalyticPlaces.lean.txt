variable (M : ℕ) [NeZero M] (g : Gamma0 M)

theorem analyticOrder_diamondFieldHom (τ : ℍ) (x : functionFieldOver (Gamma1 M) ℂ) :
    analyticOrder (Gamma1 M) τ (diamondFieldHom M g x) =
      analyticOrder (Gamma1 M) (g.val • τ) x := by
  unfold analyticOrder
  rw [meromorphicOrderAt_congr (realize_diamondFieldHom_eventuallyEq M g x τ)]
  have hd : 0 < (g.val : GL (Fin 2) ℝ).val.det := by
    have he : (g.val : GL (Fin 2) ℝ).val.det = 1 := Matrix.SpecialLinearGroup.det_coe _
    rw [he]
    exact one_pos
  simpa using (meromorphicOrderAt_comp_smul
    (f := fun z : ℍ => realize (Gamma1 M) (x : LaurentSeries ℂ) z) (τ := τ) hd)

theorem analyticPlace_comap_diamondFieldAut (hT : ModularGroup.T ∈ Gamma1 M) (τ : ℍ) :
    (analyticPlace hT τ).comapEquiv (diamondFieldAut M g) =
      analyticPlace hT (g.val • τ) := by
  apply Place.ext
  ext x
  rw [Place.mem_comapEquiv_iff, analyticPlace_mem_iff, analyticPlace_mem_iff]
  change 0 ≤ analyticOrder (Gamma1 M) τ (diamondFieldHom M g x) ↔ _
  rw [analyticOrder_diamondFieldHom]

end
