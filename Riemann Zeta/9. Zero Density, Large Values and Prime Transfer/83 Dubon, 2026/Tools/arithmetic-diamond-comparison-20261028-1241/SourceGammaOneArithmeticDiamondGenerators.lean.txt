import SourceGammaOneArithmeticDiamond

/-! DUB-17/18 -> original all-weight association -> arithmetic Gamma1
diamond Galois compatibility. The fixed original Qbar-to-C embedding
and the proved literal field square pull rationality of diamond images
back to the original arithmetic generators. No extension of a Galois
automorphism to C is required.
-/
namespace Dubon2026.SourceFunctionField.GammaOne.ConstantField
noncomputable section
open CongruenceSubgroup
local notation "Qbar" => AlgebraicClosure ℚ
variable (M : ℕ) [NeZero M] (g : Gamma0 M) (σ : Qbar ≃ₐ[ℚ] Qbar)
local notation "FQ" => functionFieldOver (Gamma1 M) Qbar

theorem arithmeticDiamond_generator_fixed (x : FQ)
    (hx : (x : LaurentSeries Qbar) ∈ intFormRatiosC Qbar (Gamma1 M)) :
    arithmeticRingAut (Gamma1 M) σ x = x ∧
      arithmeticRingAut (Gamma1 M) σ (arithmeticDiamondHom M g x) =
        arithmeticDiamondHom M g x := by
  obtain ⟨y, hygen, hy⟩ := (coeffEmb_image_intFormRatiosC (Gamma1 M) Qbar).symm ▸ hx
  have hxC : (fieldMap M x : LaurentSeries ℂ) = coeffEmb ℂ y := by
    change coeffMap OriginalConstantField.sigma.toRingHom (x : LaurentSeries Qbar) = _
    rw [← hy]
    exact OriginalConstantField.coeffMap_sigma_rational y
  have hxgenC : (fieldMap M x : LaurentSeries ℂ) ∈ intFormRatiosC ℂ (Gamma1 M) := by
    rw [hxC, ← coeffEmb_image_intFormRatiosC]
    exact ⟨y, hygen, rfl⟩
  constructor
  · apply Subtype.ext
    ext n
    change σ ((x : LaurentSeries Qbar).coeff n) = (x : LaurentSeries Qbar).coeff n
    rw [← hy, coeffEmb_coeff]
    exact σ.commutes _
  · apply Subtype.ext
    ext n
    change σ ((arithmeticDiamondHom M g x : LaurentSeries Qbar).coeff n) =
      (arithmeticDiamondHom M g x : LaurentSeries Qbar).coeff n
    obtain ⟨a, ha⟩ := diamondFieldHom_generator_coeff_rational M g (fieldMap M x) hxgenC n
    have he := congrArg (fun z : functionFieldOver (Gamma1 M) ℂ => (z : LaurentSeries ℂ).coeff n)
      (fieldMap_arithmeticDiamondHom M g x)
    have hq : (arithmeticDiamondHom M g x : LaurentSeries Qbar).coeff n =
        algebraMap ℚ Qbar a := by
      apply OriginalConstantField.sigma.injective
      exact he.trans (ha.symm.trans (OriginalConstantField.sigma.commutes a).symm)
    rw [hq]
    exact σ.commutes a

end
end Dubon2026.SourceFunctionField.GammaOne.ConstantField
