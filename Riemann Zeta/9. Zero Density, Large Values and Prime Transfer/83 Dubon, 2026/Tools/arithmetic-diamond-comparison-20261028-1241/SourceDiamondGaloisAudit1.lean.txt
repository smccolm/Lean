private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneDiamondPeriodInclusion, `SourceGammaOneBinaryDiamondTateH1, `SourceCyclotomicDiamondCovariance, `SourceRationalDiamondFraction, `SourceRationalCuspDiamondImage, `SourceRationalModularCuspQuotient, `SourceRationalDiamondQuotientGerm, `SourceRationalDiamondCoefficients, `SourceGammaOneDiamondRationalGenerators, `SourceGammaOneDiamondComplexArithmetic, `SourceGammaOneArithmeticDiamondGenerators, `SourceGammaOneArithmeticDiamondGalois, `SourceGammaOneDiamondGaloisPicard]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.arithmeticPeriodInclusion_diamond, `Dubon2026.SourceFunctionField.GammaOne.binaryTateCharacterEquiv_diamond, `Dubon2026.SourceFunctionField.GammaOne.binaryTateH1Equiv_diamond, `Dubon2026.SourceFricke.exists_gammaZero_cyclotomic_same_label, `Dubon2026.SourceFricke.gammaOne_invariant_comp_eq_of_label, `Dubon2026.SourceFricke.CyclotomicConjugates.diamond, `Dubon2026.SourceFricke.cyclotomicConjugates_self_of_rational_rep, `Dubon2026.SourceFricke.exists_rational_laurent_diamond_fraction, `Dubon2026.SourceFricke.exists_rational_cuspWeightImage_diamond_laurent, `Dubon2026.SourceFunctionField.GammaOne.exists_rational_normalized_cusp_quotient, `Dubon2026.SourceFunctionField.GammaOne.exists_rational_laurent_translated_ratio, `Dubon2026.SourceFunctionField.GammaOne.modularDiamond_ratio_coeff_rational, `Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut_integral_generator, `Dubon2026.SourceFunctionField.GammaOne.diamondFieldHom_generator_coeff_rational, `Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut_diamond_generator, `Dubon2026.SourceFunctionField.GammaOne.diamondFieldHom_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.arithmeticDiamond_generator_fixed, `Dubon2026.SourceFunctionField.GammaOne.ConstantField.arithmeticDiamondHom_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_diamond, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisor_diamond, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDiamondPicard_galois, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDiamondTate_galois]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Diamond Period postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Diamond Period theorems discovered"
  logInfo m!"SOURCE DIAMOND GALOIS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
