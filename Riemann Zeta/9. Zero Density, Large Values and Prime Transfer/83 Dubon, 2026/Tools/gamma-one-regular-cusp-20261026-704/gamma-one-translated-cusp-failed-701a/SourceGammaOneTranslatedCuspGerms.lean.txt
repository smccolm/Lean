/-! DUB-17/18 -> actual Gamma1 regular-differential/cusp converse ->
vanishing at every cusp. Native translated modular forms and the proved
principal-congruence translation width supply analytic, nonzero cusp
denominator germs. These feed the actual cusp order map; no cusp chart
or field embedding is assumed. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane Filter
open scoped MatrixGroups ModularForm Topology Manifold
variable {Γ : Subgroup (Matrix.SpecialLinearGroup (Fin 2) ℤ)}

theorem translated_modular_ne_zero {k : ℤ}
    (f : ModularForm (Γ : Subgroup (GL (Fin 2) ℝ)) k) (hf : f ≠ 0)
    (γ : SL(2, ℤ)) : ModularForm.translate f (γ : GL (Fin 2) ℝ) ≠ 0 := by
  intro he
  apply hf
  apply DFunLike.ext
  intro τ
  have hz : ((f : ℍ → ℂ) ∣[k] γ) (γ⁻¹ • τ) = 0 := by
    change ModularForm.translate f (γ : GL (Fin 2) ℝ) (γ⁻¹ • τ) = 0
    rw [he]
    rfl
  rw [ModularForm.SL_slash_apply, smul_inv_smul] at hz
  exact (mul_eq_zero.mp hz).resolve_right (zpow_ne_zero _ (denom_ne_zero γ _))

theorem analyticAt_translated_cuspFunction (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) {k : ℤ}
    (f : ModularForm (Γ : Subgroup (GL (Fin 2) ℝ)) k) (γ : SL(2, ℤ)) :
    AnalyticAt ℂ (cuspFunction N (ModularForm.translate f (γ : GL (Fin 2) ℝ))) 0 := by
  have hp := CuspAuxiliary.natCast_mem_strictPeriods_conj N hΓ γ
  exact analyticAt_cuspFunction_zero (by exact_mod_cast NeZero.pos N)
    (SlashInvariantFormClass.periodic_comp_ofComplex _ hp)
    (ModularForm.translate f (γ : GL (Fin 2) ℝ)).holo'
    (ModularFormClass.bdd_at_infty (ModularForm.translate f (γ : GL (Fin 2) ℝ)))

theorem eventually_ne_zero_translated_cuspFunction (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) {k : ℤ}
    (f : ModularForm (Γ : Subgroup (GL (Fin 2) ℝ)) k) (hf : f ≠ 0) (γ : SL(2, ℤ)) :
    ∀ᶠ q in 𝓝[≠] (0 : ℂ),
      cuspFunction N (ModularForm.translate f (γ : GL (Fin 2) ℝ)) q ≠ 0 := by
  apply CuspTaylor.eventually_ne_zero_of_series_ne_zero
    (analyticAt_translated_cuspFunction N hΓ f γ)
  rw [CuspTaylor.series_cuspFunction]
  intro he
  exact translated_modular_ne_zero f hf γ
    ((ModularForm.qExpansion_eq_zero_iff (by exact_mod_cast NeZero.pos N)
      (CuspAuxiliary.natCast_mem_strictPeriods_conj N hΓ γ)).mp he)

end
end Dubon2026.SourceFunctionField.GammaOne
