private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRatioRealization, `SourceGammaOneAnalyticGerms, `SourceGammaOneMeromorphicRealization, `SourceGammaOneRealizationAlgebra, `SourceGammaOneAnalyticOrder, `SourceGammaOneAnalyticOrderNontrivial, `SourceGammaOneAnalyticPlaces, `SourceGammaOneJRealization, `SourceGammaOneRegularCoefficientOrders, `SourceGammaOneAnalyticJOrders, `SourceGammaOneAnalyticDifferentialRegularity, `SourceGammaOneHolomorphicDifferentials]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.realize_eq_div, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.analyticAt_comp_ofComplex, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.coe_eq_zero_iff, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.eventually_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.eventually_ne_zero_ofComplex, `Dubon2026.SourceFunctionField.GammaOne.realize_eventuallyEq_of_ratio, `Dubon2026.SourceFunctionField.GammaOne.meromorphicAt_realize, `Dubon2026.SourceFunctionField.GammaOne.meromorphicOrderAt_realize_ne_top, `Dubon2026.SourceFunctionField.GammaOne.realize_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.realize_mul_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.realize_add_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_ne_top, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_zero, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_eq_top_iff, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_one, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_mul, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_add, `Dubon2026.SourceFunctionField.GammaOne.exists_analyticOrder_pos, `Dubon2026.SourceFunctionField.GammaOne.exists_analytic_place, `Dubon2026.SourceFunctionField.GammaOne.analyticRamification_pos, `Dubon2026.SourceFunctionField.GammaOne.analyticPlace_order, `Dubon2026.SourceFunctionField.GammaOne.realize_jCoordinateOver, `Dubon2026.SourceFunctionField.GammaOne.regular_coefficient_order_bound, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_j_sub, `Dubon2026.SourceFunctionField.GammaOne.analyticPlace_j_sub_pos, `Dubon2026.SourceFunctionField.GammaOne.analytic_derivative_order, `Dubon2026.SourceFunctionField.GammaOne.regular_analytic_differential_order, `Dubon2026.SourceFunctionField.GammaOne.regular_analytic_differential_bounded, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_differential, `Dubon2026.SourceFunctionField.GammaOne.regular_differential_holomorphic_realization]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.realize, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder, `Dubon2026.SourceFunctionField.GammaOne.analyticPlace, `Dubon2026.SourceFunctionField.GammaOne.analyticRamification]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Local Holomorphy postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Local Holomorphy theorems discovered"
  logInfo m!"SOURCE GAMMA ONE LOCAL HOLOMORPHY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
