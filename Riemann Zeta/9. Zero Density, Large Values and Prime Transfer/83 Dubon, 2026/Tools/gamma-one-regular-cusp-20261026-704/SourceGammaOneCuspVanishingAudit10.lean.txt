private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneTranslatedCuspGerms, `SourceGammaOneCuspRatioRealization, `SourceGammaOneCuspRealizationAlgebra, `SourceGammaOneCuspOrder, `SourceGammaOneCuspJOrder, `SourceGammaOneCuspPlaces, `SourceGammaOneCuspDerivativeOrder, `SourceGammaOneDifferentialCuspGerm, `SourceGammaOneRegularCuspOrder, `SourceGammaOneRegularCuspForms]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.translated_modular_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.analyticAt_translated_cuspFunction, `Dubon2026.SourceFunctionField.GammaOne.eventually_ne_zero_translated_cuspFunction, `Dubon2026.SourceFunctionField.GammaOne.realize_smul_eq_translated_ratio, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_eventuallyEq_of_ratio, `Dubon2026.SourceFunctionField.GammaOne.meromorphicAt_cuspRealize, `Dubon2026.SourceFunctionField.GammaOne.meromorphicOrderAt_cuspRealize_ne_top, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_const_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_mul_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_add_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_ne_top, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_zero, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_eq_top_iff, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_one, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_mul, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_add, `Dubon2026.SourceFunctionField.GammaOne.analyticAt_levelOne_cuspFunction, `Dubon2026.SourceFunctionField.GammaOne.cusp_order_zero_of_tendsto_one, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_jCoordinate, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_jCoordinate_neg, `Dubon2026.SourceFunctionField.GammaOne.exists_cuspOrder_pos, `Dubon2026.SourceFunctionField.GammaOne.exists_cusp_place, `Dubon2026.SourceFunctionField.GammaOne.cuspRamification_pos, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_order, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_j_neg, `Dubon2026.SourceFunctionField.GammaOne.cusp_raw_j_derivative_order, `Dubon2026.SourceFunctionField.GammaOne.differential_cusp_germ, `Dubon2026.SourceFunctionField.GammaOne.regular_differential_cusp_order_pos, `Dubon2026.SourceFunctionField.GammaOne.regular_differential_cusp_decay, `Dubon2026.SourceFunctionField.GammaOne.exists_regular_cusp_form, `Dubon2026.SourceFunctionField.GammaOne.regular_differential_cusp_form_realization]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.cuspRealize, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace, `Dubon2026.SourceFunctionField.GammaOne.cuspRamification]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Cusp Vanishing postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Cusp Vanishing theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CUSP VANISHING AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
