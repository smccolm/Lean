/-! DUB-17/18 -> original Gamma1 Picard/H1 and Hecke/diamond comparison ->
actual regular-differential/cusp equivalence. The two proved realization
directions identify the ranges of the injective literal Fourier maps.
The resulting complex-linear equivalence retains the exact q-expansion;
no dimension identity or matching witness is an additional assumption. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane SourceDifferential SourceCurve CongruenceSubgroup
open scoped MatrixGroups ModularForm
variable (N : ℕ) [NeZero N]

private theorem gammaOne_translation_mem : ModularGroup.T ∈ Gamma1 N := by
  simp [Gamma1_mem, ModularGroup.T]

theorem regular_cusp_qExpansion_range_eq :
    (regularQExpansion (Gamma1 N)).range =
      (cuspQExpansion (gammaOne_translation_mem N)).range := by
  apply le_antisymm
  · rintro _ ⟨ω, rfl⟩
    have hΓ : Gamma N ≤ Gamma1 N := by
      intro A hA
      rw [Gamma_mem] at hA
      rw [Gamma1_mem]
      exact ⟨hA.1, hA.2.2.2, hA.2.2.1⟩
    obtain ⟨f, hf⟩ := exists_cuspForm_of_regular_differential N hΓ
      (gammaOne_translation_mem N) ω.val ω.property
    exact ⟨f, hf⟩
  · rintro _ ⟨f, rfl⟩
    obtain ⟨ω, hω, he⟩ := exists_gammaOne_regular_differential_of_coefficients
      (RingHom.id ℂ) N f (fun n => ⟨(qExpansion 1 f).coeff n, rfl⟩)
    rw [coeffMap_id] at he
    exact ⟨⟨ω, hω⟩, he⟩

/-- The actual Gamma1 regular-differential/cusp equivalence fixed by literal Fourier expansion. -/
def regularCuspEquiv :
    regularDifferentials ℂ (functionFieldOver (Gamma1 N) ℂ) ≃ₗ[ℂ]
      CuspForm (Gamma1 N) 2 :=
  (LinearEquiv.ofInjective (regularQExpansion (Gamma1 N))
    (regularQExpansion_injective (gammaOne_translation_mem N))).trans
      ((LinearEquiv.ofEq _ _ (regular_cusp_qExpansion_range_eq N)).trans
        (LinearEquiv.ofInjective (cuspQExpansion (gammaOne_translation_mem N))
          (cuspQExpansion_injective (gammaOne_translation_mem N))).symm)

theorem regularCuspEquiv_qExpansion
    (ω : regularDifferentials ℂ (functionFieldOver (Gamma1 N) ℂ)) :
    (qExpansion 1 (regularCuspEquiv N ω) : LaurentSeries ℂ) =
      diffQExp (functionFieldOver (Gamma1 N) ℂ) ω.val := by
  have he := (LinearEquiv.ofInjective (cuspQExpansion (gammaOne_translation_mem N))
    (cuspQExpansion_injective (gammaOne_translation_mem N))).apply_symm_apply
      ((LinearEquiv.ofEq _ _ (regular_cusp_qExpansion_range_eq N))
        ((LinearEquiv.ofInjective (regularQExpansion (Gamma1 N))
          (regularQExpansion_injective (gammaOne_translation_mem N))) ω))
  exact congrArg Subtype.val he

end
end Dubon2026.SourceFunctionField.GammaOne
