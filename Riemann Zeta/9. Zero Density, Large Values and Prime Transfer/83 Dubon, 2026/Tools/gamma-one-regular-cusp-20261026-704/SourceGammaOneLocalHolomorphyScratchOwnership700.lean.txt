import SourceGammaOneRatioRealization
import SourceGammaOneAnalyticGerms
import SourceGammaOneMeromorphicRealization
import SourceGammaOneRealizationAlgebra
import SourceGammaOneAnalyticOrder
import SourceGammaOneAnalyticOrderNontrivial
import SourceGammaOneAnalyticPlaces
import SourceGammaOneJRealization
import SourceGammaOneRegularCoefficientOrders
import SourceGammaOneAnalyticJOrders
import SourceGammaOneAnalyticDifferentialRegularity
import SourceGammaOneHolomorphicDifferentials
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneRatioRealization, `SourceGammaOneAnalyticGerms, `SourceGammaOneMeromorphicRealization, `SourceGammaOneRealizationAlgebra, `SourceGammaOneAnalyticOrder, `SourceGammaOneAnalyticOrderNontrivial, `SourceGammaOneAnalyticPlaces, `SourceGammaOneJRealization, `SourceGammaOneRegularCoefficientOrders, `SourceGammaOneAnalyticJOrders, `SourceGammaOneAnalyticDifferentialRegularity, `SourceGammaOneHolomorphicDifferentials]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
