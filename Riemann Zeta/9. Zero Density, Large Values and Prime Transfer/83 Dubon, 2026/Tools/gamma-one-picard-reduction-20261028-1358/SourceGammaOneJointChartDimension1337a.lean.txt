/-! DUB-17/18 -> original Gamma1 prime-to-residue Picard injectivity ->
Kummer power lifting -> the residue genus inequality. The actual reduced j
is separating, so the existing finite integral-closure argument generates
both residue charts from finite pole spaces. This discharges the chart-
generation inputs of the native joint dimension theorem for the original
Gamma1 residue field. The total coordinate degree remains explicit and
will be supplied by the actual Kummer cover and base degree comparison. -/
namespace Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
noncomputable section
open SourceCurve SourceCurve.RegularProlongation.JointChartAssembly IntermediateField
attribute [local instance] specializationResidueAlgebra
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (M : ℕ) [NeZero M]
local notation "Δ" => CongruenceSubgroup.Gamma1 M
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Δ

theorem joint_chart_dimension_of_gammaOne_residue_coordinate
    (q : ℕ) [NeZero q] (E : Type*) [Field E] [Algebra Qbar E]
    (S : Fin q → RegularProlongation A E Fb)
    (hSinj : Function.Injective fun i => (S i).integers)
    (t : E) (ht : ∀ i, t ∈ (S i).integers)
    (hres : ∀ i, (S i).residue ⟨t, ht i⟩ = residueJCoordinate A Δ)
    (htotal : Module.finrank Qbar⟮t⟯ E =
      Module.finrank kk⟮residueJCoordinate A Δ⟯ Fb * q)
    (ht0 : t ≠ 0)
    (D : Divisor kk Fb) (hD : ∀ v, D v = max 0 (-v.ord (residueJCoordinate A Δ))) :
    ∃ m₀ : ℕ, ∀ m : ℕ, m₀ ≤ m →
      FiniteDimensional kk ↥(ST (Fb := fun _ : Fin q => Fb) S t ⊓ weightedST S t m) ∧
      Module.finrank kk ↥(ST (Fb := fun _ : Fin q => Fb) S t ⊓ weightedST S t m) + q ≤
        q * Module.finrank kk ↥(LSpace (m • D)) +
          Module.finrank kk ↥(ST (Fb := fun _ : Fin q => Fb) S t ⊓ ST S t⁻¹) := by
  classical
  haveI := valuation_residue_isAlgClosed A
  have hT : ModularGroup.T ∈ Δ := by simp [CongruenceSubgroup.Gamma1_mem, ModularGroup.T]
  haveI := (finite_residueCoordinate_and_degree_le A Δ hT).1
  haveI := isSeparable_residueCoordinate A Δ hT
  have hpos : 0 < Module.finrank kk⟮residueJCoordinate A Δ⟯ Fb := Module.finrank_pos
  letI : FiniteDimensional Qbar⟮t⟯ E := Module.finite_of_finrank_pos (by
    rw [htotal]
    exact Nat.mul_pos hpos (Nat.pos_of_ne_zero (NeZero.ne q)))
  have hy := transcendental_residueJCoordinate A Δ
  have hsupport := finite_function_support_of_separating_coordinate
    (residueJCoordinate A Δ) hy (residueJCoordinate A Δ)
  have heD : D = poleDivisor (residueJCoordinate A Δ) hsupport := by
    ext v
    rw [hD, poleDivisor_apply, max_comm]
  have hg := exists_pole_space_generating_window_of_separating_coordinate
    (residueJCoordinate A Δ) hy hsupport
  rw [← heD] at hg
  haveI : FiniteDimensional kk⟮(residueJCoordinate A Δ)⁻¹⟯ Fb := by
    rw [adjoin_singleton_inv_eq]
    infer_instance
  haveI : Algebra.IsSeparable kk⟮(residueJCoordinate A Δ)⁻¹⟯ Fb := by
    rw [adjoin_singleton_inv_eq]
    infer_instance
  have hyi : Transcendental kk (residueJCoordinate A Δ)⁻¹ :=
    fun h => hy (IsAlgebraic.inv_iff.mp h)
  have hsupporti := finite_function_support_of_separating_coordinate
    (residueJCoordinate A Δ) hy (residueJCoordinate A Δ)⁻¹
  let Di := poleDivisor (residueJCoordinate A Δ)⁻¹ hsupporti
  have hgi := exists_pole_space_generating_window_of_separating_coordinate
    (residueJCoordinate A Δ)⁻¹ hyi hsupporti
  have heq : Fintype.card (Fin q) * Module.finrank kk⟮residueJCoordinate A Δ⟯ Fb =
      Module.finrank Qbar⟮t⟯ E := by rw [Fintype.card_fin, htotal, Nat.mul_comm]
  have h := @eventual_uniform_residue_chart_dimension Qbar E inferInstance A inferInstance
    inferInstance (Fin q) inferInstance (Fb : Type) inferInstance
    (specializationResidueAlgebra A Δ)
    S hSinj t ht ht0 inferInstance (residueJCoordinate A Δ) hy hres
    inferInstance (fun D => finite_lSpace_of_separating_coordinate (residueJCoordinate A Δ) hy D)
    heq D Di hD (fun _ => max_comm _ _) hg hgi
  simpa only [Fintype.card_fin] using h

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
