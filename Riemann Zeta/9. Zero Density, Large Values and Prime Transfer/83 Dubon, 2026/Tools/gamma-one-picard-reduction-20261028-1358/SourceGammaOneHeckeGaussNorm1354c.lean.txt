/-! DUB-17/18 -> original Gamma1 geometric Tate Frobenius -> actual
Hecke norm reduction. Coordinate-degree preservation derives diamond
Gauss integrality. The exact embedding census and arithmetic diamond give
the actual norm product; integral fraction witnesses reduce it and beta
reflects integrality. No integral diamond representative, norm identity or
residue formula is supplied by the caller. Degree preservation remains the
explicit geometric input; genus and Picard/Tate compatibility remain later. -/
namespace Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
noncomputable section
open SourceCurve IntermediateField CongruenceSubgroup GaussReduction
attribute [local instance] specializationResidueAlgebra HeckeNorm.alphaAlgebra HeckeNorm.alphaModule
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (M p : ℕ) [NeZero M] [hp : Fact p.Prime]
  [CharP (IsLocalRing.ResidueField A) p]
local notation "Γ" => Gamma1 M
local notation "Δ" => Gamma1 M ⊓ Gamma0 (p * M)
local notation "FF" => laurentBaseChange Qbar (x1FunctionField M)
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Γ

theorem exists_gammaOne_hecke_gauss_norm_reduction_of_degree_eq
    (hcop : Nat.Coprime M p)
    (hdeg : Module.finrank Qbar⟮jCoordinateOver Γ Qbar⟯ (functionFieldOver Γ Qbar) =
      Module.finrank kk⟮residueJCoordinate A Γ⟯ Fb) :
    ∃ g : Gamma0 M, (p : ZMod M) * (g.val 1 1 : ZMod M) = 1 ∧
      ∀ f : coefficientGaussRing Γ A,
        ∃ hn : Algebra.norm FF (heckeBeta Qbar M p f.val) ∈ coefficientGaussRing Γ A,
          resHom A Γ ⟨Algebra.norm FF (heckeBeta Qbar M p f.val), hn⟩ =
            coeffMap (frobenius kk p) (resHom A Γ (arithmeticDiamondGauss A M hdeg g f)) *
              qExpand kk p (resHom A Γ f) := by
  haveI : Algebra.IsAlgebraic ℚ Qbar := AlgebraicClosure.isAlgebraic ℚ
  letI : NeZero (p : ℚ) := ⟨Nat.cast_ne_zero.mpr hp.out.ne_zero⟩
  obtain ⟨z, hz⟩ := HasEnoughRootsOfUnity.exists_primitiveRoot Qbar p
  let ζ : Qbarˣ := Units.map A.subtype.toMonoidHom (primeRootValuationUnit A p hz)
  have hζ : IsPrimitiveRoot (ζ : Qbar) p := by
    change IsPrimitiveRoot ((primeRootValuationUnit A p hz : A) : Qbar) p
    rw [primeRootValuationUnit_coe]
    exact hz
  obtain ⟨g, hlabel, hnorm⟩ := exists_gammaOne_hecke_norm_beta_product M p hcop ζ hζ
  refine ⟨g, hlabel, fun f => ?_⟩
  let u := arithmeticDiamondGauss A M hdeg g f
  let n : FF := Algebra.norm FF (heckeBeta Qbar M p f.val)
  obtain ⟨hub, hu⟩ := wit_spec A Γ u
  obtain ⟨hyb, hy⟩ := wit_spec A Γ f
  have hprod : qExpand Qbar p (n : LaurentSeries Qbar) =
      (∏ b ∈ Finset.range p, qTwist (ζ ^ b) (u.val : LaurentSeries Qbar)) *
        qExpand Qbar p (qExpand Qbar p (f.val : LaurentSeries Qbar)) := hnorm f.val
  obtain ⟨a, b, hb, hab, hred⟩ := HeckeNorm.exists_diamond_fraction_product_witness
    A p ζ hζ (u.val : LaurentSeries Qbar) (f.val : LaurentSeries Qbar)
    (n : LaurentSeries Qbar) hprod
    (wit A Γ u).1 (wit A Γ u).2 (wit A Γ f).1 (wit A Γ f).2 hub hyb hu hy
  have hbeta : heckeBeta Qbar M p n ∈ coefficientGaussRing Δ A := by
    change IsInt A (heckeBeta Qbar M p n : LaurentSeries Qbar)
    rw [heckeBeta_coe]
    exact ⟨a, b, hb, hab⟩
  have hn : n ∈ coefficientGaussRing Γ A := (heckeBeta_mem_gaussRing_iff A M p n).mp hbeta
  refine ⟨hn, ?_⟩
  have hβ := heckeBetaGauss_resHom A M p ⟨n, hn⟩
  have hβres : resHom A Δ (heckeBetaGauss A M p ⟨n, hn⟩) = ϖ A a / ϖ A b := by
    apply resHom_eq A Δ hb
    change qExpand Qbar p (n : LaurentSeries Qbar) * ι A b = ι A a
    exact hab
  apply qExpand_injective p
  rw [← hβ, hβres, hred, resHom_eq A Γ hub hu, resHom_eq A Γ hyb hy]

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
