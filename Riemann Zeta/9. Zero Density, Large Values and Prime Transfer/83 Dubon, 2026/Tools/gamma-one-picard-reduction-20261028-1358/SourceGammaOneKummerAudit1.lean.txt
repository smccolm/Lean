private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneKummerDegree, `SourceGammaOneKummerGenus, `SourceJointSeparableFiniteOverlap, `SourceJointSeparableConstants, `SourceJointSeparableChartDimension, `SourceJointSeparablePoleDimension, `SourceJointSeparableGenus, `SourceGammaOneKummerResidueGenus, `SourceGammaOneKummerPowerLifting, `SourceGammaOnePicardPrimeKernel, `SourceGammaOnePicardTorsion]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.gammaOne_kummer_degree_and_unramified, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_kummer_genus_sub_one, `Dubon2026.SourceCurve.RegularProlongation.JointChartConstants.finite_overlap_of_separating_residue_coordinates, `Dubon2026.SourceCurve.RegularProlongation.JointChartConstants.common_constants_of_separating_residue_coordinates, `Dubon2026.SourceCurve.RegularProlongation.JointChartAssembly.eventual_chart_dimension_of_separating_residue_coordinate, `Dubon2026.SourceCurve.RegularProlongation.JointChartAssembly.pole_dimension_bound_of_separating_residue_coordinate, `Dubon2026.SourceCurve.RegularProlongation.JointChartAssembly.genus_inequality_of_separating_residue_coordinate, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_kummer_residue_genus_inequality_of_degree_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_exists_power_of_residue_power_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_residue_lSpace_zero_eq_range, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_pic0_eq_zero_of_prime_torsion_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_pic0_eq_zero_of_torsion_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_picard_torsion_injective_of_degree_genus_eq]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOnePowerLiftOriginalAlgebra]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE KUMMER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
