The coefficient valuation is the actual valuation on the algebraic closure

common-constant hypothesis remains. -/
namespace Dubon2026.SourceCurve.RegularProlongation.JointChartConstants
noncomputable section
open IsLocalRing JointChartAssembly
local notation "Qbar" => AlgebraicClosure ℚ

theorem common_constants_of_separating_residue_coordinates
    (A : ValuationSubring Qbar) {F : Type*} [Field F] [Algebra Qbar F]
    {ι : Type*} [Fintype ι] [Nonempty ι] (Fb : ι → Type*) [∀ i, Field (Fb i)]
    [∀ i, Algebra (ResidueField A) (Fb i)]
    (R : ∀ i, RegularProlongation A F (Fb i))
    (hR : Function.Injective fun i => (R i).integers)
    (f : F) (hf : ∀ i, f ∈ (R i).integers)
    (htr : ∀ i, Transcendental (ResidueField A) ((R i).residue ⟨f, hf i⟩))
    [FiniteDimensional (IntermediateField.adjoin Qbar ({f} : Set F)) F]
    [∀ i, FiniteDimensional (IntermediateField.adjoin (ResidueField A)
      ({(R i).residue ⟨f, hf i⟩} : Set (Fb i))) (Fb i)]
    [∀ i, Algebra.IsSeparable (IntermediateField.adjoin (ResidueField A)
      ({(R i).residue ⟨f, hf i⟩} : Set (Fb i))) (Fb i)]
    (heq : ∑ i, Module.finrank (IntermediateField.adjoin (ResidueField A)
        ({(R i).residue ⟨f, hf i⟩} : Set (Fb i))) (Fb i) =
      Module.finrank (IntermediateField.adjoin Qbar ({f} : Set F)) F)
    (h : ∀ i, Fb i) (hh : h ∈ ST R f) (hhi : h ∈ ST R f⁻¹) :
    ∃ c : ResidueField A, ∀ i, h i = algebraMap (ResidueField A) (Fb i) c := by
  classical
  obtain ⟨hfi, hfinite⟩ := finite_overlap_of_separating_residue_coordinates A Fb R hR f hf htr heq
  haveI := hfinite
  haveI := algebraicClosure_rat_valuation_krullDimLE_one A
  let i₀ : ι := Classical.choice inferInstance
  have htrL : Transcendental Qbar f :=
    transcendental_of_residue_transcendental (R i₀) ⟨f, hf i₀⟩ (htr i₀)
  exact common_constants_of_finite_overlap A Fb R hR f hf htrL htr heq hfi h hh hhi

end
end Dubon2026.SourceCurve.RegularProlongation.JointChartConstants
