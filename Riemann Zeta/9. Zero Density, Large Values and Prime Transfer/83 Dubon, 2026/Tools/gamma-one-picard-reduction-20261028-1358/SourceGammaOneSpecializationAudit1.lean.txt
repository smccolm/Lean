private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneResidueGeometry, `SourceCurveSeparatingGenusMaximum, `SourceReductionGenusPoleLifting, `SourceReductionGenusIntegralCharts, `SourceGammaOneGenusSpecialization, `SourceGammaOneOriginalSpecialization]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.D_residueJCoordinate_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueImage_hasPrincipalDivisors, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueImage_place_residue_bijective, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueImage_isCurveOver, `Dubon2026.SourceCurve.finite_lSpace_of_separating_coordinate, `Dubon2026.SourceCurve.separating_coordinate_degree_dimension_maximum, `Dubon2026.SourceCurve.separating_coordinate_index_zero_eq_maximum, `Dubon2026.SourceCurve.RegularProlongation.exists_pole_space_lifts_of_degree_genus_eq, `Dubon2026.SourceCurve.RegularProlongation.exists_integral_coordinate_lift_of_degree_genus_eq, `Dubon2026.SourceCurve.RegularProlongation.integral_charts_lift_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_principal_reduction_map_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_constant_reduction_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_constant_reduction_of_degree_genus_eq]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.specializationResidueAlgebra]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE SPECIALIZATION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
