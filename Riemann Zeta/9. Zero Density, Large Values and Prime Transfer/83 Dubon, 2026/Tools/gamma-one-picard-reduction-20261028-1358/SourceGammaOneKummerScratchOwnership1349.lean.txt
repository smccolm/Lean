import SourceGammaOneKummerDegree
import SourceGammaOneKummerGenus
import SourceJointSeparableFiniteOverlap
import SourceJointSeparableConstants
import SourceJointSeparableChartDimension
import SourceJointSeparablePoleDimension
import SourceJointSeparableGenus
import SourceGammaOneKummerResidueGenus
import SourceGammaOneKummerPowerLifting
import SourceGammaOnePicardPrimeKernel
import SourceGammaOnePicardTorsion
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneKummerDegree, `SourceGammaOneKummerGenus, `SourceJointSeparableFiniteOverlap, `SourceJointSeparableConstants, `SourceJointSeparableChartDimension, `SourceJointSeparablePoleDimension, `SourceJointSeparableGenus, `SourceGammaOneKummerResidueGenus, `SourceGammaOneKummerPowerLifting, `SourceGammaOnePicardPrimeKernel, `SourceGammaOnePicardTorsion]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
