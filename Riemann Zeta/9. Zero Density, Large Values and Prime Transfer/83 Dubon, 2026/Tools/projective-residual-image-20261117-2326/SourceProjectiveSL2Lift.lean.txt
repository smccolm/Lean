/-- Projectivizing the literal subfield SL₂ image gives its standard PSL₂ image. -/
theorem subfieldSL2_projective_image {F L : Type*} [Field F] [Field L]
    (j : F →+* L) :
    (((GeneralLinearGroup.map j).comp
      (SpecialLinearGroup.toGL (n := Fin 2) (R := F))).range).map
        (QuotientGroup.mk' (Subgroup.center (GL (Fin 2) L))) =
      (PSLImageInPGL F).map (pglMap j) := by
  ext x
  constructor
  · rintro ⟨y, ⟨s, rfl⟩, rfl⟩
    exact ⟨SL2ToPGL F s, ⟨s, rfl⟩, rfl⟩
  · rintro ⟨y, ⟨s, rfl⟩, rfl⟩
    exact ⟨(GeneralLinearGroup.map j) (SpecialLinearGroup.toGL s), ⟨s, rfl⟩, rfl⟩

/-- A projective PSL₂ subfield lower bound gives the actual linear SL₂ lower bound. -/
theorem subfieldSL2_le_of_projective_image {F L : Type*} [Field F] [Field L]
    [NeZero (2 : F)] [NeZero (3 : F)] (j : F →+* L)
    (H : Subgroup (GL (Fin 2) L))
    (h : (PSLImageInPGL F).map (pglMap j) ≤
      H.map (QuotientGroup.mk' (Subgroup.center (GL (Fin 2) L)))) :
    ((GeneralLinearGroup.map j).comp
      (SpecialLinearGroup.toGL (n := Fin 2) (R := F))).range ≤ H := by
  let ψ := (GeneralLinearGroup.map j).comp
    (SpecialLinearGroup.toGL (n := Fin 2) (R := F))
  letI : Group.IsPerfect SL(2, F) := Dubon2026.fieldSL2_isPerfect
  letI : Group.IsPerfect ψ.range := Group.IsPerfect.range ψ
  apply perfect_subgroup_le_of_center_quotient_map_le ψ.range H
  rwa [subfieldSL2_projective_image]

end
end Dubon2026.SourceDickson
