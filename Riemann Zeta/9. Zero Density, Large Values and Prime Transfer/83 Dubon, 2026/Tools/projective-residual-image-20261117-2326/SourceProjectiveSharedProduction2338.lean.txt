import Dubon2026.SourceGammaOneFiniteProjectiveLinearResidual
import Mathlib.Tactic
open Lean
#print axioms Matrix.GeneralLinearGroup.mkOfDetNeZero.congr_simp
private def inspectSharedGenerated : CoreM Unit := do
  let env ← getEnv
  let name := `Matrix.GeneralLinearGroup.mkOfDetNeZero.congr_simp
  let some info := env.find? name | throwError "Missing shared declaration"
  let some idx := env.getModuleIdxFor? name | throwError "Missing module provenance"
  let owner := env.allImportedModuleNames[idx.toNat]!
  let some value := info.value? (allowOpaque := true) | throwError "Missing proof value"
  let allowed := #[``propext, ``Classical.choice, ``Quot.sound]
  let axioms ← Lean.collectAxioms name
  unless axioms.all allowed.contains do throwError "Unexpected axioms: {axioms}"
  logInfo m!"SHARED {owner} {name}: {hash info.type} {hash value}"
  logInfo m!"TYPE {info.type}"
  logInfo m!"VALUE {value}"
#eval inspectSharedGenerated
