-> residue fiber bounds. The actual scalar-extended diamond action has the

theorem reducedFieldActionOver_range_finite (L : Type*) [Field L] (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) L]
    (hℓN : ¬ ℓ ∣ N) : Finite (reducedFieldActionOver L N ℓ hℓN).range := by
  letI : (plusMinus (Gamma1 N)).FiniteIndex := Subgroup.finiteIndex_of_le (le_plusMinus _)
  letI : (reducedFieldActionOver L N ℓ hℓN).ker.FiniteIndex := by
    rw [reducedFieldActionOver_kernel]
    infer_instance
  apply Nat.finite_of_card_ne_zero
  rw [← Subgroup.index_ker]
  exact Subgroup.FiniteIndex.index_ne_zero

theorem reduced_over_fixedField_degree_eq_signed_relativeIndex
    (L : Type*) [Field L] (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) L] (hℓN : ¬ ℓ ∣ N) :
    Module.finrank (IntermediateField.fixedField (reducedFieldActionOver L N ℓ hℓN).range)
      (qExpFunctionFieldC L (Gamma1 N)) =
        (plusMinus (Gamma1 N)).relIndex (Gamma0 N) := by
  letI := reducedFieldActionOver_range_finite L N ℓ hℓN
  letI := Fintype.ofFinite (reducedFieldActionOver L N ℓ hℓN).range
  have hArtin : Module.finrank (IntermediateField.fixedField (reducedFieldActionOver L N ℓ hℓN).range)
      (qExpFunctionFieldC L (Gamma1 N)) =
        Fintype.card (reducedFieldActionOver L N ℓ hℓN).range :=
    FixedPoints.finrank_eq_card (reducedFieldActionOver L N ℓ hℓN).range _
  rw [hArtin, ← Nat.card_eq_fintype_card, ← Subgroup.index_ker,
    reducedFieldActionOver_kernel]
  rfl

end
end Dubon2026.SourceFunctionField.IntegralDiamond
