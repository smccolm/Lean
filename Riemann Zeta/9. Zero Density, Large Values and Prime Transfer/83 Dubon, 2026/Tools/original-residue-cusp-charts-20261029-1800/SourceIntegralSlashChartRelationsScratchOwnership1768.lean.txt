import SourceIntegralSlashCoefficientLocalization
import SourceIntegralSlashCoefficientReduction
import SourceIntegralSlashReducedSeries
import SourceIntegralSlashReducedScaling
import SourceIntegralSlashReducedZero
import SourceIntegralSlashReducedOperations
import SourceIntegralSlashReducedSums
import SourceIntegralSlashLinearRelation
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceIntegralSlashCoefficientLocalization, `SourceIntegralSlashCoefficientReduction, `SourceIntegralSlashReducedSeries, `SourceIntegralSlashReducedScaling, `SourceIntegralSlashReducedZero, `SourceIntegralSlashReducedOperations, `SourceIntegralSlashReducedSums, `SourceIntegralSlashLinearRelation]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
