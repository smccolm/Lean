/-! DUB-17/18 -> original Gamma1 genus preservation -> reverse genus inequality
-> residue fiber bounds. Apply the proved faithful coefficient extension to
the actual prime-field diamond action. This constructs the original action
over the residue coefficient field and proves its exact signed Gamma1 kernel;
no action or kernel hypothesis is supplied. -/
namespace Dubon2026.SourceFunctionField.IntegralDiamond
noncomputable section
open CongruenceSubgroup GammaOne ReducedDegreeTransport
open scoped MatrixGroups
attribute [local instance] originalFieldScalarAlgebra

/-- The actual diamond action on the original field over larger residue coefficients. -/
def reducedFieldActionOver (L : Type*) [Field L] (N ℓ : ℕ)
    [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) L] (hℓN : ¬ ℓ ∣ N) :
    Gamma0 N →* (qExpFunctionFieldC L (Gamma1 N) ≃ₐ[L] qExpFunctionFieldC L (Gamma1 N)) :=
  (originalFieldAutomorphismExtensionHom (K := ZMod ℓ) (L := L) (Gamma1 N)).comp
    (reducedFieldAction N ℓ hℓN)

theorem reducedFieldActionOver_coeff (L : Type*) [Field L] (N ℓ : ℕ)
    [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) L] (hℓN : ¬ ℓ ∣ N)
    (γ : Gamma0 N) (x : qExpFunctionFieldC (ZMod ℓ) (Gamma1 N)) :
    reducedFieldActionOver L N ℓ hℓN γ (originalFieldCoefficientHom (L := L) (Gamma1 N) x) =
      originalFieldCoefficientHom (L := L) (Gamma1 N) (reducedFieldAction N ℓ hℓN γ x) :=
  originalFieldAutomorphismExtension_coeff (Gamma1 N) (reducedFieldAction N ℓ hℓN γ) x

theorem reducedFieldActionOver_eq_one_iff (L : Type*) [Field L] (N ℓ : ℕ)
    [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) L] (hℓN : ¬ ℓ ∣ N) (γ : Gamma0 N) :
    reducedFieldActionOver L N ℓ hℓN γ = 1 ↔ γ.val ∈ plusMinus (Gamma1 N) := by
  let f := originalFieldAutomorphismExtensionHom (K := ZMod ℓ) (L := L) (Gamma1 N)
  change f (reducedFieldAction N ℓ hℓN γ) = 1 ↔ _
  rw [← map_one f, (originalFieldAutomorphismExtensionHom_injective (L := L) (Gamma1 N)).eq_iff]
  exact reducedFieldAction_eq_one_iff N ℓ hℓN γ

theorem reducedFieldActionOver_kernel (L : Type*) [Field L] (N ℓ : ℕ)
    [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) L] (hℓN : ¬ ℓ ∣ N) :
    (reducedFieldActionOver L N ℓ hℓN).ker = (plusMinus (Gamma1 N)).subgroupOf (Gamma0 N) := by
  rw [← reducedFieldAction_kernel N ℓ hℓN]
  ext γ
  change reducedFieldActionOver L N ℓ hℓN γ = 1 ↔ reducedFieldAction N ℓ hℓN γ = 1
  exact (reducedFieldActionOver_eq_one_iff L N ℓ hℓN γ).trans
    (reducedFieldAction_eq_one_iff N ℓ hℓN γ).symm

end
end Dubon2026.SourceFunctionField.IntegralDiamond
