private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralSlashReducedTranslation1789b, `SourceIntegralSlashReducedTranslationSeries1791a, `SourceIntegralSlashDiamondChart1792a, `SourceIntegralSlashNormalizedDiamond1793a]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.IntegralSlash.levelReduction_zeta, `Dubon2026.SourceFunctionField.IntegralSlash.reducedSlashSeries_mul_T_zpow, `Dubon2026.SourceFunctionField.IntegralSlash.ofPowerSeries_reducedSlashSeries_mul_T_zpow, `Dubon2026.SourceFunctionField.IntegralSlash.normalizedSlashValue_mul_T_zpow, `Dubon2026.SourceFunctionField.IntegralSlash.reducedSlashSeries_neg_matrix, `Dubon2026.SourceFunctionField.IntegralSlash.normalizedSlashValue_neg_matrix, `Dubon2026.SourceFunctionField.IntegralSlash.reducedSlashSeries_scaledDiamond, `Dubon2026.SourceFunctionField.IntegralSlash.ofPowerSeries_reducedSlashSeries_discriminant, `Dubon2026.SourceFunctionField.IntegralSlash.normalizedSlashValue_scaledDiamond]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue slash embedding postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue slash embedding theorems discovered"
  logInfo m!"SOURCE REDUCED CUSP TRANSLATION PREFLIGHT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
