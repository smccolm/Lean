private instance originalFieldTensorScalarTower {K L : Type*}
    [Field K] [Field L] [Algebra K L] (Γ : Subgroup SL(2, ℤ)) :
    IsScalarTower L (L ⊗[K] qExpFunctionFieldC K Γ) (qExpFunctionFieldC L Γ) :=
  IsScalarTower.of_algebraMap_eq (R := L)
    (S := L ⊗[K] qExpFunctionFieldC K Γ) (A := qExpFunctionFieldC L Γ)
    fun a => (originalFieldTensorEval (K := K) (L := L) Γ).commutes a |>.symm

/-- Extend an actual original-field automorphism through faithful scalar tensor and fractions. -/
def originalFieldAutomorphismExtension {K L : Type*} [Field K] [Field L] [Algebra K L]
    (Γ : Subgroup SL(2, ℤ)) (σ : qExpFunctionFieldC K Γ ≃ₐ[K] qExpFunctionFieldC K Γ) :
    qExpFunctionFieldC L Γ ≃ₐ[L] qExpFunctionFieldC L Γ := by
  letI := originalField_isFractionRing (K := K) (L := L) Γ
  exact IsFractionRing.algEquivOfAlgEquiv
    (Algebra.TensorProduct.congr (AlgEquiv.refl : L ≃ₐ[L] L) σ)

theorem originalFieldAutomorphismExtension_tensor {K L : Type*}
    [Field K] [Field L] [Algebra K L] (Γ : Subgroup SL(2, ℤ))
    (σ : qExpFunctionFieldC K Γ ≃ₐ[K] qExpFunctionFieldC K Γ)
    (t : L ⊗[K] qExpFunctionFieldC K Γ) :
    originalFieldAutomorphismExtension (L := L) Γ σ (originalFieldTensorEval Γ t) =
      originalFieldTensorEval Γ
        (Algebra.TensorProduct.congr (AlgEquiv.refl : L ≃ₐ[L] L) σ t) := by
  letI := originalField_isFractionRing (K := K) (L := L) Γ
  exact IsFractionRing.algEquivOfAlgEquiv_algebraMap
    (Algebra.TensorProduct.congr (AlgEquiv.refl : L ≃ₐ[L] L) σ) t

theorem originalFieldAutomorphismExtension_coeff {K L : Type*}
    [Field K] [Field L] [Algebra K L] (Γ : Subgroup SL(2, ℤ))
    (σ : qExpFunctionFieldC K Γ ≃ₐ[K] qExpFunctionFieldC K Γ)
    (x : qExpFunctionFieldC K Γ) :
    originalFieldAutomorphismExtension (L := L) Γ σ (originalFieldCoefficientHom Γ x) =
      originalFieldCoefficientHom Γ (σ x) := by
  have h := originalFieldAutomorphismExtension_tensor (L := L) Γ σ (1 ⊗ₜ[K] x)
  simpa only [Algebra.TensorProduct.congr_apply, Algebra.TensorProduct.map_tmul,
    AlgEquiv.coe_algHom, map_one, originalFieldTensorEval_tmul, one_smul] using h

end
