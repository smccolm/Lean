import SourceReducedFieldScalarGeneration
import SourceReducedFieldTensorEmbedding
import SourceReducedFieldTensorFraction
import SourceReducedFieldAutomorphismExtension
import SourceReducedFieldAutomorphismExtensionLaws
import SourceIntegralDiamondScalarAction
import SourceIntegralDiamondScalarBaseFixed
import SourceIntegralDiamondScalarFixedDegree
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceReducedFieldScalarGeneration, `SourceReducedFieldTensorEmbedding, `SourceReducedFieldTensorFraction, `SourceReducedFieldAutomorphismExtension, `SourceReducedFieldAutomorphismExtensionLaws, `SourceIntegralDiamondScalarAction, `SourceIntegralDiamondScalarBaseFixed, `SourceIntegralDiamondScalarFixedDegree]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
