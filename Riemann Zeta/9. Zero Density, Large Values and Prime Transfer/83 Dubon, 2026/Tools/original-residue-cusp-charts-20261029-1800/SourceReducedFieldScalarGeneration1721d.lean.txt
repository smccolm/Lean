theorem originalField_eq_adjoin_coeffMap {K L : Type*} [Field K] [Field L]
    (σ : K →+* L) (Γ : Subgroup SL(2, ℤ)) :
    qExpFunctionFieldC L Γ = IntermediateField.adjoin L
      (coeffMap σ '' (qExpFunctionFieldC K Γ : Set (LaurentSeries K))) := by
  apply le_antisymm
  · rw [qExpFunctionFieldC, adjoin_le_iff]
    rintro y ⟨k, f, g, p, q, hp, hq, hq0, rfl⟩
    have hqK : intSeriesC K q ≠ 0 := by
      intro hz
      apply hq0
      rw [← coeffMap_intSeries σ, hz, map_zero]
    refine subset_adjoin _ _ ⟨intSeriesC K p / intSeriesC K q,
      div_mem_qExpFunctionFieldC f g hp hq hqK, ?_⟩
    rw [map_div₀, coeffMap_intSeries, coeffMap_intSeries]
  · rw [adjoin_le_iff]
    rintro y ⟨x, hx, rfl⟩
    exact coeffMap_mem_originalField σ Γ hx

private instance laurentScalarTower {K L : Type*} [Field K] [Field L] [Algebra K L] :
    IsScalarTower K L (LaurentSeries L) where
  smul_assoc a b f := by
    apply HahnSeries.ext
    funext n
    simp only [HahnSeries.coeff_smul]
    exact smul_assoc a b (f.coeff n)

/-- Literal coefficient embedding of the original integral-ratio field. -/
def originalFieldCoefficientHom {K L : Type*} [Field K] [Field L] [Algebra K L]
    (Γ : Subgroup SL(2, ℤ)) : qExpFunctionFieldC K Γ →ₐ[K] (qExpFunctionFieldC L Γ).restrictScalars K where
  toFun x := ⟨coeffMap (algebraMap K L) x.val,
    coeffMap_mem_originalField (algebraMap K L) Γ x.property⟩
  map_one' := Subtype.ext (map_one _)
  map_mul' x y := Subtype.ext (map_mul _ x.val y.val)
  map_zero' := Subtype.ext (map_zero _)
  map_add' x y := Subtype.ext (map_add _ x.val y.val)
  commutes' a := by
    apply Subtype.ext
    change coeffMap (algebraMap K L) (algebraMap K (LaurentSeries K) a) =
      algebraMap K (LaurentSeries L) a
    rw [IsScalarTower.algebraMap_apply K L (LaurentSeries L),
      SourceDifferential.algebraMap_laurentSeries_apply_alt,
      SourceDifferential.algebraMap_laurentSeries_apply_alt, coeffMap_single]

theorem originalFieldCoefficientHom_coe {K L : Type*} [Field K] [Field L] [Algebra K L]
    (Γ : Subgroup SL(2, ℤ)) (x : qExpFunctionFieldC K Γ) :
    (originalFieldCoefficientHom (L := L) Γ x : LaurentSeries L) =
      coeffMap (algebraMap K L) x.val := rfl

end
