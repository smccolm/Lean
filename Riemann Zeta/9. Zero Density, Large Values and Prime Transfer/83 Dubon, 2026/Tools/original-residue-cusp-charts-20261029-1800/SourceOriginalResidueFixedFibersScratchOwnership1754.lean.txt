import SourceIntegralDiamondScalarFixedField
import SourceOriginalResidueCoordinateIndex
import SourceIntegralDiamondFixedPlaceTransitivity
import SourceOriginalResidueDiamondFixedField
import SourceIntegralDiamondBaseFiberTransitivity
import SourceOriginalResidueDiamondFiberTransitivity
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceIntegralDiamondScalarFixedField, `SourceOriginalResidueCoordinateIndex, `SourceIntegralDiamondFixedPlaceTransitivity, `SourceOriginalResidueDiamondFixedField, `SourceIntegralDiamondBaseFiberTransitivity, `SourceOriginalResidueDiamondFiberTransitivity]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
