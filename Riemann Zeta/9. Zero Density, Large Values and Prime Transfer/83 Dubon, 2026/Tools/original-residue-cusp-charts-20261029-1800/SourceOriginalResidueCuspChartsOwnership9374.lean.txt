private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceReducedFieldScalarGeneration, `Dubon2026.SourceReducedFieldTensorEmbedding, `Dubon2026.SourceReducedFieldTensorFraction, `Dubon2026.SourceReducedFieldAutomorphismExtension, `Dubon2026.SourceReducedFieldAutomorphismExtensionLaws, `Dubon2026.SourceIntegralDiamondScalarAction, `Dubon2026.SourceIntegralDiamondScalarBaseFixed, `Dubon2026.SourceIntegralDiamondScalarFixedDegree, `Dubon2026.SourceIntegralDiamondScalarFixedField, `Dubon2026.SourceOriginalResidueCoordinateIndex, `Dubon2026.SourceIntegralDiamondFixedPlaceTransitivity, `Dubon2026.SourceOriginalResidueDiamondFixedField, `Dubon2026.SourceIntegralDiamondBaseFiberTransitivity, `Dubon2026.SourceOriginalResidueDiamondFiberTransitivity, `Dubon2026.SourceIntegralSlashCoefficientLocalization, `Dubon2026.SourceIntegralSlashCoefficientReduction, `Dubon2026.SourceIntegralSlashReducedSeries, `Dubon2026.SourceIntegralSlashReducedScaling, `Dubon2026.SourceIntegralSlashReducedZero, `Dubon2026.SourceIntegralSlashReducedOperations, `Dubon2026.SourceIntegralSlashReducedSums, `Dubon2026.SourceIntegralSlashLinearRelation, `Dubon2026.SourceIntegralSlashLevelOne, `Dubon2026.SourceIntegralSlashDiscriminantNormalization, `Dubon2026.SourceCoordinateAlgebraicChartInjectivity, `Dubon2026.SourceIntegralSlashReducedPowers, `Dubon2026.SourceIntegralSlashReducedEquality, `Dubon2026.SourceIntegralSlashNormalizedGenerators, `Dubon2026.SourceIntegralSlashNormalizedRatioEquality, `Dubon2026.SourceIntegralSlashNormalizedMap, `Dubon2026.SourceIntegralSlashNormalizedLinearRelations, `Dubon2026.SourceIntegralSlashNormalizedCoordinate, `Dubon2026.SourceMonoidGeneratorAlgebraLift, `Dubon2026.SourceIntegralSlashNormalizedAlgebra, `Dubon2026.SourceCoordinateChartFractionExtension, `Dubon2026.SourceIntegralSlashFieldEmbedding, `Dubon2026.SourceOriginalResidueSlashFieldEmbedding, `Dubon2026.SourcePrescribedPrimitiveRootReduction, `Dubon2026.SourceOriginalResiduePrescribedCuspChart, `Dubon2026.SourceIntegralSlashReducedTranslation, `Dubon2026.SourceIntegralSlashReducedTranslationSeries, `Dubon2026.SourceIntegralSlashDiamondChart, `Dubon2026.SourceIntegralSlashNormalizedDiamond, `Dubon2026.SourceIntegralSlashFieldChartIdentities]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
