private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralDiamondScalarFixedField1739a, `SourceOriginalResidueCoordinateIndex1742a, `SourceIntegralDiamondFixedPlaceTransitivity1744c, `SourceOriginalResidueDiamondFixedField1745b, `SourceIntegralDiamondBaseFiberTransitivity1747a, `SourceOriginalResidueDiamondFiberTransitivity1748b]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.IntegralDiamond.scalar_fixedField_eq_fullBase_of_coordinate_degree, `Dubon2026.SourceFunctionField.IntegralDiamond.residue_original_coordinate_degree_eq_signed_index, `Dubon2026.SourceFunctionField.IntegralDiamond.diamond_exists_comapEquiv_of_restrict_fixed_eq, `Dubon2026.SourceFunctionField.IntegralDiamond.originalResidueDiamond_fixedField_eq_fullBase, `Dubon2026.SourceFunctionField.IntegralDiamond.diamond_exists_comapEquiv_of_fullBase_comap_eq, `Dubon2026.SourceFunctionField.IntegralDiamond.originalResidueDiamond_exists_comapEquiv_of_fullBase_comap_eq]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.IntegralDiamond.originalResidueDiamondAction]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue fixed fibers postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue fixed fibers theorems discovered"
  logInfo m!"SOURCE ORIGINAL RESIDUE FIXED FIBERS PREFLIGHT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
