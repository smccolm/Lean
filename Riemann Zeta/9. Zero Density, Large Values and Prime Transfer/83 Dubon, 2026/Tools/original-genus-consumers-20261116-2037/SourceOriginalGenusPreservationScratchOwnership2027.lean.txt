open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceCurveSeparatingCoordinateGeometry, `SourceCurveSeparatingCoordinateCanonical, `SourceCurveSeparatingCoordinateGlobalResidues, `SourceCurveSeparatingCoordinateResidueDuality, `SourceCurveSeparatingCoordinateCanonicalDegree, `SourceCurveSeparatingCoordinateHurwitz, `SourceCurveSeparatingHurwitzBound, `SourceCurveSeparatingPoleSums, `SourceCurveSeparatingHurwitzCuspBound, `SourceOriginalResidueHurwitz, `SourceOriginalWildGenusLower, `SourceCurveSeparatingFiberSums, `SourceOriginalTameGenusCriterion, `SourceOriginalTameGenusLower, `SourceOriginalLargeLevelGenusLower, `SourceGammaOneSmallLevelIndex, `SourceGammaOneSmallLevelGenus, `SourceOriginalGammaOneGenusPreservation]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
