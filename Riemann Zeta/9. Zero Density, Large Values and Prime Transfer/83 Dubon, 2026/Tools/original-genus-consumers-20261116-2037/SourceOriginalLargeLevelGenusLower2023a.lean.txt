/-! DUB-17/18 -> original all-weight association -> homogeneous Gamma1
realization -> original genus preservation. Assemble the proved wild and tame
reverse genus inequalities for every residue prime away from N and every
N > 3. There is no exceptional-prime exclusion or geometric certificate.
The remaining small levels are handled separately to preserve the full contract. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open CongruenceSubgroup GammaOne SourceCurve
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
  [CharP (IsLocalRing.ResidueField A) ℓ]
local notation "kk" => IsLocalRing.ResidueField A
local notation "F" => qExpFunctionFieldC kk (Gamma1 N)
private local instance : Algebra kk F := (F).toSubalgebra.algebra

theorem original_genus_lower_of_three_lt (hℓN : ¬ ℓ ∣ N) (hN : 3 < N) :
    genusFF Qbar (functionFieldOver (Gamma1 N) Qbar) ≤ genusFF kk F := by
  by_cases h2 : ℓ = 2
  · subst ℓ
    exact original_genus_lower_of_char_two A N hℓN hN
  by_cases h3 : ℓ = 3
  · subst ℓ
    exact original_genus_lower_of_char_three A N hℓN hN
  have hp : ℓ.Prime := Fact.out
  have h4 : ℓ ≠ 4 := by
    intro h
    rw [h] at hp
    norm_num at hp
  exact original_genus_lower_of_five_le A N ℓ hℓN hN (by have := hp.two_le; omega)

end
end Dubon2026.SourceFunctionField.MarkedChart
