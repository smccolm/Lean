private def runOriginalGenusPreservationAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCurveSeparatingCoordinateGeometry, `SourceCurveSeparatingCoordinateCanonical, `SourceCurveSeparatingCoordinateGlobalResidues, `SourceCurveSeparatingCoordinateResidueDuality, `SourceCurveSeparatingCoordinateCanonicalDegree, `SourceCurveSeparatingCoordinateHurwitz, `SourceCurveSeparatingHurwitzBound, `SourceCurveSeparatingPoleSums, `SourceCurveSeparatingHurwitzCuspBound, `SourceOriginalResidueHurwitz, `SourceOriginalWildGenusLower, `SourceCurveSeparatingFiberSums, `SourceOriginalTameGenusCriterion, `SourceOriginalTameGenusLower, `SourceOriginalLargeLevelGenusLower, `SourceGammaOneSmallLevelIndex, `SourceGammaOneSmallLevelGenus, `SourceOriginalGammaOneGenusPreservation]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.separating_coordinate_place_essFiniteType, `Dubon2026.SourceCurve.separating_coordinate_place_differentials, `Dubon2026.SourceCurve.separating_coordinate_place_dCoordGenerates, `Dubon2026.SourceCurve.isCurveOver_separating_coordinate, `Dubon2026.SourceCurve.separating_coordinate_D_support, `Dubon2026.SourceCurve.separating_coordinate_differential_support, `Dubon2026.SourceCurve.hasCanonicalDivisor_separating_coordinate, `Dubon2026.SourceCurve.separating_coordinate_tate_support, `Dubon2026.SourceCurve.separating_coordinate_global_tateResidue_zero, `Dubon2026.SourceCurve.separating_coordinate_global_residue_sum_zero, `Dubon2026.SourceCurve.separating_coordinate_index_eq_canonical_complement, `Dubon2026.SourceCurve.separating_coordinate_canonical_degree_eq_genusFF, `Dubon2026.SourceCurve.separatingHurwitzDivisor_apply, `Dubon2026.SourceCurve.separating_coordinate_ord_sub_one_le_D, `Dubon2026.SourceCurve.separatingHurwitzDivisor_nonneg, `Dubon2026.SourceCurve.degree_separatingHurwitzDivisor, `Dubon2026.SourceCurve.separating_hurwitz_differential_bound, `Dubon2026.SourceCurve.separating_hurwitz_ramification_bound, `Dubon2026.SourceCurve.sum_pole_orders_eq_finrank_separating, `Dubon2026.SourceCurve.sum_pole_orders_finset_separating, `Dubon2026.SourceCurve.separating_hurwitz_differential_cusp_bound, `Dubon2026.SourceCurve.separating_hurwitz_ramification_cusp_bound, `Dubon2026.SourceFunctionField.MarkedChart.original_hurwitz_differential_cusp_bound, `Dubon2026.SourceFunctionField.MarkedChart.original_hurwitz_ramification_cusp_bound, `Dubon2026.SourceFunctionField.MarkedChart.original_genus_lower_of_char_two, `Dubon2026.SourceFunctionField.MarkedChart.original_genus_lower_of_char_three, `Dubon2026.SourceCurve.sum_fiber_orders_eq_finrank_separating, `Dubon2026.SourceCurve.sum_fiber_orders_finset_separating, `Dubon2026.SourceFunctionField.MarkedChart.original_genus_lower_of_tame_fiber_bounds, `Dubon2026.SourceFunctionField.MarkedChart.original_genus_lower_of_five_le, `Dubon2026.SourceFunctionField.MarkedChart.original_genus_lower_of_three_lt, `Dubon2026.SourceFunctionField.GammaOne.gamma1_small_signed_index_le_four, `Dubon2026.SourceFunctionField.GammaOne.gamma1_arithmetic_genus_zero_of_le_three, `Dubon2026.SourceFunctionField.IntegralDiamond.gammaOne_residue_genus_eq_away_level]
  let constructions : Array Name := #[`Dubon2026.SourceCurve.separatingHurwitzDivisor]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE ORIGINAL GENUS PRESERVATION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runOriginalGenusPreservationAudit
