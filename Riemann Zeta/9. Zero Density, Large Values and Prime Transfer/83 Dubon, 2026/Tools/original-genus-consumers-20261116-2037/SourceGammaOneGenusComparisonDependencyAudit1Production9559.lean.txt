  let localModules : Array Name := #[`Dubon2026.SourceGammaOneConormDivisorSpaces, `Dubon2026.SourceGammaOneConormDimension, `Dubon2026.SourceGammaOneArithmeticGenus, `Dubon2026.SourceGammaOneArithmeticGenusBound]
