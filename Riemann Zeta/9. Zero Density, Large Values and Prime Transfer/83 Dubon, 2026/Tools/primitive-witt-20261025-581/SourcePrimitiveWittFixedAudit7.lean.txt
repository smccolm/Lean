private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourcePadicWittCoefficient, `SourceArithmeticPadicCyclotomic, `SourceArithmeticPadicCyclotomicReduction, `SourceArithmeticWittCyclotomic, `SourceResidualWittFixedDeterminant, `SourceResidualWittFixedRing, `SourcePrimitiveWittFixedResidual]
  let theorems : Array Name := #[`Dubon2026.sourcePadicInt_charP_map, `Dubon2026.sourcePadicWittCoefficient_injective, `Dubon2026.sourcePadicWittCoefficient_residue, `Dubon2026.sourceArithmetic_exists_primePower_root, `Dubon2026.sourceArithmeticPadicCyclotomicCharacter_root, `Dubon2026.sourcePadicInt_toZMod_from_pow_one, `Dubon2026.sourceArithmeticPadicCyclotomicCharacter_reduction, `Dubon2026.sourceArithmeticWittCyclotomicCharacter_reduction, `Dubon2026.sourceResidualWitt_fixedDeterminant_compatibility, `Dubon2026.sourceResidualWittFixed_localData, `Dubon2026.sourcePrimitive_exists_originalSupport_wittFixedResidualGalois_charP]
  let constructions : Array Name := #[`Dubon2026.sourcePadicWittCoefficient, `Dubon2026.sourceArithmeticPadicCyclotomicCharacter, `Dubon2026.sourceArithmeticWittCyclotomicCharacter, `Dubon2026.sourceResidualWittFixedData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE PRIMITIVE WITT FIXED AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
