import SourcePadicWittCoefficient
import SourceArithmeticPadicCyclotomic
import SourceArithmeticPadicCyclotomicReduction
import SourceArithmeticWittCyclotomic
import SourceResidualWittFixedDeterminant
import SourceResidualWittFixedRing
import SourcePrimitiveWittFixedResidual
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourcePadicWittCoefficient, `SourceArithmeticPadicCyclotomic, `SourceArithmeticPadicCyclotomicReduction, `SourceArithmeticWittCyclotomic, `SourceResidualWittFixedDeterminant, `SourceResidualWittFixedRing, `SourcePrimitiveWittFixedResidual]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
