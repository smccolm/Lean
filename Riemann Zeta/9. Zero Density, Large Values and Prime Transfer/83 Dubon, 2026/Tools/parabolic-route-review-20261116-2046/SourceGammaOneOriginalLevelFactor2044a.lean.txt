/-! DUB-17/18 -> original all-weight association -> Gamma1 character
realization -> scope check of the proposed parabolic-transfer route.
The actual diagonal cover factors through Gamma1 of the original level.
Consequently a universal surjectivity claim must account for original
Shimura-cover obstructions; full-H1 surjectivity alone does not remove them.
This exact factorization changes no source or accepted theorem. -/
namespace Dubon2026
noncomputable section
open CongruenceSubgroup
variable (Q N : ℕ) [NeZero N]

theorem sourceGammaOneDiagonalHom_mem_original_gammaOne
    (g : Gamma1 (Q * N ^ 2)) : (sourceGammaOneDiagonalHom Q N g).val ∈ Gamma1 Q := by
  have hg := (Gamma1_mem (Q * N ^ 2) g.val).mp g.property
  let π : ZMod (Q * N ^ 2) →+* ZMod Q := ZMod.castHom (dvd_mul_right Q (N ^ 2)) (ZMod Q)
  apply (Gamma1_mem Q _).mpr
  refine ⟨?_, ?_, Gamma0_mem.mp (sourceGammaOneDiagonalHom Q N g).property⟩
  · change (g.val 0 0 : ZMod Q) = 1
    simpa only [map_intCast, map_one] using congrArg π hg.1
  · change (g.val 1 1 : ZMod Q) = 1
    simpa only [map_intCast, map_one] using congrArg π hg.2.1

/-- The literal diagonal map with its proved original Gamma1 codomain. -/
def sourceGammaOneOriginalLevelHom : Gamma1 (Q * N ^ 2) →* Gamma1 Q where
  toFun g := ⟨(sourceGammaOneDiagonalHom Q N g).val,
    sourceGammaOneDiagonalHom_mem_original_gammaOne Q N g⟩
  map_one' := Subtype.ext (congrArg Subtype.val (map_one (sourceGammaOneDiagonalHom Q N)))
  map_mul' g h := Subtype.ext (congrArg Subtype.val (map_mul (sourceGammaOneDiagonalHom Q N) g h))

theorem sourceGammaOneDiagonalHom_original_level_factor :
    (Subgroup.inclusion (Gamma1_in_Gamma0 Q)).comp (sourceGammaOneOriginalLevelHom Q N) =
      sourceGammaOneDiagonalHom Q N := rfl

end
end Dubon2026
