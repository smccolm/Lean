private def runGammaOneParabolicShapiroAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceParabolicShapiroPotentials, `SourceOriginalCoinducedLocalCoboundary, `SourceParabolicTracePowers, `SourceGammaOneParabolicShapiro, `SourceGammaOneParabolicShapiroComparison]
  let theorems : Array Name := #[`Dubon2026.sourceTrivialShapiro_local_coboundary, `Dubon2026.sourceOriginalShapiro_local_coboundary, `Dubon2026.SourceParabolic.trace_sq_zpow, `Dubon2026.SourceParabolic.trace_sq_conjugate_zpow, `Dubon2026.sourceGammaOneShapiroLift_parabolic, `Dubon2026.sourceGammaOneShapiroH1_parabolic_of_evaluation, `Dubon2026.sourceGammaOneConstantH1Iso_parabolic_preimage, `Dubon2026.sourceGammaOneConstantH1Transfer_parabolic]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE GAMMAONE PARABOLIC SHAPIRO AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runGammaOneParabolicShapiroAudit
