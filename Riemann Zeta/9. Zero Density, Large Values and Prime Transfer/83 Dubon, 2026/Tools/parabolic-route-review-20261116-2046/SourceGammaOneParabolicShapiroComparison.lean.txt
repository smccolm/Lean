private theorem subgroupIso_cocycle_value (c : cocycles₁ (Rep.res (S).subtype A))
    (γ : Gamma1 (Q * N ^ 2)) :
    (H1IsoOfIsTrivial C).hom ((sourceGammaOneSubgroupH1Iso Q N w).hom (H1π _ c))
      (Additive.ofMul γ) = c (sourceGammaOneRangeEquiv Q N γ) := by
  have hm := H1π_comp_map_apply (sourceGammaOneRangeEquiv Q N).toMonoidHom
    ((Rep.resFunctor (sourceGammaOneRangeEquiv Q N).toMonoidHom).map
      (sourceGammaOneCoefficientTrivialIso Q N w).hom) c
  rw [sourceGammaOneSubgroupH1Iso_hom_eq_map]
  have hv := congrArg (fun x : H1 C => (H1IsoOfIsTrivial C).hom x (Additive.ofMul γ)) hm
  exact hv.trans (H1IsoOfIsTrivial_H1π_apply_apply _ (Additive.ofMul γ))

private theorem constantIso_cocycle_value (c : cocycles₁ B)
    (γ : Gamma1 (Q * N ^ 2)) :

      (Additive.ofMul γ) =
        sourceShapiroEvaluationCocycles A S c (sourceGammaOneRangeEquiv Q N γ) := by

  change (H1IsoOfIsTrivial C).hom
    ((sourceGammaOneSubgroupH1Iso Q N w).hom
      ((coindIso (Rep.res (S).subtype A) 1).hom (H1π _ c))) _ = _

theorem sourceGammaOneConstantH1Iso_parabolic_preimage
    (x : H1 B) (hx : (sourceGammaOneConstantH1Iso Q N w).hom x ∈ binaryParabolicH1 Q N w) :
    x ∈ sourceParabolicH1 (Gamma0 Q) B := by
  obtain ⟨c, rfl⟩ := (ModuleCat.epi_iff_surjective (H1π B)).mp inferInstance x
  apply sourceGammaOneShapiroH1_parabolic_of_evaluation Q N w c
  intro s hs
  obtain ⟨γ, rfl⟩ := (sourceGammaOneRangeEquiv Q N).surjective s
  have hγ : γ.val.val.trace ^ 2 = 4 := by
    change (sourceGammaOneDiagonalHom Q N γ).val.val.trace ^ 2 = 4 at hs
    rw [sourceGammaOneDiagonalHom_trace] at hs
    exact hs
  have hv := hx γ hγ
  rw [constantIso_cocycle_value] at hv
  exact hv

private theorem parabolicH1_coefficient_map {R : Type} [CommRing R]
    (Γ : Subgroup SL(2, ℤ)) (U V : Rep R Γ) (f : U ⟶ V)
    {x : H1 U} (hx : x ∈ sourceParabolicH1 Γ U) :
    (groupCohomology.functor R Γ 1).map f x ∈ sourceParabolicH1 Γ V := by
  obtain ⟨c, hc, rfl⟩ := hx
  change groupCohomology.map (MonoidHom.id Γ) f 1 (H1π U c) ∈ sourceParabolicH1 Γ V
  rw [H1π_comp_map_apply (MonoidHom.id Γ) f c]
  exact (sourceParabolicH1_cocycle_iff Γ V _).mpr
    (sourceParabolicCocycles_coefficientMap Γ U V f hc)

    simpa only [Iso.inv_hom_id_apply] using hy
  have h := parabolicH1_coefficient_map (Gamma0 Q) B A
    ((Rep.coindResAdjunction (ZMod N) S).counit.app A) hx
  unfold sourceGammaOneConstantH1Transfer
  exact h
