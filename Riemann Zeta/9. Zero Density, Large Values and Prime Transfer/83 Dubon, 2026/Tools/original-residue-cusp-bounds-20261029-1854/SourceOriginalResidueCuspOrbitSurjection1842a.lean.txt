theorem originalResidueOrbitCuspPlace_surjective (hℓN : ¬ ℓ ∣ N) (φ : Zbar →+* kk)
    (ζ : kkˣ) (hζ : IsPrimitiveRoot (ζ : kk) N)
    (hφ : ∀ z : Zbar,
      (z : ℂ) = Complex.exp (2 * Real.pi * Complex.I / N) → φ z = ζ) :
    Function.Surjective (fun q : Orbits (Γ := Gamma1 N) =>
      (⟨originalResidueOrbitCuspPlace A N ℓ hℓN φ ζ hφ q,
        originalResidueOrbitCuspPlace_j_neg A N ℓ hℓN φ ζ hφ q⟩ :
        {w : Place kk (qExpFunctionFieldC kk (Gamma1 N)) //
          w.ord ⟨jqModC kk, jqModC_mem_integralRatioField kk (Gamma1 N)⟩ < 0})) := by
  rintro ⟨w, hw⟩
  obtain ⟨β, hβ⟩ := exists_originalResidueCuspPlace_of_j_pole A N ℓ hℓN φ ζ hζ hφ w hw
  refine ⟨SourceManin.orbQ (translation (Gamma1 N)) (β⁻¹ : Cosets (Gamma1 N)), Subtype.ext ?_⟩
  rw [originalResidueOrbitCuspPlace_mk, originalResidueCosetCuspPlace_mk, inv_inv]
  exact hβ

theorem originalResidue_j_pole_card_le_complex_of_reduction (hℓN : ¬ ℓ ∣ N)
    (φ : Zbar →+* kk) (ζ : kkˣ) (hζ : IsPrimitiveRoot (ζ : kk) N)
    (hφ : ∀ z : Zbar,
      (z : ℂ) = Complex.exp (2 * Real.pi * Complex.I / N) → φ z = ζ) :
    Nat.card {w : Place kk (qExpFunctionFieldC kk (Gamma1 N)) //
      w.ord ⟨jqModC kk, jqModC_mem_integralRatioField kk (Gamma1 N)⟩ < 0} ≤
      Nat.card (PolePlaces (Gamma1 N)) := by
  letI : (plusMinus (Gamma1 N)).FiniteIndex := Subgroup.finiteIndex_of_le (le_plusMinus _)
  rw [gammaOne_j_pole_card_eq_translation_orbits N]
  exact Nat.card_le_card_of_surjective _
    (originalResidueOrbitCuspPlace_surjective A N ℓ hℓN φ ζ hζ hφ)

end
