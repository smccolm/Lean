import SourceIntegralDiamondNormalizedAction
import SourceIntegralSlashFieldNormalizedRatio
import SourceIntegralDiamondScalarNormalizedAction
import SourceIntegralSlashDiamondGenerator
import SourceIntegralSlashDiamondStabilizer
import SourceOriginalResidueDiamondCuspChart
import SourceOriginalResidueCuspStabilizerPlace
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceIntegralDiamondNormalizedAction, `SourceIntegralSlashFieldNormalizedRatio, `SourceIntegralDiamondScalarNormalizedAction, `SourceIntegralSlashDiamondGenerator, `SourceIntegralSlashDiamondStabilizer, `SourceOriginalResidueDiamondCuspChart, `SourceOriginalResidueCuspStabilizerPlace]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
