open CongruenceSubgroup GammaOne IntegralDiamond SourceCurve

local notation "Zbar" => integralClosure ℤ ℂ
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
  [CharP (IsLocalRing.ResidueField A) ℓ]
local notation "kk" => IsLocalRing.ResidueField A
private local instance : Algebra kk (qExpFunctionFieldC kk (Gamma1 N)) :=
  (qExpFunctionFieldC kk (Gamma1 N)).toSubalgebra.algebra
private local instance : Algebra kk (modularFunctionFieldCoeff N kk) :=
  (modularFunctionFieldCoeff N kk).toSubalgebra.algebra

theorem exists_originalResidueCuspPlace_of_j_pole (hℓN : ¬ ℓ ∣ N) (φ : Zbar →+* kk)
    (ζ : kkˣ) (hζ : IsPrimitiveRoot (ζ : kk) N)
    (hφ : ∀ z : Zbar,
      (z : ℂ) = Complex.exp (2 * Real.pi * Complex.I / N) → φ z = ζ)
    (w : Place kk (qExpFunctionFieldC kk (Gamma1 N)))
    (hw : w.ord ⟨jqModC kk, jqModC_mem_integralRatioField kk (Gamma1 N)⟩ < 0) :
    ∃ β : SL(2, ℤ), originalResidueCuspPlace A N ℓ hℓN φ β = w := by
  obtain ⟨s, hs⟩ := originalResiduePole_lies_over_labeledCusp A N ℓ hℓN ζ hζ w hw
  obtain ⟨β, hβ⟩ := exists_originalResidueCuspFamily_over_label A N ℓ hℓN φ ζ hζ hφ s
  obtain ⟨γ, hγ⟩ := originalResidueDiamond_exists_comapEquiv_of_fullBase_comap_eq A N ℓ hℓN
    (originalResidueCuspPlace A N ℓ hℓN φ β) w (hs.trans hβ.symm)
  exact ⟨(γ⁻¹).val * β, (originalResidueCuspPlace_diamond_mul A N ℓ hℓN φ β γ).trans hγ⟩

end
end Dubon2026.SourceFunctionField.IntegralSlash
