the exact entry bridge for native diamond transitivity and orbit surjectivity. -/

theorem exists_originalResidueCuspFamily_over_label (hℓN : ¬ ℓ ∣ N) (φ : Zbar →+* kk)
    (ζ : kkˣ) (hζ : IsPrimitiveRoot (ζ : kk) N)
    (hφ : ∀ z : Zbar,
      (z : ℂ) = Complex.exp (2 * Real.pi * Complex.I / N) → φ z = ζ)
    (s : {s // s ∈ CuspLabels.labels N}) :
    let hN : (N : kk) ≠ 0 := fun h => hℓN ((CharP.cast_eq_zero_iff kk ℓ N).mp h)
    ∃ β : SL(2, ℤ),
      (originalResidueCuspPlace A N ℓ hℓN φ β).toValuationSubring.comap
        (IntermediateField.inclusion (fullBase_le_originalField kk N)).toRingHom =
          (CoefficientCuspLabels.place N kk hN ζ hζ s).toValuationSubring := by
  dsimp only
  let hN : (N : kk) ≠ 0 := fun h => hℓN ((CharP.cast_eq_zero_iff kk ℓ N).mp h)
  have hs := (CuspLabels.mem_labels N s.1).mp s.2
  letI : NeZero s.1.1 := ⟨index_fst_ne_zero (CuspLabels.label_mem_indices N s.1 s.2)⟩
  obtain ⟨β, hβ, hb⟩ := exists_primitiveCuspMatrix N s.1.1 s.1.2 hs.1 hs.2.2.symm
  let Φ := originalResidueCuspHom A N ℓ hℓN φ β
  let θ := CoefficientCuspLabels.embedding N kk hN ζ hζ s
  let ι := IntermediateField.inclusion (fullBase_le_originalField kk N)
  have hchart := CoefficientCuspLabels.embedding_spec N kk hN ζ hζ s
  have hj : θ (jCoeff N kk) = qExpand kk N (jqModC kk) := by
    simpa only [TS, qTwist_one_apply] using hchart.1
  have hjN : θ ⟨jqNModC kk N, jqd_mem_coeff_field N kk (dvd_refl N)⟩ =
      qExpand kk (s.1.1 * s.1.1) (qTwist (ζ ^ ((s.1.2 : ℤ) * (s.1.1 : ℤ))) (jqModC kk)) := by
    simpa only [slotValue_eq, TS, ← Nat.cast_mul, zpow_natCast] using hchart.2.1
  have hζN : ζ ^ N = 1 := Units.ext (by
    rw [Units.val_pow_eq_pow_val, Units.val_one]
    exact hζ.pow_eq_one)
  have hΦN := originalReducedSlashFieldHom_jqN N ℓ φ hN ζ hζN hφ β s.1.1 (s.1.2 : ℤ)
    (hβ ▸ dvd_refl _) hs.1 hb Φ (originalResidueCuspHom_normalized A N ℓ hℓN φ hN β)
  have hbase : Φ.comp ι = θ := fullBase_algHom_ext N kk hN _ _
    ((originalResidueCuspHom_j A N ℓ hℓN φ β).trans hj.symm) (hΦN.trans hjN.symm)
  refine ⟨β, ?_⟩
  ext x
  rw [ValuationSubring.mem_comap, originalResidueCuspPlace_mem_iff, Place.mem_iff_ord_nonneg]
  change 0 ≤ (Φ (ι x)).order ↔ _
  rw [show Φ (ι x) = θ x from AlgHom.congr_fun hbase x, ← hchart.2.2.1 x]
  exact mul_nonneg_iff_of_pos_right (show (0 : ℤ) <
    ((s.1.1 * Nat.gcd s.1.1 (N / s.1.1) : ℕ) : ℤ) by
      exact_mod_cast Nat.mul_pos (NeZero.pos s.1.1)
        (Nat.pos_of_ne_zero (Nat.gcd_ne_zero_left (NeZero.ne s.1.1))))

end
