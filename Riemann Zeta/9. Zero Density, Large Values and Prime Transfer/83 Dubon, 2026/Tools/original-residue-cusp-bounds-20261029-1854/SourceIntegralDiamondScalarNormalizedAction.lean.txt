theorem reducedFieldActionOver_normalized (K : Type*) [Field K] (N ℓ : ℕ)
    [NeZero N] [Fact ℓ.Prime] [Algebra (ZMod ℓ) K] (hℓN : ¬ ℓ ∣ N)
    (γ : Gamma0 N) {r : ℕ} (f : ModularForm (Gamma1 N) (12 * (r : ℤ)))
    {p P : PowerSeries ℤ} (hp : IsIntegralQExp f p) (a : ℕ)
    (hP : IsIntegralQExp (scaledDiamond N γ⁻¹ a f) P) :
    (reducedFieldActionOver K N ℓ hℓN γ
      ⟨intSeriesC K p / intSeriesC K discriminantSeries ^ r,
        normalizedGenerators_subset N K ⟨r, f, p, hp, rfl⟩⟩ : LaurentSeries K) =
      (intSeriesC K P / intSeriesC K discriminantSeries ^ r) /
        (N : LaurentSeries K) ^ a := by
  let x₀ : qExpFunctionFieldC (ZMod ℓ) (Gamma1 N) :=
    ⟨intSeriesC (ZMod ℓ) p / intSeriesC (ZMod ℓ) discriminantSeries ^ r,
      normalizedGenerators_subset N (ZMod ℓ) ⟨r, f, p, hp, rfl⟩⟩
  have he : originalFieldCoefficientHom (L := K) (Gamma1 N) x₀ =
      ⟨intSeriesC K p / intSeriesC K discriminantSeries ^ r,
        normalizedGenerators_subset N K ⟨r, f, p, hp, rfl⟩⟩ := by
    apply Subtype.ext
    change coeffMap (algebraMap (ZMod ℓ) K)
      (intSeriesC (ZMod ℓ) p / intSeriesC (ZMod ℓ) discriminantSeries ^ r) = _
    rw [map_div₀, map_pow, coeffMap_intSeries, coeffMap_intSeries]
  rw [← he, reducedFieldActionOver_coeff, originalFieldCoefficientHom_coe]
  change coeffMap (algebraMap (ZMod ℓ) K)
    (reducedFieldAction N ℓ hℓN γ
      ⟨intSeriesC (ZMod ℓ) p / intSeriesC (ZMod ℓ) discriminantSeries ^ r, _⟩) = _
  rw [reducedFieldAction_normalized N ℓ hℓN γ f hp a hP,
    map_div₀, map_div₀, map_pow, map_pow, coeffMap_intSeries, coeffMap_intSeries, map_natCast]

end
