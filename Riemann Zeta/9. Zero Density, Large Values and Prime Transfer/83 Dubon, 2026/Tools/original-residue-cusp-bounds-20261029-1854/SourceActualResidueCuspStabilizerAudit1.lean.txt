private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralDiamondNormalizedAction, `SourceIntegralSlashFieldNormalizedRatio, `SourceIntegralDiamondScalarNormalizedAction, `SourceIntegralSlashDiamondGenerator, `SourceIntegralSlashDiamondStabilizer, `SourceOriginalResidueDiamondCuspChart, `SourceOriginalResidueCuspStabilizerPlace]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.IntegralDiamond.gammaOneDiscriminantPow_slash, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldAction_normalized, `Dubon2026.SourceFunctionField.IntegralSlash.originalReducedSlashFieldHom_normalized_ratio, `Dubon2026.SourceFunctionField.IntegralDiamond.reducedFieldActionOver_normalized, `Dubon2026.SourceFunctionField.IntegralSlash.originalReducedSlashFieldHom_diamond_generator, `Dubon2026.SourceFunctionField.IntegralSlash.originalReducedSlashFieldHom_diamond_stabilizer, `Dubon2026.SourceFunctionField.IntegralSlash.exists_originalResidueDiamondCuspFieldHom, `Dubon2026.SourceFunctionField.IntegralSlash.exists_originalResidueCuspStabilizerPlace, `Dubon2026.SourceFunctionField.IntegralSlash.exists_originalResidueCuspFixedPlace]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue slash embedding postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue slash embedding theorems discovered"
  logInfo m!"SOURCE ACTUAL RESIDUE CUSP STABILIZER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
