local notation "Qbar" => AlgebraicClosure ℚ

  [CharP (IsLocalRing.ResidueField A) ℓ]
local notation "kk" => IsLocalRing.ResidueField A
private local instance : IsAlgClosed kk := valuation_residue_isAlgClosed A
private local instance : Algebra kk (qExpFunctionFieldC kk (Gamma1 N)) :=
  (qExpFunctionFieldC kk (Gamma1 N)).toSubalgebra.algebra
private local instance : Algebra kk (modularFunctionFieldCoeff N kk) :=
  (modularFunctionFieldCoeff N kk).toSubalgebra.algebra

theorem exists_originalResidueLabeledCuspFixedPlace (hℓN : ¬ ℓ ∣ N)
    (ζ : kkˣ) (hζ : IsPrimitiveRoot (ζ : kk) N) (s : {s // s ∈ CuspLabels.labels N}) :
    let hN : (N : kk) ≠ 0 := fun h => hℓN ((CharP.cast_eq_zero_iff kk ℓ N).mp h)
    ∃ (g : SL(2, ℤ)) (w : Place kk (qExpFunctionFieldC kk (Gamma1 N))),
      g 1 0 = (s.1.1 : ℤ) ∧
      w.toValuationSubring.comap
        (IntermediateField.inclusion (fullBase_le_originalField kk N)).toRingHom =
          (CoefficientCuspLabels.place N kk hN ζ hζ s).toValuationSubring ∧
      w.ord ⟨jqModC kk, jqModC_mem_integralRatioField kk (Gamma1 N)⟩ < 0 ∧
      ∀ (γ : Gamma0 N) (j : ℤ),
        (g⁻¹ * γ.val * g = ModularGroup.T ^ j ∨
          g⁻¹ * γ.val * g = -(ModularGroup.T ^ j)) →
        w.comapEquiv (originalResidueDiamondAction A N ℓ hℓN γ⁻¹) = w := by
  dsimp only
  let hN : (N : kk) ≠ 0 := fun h => hℓN ((CharP.cast_eq_zero_iff kk ℓ N).mp h)
  have hs := (CuspLabels.mem_labels N s.1).mp s.2
  letI : NeZero s.1.1 := ⟨index_fst_ne_zero (CuspLabels.label_mem_indices N s.1 s.2)⟩
  obtain ⟨g, hg, hb⟩ := exists_primitiveCuspMatrix N s.1.1 s.1.2 hs.1 hs.2.2.symm
  have hchart := CoefficientCuspLabels.embedding_spec N kk hN ζ hζ s
  have hj : CoefficientCuspLabels.embedding N kk hN ζ hζ s (jCoeff N kk) =
      qExpand kk N (jqModC kk) := by
    simpa only [TS, qTwist_one_apply] using hchart.1
  have hjN : CoefficientCuspLabels.embedding N kk hN ζ hζ s
      ⟨jqNModC kk N, jqd_mem_coeff_field N kk (dvd_refl N)⟩ =
      qExpand kk (s.1.1 * s.1.1) (qTwist (ζ ^ ((s.1.2 : ℤ) * (s.1.1 : ℤ))) (jqModC kk)) := by
    simpa only [slotValue_eq, TS, ← Nat.cast_mul, zpow_natCast] using hchart.2.1
  have he : 0 < s.1.1 * Nat.gcd s.1.1 (N / s.1.1) :=
    Nat.mul_pos (NeZero.pos s.1.1) (Nat.pos_of_ne_zero (Nat.gcd_ne_zero_left (NeZero.ne s.1.1)))
  obtain ⟨w, hw, hp, hfix⟩ := exists_originalResidueCuspFixedPlace_over_chart A N ℓ hℓN ζ hζ
    g s.1.1 (s.1.2 : ℤ) (hg ▸ dvd_refl _) hs.1 hb
    (CoefficientCuspLabels.embedding N kk hN ζ hζ s) hj hjN
    (CoefficientCuspLabels.place N kk hN ζ hζ s) _ he hchart.2.2.1
  exact ⟨g, w, hg, hw, hp, hfix⟩

end
end Dubon2026.SourceFunctionField.IntegralSlash
