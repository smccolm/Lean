import Dubon2026.SourceGammaOneBoundaryWordKernel
import Dubon2026.SourcePrimitiveH1SquareCongruence
import Dubon2026.SourceOperatorRangeCharacter
import Dubon2026.SourceGammaOneBoundaryQuotient
import Dubon2026.SourceGammaOneExactJointRing
import Dubon2026.SourceGammaOneExactJointSquare
import Dubon2026.SourceGammaOneExactJointFinite
import Dubon2026.SourceGammaOneExactJointCharacter
import Dubon2026.SourceGammaOneGeometricExactJointCharacter
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOneBoundaryWordKernel, `Dubon2026.SourcePrimitiveH1SquareCongruence, `Dubon2026.SourceOperatorRangeCharacter, `Dubon2026.SourceGammaOneBoundaryQuotient, `Dubon2026.SourceGammaOneExactJointRing, `Dubon2026.SourceGammaOneExactJointSquare, `Dubon2026.SourceGammaOneExactJointFinite, `Dubon2026.SourceGammaOneExactJointCharacter, `Dubon2026.SourceGammaOneGeometricExactJointCharacter]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
