private def runGammaOneExactJointCharacterAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneBoundaryWordKernel, `SourcePrimitiveH1SquareCongruence, `SourceOperatorRangeCharacter, `SourceGammaOneBoundaryQuotient, `SourceGammaOneExactJointRing, `SourceGammaOneExactJointSquare, `SourceGammaOneExactJointFinite, `SourceGammaOneExactJointCharacter, `SourceGammaOneGeometricExactJointCharacter]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientOriginalTransfer_boundary_word_kernel, `Dubon2026.sourcePrimitiveH1Hecke_square_congruence_integral, `Dubon2026.sourcePrimitiveH1AdicQuotientCharacter_zero_of_operator_square, `Dubon2026.sourceOperatorRangeCharacterData_of_kernel, `Dubon2026.SourceFunctionField.GammaOne.binaryBoundaryQuotient_word, `Dubon2026.SourceFunctionField.GammaOne.binaryBoundaryQuotient_word_zero_parabolic, `Dubon2026.SourceFunctionField.GammaOne.binaryBoundaryQuotient_finite, `Dubon2026.SourceFunctionField.GammaOne.binaryExactJointHeckeWord_eq_zero_iff, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientOriginalTransfer_exact_joint_kernel_square, `Dubon2026.SourceFunctionField.GammaOne.binaryExactJointOperatorRing_finite, `Dubon2026.SourceFunctionField.GammaOne.binaryExactJoint_original_character, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_exact_joint_original_character]
  let constructions : Array Name := #[`Dubon2026.sourceOperatorRangeCharacterData, `Dubon2026.SourceFunctionField.GammaOne.binaryBoundaryQuotient, `Dubon2026.SourceFunctionField.GammaOne.binaryBoundaryQuotientPrimeOperators, `Dubon2026.SourceFunctionField.GammaOne.binaryExactJointOperatorRing, `Dubon2026.SourceFunctionField.GammaOne.instRingBinaryExactJointOperatorRing, `Dubon2026.SourceFunctionField.GammaOne.binaryExactJointHeckeWord]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE GAMMAONE EXACT JOINT CHARACTER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runGammaOneExactJointCharacterAudit
