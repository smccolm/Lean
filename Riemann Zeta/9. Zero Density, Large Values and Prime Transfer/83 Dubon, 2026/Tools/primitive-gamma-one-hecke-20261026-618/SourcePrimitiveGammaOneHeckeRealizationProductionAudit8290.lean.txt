  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`Dubon2026.SourceH1TraceBaseChangeCoefficient, `Dubon2026.SourceGammaOneCoverBinaryHeckeTrace, `Dubon2026.SourceGammaOneHeckeUpperImage, `Dubon2026.SourceGammaOneSubgroupCohomologyMap, `Dubon2026.SourceGammaOneHeckePullbackSquare, `Dubon2026.SourceGammaOneHeckeUpperTraceReindex, `Dubon2026.SourceH1TraceIntersection, `Dubon2026.SourceGammaOneIntersectionTraceExchange, `Dubon2026.SourceGammaOneUpperH1Composition, `Dubon2026.SourceGammaOneSubgroupTraceHecke, `Dubon2026.SourceGammaOneTransferHecke, `Dubon2026.SourceGammaOneHeckeWords, `Dubon2026.SourceGammaOneOperatorDescent, `Dubon2026.SourceGammaOneHeckeFinite, `Dubon2026.SourcePrimitiveGammaOneHeckeCharacter, `Dubon2026.SourcePrimitiveGammaOneH1Realization]
  let theorems : Array Name := #[`Dubon2026.sourceSubgroupH1Trace_pullback_coefficient, `Dubon2026.sourceGammaOneCoverBinaryHeckeTrace_pullback, `Dubon2026.sourceGammaOneCoverBinaryHeckeTrace_iterated, `Dubon2026.sourceGammaOneDiagonalHom_heckeUpper_iff, `Dubon2026.sourceGammaOneCoverHeckeUpperHom_injective, `Dubon2026.sourceGammaOneCoverHeckeUpperHom_range, `Dubon2026.sourceGammaOneCoverHeckeUpperEquiv_apply, `Dubon2026.sourceGammaOneCoverHeckeUpperEquiv_conjugation, `Dubon2026.sourceGammaOneSubgroupH1Iso_hom_eq_map, `Dubon2026.sourceGammaOneConstantH1Transfer_subgroupIso, `Dubon2026.sourceGammaOneHeckeH1Pullback_square, `Dubon2026.sourceGammaOneRangeEquiv_heckeUpper_comap, `Dubon2026.sourceGammaOneHeckeUpperTrace_reindex, `Dubon2026.sourceSubgroupH1Trace_intersection_swap, `Dubon2026.sourceGammaOneHeckeIntersectionSwap_cover, `Dubon2026.sourceGammaOneCoverTrace_exchange, `Dubon2026.sourceGammaOneUpperH1Composition, `Dubon2026.sourceGammaOneSubgroupTrace_hecke_square, `Dubon2026.sourceGammaOneConstantH1Transfer_hecke, `Dubon2026.sourceGammaOneOriginalHeckeWord_val, `Dubon2026.sourceGammaOneHeckeWord_transfer, `Dubon2026.sourceGammaOneOriginalHeckeWord_change, `Dubon2026.sourceGammaOneHeckeWord_modulus_zero, `Dubon2026.sourceGammaOneHeckeWord_modulus_kernel, `Dubon2026.sourceGammaOneToModulus_comp, `Dubon2026.sourceGammaOneConstantH1_finite, `Dubon2026.sourceGammaOneHeckeAlgebra_finite, `Dubon2026.sourcePrimitiveGammaOneHeckeCharacter_comp, `Dubon2026.sourcePrimitiveGammaOneHeckeCharacter_hecke, `Dubon2026.sourcePrimitive_exists_gammaOneH1HeckeCharacters]
  let constructions : Array Name := #[`Dubon2026.sourceGammaOneCoverHeckeUpperEquiv, `Dubon2026.sourceGammaOneUpperRangeHom, `Dubon2026.sourceSubgroupIntersectionSwap, `Dubon2026.sourceGammaOneHeckeIntersectionSwap, `Dubon2026.sourceGammaOneHeckePrimeForget, `Dubon2026.sourceGammaOnePrimeOperators, `Dubon2026.sourceGammaOneHeckeAlgebra, `Dubon2026.sourceGammaOneOriginalHeckeWord, `Dubon2026.sourceGammaOneToModulus, `Dubon2026.sourcePrimitiveGammaOneHeckeCharacter, `Dubon2026.sourcePrimitiveGammaOneH1CharacterData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue functional postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue functional theorems discovered"
  logInfo m!"SOURCE PRIMITIVE GAMMA ONE HECKE REALIZATION AUDIT PASS: {count} theorems including private/generated declarations. The actual surjective Gamma1 transfer intertwines full Hecke operators, and every original primitive form supplies finite Gamma1 operator characters with original Fourier values. Arithmetic geometric and diamond comparison, Galois realization, all-weight association and higher continuation remain open; 16/20 gates."
