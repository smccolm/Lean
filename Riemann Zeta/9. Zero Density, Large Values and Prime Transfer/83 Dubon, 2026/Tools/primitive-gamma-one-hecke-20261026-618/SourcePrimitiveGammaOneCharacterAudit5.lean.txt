private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneHeckeWords, `SourceGammaOneOperatorDescent, `SourceGammaOneHeckeFinite, `SourcePrimitiveGammaOneHeckeCharacter, `SourcePrimitiveGammaOneH1Realization]
  let theorems : Array Name := #[`Dubon2026.sourceGammaOneOriginalHeckeWord_val, `Dubon2026.sourceGammaOneHeckeWord_transfer, `Dubon2026.sourceGammaOneOriginalHeckeWord_change, `Dubon2026.sourceGammaOneHeckeWord_modulus_zero, `Dubon2026.sourceGammaOneHeckeWord_modulus_kernel, `Dubon2026.sourceGammaOneToModulus_comp, `Dubon2026.sourceGammaOneConstantH1_finite, `Dubon2026.sourceGammaOneHeckeAlgebra_finite, `Dubon2026.sourcePrimitiveGammaOneHeckeCharacter_comp, `Dubon2026.sourcePrimitiveGammaOneHeckeCharacter_hecke, `Dubon2026.sourcePrimitive_exists_gammaOneH1HeckeCharacters]
  let constructions : Array Name := #[`Dubon2026.sourceGammaOneHeckePrimeForget, `Dubon2026.sourceGammaOnePrimeOperators, `Dubon2026.sourceGammaOneHeckeAlgebra, `Dubon2026.sourceGammaOneOriginalHeckeWord, `Dubon2026.sourceGammaOneToModulus, `Dubon2026.sourcePrimitiveGammaOneHeckeCharacter, `Dubon2026.sourcePrimitiveGammaOneH1CharacterData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE PRIMITIVE GAMMA ONE CHARACTER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
