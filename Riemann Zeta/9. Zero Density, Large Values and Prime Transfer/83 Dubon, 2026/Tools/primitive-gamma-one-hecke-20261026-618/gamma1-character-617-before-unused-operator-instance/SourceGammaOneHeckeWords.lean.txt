/-! DUB-17/18 -> original all-weight adic association -> actual finite Gamma1
Hecke characters. Restrict to primes coprime to level times modulus and reuse
the existing additive word intertwiner for the genuine surjective transfer
and original integral coefficient reduction. -/

open CongruenceSubgroup

/-- A prime away from level times modulus is an original good prime. -/
def sourceGammaOneHeckePrimeForget (Q N : ℕ) :
    sourceBinaryHeckePrime (Q * N) → sourceBinaryHeckePrime Q :=
  fun ell => ⟨ell.val, ell.property.1,
    ell.property.2.of_dvd_right (dvd_mul_right Q N)⟩

/-- The actual full Gamma1 Hecke operators, with the original homogeneous
coefficient action, at primes away from level times modulus. -/
def sourceGammaOnePrimeOperators (Q N w : ℕ) [NeZero N] :
    sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N)
      (groupCohomology.H1 (Rep.trivial (ZMod N) (Gamma1 (Q * N ^ 2))
        (sourceBinaryCoefficientRep (ZMod N) Q w))) :=
  fun ell => letI : NeZero ell.val := ⟨ell.property.1.ne_zero⟩
    sourceGammaOneConstantHeckeH1 (ZMod N) (sourceBinaryCoefficientRep (ZMod N) Q w)
      (Q * N ^ 2) ell.val
      (homogeneousMatrixAction w (Matrix.diagonal ![(ell.val : ZMod N), 1]))

/-- The ring of actual Gamma1 H1 operators generated by these genuine
prime correspondences. It is the literal word-action image. -/
def sourceGammaOneHeckeAlgebra (Q N w : ℕ) [NeZero N] :=
  (FreeAlgebra.lift ℤ (sourceGammaOnePrimeOperators Q N w)).range

/-- The same restricted prime words in the original integral H1 algebra. -/
def sourceGammaOneOriginalHeckeWord (Q N w : ℕ) :
    FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N)) →ₐ[ℤ] sourceBinaryH1HeckeAlgebra Q w :=
  FreeAlgebra.lift ℤ (fun ell =>
    ⟨sourceBinaryH1PrimeOperators ℤ Q w (sourceGammaOneHeckePrimeForget Q N ell),
      Algebra.subset_adjoin ⟨sourceGammaOneHeckePrimeForget Q N ell, rfl⟩⟩)

/-- The original restricted word has its literal full operator value. -/
theorem sourceGammaOneOriginalHeckeWord_val (Q N w : ℕ)
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N))) :
    (sourceGammaOneOriginalHeckeWord Q N w a).val =
      FreeAlgebra.lift ℤ
        (fun ell => sourceBinaryH1PrimeOperators ℤ Q w (sourceGammaOneHeckePrimeForget Q N ell)) a := by
  have he : (sourceBinaryH1HeckeAlgebra Q w).val.comp (sourceGammaOneOriginalHeckeWord Q N w) =
      FreeAlgebra.lift ℤ
        (fun ell => sourceBinaryH1PrimeOperators ℤ Q w (sourceGammaOneHeckePrimeForget Q N ell)) := by
    apply FreeAlgebra.hom_ext
    funext ell
    simp only [Function.comp_apply, AlgHom.comp_apply, sourceGammaOneOriginalHeckeWord, FreeAlgebra.lift_ι_apply]
    rfl
  exact AlgHom.congr_fun he a

/-- The original surjective transfer intertwines every integral word in
the actual good-prime Gamma1 correspondences. -/
theorem sourceGammaOneHeckeWord_transfer (Q N w : ℕ) [NeZero Q] [NeZero N]
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N)))
    (x : groupCohomology.H1 (Rep.trivial (ZMod N) (Gamma1 (Q * N ^ 2))
      (sourceBinaryCoefficientRep (ZMod N) Q w))) :
    sourceGammaOneConstantH1Transfer Q N w
        (FreeAlgebra.lift ℤ (sourceGammaOnePrimeOperators Q N w) a x) =
      FreeAlgebra.lift ℤ
        (fun ell => sourceBinaryH1PrimeOperators (ZMod N) Q w
          (sourceGammaOneHeckePrimeForget Q N ell)) a
        (sourceGammaOneConstantH1Transfer Q N w x) := by
  apply sourceHeckeWord_intertwine_addHom _ _
    (sourceGammaOneConstantH1Transfer Q N w).toAddMonoidHom
  intro ell y
  letI : NeZero ell.val := ⟨ell.property.1.ne_zero⟩
  exact (LinearMap.congr_fun
    (sourceGammaOneConstantH1Transfer_hecke Q N ell.val w ell.property.1 ell.property.2) y).symm

/-- Original integral coefficient reduction intertwines the same full
restricted prime-word action. -/
theorem sourceGammaOneOriginalHeckeWord_change (Q N w : ℕ)
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N)))
    (x : groupCohomology.H1 (sourceBinaryCoefficientRep ℤ Q w)) :
    sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w
        ((sourceGammaOneOriginalHeckeWord Q N w a).val x) =
      FreeAlgebra.lift ℤ
        (fun ell => sourceBinaryH1PrimeOperators (ZMod N) Q w
          (sourceGammaOneHeckePrimeForget Q N ell)) a
        (sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w x) := by
  rw [sourceGammaOneOriginalHeckeWord_val]
  apply sourceHeckeWord_intertwine_addHom _ _
    (sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w).toAddMonoidHom
  intro ell y
  letI : NeZero ell.val := ⟨ell.property.1.ne_zero⟩
  exact sourceBinaryH1Change_hecke (Int.castRingHom (ZMod N)) ell.property.1 w y

end
