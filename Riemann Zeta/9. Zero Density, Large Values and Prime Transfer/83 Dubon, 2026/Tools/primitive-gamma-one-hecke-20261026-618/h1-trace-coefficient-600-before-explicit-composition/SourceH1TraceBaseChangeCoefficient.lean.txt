          subgroupCohomologyTrace B (S.comap φ) 1 := by
  have hmap : groupCohomology.map φ f 1 =
      groupCohomology.map φ (𝟙 (Rep.res φ A)) 1 ≫
        (groupCohomology.functor R D 1).map f := by
    change groupCohomology.map φ f 1 =
      groupCohomology.map φ (𝟙 (Rep.res φ A)) 1 ≫
        groupCohomology.map (MonoidHom.id D) f 1
    rw [← groupCohomology.map_comp]
    rfl
  rw [hmap]
  erw [← Category.assoc, sourceSubgroupH1Trace_pullback A φ S hcosets]
  erw [Category.assoc, ← originalCohomologyTrace_coefficient_naturality (S.comap φ) f 1,
    ← Category.assoc]
  change (groupCohomology.map (sourceTracePullbackSubgroupHom φ S)
      (𝟙 (Rep.res (S.comap φ).subtype (Rep.res φ A))) 1 ≫
      groupCohomology.map (MonoidHom.id (S.comap φ))
        ((Rep.resFunctor (S.comap φ).subtype).map f) 1) ≫ _ = _
  rw [← groupCohomology.map_comp]
  rfl

end
