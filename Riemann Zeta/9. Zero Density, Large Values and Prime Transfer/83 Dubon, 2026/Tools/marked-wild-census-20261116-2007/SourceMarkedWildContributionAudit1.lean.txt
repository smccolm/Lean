private def runMarkedWildContributionAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMarkedDihedralFiberBound, `SourceMarkedSignFiberSymmetries, `SourceMarkedOriginalWildFiberThree, `SourceMarkedOrderTwelveFiberBound, `SourceMarkedSignActionHom, `SourceMarkedCharTwoSignAction, `SourceMarkedOriginalWildFiberTwo, `SourceMarkedNormalOrderDescent, `SourceMarkedOriginalOrderBoundTwo, `SourceMarkedOriginalOrderBoundThree, `SourceMarkedOriginalPositiveDegree, `SourceMarkedFiberOrderRigidity, `SourceMarkedOriginalExactOrderTwo, `SourceMarkedOriginalExactOrderThree, `SourceMarkedOriginalWildContributionTwo, `SourceMarkedOriginalWildContributionThree]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.MarkedTorsion.six_mul_card_fiber_le_of_free_dihedral, `Dubon2026.SourceFunctionField.MarkedTorsion.signClass_fiber_factor_invariant, `Dubon2026.SourceFunctionField.MarkedTorsion.signClassesEquiv_conjugation, `Dubon2026.SourceFunctionField.MarkedChart.original_six_mul_card_j_zero_fiber_le_of_char_three, `Dubon2026.SourceFunctionField.MarkedTorsion.twelve_mul_card_fiber_le_of_free_marks, `Dubon2026.SourceFunctionField.MarkedTorsion.signClassesEquiv_eq_self_of_negation, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwo_sign_le_marked_action_kernel, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwo_signProjection_marked_smul, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwo_quartic_marked_ne_self, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwo_cubic_marked_ne_self, `Dubon2026.SourceFunctionField.MarkedChart.original_twelve_mul_card_j_zero_fiber_le_of_char_two, `Dubon2026.SourceFunctionField.MarkedChart.coordinate_order_dvd_of_normal_embedding, `Dubon2026.SourceFunctionField.MarkedChart.original_positive_j_order_dvd_twelve, `Dubon2026.SourceFunctionField.MarkedChart.original_positive_j_order_dvd_six, `Dubon2026.SourceFunctionField.MarkedChart.original_positive_j_fiber_finite_and_degree, `Dubon2026.SourceFunctionField.MarkedChart.eq_const_of_finsum_and_card_bound, `Dubon2026.SourceFunctionField.MarkedChart.original_positive_j_order_eq_twelve, `Dubon2026.SourceFunctionField.MarkedChart.original_positive_j_order_eq_six, `Dubon2026.SourceFunctionField.MarkedChart.original_char_two_differential_contribution, `Dubon2026.SourceFunctionField.MarkedChart.original_char_three_differential_contribution]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.MarkedTorsion.signClassesPermHom, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwoMarkedStabilizerAction, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwoMarkedSignAction]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE MARKED WILD CONTRIBUTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runMarkedWildContributionAudit
