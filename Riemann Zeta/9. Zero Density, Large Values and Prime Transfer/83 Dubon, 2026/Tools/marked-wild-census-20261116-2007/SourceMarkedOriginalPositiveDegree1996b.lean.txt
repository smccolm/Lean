/-! DUB-17/18 -> reverse original genus -> exact wild fiber orders.
The original field's proved separating j coordinate has total positive
order equal to the signed Gamma1 index. Its positive support is finite.
These native degree and support facts will turn the wild fiber cardinal
bounds and normal-field order bounds into exact local orders. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open CongruenceSubgroup GammaOne SourceCurve IntermediateField
local notation "Qbar" => AlgebraicClosure ℚ
variable (A : ValuationSubring Qbar) (N ℓ : ℕ) [NeZero N] [Fact ℓ.Prime]
    [CharP (IsLocalRing.ResidueField A) ℓ]
local notation "kk" => IsLocalRing.ResidueField A
local notation "F" => qExpFunctionFieldC kk (Gamma1 N)
private local instance : Algebra kk F := (F).toSubalgebra.algebra

theorem original_positive_j_fiber_finite_and_degree (hℓN : ¬ ℓ ∣ N)
    (x : F) (hx : (x : LaurentSeries kk) = jqModC kk) :
    Finite {v : Place kk F // 0 < v.ord x} ∧
      (∑ᶠ v : {v : Place kk F // 0 < v.ord x}, v.val.ord x) =
        ((plusMinus (Gamma1 N)).index : ℤ) := by
  letI := valuation_residue_isAlgClosed A
  letI := original_coordinate_finite A N ℓ hℓN x hx
  letI := original_coordinate_isSeparable A N ℓ hℓN x hx
  have hxtr : Transcendental kk x := by
    intro h
    apply transcendental_jqModC kk
    rw [← hx]
    exact h.algHom (F).val
  have hfinite : {v : Place kk F | 0 < v.ord x}.Finite :=
    (finite_function_support_of_separating_coordinate x hxtr x).subset
      (fun _ hv => ne_of_gt hv)
  refine ⟨hfinite.to_subtype, ?_⟩
  have h := sum_positive_orders_eq_finrank_separating x hxtr
  have hr (v : Place kk F) :=
    Module.finrank_of_bijective_algebraMap (place_residue_bijective_of_finite_coordinate x v)
  simp only [hr, Nat.cast_one, mul_one] at h
  rwa [original_coordinate_finrank_eq_index A N ℓ hℓN x hx] at h

end
end Dubon2026.SourceFunctionField.MarkedChart
