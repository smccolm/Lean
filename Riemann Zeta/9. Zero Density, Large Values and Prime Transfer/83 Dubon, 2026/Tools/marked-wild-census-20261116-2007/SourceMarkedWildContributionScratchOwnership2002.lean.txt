import SourceMarkedDihedralFiberBound
import SourceMarkedSignFiberSymmetries
import SourceMarkedOriginalWildFiberThree
import SourceMarkedOrderTwelveFiberBound
import SourceMarkedSignActionHom
import SourceMarkedCharTwoSignAction
import SourceMarkedOriginalWildFiberTwo
import SourceMarkedNormalOrderDescent
import SourceMarkedOriginalOrderBoundTwo
import SourceMarkedOriginalOrderBoundThree
import SourceMarkedOriginalPositiveDegree
import SourceMarkedFiberOrderRigidity
import SourceMarkedOriginalExactOrderTwo
import SourceMarkedOriginalExactOrderThree
import SourceMarkedOriginalWildContributionTwo
import SourceMarkedOriginalWildContributionThree
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceMarkedDihedralFiberBound, `SourceMarkedSignFiberSymmetries, `SourceMarkedOriginalWildFiberThree, `SourceMarkedOrderTwelveFiberBound, `SourceMarkedSignActionHom, `SourceMarkedCharTwoSignAction, `SourceMarkedOriginalWildFiberTwo, `SourceMarkedNormalOrderDescent, `SourceMarkedOriginalOrderBoundTwo, `SourceMarkedOriginalOrderBoundThree, `SourceMarkedOriginalPositiveDegree, `SourceMarkedFiberOrderRigidity, `SourceMarkedOriginalExactOrderTwo, `SourceMarkedOriginalExactOrderThree, `SourceMarkedOriginalWildContributionTwo, `SourceMarkedOriginalWildContributionThree]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
