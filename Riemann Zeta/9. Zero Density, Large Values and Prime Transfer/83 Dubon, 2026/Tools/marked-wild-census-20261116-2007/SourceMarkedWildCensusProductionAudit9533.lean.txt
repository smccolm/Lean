  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.MarkedChart.normalized_embeddings_fixed_of_torsion_sign, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_normal_field_of_torsion_sign_kernel, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_normal_field_j_order_dvd_twelve, `Dubon2026.SourceFunctionField.MarkedChart.original_char_three_differential_contribution]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.MarkedTorsion.signClassesPermHom, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwoMarkedStabilizerAction, `Dubon2026.SourceFunctionField.MarkedTorsion.charTwoMarkedSignAction]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue functional postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue functional theorems discovered"
  logInfo m!"SOURCE MARKED WILD CENSUS AUDIT PASS: {count} theorems including private/generated declarations. Actual original wild orders and summed different contributions proved. Hurwitz, small levels, reverse genus and source gates OPEN;16/20 gates."
