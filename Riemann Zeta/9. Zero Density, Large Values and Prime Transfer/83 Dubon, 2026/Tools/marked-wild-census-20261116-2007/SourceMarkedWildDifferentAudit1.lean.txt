private def runMarkedWildDifferentAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMarkedEmbeddingSignKernel, `SourceMarkedOriginalNormalField, `SourceMarkedOriginalNormalFieldTwo, `SourceMarkedOriginalNormalFieldThree, `SourceMarkedOriginalLocalDifferentials, `SourceMarkedOriginalWildDifferent, `SourceMarkedOriginalWildDifferentTwo, `SourceMarkedOriginalWildDifferentThree]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.MarkedChart.normalized_embeddings_fixed_of_torsion_sign, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_normal_field_of_torsion_sign_kernel, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_normal_field_j_order_dvd_twelve, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_normal_field_j_order_dvd_six, `Dubon2026.SourceFunctionField.MarkedChart.original_j_D_ne_zero, `Dubon2026.SourceFunctionField.MarkedChart.original_j_differentials_finrank, `Dubon2026.SourceFunctionField.MarkedChart.original_positive_place_dCoordGenerates, `Dubon2026.SourceFunctionField.MarkedChart.original_wild_different_of_normal_order_bound, `Dubon2026.SourceFunctionField.MarkedChart.original_fourteen_le_different_of_j_order_twelve, `Dubon2026.SourceFunctionField.MarkedChart.original_seven_le_different_of_j_order_six]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE MARKED WILD DIFFERENT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runMarkedWildDifferentAudit
