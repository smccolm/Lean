/-! DUB-17/18 -> actual Gamma1 Picard/H1 -> principal-divisor converse ->
meromorphic construction -> analytic residue realization. Apply the already
proved logarithmic-derivative expansion to the actual uniformizer and
the analytic representative of its pole-cleared differential coefficient.
This avoids a new ramified normal-form calculation and also handles the
zero differential. All residue and ramification data are the original ones. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane SourceCurve KaehlerDifferential Filter RealizationDerivative
open scoped MatrixGroups Topology
variable {Γ : Subgroup SL(2, ℤ)} [Γ.FiniteIndex] (hT : ModularGroup.T ∈ Γ)

theorem analytic_differential_residue_expansion (a : functionFieldOver Γ ℂ) (τ : ℍ)
    (hord : -1 ≤ (analyticPlace hT τ).ordDifferential
      (a • KaehlerDifferential.D ℂ (functionFieldOver Γ ℂ) (jCoordinateOver Γ ℂ))) :
    ∃ g : ℂ → ℂ, AnalyticAt ℂ g (τ : ℂ) ∧
      ∀ᶠ z in 𝓝[≠] (τ : ℂ),
        rz (Γ := Γ) a z * deriv (JMultiplicity.jf ∘ ofComplex) z =
          (analyticRamification hT τ : ℂ) *
            placeEval hT (analyticPlace hT τ)
              ((analyticPlace hT τ).uniformizer *
                (analyticPlace hT τ).differentialCoeff
                  (a • KaehlerDifferential.D ℂ (functionFieldOver Γ ℂ) (jCoordinateOver Γ ℂ))) /
                    (z - τ) + g z := by
  let η : Ω[functionFieldOver Γ ℂ⁄ℂ] := a • KaehlerDifferential.D ℂ (functionFieldOver Γ ℂ) (jCoordinateOver Γ ℂ)
  let v := analyticPlace hT τ
  let π := v.uniformizer
  let b := v.differentialCoeff η
  letI := functionFieldOver_dCoordGenerates Γ ℂ hT v
  have hd : a • KaehlerDifferential.D ℂ (functionFieldOver Γ ℂ) (jCoordinateOver Γ ℂ) =
      b • KaehlerDifferential.D ℂ (functionFieldOver Γ ℂ) π :=
    (v.differentialCoeff_smul_dCoord η).symm
  have hkey := eventuallyEq_of_smul_D_eq_smul_D hT τ hd
  have hπ : MeromorphicAt (rz (Γ := Γ) π) (τ : ℂ) := meromorphicAt_realize hT π τ
  have horder : meromorphicOrderAt (rz (Γ := Γ) π) (τ : ℂ) =
      ((analyticRamification hT τ : ℤ) : WithTop ℤ) := by
    change analyticOrder Γ τ π = _
    rw [analyticPlace_order hT τ π v.uniformizer_ne_zero]
    change (((analyticRamification hT τ : ℤ) * v.ord v.uniformizer : ℤ) : WithTop ℤ) = _
    rw [v.ord_uniformizer, mul_one]
  have hclear : π * b ∈ v.toValuationSubring := (simple_pole_bound_iff v η).mp hord
  obtain ⟨E, hE, hEval, hEE⟩ := regular_analytic_representative hT τ (π * b) hclear
  obtain ⟨g, hg, hloc⟩ := SourceComplexReciprocity.LogDerivStokes.hloc_logDeriv hπ horder hE
  have hnonzero : ∀ᶠ z in 𝓝[≠] (τ : ℂ), rz (Γ := Γ) π z ≠ 0 :=
    (meromorphicOrderAt_ne_top_iff_eventually_ne_zero hπ).mp
      (by rw [horder]; exact WithTop.coe_ne_top)
  have hprod := realize_mul_eventuallyEq hT π b τ
  have hj : rz (Γ := Γ) (jCoordinateOver Γ ℂ) = JMultiplicity.jf ∘ ofComplex := by
    funext z
    exact realize_jCoordinateOver hT (ofComplex z)
  refine ⟨g, hg, ?_⟩
  filter_upwards [hkey, hEE, hprod, hnonzero, hloc] with z hk hEeq hp hn hl
  change rz (Γ := Γ) a z * deriv (rz (Γ := Γ) (jCoordinateOver Γ ℂ)) z =
    rz (Γ := Γ) b z * deriv (rz (Γ := Γ) π) z at hk
  change E z = rz (Γ := Γ) (π * b) z at hEeq
  change rz (Γ := Γ) (π * b) z = rz (Γ := Γ) π z * rz (Γ := Γ) b z at hp
  rw [hj] at hk
  rw [hk]
  calc
    rz (Γ := Γ) b z * deriv (rz (Γ := Γ) π) z = E z * (deriv (rz (Γ := Γ) π) z / rz (Γ := Γ) π z) := by
      rw [hEeq, hp]
      field_simp
    _ = _ := by
      rw [hl, hEval]
      push_cast
      rfl

end
end Dubon2026.SourceFunctionField.GammaOne
