private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRealizationDerivation, `SourceGammaOneAnalyticDifferentialResidue]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.realizeGerm_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.germ_smul, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.germ_scalarTower, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.rz_smul, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.eventually_differentiableAt_rz, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.eventuallyEq_of_smul_D_eq_smul_D, `Dubon2026.SourceFunctionField.GammaOne.analytic_differential_residue_expansion]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.rz, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.realizeGerm, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.germAlgebra, `Dubon2026.SourceFunctionField.GammaOne.RealizationDerivative.realizeDerivation]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Analytic Residue postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Analytic Residue theorems discovered"
  logInfo m!"SOURCE GAMMA ONE ANALYTIC RESIDUE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
