/-- Real-part projection of multiplication by the actual complex derivative. -/
def realMul (a : ℂ) : ℂ →L[ℝ] ℂ :=
  Complex.ofRealCLM.comp (Complex.reCLM.comp (a • (1 : ℂ →L[ℝ] ℂ)))

theorem realMul_apply (a v : ℂ) : realMul a v = ((a * v).re : ℂ) := by
  simp [realMul, smul_eq_mul]

theorem norm_realMul_le (a : ℂ) : ‖realMul a‖ ≤ ‖a‖ := by
  apply ContinuousLinearMap.opNorm_le_bound (norm_nonneg a)
  intro v
  rw [realMul_apply, Complex.norm_real, Real.norm_eq_abs]
  exact (Complex.abs_re_le_norm (a * v)).trans (norm_mul_le a v)

theorem dbar_realMul (a : ℂ) :
    (realMul a 1 + Complex.I * realMul a Complex.I) / 2 = conj a / 2 := by
  apply Complex.ext <;> simp [realMul_apply] <;> ring

variable {Γ : Subgroup SL(2, ℤ)} [Γ.FiniteIndex]
    (k : CuspForm (Γ.map (Matrix.SpecialLinearGroup.mapGL ℝ)) 2)

/-- The real part of the original normalized complex endpoint primitive. -/
def realEk (z : ℂ) : ℂ := ((Ek k z).re : ℂ)

theorem hasFDerivAt_realEk {z : ℂ} (hz : 0 < z.im) :
    HasFDerivAt (realEk k) (realMul (k (ofComplex z))) z :=
  Complex.ofRealCLM.hasFDerivAt.comp z
    (Complex.reCLM.hasFDerivAt.comp z (Ek_hasDerivAt k hz).complexToReal_fderiv)

theorem continuousOn_realEk_derivative :
    ContinuousOn (fun z : ℂ => realMul (k (ofComplex z))) upperHalfPlaneSet := by
  intro z hz
  have hk := ((UpperHalfPlane.mdifferentiable_iff.mp (ModularFormClass.holo k)).differentiableAt
    (isOpen_upperHalfPlaneSet.mem_nhds hz)).continuousAt
  exact (continuousAt_const.clm_comp
    (continuousAt_const.clm_comp (hk.smul continuousAt_const))).continuousWithinAt

theorem realEk_smul (γ : Γ) (τ : ℍ) :
    realEk k (((γ : SL(2, ℤ)) • τ : ℍ) : ℂ) =
      realEk k τ + ((sourceWeightTwoPeriodValue Γ k γ).re : ℂ) := by
  simp only [realEk, Ek_smul, Complex.add_re, Complex.ofReal_add]

theorem hasFDerivAt_realEp (σ : SL(2, ℤ)) {z : ℂ} (hz : 0 < z.im) :
    HasFDerivAt (fun u : ℂ => realEk k ((σ • ofComplex u : ℍ) : ℂ))
      (realMul (tilePull k σ z)) z := by
  exact Complex.ofRealCLM.hasFDerivAt.comp z
    (Complex.reCLM.hasFDerivAt.comp z (hasDerivAt_Ep k σ hz).complexToReal_fderiv)

theorem realEk_cusp_bound (σ : SL(2, ℤ)) :
    ∃ C Y : ℝ, ∀ z : ℂ, Y ≤ z.im →
      ‖realEk k ((σ • ofComplex z : ℍ) : ℂ)‖ ≤ C ∧
        ‖fderiv ℝ (fun u : ℂ => realEk k ((σ • ofComplex u : ℍ) : ℂ)) z‖ ≤ C := by
  obtain ⟨L, hL⟩ := Ek_exists_cusp_limit k σ
  have hr : Tendsto (fun τ : ℍ => realEk k ((σ • τ : ℍ) : ℂ)) atImInfty (𝓝 (L.re : ℂ)) :=
    hL.re.ofReal
  have hb := hr.norm.eventually (Iic_mem_nhds (lt_add_one ‖(L.re : ℂ)‖))
  obtain ⟨A, hA⟩ := (atImInfty_mem _).mp hb
  obtain ⟨c, hc, C, hC0, hC⟩ := tilePull_decay k σ (show (0 : ℝ) < 1 by norm_num)
  refine ⟨max (‖(L.re : ℂ)‖ + 1) C, max A 1, fun z hz => ?_⟩
  have hz1 : 1 ≤ z.im := (le_max_right A 1).trans hz
  have hz0 : 0 < z.im := one_pos.trans_le hz1
  have hi : (ofComplex z).im = z.im := by rw [ofComplex_apply_of_im_pos hz0]; rfl
  have hv := hA (ofComplex z) (by rw [hi]; exact (le_max_left A 1).trans hz)
  refine ⟨hv.trans (le_max_left _ _), ?_⟩
  rw [(hasFDerivAt_realEp k σ hz0).fderiv]
  calc
    ‖realMul (tilePull k σ z)‖ ≤ ‖tilePull k σ z‖ := norm_realMul_le _
    _ ≤ C * Real.exp (-c * z.im) := hC z hz1
    _ ≤ C * 1 := mul_le_mul_of_nonneg_left
      (Real.exp_le_one_iff.mpr (by nlinarith)) hC0
    _ ≤ max (‖(L.re : ℂ)‖ + 1) C := by simpa using le_max_right (‖(L.re : ℂ)‖ + 1) C

end
