variable {Γ : Subgroup SL(2, ℤ)} [Γ.FiniteIndex]
local notation "EE" => functionFieldOver Γ ℂ
private local instance coefficientCommutes : SMulCommClass ℂ EE EE := ⟨fun c x y => by
  change algebraMap ℂ EE c * (x * y) = x * (algebraMap ℂ EE c * y)
  ring⟩
private local instance differentialModule : Module EE Ω[EE⁄ℂ] :=
  @KaehlerDifferential.module' ℂ EE inferInstance inferInstance inferInstance
    EE inferInstance inferInstance (coefficientCommutes (Γ := Γ))

/-- The actual ramification-weighted order prescribed by the finite analytic divisor. -/
def orders (hT : ModularGroup.T ∈ Γ) (c : ℍ →₀ ℤ) (τ : ℍ) : ℤ :=
  (analyticRamification hT τ : ℤ) *
    Finsupp.mapDomain (analyticPlace hT) c (analyticPlace hT τ)

theorem exists_residue_coefficient (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) (hT : ModularGroup.T ∈ Γ)
    (c : ℍ →₀ ℤ) (hc : c.sum (fun _ n => n) = 0) :
    ∃ ω : ℍ → ℂ, Inv Γ ω ∧ Cusp ω ∧
      ∀ τ : ℍ, ∃ u : ℂ → ℂ, AnalyticAt ℂ u (τ : ℂ) ∧
        (∀ᶠ z in 𝓝[≠] (τ : ℂ), ω (ofComplex z) =
          (orders hT c τ : ℂ) / (z - τ) + u z) ∧
        (orders hT c τ = 0 → ω τ = u τ) := by
  obtain ⟨theta, htheta, hthetac⟩ := exists_analytic_divisor_differential hT c hc
  letI : IsCurveOver ℂ EE := functionFieldOver_isCurveOver Γ ℂ hT
  obtain ⟨a, hatheta⟩ := RegularDifferentialsComparison.exists_smul_eq
    (D_jCoordinateOver_ne_zero Γ ℂ) theta
  by_cases ha : a = 0
  · have htheta0 : theta = 0 := by simpa only [ha, zero_smul] using hatheta.symm
    have hm : ∀ τ : ℍ, orders hT c τ = 0 := by
      intro τ
      have he := (htheta (analyticPlace hT τ)).2
      rw [htheta0, Place.differentialCoeff_zero, mul_zero, placeEval,
        Place.rationalEval_zero] at he
      have hd : Finsupp.mapDomain (analyticPlace hT) c (analyticPlace hT τ) = 0 := by
        exact_mod_cast he.symm
      simp only [orders, hd, mul_zero]
    refine ⟨0, fun γ _ => SlashAction.zero_slash _ _, fun σ => ?_, fun τ => ?_⟩
    · refine ⟨1, one_pos, ?_⟩
      rw [SlashAction.zero_slash]
      exact Asymptotics.isBigO_zero _ _
    · refine ⟨0, analyticAt_const, ?_, fun _ => rfl⟩
      exact Eventually.of_forall fun z => by simp only [hm, Int.cast_zero, zero_div, zero_add]; rfl
  · obtain ⟨k, g, h, hh, hr⟩ := functionFieldOver_modular_ratio hT a
    have hinv : Inv Γ (rawDifferential g h) := fun γ hγ => rawDifferential_slash g h γ hγ
    have hcusp : Cusp (rawDifferential g h) := by
      intro γ
      apply rawDifferential_cusp_exp_decay N hΓ hT a ha g h hh hr γ
      rw [hatheta]
      exact hthetac _ (cuspPlace_j_neg N hΓ hT γ)
    have hloc : Loc (rawDifferential g h) (fun τ => (orders hT c τ : ℂ)) := by
      intro τ
      obtain ⟨u, hu, he⟩ := analytic_differential_residue_expansion hT a τ
        (by rw [hatheta]; exact (htheta (analyticPlace hT τ)).1)
      refine ⟨u, hu, ?_⟩
      filter_upwards [rawDifferential_germ hT g h hh a hr τ, he] with z hz hez
      rw [hz, hez, hatheta, (htheta (analyticPlace hT τ)).2]
      simp only [orders, Int.cast_mul, Int.cast_natCast]
    exact exists_regularized_differential Γ (orders hT c) (rawDifferential g h) hinv hcusp hloc

end
