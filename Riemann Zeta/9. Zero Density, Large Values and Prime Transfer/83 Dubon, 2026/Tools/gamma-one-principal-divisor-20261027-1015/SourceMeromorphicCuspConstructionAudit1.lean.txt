private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMeromorphicPrimitiveSpecification, `SourceMeromorphicPrimitiveGroupAction, `SourceMeromorphicPrimitiveMultiplier, `SourceMeromorphicCuspAnalyticData, `SourceMeromorphicCuspPrimitive, `SourceMeromorphicCuspPrimitiveDecay, `SourceMeromorphicPrimitiveCusps]
  let theorems : Array Name := #[`Dubon2026.SourceMeromorphicPrimitiveMultiplier.mC_of_im_pos, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.mC_coe, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.coe_ofComplex_of_im_pos, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.res_lift, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.exists_GSpec, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.eq_zero_iff, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.ne_zero_iff, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.eventually_ne_zero, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.analyticAt, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.hasDerivAt, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.eventuallyEq_ofComplex, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.meromorphicAt_ofComplex, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.meromorphicOrderAt_ofComplex, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.μ_coe, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.μ_im_pos, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.μ_of_im_pos, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.hasDerivAt_μ, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.continuousAt_μ, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.slash_two_apply, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.eventually_μ_ne, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.tendsto_μ_nhdsNE, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.μ_inv_μ, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.hasDerivAt_comp_μ, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.goodSet_subset, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.eventually_mem_goodSet, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.isOpen_goodSet, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.countable_zeroSet, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.upperHalfPlane_eq_range, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.isPreconnected_upperHalfPlane_diff, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.isPreconnected_goodSet, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.exists_multiplier, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.exists_meromorphic_multiplier, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.analyticAt_omega, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.tendsto_norm_omega_atTop, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.exists_cuspBound, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.continuous_denom, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.differentiableAt_denom, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.exists_primitive_halfPlane, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.norm_sub_le_of_exp_decay, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.exists_uniform_limit_of_exp_decay, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.m_smul_eq_zero_of_lt_im, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec.exists_limit_cusp, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.exists_meromorphic_multiplier_cusp_orders]
  let constructions : Array Name := #[`Dubon2026.SourceMeromorphicPrimitiveMultiplier.mC, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.GSpec, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.μ, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.goodSet, `Dubon2026.SourceMeromorphicPrimitiveMultiplier.CuspBound]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Meromorphic Cusp Construction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Meromorphic Cusp Construction theorems discovered"
  logInfo m!"SOURCE MEROMORPHIC CUSP CONSTRUCTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
