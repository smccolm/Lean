  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralCosetCocycle, `SourceIntegralCosetParabolic, `SourceIntegralCosetWord, `SourceIntegralPeriodCuspLimits, `SourceIntegralPeriodGeneratorDual, `SourceRealPrimitiveGreenData, `SourceIntegralPeriodLatticeDual, `SourceRealPeriodPeterssonPairing, `SourceClosedPeterssonNondegenerate, `SourceSignedPeriodLatticeLift, `SourceGammaOneSignedRealPeriods, `SourceGammaOnePeterssonIntegralRealPeriods, `SourceGammaOneUnitaryDivisorPrincipal]

`Dubon2026.SourceIntegralPeriodDual.cocycle_neg_one_apply, `Dubon2026.SourceIntegralPeriodDual.hom_apply, `Dubon2026.SourceIntegralPeriodDual.hom_transition, `Dubon2026.SourceIntegralPeriodDual.hom_conj, `Dubon2026.SourceIntegralPeriodDual.cocycle_eps_mul_apply, `Dubon2026.SourceIntegralPeriodDual.cocycle_T_apply, `Dubon2026.SourceIntegralPeriodDual.hom_transition_S, `Dubon2026.SourceIntegralPeriodDual.hom_transition_T, `Dubon2026.SourceIntegralPeriodDual.cocycle_T_inv_pow_apply, `Dubon2026.SourceIntegralPeriodDual.cocycle_T_zpow_eq_zero, `Dubon2026.SourceIntegralPeriodDual.hom_parabolic, `Dubon2026.SourceIntegralPeriodDual.sum_ind_mul, `Dubon2026.SourceIntegralPeriodDual.wordProp_mul_T_inv, `Dubon2026.SourceIntegralPeriodDual.wordProp_mul_S, `Dubon2026.SourceIntegralPeriodDual.wordProp_mul_S_inv, `Dubon2026.SourceIntegralPeriodDual.wordProp, `Dubon2026.SourceIntegralPeriodDual.lim_mul_T, `Dubon2026.SourceIntegralPeriodDual.lim_neg, `Dubon2026.SourceIntegralPeriodDual.lim_coe_mul, `Dubon2026.SourceFunctionField.GammaOne.analytic_divisor_isPrincipal_of_unitary_meromorphic_function]
  let constructions : Array Name := #[`Dubon2026.SourceIntegralPeriodDual.coind, `Dubon2026.SourceIntegralPeriodDual.hom, `Dubon2026.SourceIntegralPeriodDual.em, `Dubon2026.SourceIntegralPeriodDual.ind, `Dubon2026.SourceIntegralPeriodDual.WordProp, `Dubon2026.SourceRealPrimitive.realMul, `Dubon2026.SourceRealPrimitive.realEk, `Dubon2026.SourceIntegralPeriodDual.pairing, `Dubon2026.SourceFunctionField.GammaOne.SignedPeriods.restrictReal]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Integral Period Duality postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Integral Period Duality theorems discovered"
  logInfo m!"SOURCE ORIGINAL INTEGRAL PERIOD DUALITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
