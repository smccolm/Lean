private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMeromorphicDifferentialRegularization, `SourceGammaOneRawDifferential, `SourceGammaOneLocalRegularDifferentialOrder, `SourceMeromorphicCuspExponentialDecay, `SourceGammaOneRawDifferentialDecay, `SourceGammaOnePrescribedResidueCoefficient, `SourceGammaOnePrincipalDivisorConverse]
  let theorems : Array Name := #[`Dubon2026.SourceMeromorphicDifferentialRegularization.eventually_im_pos, `Dubon2026.SourceMeromorphicDifferentialRegularization.im_ofComplex_of_im_pos, `Dubon2026.SourceMeromorphicDifferentialRegularization.reg_eq_of_hasLim, `Dubon2026.SourceMeromorphicDifferentialRegularization.reg_eq_of_not, `Dubon2026.SourceMeromorphicDifferentialRegularization.hasLim_slash, `Dubon2026.SourceMeromorphicDifferentialRegularization.inv_reg, `Dubon2026.SourceMeromorphicDifferentialRegularization.cusp_reg, `Dubon2026.SourceMeromorphicDifferentialRegularization.reg_ofComplex_eventuallyEq, `Dubon2026.SourceMeromorphicDifferentialRegularization.loc_reg, `Dubon2026.SourceMeromorphicDifferentialRegularization.exists_regularized_differential, `Dubon2026.SourceFunctionField.GammaOne.rawDifferential_slash, `Dubon2026.SourceFunctionField.GammaOne.rawDifferential_germ, `Dubon2026.SourceFunctionField.GammaOne.rawDifferential_cusp_germ, `Dubon2026.SourceFunctionField.GammaOne.locally_regular_coefficient_order_bound, `Dubon2026.SourceFunctionField.GammaOne.rawDifferential_cusp_order_pos, `Dubon2026.SourceMeromorphicCuspDecay.exp_decay_of_cusp_order_pos, `Dubon2026.SourceFunctionField.GammaOne.rawDifferential_cusp_exp_decay, `Dubon2026.SourceFunctionField.GammaOne.PrescribedMeromorphic.exists_residue_coefficient, `Dubon2026.SourceFunctionField.GammaOne.exists_analytic_divisor_meromorphic_multiplier, `Dubon2026.SourceFunctionField.GammaOne.analytic_divisor_isPrincipal_of_period_mem]
  let constructions : Array Name := #[`Dubon2026.SourceMeromorphicDifferentialRegularization.Inv, `Dubon2026.SourceMeromorphicDifferentialRegularization.Cusp, `Dubon2026.SourceMeromorphicDifferentialRegularization.Loc, `Dubon2026.SourceMeromorphicDifferentialRegularization.HasLim, `Dubon2026.SourceMeromorphicDifferentialRegularization.reg, `Dubon2026.SourceFunctionField.GammaOne.rawDifferential, `Dubon2026.SourceFunctionField.GammaOne.PrescribedMeromorphic.orders]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Principal Divisor Converse postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Principal Divisor Converse theorems discovered"
  logInfo m!"SOURCE GAMMA ONE PRINCIPAL DIVISOR CONVERSE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
