import Mathlib.LinearAlgebra.Matrix.Charpoly.Coeff
import Mathlib.LinearAlgebra.Matrix.GeneralLinearGroup.Basic

/-! DUB-17/18 -> original dihedral image exclusion -> zero traces on reflection
classes. Native rank-two Cayley-Hamilton shows that a nonscalar invertible
matrix whose square is scalar has zero trace. This is the exact matrix input
for a nonidentity projective involution; no eigenvalue framework is introduced. -/
namespace Dubon2026.SourceDickson
noncomputable section
open Matrix Polynomial

/-- A nonscalar original rank-two matrix with scalar square has zero trace. -/
theorem noncentral_scalar_square_trace_zero
    {F : Type*} [Field F] (A : GL (Fin 2) F) (c : F)
    (hsq : A.val ^ 2 = c • (1 : Matrix (Fin 2) (Fin 2) F))
    (hn : A ∉ Subgroup.center (GL (Fin 2) F)) : A.val.trace = 0 := by
  by_contra ht
  have hc := Matrix.aeval_self_charpoly A.val
  rw [Matrix.charpoly_fin_two] at hc
  simp only [map_add, map_sub, map_mul, map_pow, aeval_X, aeval_C,
    Algebra.algebraMap_eq_smul_one, smul_mul_assoc, one_mul] at hc
  have hsum : A.val ^ 2 + A.val.det • 1 - A.val.trace • A.val = 0 := by
    calc
      _ = A.val ^ 2 - A.val.trace • A.val + A.val.det • 1 := by abel
      _ = 0 := hc
  have he := (sub_eq_zero.mp hsum).symm
  rw [hsq, ← add_smul] at he
  have hscalar : A.val = (A.val.trace⁻¹ * (c + A.val.det)) • 1 := by
    calc
      A.val = A.val.trace⁻¹ • (A.val.trace • A.val) := by
        rw [smul_smul, inv_mul_cancel₀ ht, one_smul]
      _ = _ := by rw [he, smul_smul]
  apply hn
  apply Subgroup.mem_center_iff.mpr
  intro B
  apply Units.ext
  simp only [Units.val_mul, hscalar, smul_mul_assoc, mul_smul_comm, one_mul, mul_one]

end
end Dubon2026.SourceDickson
