import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceFrickeLevelIntegralRing, `Dubon2026.SourceFrickeGammaOneNormalizer, `Dubon2026.SourceFrickeTraceSubgroup, `Dubon2026.SourceFrickeTraceForms, `Dubon2026.SourceFrickeDenominatorUniformization, `Dubon2026.SourceEvenWeightFrickeTrace, `Dubon2026.SourceIntegralSeriesScalarDescent, `Dubon2026.SourceIntegralSeriesCommonDenominator, `Dubon2026.SourceGammaOneAlgebraicCuspDenominator, `Dubon2026.SourceIntegralDiscriminantCancellation, `Dubon2026.SourceGammaOneAlgebraicFormDenominator, `Dubon2026.SourceEvenWeightBalance, `Dubon2026.SourceEvenWeightFrickeCommonDenominator, `Dubon2026.SourceEvenWeightUniformFrickeIntegrality, `Dubon2026.SourceIntegralSlashCuspSubgroup, `Dubon2026.SourceIntegralSlashCuspChart, `Dubon2026.SourceIntegralSlashFrickeChart, `Dubon2026.SourceIntegralSlashSMove, `Dubon2026.SourceIntegralSlashTMove, `Dubon2026.SourceEvenWeightUniformSlashIntegrality, `Dubon2026.SourceRationalCuspDiamondFourier, `Dubon2026.SourceRationalDiscriminantCancellation, `Dubon2026.SourceRationalModularDiamondFourier, `Dubon2026.SourceEvenWeightIntegralDiamondForm]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
