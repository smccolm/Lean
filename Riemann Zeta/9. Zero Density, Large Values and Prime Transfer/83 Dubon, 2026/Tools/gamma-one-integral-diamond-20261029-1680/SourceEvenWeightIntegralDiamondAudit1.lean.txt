private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRationalCuspDiamondFourier, `SourceRationalDiscriminantCancellation, `SourceRationalModularDiamondFourier, `SourceEvenWeightIntegralDiamondForm]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.IntegralSlash.exists_rational_cusp_diamond_laurent, `Dubon2026.SourceFunctionField.IntegralSlash.rational_cusp_diamond_coeff, `Dubon2026.SourceFunctionField.IntegralSlash.rational_coeff_of_discriminant_mul, `Dubon2026.SourceFunctionField.IntegralSlash.rational_modular_diamond_coeff, `Dubon2026.SourceFunctionField.IntegralSlash.exists_level_power_integral_even_diamond_form]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Siegel expansion postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Siegel expansion theorems discovered"
  logInfo m!"SOURCE EVEN WEIGHT INTEGRAL DIAMOND AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
