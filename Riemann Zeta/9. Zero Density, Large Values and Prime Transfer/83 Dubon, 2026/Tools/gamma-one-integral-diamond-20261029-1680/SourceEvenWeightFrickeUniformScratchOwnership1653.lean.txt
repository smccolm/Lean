import SourceIntegralSeriesScalarDescent
import SourceIntegralSeriesCommonDenominator
import SourceGammaOneAlgebraicCuspDenominator
import SourceIntegralDiscriminantCancellation
import SourceGammaOneAlgebraicFormDenominator
import SourceEvenWeightBalance
import SourceEvenWeightFrickeCommonDenominator
import SourceEvenWeightUniformFrickeIntegrality
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceIntegralSeriesScalarDescent, `SourceIntegralSeriesCommonDenominator, `SourceGammaOneAlgebraicCuspDenominator, `SourceIntegralDiscriminantCancellation, `SourceGammaOneAlgebraicFormDenominator, `SourceEvenWeightBalance, `SourceEvenWeightFrickeCommonDenominator, `SourceEvenWeightUniformFrickeIntegrality]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
