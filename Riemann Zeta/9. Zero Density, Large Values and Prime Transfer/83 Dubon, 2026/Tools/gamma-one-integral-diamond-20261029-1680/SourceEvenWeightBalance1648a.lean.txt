/-! DUB-17/18 -> original Gamma1 degree/genus -> uniform level-power
slash integrality -> common denominator from integral cusp spanning.
Discharge the exact weight-balance hypothesis of the existing cusp
spanning theorem for every even integer weight, including negative
weights. Only the required weights four, six and twelve are used. -/
namespace Dubon2026.SourceFunctionField.IntegralFricke

theorem exists_even_weight_balance {k : ℤ} (hk : Even k) :
    ∃ a b m : ℕ, k + 4 * a + 6 * b = 12 * m := by
  obtain ⟨z, hz⟩ := hk
  let a : ℕ := (z % 3).toNat + 3 * z.natAbs
  let b : ℕ := (z % 2).toNat
  let q : ℤ := z + 2 * a + 3 * b
  have ha : (a : ℤ) = z % 3 + 3 * (z.natAbs : ℤ) := by
    dsimp only [a]
    rw [Nat.cast_add, Nat.cast_mul, Nat.cast_ofNat,
      Int.toNat_of_nonneg (Int.emod_nonneg z (by norm_num : (3 : ℤ) ≠ 0))]
  have hb : (b : ℤ) = z % 2 :=
    Int.toNat_of_nonneg (Int.emod_nonneg z (by norm_num : (2 : ℤ) ≠ 0))
  have hbound : -(z.natAbs : ℤ) ≤ z := Int.neg_natAbs_le z
  have hq : 0 ≤ q := by
    dsimp only [q]
    rw [ha, hb]
    omega
  have hq6 : q % 6 = 0 := by
    dsimp only [q]
    rw [ha, hb]
    omega
  refine ⟨a, b, (q / 6).toNat, ?_⟩
  rw [Int.toNat_of_nonneg (Int.ediv_nonneg hq (by norm_num))]
  have := Int.emod_add_ediv q 6
  dsimp only [q] at this hq6
  omega

end Dubon2026.SourceFunctionField.IntegralFricke
