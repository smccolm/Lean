private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralSlashCuspSubgroup1654c, `SourceIntegralSlashCuspChart1655b, `SourceIntegralSlashFrickeChart1656d, `SourceIntegralSlashSMove1657b, `SourceIntegralSlashTMove1658b, `SourceEvenWeightUniformSlashIntegrality1659b]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.IntegralSlash.levelRaise_dvd_of_Gamma1_sq, `Dubon2026.SourceFunctionField.IntegralSlash.levelRaiseConj_mem_Gamma, `Dubon2026.SourceFunctionField.IntegralSlash.Gamma1_sq_le_conj_cusp, `Dubon2026.SourceFunctionField.IntegralSlash.cusp_strictPeriod, `Dubon2026.SourceFunctionField.IntegralSlash.hasSum_cusp, `Dubon2026.SourceFunctionField.IntegralSlash.coe_cuspChart, `Dubon2026.SourceFunctionField.IntegralSlash.slash_levelRaise_apply, `Dubon2026.SourceFunctionField.IntegralSlash.cuspChart_apply, `Dubon2026.SourceFunctionField.IntegralSlash.qParam_levelRaise_smul, `Dubon2026.SourceFunctionField.IntegralSlash.coeff_qExpansion_cuspChart, `Dubon2026.SourceFunctionField.IntegralSlash.levelRaise_mul_fricke_sq, `Dubon2026.SourceFunctionField.IntegralSlash.fricke_cuspChart_eq, `Dubon2026.SourceFunctionField.IntegralSlash.coeff_qExpansion_fricke_cuspChart, `Dubon2026.SourceFunctionField.IntegralSlash.integral_level_pow_of_zpow, `Dubon2026.SourceFunctionField.IntegralSlash.exists_level_power_cusp_mul_S, `Dubon2026.SourceFunctionField.IntegralSlash.coeff_cusp_mul_T_zpow, `Dubon2026.SourceFunctionField.IntegralSlash.exists_level_power_cusp_mul_T_zpow, `Dubon2026.SourceFunctionField.IntegralSlash.exists_level_power_cusp_one, `Dubon2026.SourceFunctionField.IntegralSlash.exists_level_power_integral_even_slash_of_level_power, `Dubon2026.SourceFunctionField.IntegralSlash.exists_level_power_integral_even_slash_coefficients]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.IntegralSlash.cuspChart]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Siegel expansion postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Siegel expansion theorems discovered"
  logInfo m!"SOURCE EVEN WEIGHT FULL SLASH PREFLIGHT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
