/-! DUB-17/18 -> original Gamma1 degree/genus -> uniform level-power
slash integrality -> common integer denominator. Reuse the proved
integral cusp-form span, extract its actual finite combination and
descend its scalars to the algebraic coefficient field. One nonzero
integer then clears every original Fourier coefficient. The weight
balance is precisely the existing spanning theorem's explicit input;
the discriminant multiplication/cancellation consumer remains separate. -/
namespace Dubon2026.SourceFunctionField.IntegralFricke
noncomputable section
open UpperHalfPlane CongruenceSubgroup
open scoped MatrixGroups ModularForm

theorem exists_common_integer_denominator_of_algebraic_cusp_coefficients
    (N : ℕ) [NeZero N] (k : ℤ) (a b m : ℕ)
    (hk : k + 4 * a + 6 * b = 12 * m)
    (F : CuspForm (Gamma1 N) k)
    (hF : ∀ n : ℕ, IsAlgebraic ℚ ((qExpansion 1 F).coeff n)) :
    ∃ d : ℤ, d ≠ 0 ∧ ∀ n : ℕ, IsIntegral ℤ ((d : ℂ) * (qExpansion 1 F).coeff n) := by
  classical
  have hspan : F ∈ Submodule.span ℂ {f : CuspForm (Gamma1 N) k |
      ∃ p : PowerSeries ℤ, GammaOne.IsIntegralQExp f p} := by
    rw [SourceFricke.span_gammaOne_integral_cuspForms_eq_top N k a b m hk]
    trivial
  obtain ⟨c, t, ht, _, he⟩ := Submodule.mem_span_iff_exists_finset_subset.mp hspan
  have hrepr : (∑ i : t, c i.val • i.val) = F := by
    exact (Finset.sum_coe_sort t (fun x => c x • x)).trans he
  choose r hr using fun i : t => ht i.property
  have hq : qExpansion 1 F = ∑ i : t, c i.val • (r i).map (Int.castRingHom ℂ) := by
    ext n
    change SourceFricke.gammaOneQCoeffLinear N k n F = _
    rw [← hrepr, map_sum, map_sum]
    apply Finset.sum_congr rfl
    intro i _
    rw [map_smul, PowerSeries.coeff_smul]
    exact congrArg (fun z : ℂ => c i.val • z)
      (congrArg (fun p : PowerSeries ℂ => p.coeff n) (hr i).symm)
  obtain ⟨d, hd, h⟩ := exists_common_integer_denominator_of_integral_series_sum
    r (fun i : t => c i.val) (fun n => by rw [← hq]; exact hF n)
  exact ⟨d, hd, fun n => by rw [hq]; exact h n⟩

end
end Dubon2026.SourceFunctionField.IntegralFricke
