local notation "Γ₁(" M ")" => ((Gamma1 M : Subgroup SL(2, ℤ)) : Subgroup (GL (Fin 2) ℝ))

/-- The actual S-conjugate of Gamma1 in the integral special linear group. -/
def GammaS (N : ℕ) : Subgroup SL(2, ℤ) :=
  (Gamma1 N).comap (MulAut.conj ModularGroup.S).toMonoidHom

theorem mem_GammaS_iff {N : ℕ} {γ : SL(2, ℤ)} :
    γ ∈ GammaS N ↔ ModularGroup.S * γ * ModularGroup.S⁻¹ ∈ Gamma1 N := Iff.rfl

theorem Gamma_le_GammaS (N : ℕ) : CongruenceSubgroup.Gamma N ≤ GammaS N := by
  intro γ hγ
  have h := (Gamma_normal N).conj_mem γ hγ ModularGroup.S
  rw [mem_GammaS_iff]
  simp_all [Gamma1_mem, Gamma_mem]

instance instFiniteIndexGammaS (N : ℕ) [NeZero N] : (GammaS N).FiniteIndex :=
  Subgroup.finiteIndex_of_le (Gamma_le_GammaS N)

theorem T_pow_mem_GammaS (N : ℕ) : ModularGroup.T ^ N ∈ GammaS N := by
  apply Gamma_le_GammaS N
  simpa using ModularGroup_T_pow_mem_Gamma (N : ℤ) (N : ℤ) dvd_rfl

theorem STS_mem_GammaS (N : ℕ) :
    ModularGroup.S * ModularGroup.T * ModularGroup.S⁻¹ ∈ GammaS N := by
  rw [mem_GammaS_iff]
  have h : ModularGroup.S * (ModularGroup.S * ModularGroup.T * ModularGroup.S⁻¹) *
      ModularGroup.S⁻¹ = ModularGroup.T := by
    rw [ModularGroup.S_inv]
    simp only [mul_neg, neg_mul, neg_neg, ← mul_assoc, SourceSL2Cocycle.S_mul_S, one_mul]
    rw [mul_assoc, SourceSL2Cocycle.S_mul_S]
    simp
  rw [h]
  simp [Gamma1_mem, ModularGroup.T]

theorem mem_conj_iff (g x : GL (Fin 2) ℝ) (Γ : Subgroup (GL (Fin 2) ℝ)) :
    x ∈ ConjAct.toConjAct g⁻¹ • Γ ↔ g * x * g⁻¹ ∈ Γ := by
  rw [Subgroup.mem_pointwise_smul_iff_inv_smul_mem, ← ConjAct.toConjAct_inv, inv_inv,
    ConjAct.toConjAct_smul]

theorem GammaS_le_conjS (N : ℕ) :
    ((GammaS N : Subgroup SL(2, ℤ)) : Subgroup (GL (Fin 2) ℝ)) ≤
      ConjAct.toConjAct ((ModularGroup.S : SL(2, ℤ)) : GL (Fin 2) ℝ)⁻¹ • Γ₁(N) := by
  rintro x ⟨γ, hγ, rfl⟩
  rw [mem_conj_iff]
  refine ⟨ModularGroup.S * γ * ModularGroup.S⁻¹, hγ, ?_⟩
  simp only [map_mul, map_inv]
  rfl

theorem GammaS_le_conj_g₀ (N : ℕ) [NeZero N] :
    ((GammaS N : Subgroup SL(2, ℤ)) : Subgroup (GL (Fin 2) ℝ)) ≤
      ConjAct.toConjAct (frickeMatrix N *
        ((ModularGroup.S : SL(2, ℤ)) : GL (Fin 2) ℝ)⁻¹)⁻¹ • Γ₁(N) := by
  rintro x ⟨γ, hγ, rfl⟩
  rw [mem_conj_iff]
  have he : ModularGroup.S⁻¹ * γ * ModularGroup.S =
      ModularGroup.S * γ * ModularGroup.S⁻¹ := by
    rw [ModularGroup.S_inv, neg_mul, neg_mul, mul_neg]
  have hmem := fricke_conj_mem (mem_GammaS_iff.mp hγ)
  rw [← he] at hmem
  simpa only [map_mul, map_inv, mul_inv_rev, inv_inv, mul_assoc] using hmem

theorem natCast_mem_strictPeriods_GammaS (N : ℕ) :
    (N : ℝ) ∈ ((GammaS N : Subgroup SL(2, ℤ)) : Subgroup (GL (Fin 2) ℝ)).strictPeriods := by
  rw [Subgroup.mem_strictPeriods_iff]
  refine ⟨ModularGroup.T ^ N, T_pow_mem_GammaS N, ?_⟩
  simpa only [zpow_natCast, Int.cast_natCast] using GammaOne.CuspAuxiliary.mapGL_T_zpow (N : ℤ)

end
