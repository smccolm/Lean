import SourceFrickeLevelIntegralRing
import SourceFrickeGammaOneNormalizer
import SourceFrickeTraceSubgroup
import SourceFrickeTraceForms
import SourceFrickeDenominatorUniformization
import SourceEvenWeightFrickeTrace
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceFrickeLevelIntegralRing, `SourceFrickeGammaOneNormalizer, `SourceFrickeTraceSubgroup, `SourceFrickeTraceForms, `SourceFrickeDenominatorUniformization, `SourceEvenWeightFrickeTrace]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
