private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceHomogeneousHeckeDiamond, `SourceGammaOneHeckeDiamondCoefficient, `SourceGammaOneArithmeticTateOperators, `SourceTateDualQuadratic, `SourceGammaOneBinaryFrobeniusNormalization, `SourceCoefficientDualAction, `SourceGammaOneBinaryGaloisHecke]
  let theorems : Array Name := #[`Dubon2026.homogeneousHeckeDiamond_square, `Dubon2026.sourceGammaOneHeckeDiamondCoefficient_square, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGaloisRep_hecke, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGaloisRep_diamond, `Dubon2026.sourceTateDual_quadratic, `Dubon2026.SourceFunctionField.GammaOne.binaryTateFrobeniusOperator_quadratic_of_picard, `Dubon2026.sourceCoefficientDualAction_operator, `Dubon2026.SourceFunctionField.GammaOne.binaryTateGaloisRep_prime]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGaloisRep, `Dubon2026.SourceFunctionField.GammaOne.arithmeticHeckeTate, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDiamondTate, `Dubon2026.SourceFunctionField.GammaOne.binaryTateFrobeniusOperator, `Dubon2026.sourceCoefficientDualAction, `Dubon2026.SourceFunctionField.GammaOne.binaryTateGaloisRep]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE FROBENIUS NORMALIZATION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
