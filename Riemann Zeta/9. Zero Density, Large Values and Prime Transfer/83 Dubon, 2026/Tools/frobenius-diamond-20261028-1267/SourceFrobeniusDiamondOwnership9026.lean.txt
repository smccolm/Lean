import Dubon2026.SourcePrimitiveGammaOneHeckeDiamondCharacter
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceHomogeneousHeckeDiamond, `Dubon2026.SourceGammaOneHeckeDiamondCoefficient, `Dubon2026.SourceGammaOneArithmeticTateOperators, `Dubon2026.SourceTateDualQuadratic, `Dubon2026.SourceGammaOneBinaryFrobeniusNormalization, `Dubon2026.SourceCoefficientDualAction, `Dubon2026.SourceGammaOneBinaryGaloisHecke, `Dubon2026.SourceInnerH1Identity, `Dubon2026.SourceH1TraceInner, `Dubon2026.SourceGammaOneDiamondRange, `Dubon2026.SourceH1TraceSubgroupEquality, `Dubon2026.SourceNormalizerH1Trace, `Dubon2026.SourceH1CompositionSquare, `Dubon2026.SourceNormalizerCoefficientValue, `Dubon2026.SourceGammaOneDiamondSubgroupSquare, `Dubon2026.SourceGammaOneDiamondTransfer, `Dubon2026.SourceGammaOneHeckeDiamondWords, `Dubon2026.SourceGammaOneHeckeDiamondDescent, `Dubon2026.SourcePrimitiveGammaOneHeckeDiamondCharacter]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
