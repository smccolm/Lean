import SourceInnerH1Identity
import SourceH1TraceInner
import SourceGammaOneDiamondRange
import SourceH1TraceSubgroupEquality
import SourceNormalizerH1Trace
import SourceH1CompositionSquare
import SourceNormalizerCoefficientValue
import SourceGammaOneDiamondSubgroupSquare
import SourceGammaOneDiamondTransfer
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceInnerH1Identity, `SourceH1TraceInner, `SourceGammaOneDiamondRange, `SourceH1TraceSubgroupEquality, `SourceNormalizerH1Trace, `SourceH1CompositionSquare, `SourceNormalizerCoefficientValue, `SourceGammaOneDiamondSubgroupSquare, `SourceGammaOneDiamondTransfer]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
