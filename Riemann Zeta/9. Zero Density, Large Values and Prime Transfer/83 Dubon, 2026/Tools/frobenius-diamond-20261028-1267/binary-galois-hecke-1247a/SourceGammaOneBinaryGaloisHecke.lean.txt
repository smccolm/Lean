/-! DUB-17/18 -> original all-weight association -> finite Gamma1
arithmetic Hecke-character realization. The actual contragredient Galois
action on the original homogeneous-valued Tate dual commutes with every
native Hecke word. This proves stability of the eventual character spaces;
neither character descent nor Frobenius/inertia is assumed discharged.
-/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open SourceCurve CongruenceSubgroup
variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "T" => TateModule p (Pic0 Qbar (functionFieldOver (Gamma1 M) Qbar))

private local instance : Algebra ℤ_[p] (ZMod N) :=
  (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V :=
  Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

/-- The genuine dual arithmetic Galois action, including the required inverse. -/
def binaryTateGaloisRep : (Qbar ≃ₐ[ℚ] Qbar) →*
    Module.End (ZMod N) (T →ₗ[ℤ_[p]] V) where
  toFun σ :=
    { toFun := fun φ => φ.comp (arithmeticTateGaloisRep M p σ⁻¹)
      map_add' := fun _ _ => rfl
      map_smul' := fun _ _ => rfl }
  map_one' := by
    apply LinearMap.ext
    intro φ
    apply LinearMap.ext
    intro x
    change φ (arithmeticTateGaloisRep M p (1⁻¹) x) = φ x
    rw [inv_one, map_one]
    rfl
  map_mul' σ τ := by
    apply LinearMap.ext
    intro φ
    apply LinearMap.ext
    intro x
    change φ (arithmeticTateGaloisRep M p ((σ * τ)⁻¹) x) =
      φ (arithmeticTateGaloisRep M p τ⁻¹ (arithmeticTateGaloisRep M p σ⁻¹ x))
    rw [mul_inv_rev, map_mul]
    rfl

theorem binaryTateGaloisRep_prime (σ : Qbar ≃ₐ[ℚ] Qbar)
    (ell : sourceBinaryHeckePrime (Q * N)) (φ : T →ₗ[ℤ_[p]] V) :
    binaryTateGaloisRep Q p n w σ (binaryTatePrimeOperators Q p n w ell φ) =
      binaryTatePrimeOperators Q p n w ell (binaryTateGaloisRep Q p n w σ φ) := by
  letI : Fact ell.val.Prime := ⟨ell.property.1⟩
  apply LinearMap.ext
  intro x
  exact congrArg (fun z => homogeneousMatrixAction w
    (Matrix.diagonal ![(ell.val : ZMod N), 1]) (φ z))
      (LinearMap.congr_fun (arithmeticTateGaloisRep_hecke M p ell.val σ⁻¹) x).symm

theorem binaryTateGaloisRep_word (σ : Qbar ≃ₐ[ℚ] Qbar)
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N))) (φ : T →ₗ[ℤ_[p]] V) :
    binaryTateGaloisRep Q p n w σ
        (FreeAlgebra.lift ℤ (binaryTatePrimeOperators Q p n w) a φ) =
      FreeAlgebra.lift ℤ (binaryTatePrimeOperators Q p n w) a
        (binaryTateGaloisRep Q p n w σ φ) :=
  sourceHeckeWord_intertwine_addHom (binaryTatePrimeOperators Q p n w)
    (binaryTatePrimeOperators Q p n w)
    (binaryTateGaloisRep Q p n w σ).toAddMonoidHom
    (binaryTateGaloisRep_prime Q p n w σ) a φ

end
