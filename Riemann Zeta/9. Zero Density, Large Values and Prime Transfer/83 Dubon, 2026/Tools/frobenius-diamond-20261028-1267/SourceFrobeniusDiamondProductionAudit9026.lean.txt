private def runSourceResidueFunctionalAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`Dubon2026.SourceHomogeneousHeckeDiamond, `Dubon2026.SourceGammaOneHeckeDiamondCoefficient, `Dubon2026.SourceGammaOneArithmeticTateOperators, `Dubon2026.SourceTateDualQuadratic, `Dubon2026.SourceGammaOneBinaryFrobeniusNormalization, `Dubon2026.SourceCoefficientDualAction, `Dubon2026.SourceGammaOneBinaryGaloisHecke, `Dubon2026.SourceInnerH1Identity, `Dubon2026.SourceH1TraceInner, `Dubon2026.SourceGammaOneDiamondRange, `Dubon2026.SourceH1TraceSubgroupEquality, `Dubon2026.SourceNormalizerH1Trace, `Dubon2026.SourceH1CompositionSquare, `Dubon2026.SourceNormalizerCoefficientValue, `Dubon2026.SourceGammaOneDiamondSubgroupSquare, `Dubon2026.SourceGammaOneDiamondTransfer, `Dubon2026.SourceGammaOneHeckeDiamondWords, `Dubon2026.SourceGammaOneHeckeDiamondDescent, `Dubon2026.SourcePrimitiveGammaOneHeckeDiamondCharacter]
  let theorems : Array Name := #[`Dubon2026.homogeneousHeckeDiamond_square, `Dubon2026.sourceGammaOneHeckeDiamondCoefficient_square, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGaloisRep_hecke, `Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGaloisRep_diamond, `Dubon2026.sourceTateDual_quadratic, `Dubon2026.SourceFunctionField.GammaOne.binaryTateFrobeniusOperator_quadratic_of_picard, `Dubon2026.sourceCoefficientDualAction_operator, `Dubon2026.SourceFunctionField.GammaOne.binaryTateGaloisRep_prime, `Dubon2026.sourceInnerH1_identity, `Dubon2026.sourceSubgroupH1Trace_inner, `Dubon2026.sourceGammaOneDiagonalRange_conj, `Dubon2026.sourceSubgroupH1Trace_equality, `Dubon2026.sourceSubgroupH1Trace_normalizer, `Dubon2026.sourceH1Composition_square, `Dubon2026.sourceSubgroupEqualityCoefficient_linear, `Dubon2026.sourceNormalizerCoefficientMorphism_linear, `Dubon2026.sourceGammaOneSubgroupH1Iso_diamond_square, `Dubon2026.sourceGammaOneConstantH1Transfer_diamond, `Dubon2026.sourceGammaOneOriginalHeckeDiamondWord_val, `Dubon2026.sourceGammaOneHeckeDiamondWord_transfer, `Dubon2026.sourceGammaOneOriginalHeckeDiamondWord_change, `Dubon2026.sourceGammaOneHeckeDiamondWord_modulus_zero, `Dubon2026.sourceGammaOneHeckeDiamondWord_modulus_kernel, `Dubon2026.sourceGammaOneDiamondToModulus_comp, `Dubon2026.sourcePrimitiveGammaOneHeckeDiamondCharacter_comp, `Dubon2026.sourcePrimitiveGammaOneHeckeDiamondCharacter_hecke, `Dubon2026.sourcePrimitiveGammaOneHeckeDiamondCharacter_diamond]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.arithmeticTateGaloisRep, `Dubon2026.SourceFunctionField.GammaOne.arithmeticHeckeTate, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDiamondTate, `Dubon2026.SourceFunctionField.GammaOne.binaryTateFrobeniusOperator, `Dubon2026.sourceCoefficientDualAction, `Dubon2026.SourceFunctionField.GammaOne.binaryTateGaloisRep, `Dubon2026.sourceInnerCoefficientMorphism, `Dubon2026.sourceSubgroupEqualityCoefficient, `Dubon2026.sourceNormalizerSubgroupHom, `Dubon2026.sourceNormalizerCoefficientMorphism, `Dubon2026.sourceGammaOneHeckeDiamondOperators, `Dubon2026.sourceGammaOneHeckeDiamondAlgebra, `Dubon2026.sourceGammaOneOriginalHeckeDiamondWord, `Dubon2026.sourceGammaOneDiamondToModulus, `Dubon2026.sourcePrimitiveGammaOneHeckeDiamondCharacter]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue functional postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue functional theorems discovered"
  logInfo m!"SOURCE FROBENIUS DIAMOND AUDIT PASS: {count} theorems including private/generated declarations. Exact conditional Frobenius normalization, original Galois/Hecke compatibility, native transfer invariance and full-H1 diamond-one character. Geometric Frobenius/inertia, parabolic and characteristic-zero realization remain;16/20 gates."

#eval runSourceResidueFunctionalAudit
