private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceInnerH1Identity, `SourceH1TraceInner, `SourceGammaOneDiamondRange, `SourceH1TraceSubgroupEquality, `SourceNormalizerH1Trace, `SourceH1CompositionSquare, `SourceNormalizerCoefficientValue, `SourceGammaOneDiamondSubgroupSquare, `SourceGammaOneDiamondTransfer]
  let theorems : Array Name := #[`Dubon2026.sourceInnerH1_identity, `Dubon2026.sourceSubgroupH1Trace_inner, `Dubon2026.sourceGammaOneDiagonalRange_conj, `Dubon2026.sourceSubgroupH1Trace_equality, `Dubon2026.sourceSubgroupH1Trace_normalizer, `Dubon2026.sourceH1Composition_square, `Dubon2026.sourceSubgroupEqualityCoefficient_linear, `Dubon2026.sourceNormalizerCoefficientMorphism_linear, `Dubon2026.sourceGammaOneSubgroupH1Iso_diamond_square, `Dubon2026.sourceGammaOneConstantH1Transfer_diamond]
  let constructions : Array Name := #[`Dubon2026.sourceInnerCoefficientMorphism, `Dubon2026.sourceSubgroupEqualityCoefficient, `Dubon2026.sourceNormalizerSubgroupHom, `Dubon2026.sourceNormalizerCoefficientMorphism]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE DIAMOND TRANSFER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
