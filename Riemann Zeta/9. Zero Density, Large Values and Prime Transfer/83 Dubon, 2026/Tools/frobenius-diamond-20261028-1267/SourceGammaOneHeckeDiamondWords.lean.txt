/-- Original prime correspondences and original homogeneous normalizer diamonds. -/
def sourceGammaOneHeckeDiamondOperators (Q N w : ℕ) [NeZero N] :
    sourceBinaryHeckePrime (Q * N) ⊕ Gamma0 (Q * N ^ 2) → Module.End (ZMod N)
      (groupCohomology.H1 (Rep.trivial (ZMod N) (Gamma1 (Q * N ^ 2))

  Sum.elim (sourceGammaOnePrimeOperators Q N w) (fun g =>
    sourceGammaOneDiamondH1 (ZMod N) (sourceBinaryCoefficientRep (ZMod N) Q w)

/-- The literal operator image, including diamonds needed by Frobenius normalization. -/
def sourceGammaOneHeckeDiamondAlgebra (Q N w : ℕ) [NeZero N] :=
  (FreeAlgebra.lift ℤ (sourceGammaOneHeckeDiamondOperators Q N w)).range

/-- Original Gamma0 integral words; each original normalizer diamond maps to one. -/

    FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N) ⊕ Gamma0 (Q * N ^ 2)) →ₐ[ℤ]
      sourceBinaryH1HeckeAlgebra Q w :=
  FreeAlgebra.lift ℤ (Sum.elim (fun ell =>
    ⟨sourceBinaryH1PrimeOperators ℤ Q w (sourceGammaOneHeckePrimeForget Q N ell),
      Algebra.subset_adjoin ⟨sourceGammaOneHeckePrimeForget Q N ell, rfl⟩⟩) (fun _ => 1))

      FreeAlgebra.lift ℤ (Sum.elim
        (fun ell => sourceBinaryH1PrimeOperators ℤ Q w (sourceGammaOneHeckePrimeForget Q N ell))
        (fun _ => 1)) a := by
  have he : (sourceBinaryH1HeckeAlgebra Q w).val.comp
      (sourceGammaOneOriginalHeckeDiamondWord Q N w) =
      FreeAlgebra.lift ℤ (Sum.elim
        (fun ell => sourceBinaryH1PrimeOperators ℤ Q w (sourceGammaOneHeckePrimeForget Q N ell))
        (fun _ => 1)) := by
    apply FreeAlgebra.hom_ext
    funext s
    cases s <;>
      simp only [Function.comp_apply, AlgHom.comp_apply,
        sourceGammaOneOriginalHeckeDiamondWord, FreeAlgebra.lift_ι_apply] <;> rfl
  exact AlgHom.congr_fun he a
