open scoped NumberField

/-- Root-fixing regular detection supplies actual good congruence primes,
with the same finite factor and nonzero original local H1 restriction. -/
theorem rationalArithmetic_regular_primes_of_root_detection
    (a : 𝓞 ℚ) (ha : a ≠ 0) (p level : ℕ) [NeZero p]
    (hpa : (p : 𝓞 ℚ) ∣ a)
    {K ι : Type} [Field K] [TopologicalSpace K] [DiscreteTopology K]
    [Fintype ι] [DecidableEq ι]
    (A : Rep.{0} K (rationalArithmeticGaloisGroup a))
    [Finite A] [TopologicalSpace A] [DiscreteTopology A] [ContinuousSMul K A]
    (hA : Continuous (fun z : rationalArithmeticGaloisGroup a × A => A.ρ z.1 z.2))
    (ρ : rationalArithmeticGaloisGroup a →ₜ* Matrix.GeneralLinearGroup ι K)
    (c : continuousRepresentationCocyclesOne A)
    (hroot : HasArithmeticRegularRootDetection a p level A ρ.toMonoidHom c) :
    HasArithmeticRegularDetectingPrimes a p level A hA ρ.toMonoidHom c := by
  dsimp only [HasArithmeticRegularDetectingPrimes]
  let U := rationalUnramifiedExtension a
  letI : Algebra ℚ U := U.algebra'
  letI : IsGalois ℚ U := originalUnramifiedGaloisUnion_isGalois a
  have hprimes := rationalArithmetic_exists_regular_detecting_cocycle
    a ha p (level + 1) hpa A hA ρ c
  obtain ⟨ζ, hζ, D, ψ, hψ, hD⟩ := hprimes
  obtain ⟨σ, hσζ, hs, hdetect⟩ := hroot ζ hζ
  letI : Algebra ℚ D := D.algebra'
  letI : NumberField D := NumberField.of_module_finite ℚ D
  exact ⟨D, ψ, hψ, hD σ hσζ hdetect hs⟩

end
end Dubon2026
