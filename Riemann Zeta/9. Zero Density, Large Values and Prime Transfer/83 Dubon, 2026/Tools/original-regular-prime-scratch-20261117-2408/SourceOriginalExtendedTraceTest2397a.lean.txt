/-- An extended-field trace witness, in the constructed conjugated basis,
detects the original trace-zero vector outside the original coboundary range. -/
theorem traceZero_notMem_coboundary_of_extended_conjugate_test
    {G F E ι : Type} [Group G] [Field F] [Field E] [Fintype ι] [DecidableEq ι]
    (j : F →+* E) (ρ : G →* GeneralLinearGroup ι F) (χ : G →* F)
    (C : GeneralLinearGroup ι E) (g : G) (hχ : χ g = 1)
    (X : matrixTraceZeroSubmodule ι F) (M : matrixTraceZeroSubmodule ι E)
    (hM :
      (((MulEquiv.toMonoidHom (MulAut.conj C)).comp
        ((GeneralLinearGroup.map j).comp ρ)) g).val * M.val =
      M.val * (((MulEquiv.toMonoidHom (MulAut.conj C)).comp
        ((GeneralLinearGroup.map j).comp ρ)) g).val)
    (htrace : Matrix.trace
      ((((matrixTraceZeroAdjointRep (MonoidHom.id (GeneralLinearGroup ι E))).ρ C)
        (traceZeroCoefficientEmbedding j X)).val * M.val) ≠ 0) :
    X ∉ LinearMap.range ((matrixTraceZeroCharacterRep ρ χ).ρ g - 1) := by
  let ρE := (MulEquiv.toMonoidHom (MulAut.conj C)).comp
    ((GeneralLinearGroup.map j).comp ρ)
  let χE := j.toMonoidHom.comp χ
  let T := (matrixTraceZeroAdjointRep (MonoidHom.id (GeneralLinearGroup ι E))).ρ
  let Φ := (T C).toAddMonoidHom.comp (traceZeroCoefficientEmbedding (ι := ι) j)
  let ℓ := matrixTraceZeroTracePairing.flip M
  have hΦ (Y : matrixTraceZeroSubmodule ι F) :
      Φ ((matrixTraceZeroCharacterRep ρ χ).ρ g Y) =
        (matrixTraceZeroCharacterRep ρE χE).ρ g (Φ Y) := by
    change T C (traceZeroCoefficientEmbedding j ((matrixTraceZeroCharacterRep ρ χ).ρ g Y)) = _
    rw [traceZeroCoefficientEmbedding_action]
    exact (traceZero_twisted_conjugate_action ((GeneralLinearGroup.map j).comp ρ)
      χE C g (traceZeroCoefficientEmbedding j Y)).symm
  have hχE : χE g = 1 := by change j (χ g) = 1; rw [hχ, map_one]
  have hfixed (Y : matrixTraceZeroSubmodule ι F) :
      ℓ (Φ ((matrixTraceZeroCharacterRep ρ χ).ρ g Y)) = ℓ (Φ Y) := by
    rw [hΦ]
    exact matrixTraceZero_traceFunctional_fixed ρE χE g hχE M hM (Φ Y)
  change ℓ (Φ X) ≠ 0 at htrace
  rintro ⟨Y, hY⟩
  apply htrace
  rw [← hY]
  change ℓ (Φ ((matrixTraceZeroCharacterRep ρ χ).ρ g Y - Y)) = 0
  rw [map_sub, map_sub, hfixed, sub_self]

end
end Dubon2026
