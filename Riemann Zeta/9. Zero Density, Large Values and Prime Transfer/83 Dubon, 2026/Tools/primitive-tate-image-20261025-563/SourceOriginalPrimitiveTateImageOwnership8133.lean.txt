open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceFiniteBilinearBaseChange, `Dubon2026.SourceOriginalCoefficientWeilNondegeneracy, `Dubon2026.SourceBilinearBaseChangeAlternating, `Dubon2026.SourceOriginalCoefficientWeilLaws, `Dubon2026.SourceBilinearPolynomialAdjoint, `Dubon2026.SourceIdempotentBilinearRestriction, `Dubon2026.SourceHeckeCoordinateRelations, `Dubon2026.SourceOriginalCuspRationalTateHecke, `Dubon2026.SourceIntegerCoefficientExtensionOne, `Dubon2026.SourceIntegerCoefficientEndExtension, `Dubon2026.SourcePrimitiveOriginalTateEvaluation, `Dubon2026.SourcePrimitiveOriginalTateAdjoint, `Dubon2026.SourceEndIdempotentPairingRestriction, `Dubon2026.SourcePrimitiveOriginalTateRestriction, `Dubon2026.SourceCharacterIdempotentLine, `Dubon2026.SourceCharacterIdempotentRankTwo, `Dubon2026.SourceGeneratedCharacterIdempotent, `Dubon2026.SourceTensorCharacterIdempotent, `Dubon2026.SourceOriginalCuspPadicHeckeMap, `Dubon2026.SourceIntegerCoefficientExtensionCharacter, `Dubon2026.SourcePrimitiveHeckeAlgebraEvaluation, `Dubon2026.SourcePrimitiveHeckeAlgebraProjector, `Dubon2026.SourceRankTwoSurjectiveDescent, `Dubon2026.SourceOriginalCuspTateRankTwo, `Dubon2026.SourceRankTwoBaseChange, `Dubon2026.SourceIntegerCoefficientRankTwo, `Dubon2026.SourceIntertwinedRangeEquiv, `Dubon2026.SourceRankTwoCharacterRange, `Dubon2026.SourcePrimitiveHeckeAlgebraRankTwo, `Dubon2026.SourcePrimitiveOriginalTateRankTwo, `Dubon2026.SourcePrimitiveOriginalTateNondegenerateRankTwo]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
