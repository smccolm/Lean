private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFiniteBilinearBaseChange, `SourceOriginalCoefficientWeilNondegeneracy, `SourceBilinearBaseChangeAlternating, `SourceOriginalCoefficientWeilLaws, `SourceBilinearPolynomialAdjoint, `SourceIdempotentBilinearRestriction]
  let theorems : Array Name := #[`Dubon2026.finite_bilinear_baseChange_injective, `Dubon2026.SourceFunctionField.originalCoefficientWeilBilinear_injective, `Dubon2026.sourceBilinear_baseChange_alternating, `Dubon2026.SourceFunctionField.originalCoefficientWeilBilinear_self, `Dubon2026.SourceFunctionField.originalCoefficientWeilBilinear_hecke, `Dubon2026.SourceFunctionField.originalCoefficientWeilBilinear_cyclotomic, `Dubon2026.bilinear_selfAdjoint_pow, `Dubon2026.bilinear_selfAdjoint_aeval, `Dubon2026.bilinear_selfAdjoint_sum_shifted_squares, `Dubon2026.idempotent_bilinear_restriction_injective]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.originalCoefficientWeilBilinear]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Coefficient Weil postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Coefficient Weil theorems discovered"
  logInfo m!"SOURCE ORIGINAL COEFFICIENT WEIL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
