import SourcePrimitiveH1CoefficientField
import SourcePrimitiveLocalH1Character
import SourcePrimitiveAdicH1Quotients
import SourcePrimitiveCompatibleH1AdicData
import SourcePrimitiveH1AdicRealization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourcePrimitiveH1CoefficientField, `SourcePrimitiveLocalH1Character, `SourcePrimitiveAdicH1Quotients, `SourcePrimitiveCompatibleH1AdicData, `SourcePrimitiveH1AdicRealization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
