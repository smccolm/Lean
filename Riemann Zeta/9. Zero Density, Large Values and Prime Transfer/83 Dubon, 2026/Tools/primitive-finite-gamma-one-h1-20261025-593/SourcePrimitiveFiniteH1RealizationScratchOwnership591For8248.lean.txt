import SourceGammaOneCongruenceCover
import SourceBinaryStabilizerGammaOneCover
import SourceGammaOneDiagonalHom
import SourceGammaOneCoverFiniteIndex
import SourceGammaOneCohomologyTransfer
import SourceGammaOneConstantH1Realization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneCongruenceCover, `SourceBinaryStabilizerGammaOneCover, `SourceGammaOneDiagonalHom, `SourceGammaOneCoverFiniteIndex, `SourceGammaOneCohomologyTransfer, `SourceGammaOneConstantH1Realization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
