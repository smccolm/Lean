private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceBinaryH1ModulusKernel, `SourceBinaryPermutationHeckeWords, `SourceBinaryPermutationOperatorKernel, `SourceBinaryPermutationHeckeFinite, `SourceBinaryPermutationOperatorDescent, `SourcePrimitivePermutationH1HeckeCharacter, `SourcePrimitivePermutationH1Realization]
  let theorems : Array Name := #[`Dubon2026.sourceBinaryH1ModulusOperatorMap_zero_of_coefficients, `Dubon2026.sourceBinaryH1ModulusOperatorMap_zero_iff_coefficients, `Dubon2026.sourceBinaryH1HeckeWord_surjective, `Dubon2026.sourceBinaryPermutationHeckeWord_evaluation, `Dubon2026.sourceBinaryH1HeckeWord_change, `Dubon2026.sourceBinaryPermutationHeckeWord_modulus_zero, `Dubon2026.sourceBinaryPrimePowerPermutationH1_finite, `Dubon2026.sourceBinaryPrimePowerPermutationHeckeAlgebra_finite, `Dubon2026.sourceBinaryPermutationHeckeWord_modulus_kernel, `Dubon2026.sourceBinaryPermutationToModulus_comp, `Dubon2026.sourceBinaryPermutationToModulus_surjective, `Dubon2026.sourcePrimitivePermutationH1HeckeCharacter_comp, `Dubon2026.sourcePrimitivePermutationH1HeckeCharacter_hecke, `Dubon2026.sourcePrimitive_exists_permutationH1HeckeCharacters]
  let constructions : Array Name := #[`Dubon2026.sourceBinaryH1HeckeWord, `Dubon2026.sourceBinaryPermutationPrimeOperators, `Dubon2026.sourceBinaryPermutationHeckeAlgebra, `Dubon2026.sourceBinaryPermutationToModulus, `Dubon2026.sourcePrimitivePermutationH1HeckeCharacter, `Dubon2026.sourcePrimitivePermutationH1CharacterData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE PRIMITIVE PERMUTATION OPERATOR CHARACTER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
