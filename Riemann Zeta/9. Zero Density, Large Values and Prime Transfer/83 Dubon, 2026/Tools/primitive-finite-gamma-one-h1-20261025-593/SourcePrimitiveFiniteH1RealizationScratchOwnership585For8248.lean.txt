import SourceFaithfulCharacterDivisibility
import SourceBinaryH1PrimePowerKernel
import SourcePrimitiveH1HeckeCongruence
import SourcePrimitiveH1AdicOperatorKernel
import SourcePrimitiveCongruentH1AdicRealization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceFaithfulCharacterDivisibility, `SourceBinaryH1PrimePowerKernel, `SourcePrimitiveH1HeckeCongruence, `SourcePrimitiveH1AdicOperatorKernel, `SourcePrimitiveCongruentH1AdicRealization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
