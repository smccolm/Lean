private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourcePrimitiveH1CoefficientField, `SourcePrimitiveLocalH1Character, `SourcePrimitiveAdicH1Quotients, `SourcePrimitiveCompatibleH1AdicData, `SourcePrimitiveH1AdicRealization]
  let theorems : Array Name := #[`Dubon2026.sourcePrimitiveH1HeckeCharacter_mem_coefficientRing, `Dubon2026.sourcePrimitiveFieldH1HeckeCharacter_hecke, `Dubon2026.sourcePrimitiveFieldH1HeckeCharacter_isIntegral, `Dubon2026.sourcePrimitiveLocalH1HeckeCharacter_coe, `Dubon2026.sourcePrimitiveLocalH1HeckeCharacter_hecke, `Dubon2026.sourcePrimitiveLocalH1HeckeCharacter_residue, `Dubon2026.sourcePrimitiveH1AdicQuotientCharacter_hecke, `Dubon2026.sourcePrimitiveH1AdicQuotientCharacter_compatible, `Dubon2026.sourceLocalH1AdicQuotient_finite, `Dubon2026.sourcePrimitiveCompatibleH1AdicData_of_local, `Dubon2026.sourcePrimitive_exists_compatibleH1AdicRealization]
  let constructions : Array Name := #[`Dubon2026.sourcePrimitiveFieldH1HeckeCharacter, `Dubon2026.sourcePrimitiveLocalH1HeckeCharacter, `Dubon2026.sourcePrimitiveH1AdicQuotientCharacter, `Dubon2026.sourcePrimitiveCompatibleH1AdicData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE PRIMITIVE H1 ADIC AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
