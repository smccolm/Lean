private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.sourceIntegralBinary_exists_stabilizerH1Realization]
  let localModules : Array Name := #[`SourceIntegralBinaryStabilizerRealization, `SourceIntegralBinaryPermutationRealization, `SourceBinaryPermutationCongruence, `SourceBinaryPrimePowerPermutation, `SourcePermutationCoefficientQuotient, `SourceBinaryPermutationHecke, `SourceCoefficientStabilizerCohomology, `SourceCoefficientPermutationCoordinates]

  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue

          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE BINARY STABILIZER H1 DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the actual original integral binary stabilizer H1 consumer. Only declaration values are traversed. The original finite permutation module is identified with its actual finite-index congruence orbit-stabilizer H1 groups by existing permuted-basis, orbit, product and Shapiro maps. The same original integral classes and their prime Hecke transforms are retained. Identification with arithmetic geometric correspondences and Galois realization remain required. DUB-17--20 OPEN; 16/20."
