private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCongruenceCover, `SourceBinaryStabilizerGammaOneCover, `SourceGammaOneDiagonalHom, `SourceGammaOneCoverFiniteIndex, `SourceGammaOneCohomologyTransfer, `SourceGammaOneConstantH1Realization]
  let theorems : Array Name := #[`Dubon2026.sourceGammaOne_principal_diagonal_transition, `Dubon2026.sourceGammaOne_le_conj_principalLevel, `Dubon2026.sourceBinaryCoefficientStabilizer_gammaOneCover, `Dubon2026.sourceBinaryCoefficientStabilizer_conj_finiteIndex, `Dubon2026.sourceGammaOne_le_raisedGammaZero, `Dubon2026.sourceGammaOneDiagonalHom_matrix, `Dubon2026.sourceGammaOneDiagonalHom_injective, `Dubon2026.sourceGammaOneDiagonalHom_mem_principal, `Dubon2026.sourceGammaOneDiagonalHom_binary_trivial, `Dubon2026.sourceRaisedLevelConjugation_range, `Dubon2026.sourceRaisedLevelConjugation_range_finiteIndex, `Dubon2026.sourceGammaOneDiagonalHom_range_finiteIndex, `Dubon2026.sourceGammaOne_coindH1_surjective, `Dubon2026.sourceGammaOneConstantH1Transfer_surjective, `Dubon2026.sourceIntegralBinary_exists_gammaOneH1Realization]
  let constructions : Array Name := #[`Dubon2026.sourceGammaOneDiagonalHom, `Dubon2026.sourceGammaOneCoefficientTrivialIso, `Dubon2026.sourceGammaOneRangeEquiv, `Dubon2026.sourceGammaOneConstantH1Iso, `Dubon2026.sourceGammaOneConstantH1Transfer]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE COVER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
