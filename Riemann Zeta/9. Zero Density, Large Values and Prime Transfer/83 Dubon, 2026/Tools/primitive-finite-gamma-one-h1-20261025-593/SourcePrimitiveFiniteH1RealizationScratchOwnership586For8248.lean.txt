import SourceBinaryH1ModulusOperatorAlgebra
import SourceFiniteOperatorCharacterDescent
import SourcePrimitiveFiniteH1HeckeCharacter
import SourcePrimitiveFiniteH1OperatorRealization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceBinaryH1ModulusOperatorAlgebra, `SourceFiniteOperatorCharacterDescent, `SourcePrimitiveFiniteH1HeckeCharacter, `SourcePrimitiveFiniteH1OperatorRealization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
