import SourceGammaOneHeckeSubgroup
import SourceGammaOneConstantHecke
import SourceNativeCoinductionCosetTrace
import SourceShapiroH1CocycleComparison
import SourceShapiroH1CosetTrace
import SourceGammaOneTransferCosetFormula
import SourceGammaOneLiteralTransferSurjectivity
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneHeckeSubgroup, `SourceGammaOneConstantHecke, `SourceNativeCoinductionCosetTrace, `SourceShapiroH1CocycleComparison, `SourceShapiroH1CosetTrace, `SourceGammaOneTransferCosetFormula, `SourceGammaOneLiteralTransferSurjectivity]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
