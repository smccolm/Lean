private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneHeckeSubgroup, `SourceGammaOneConstantHecke, `SourceNativeCoinductionCosetTrace, `SourceShapiroH1CocycleComparison, `SourceShapiroH1CosetTrace, `SourceGammaOneTransferCosetFormula, `SourceGammaOneLiteralTransferSurjectivity]
  let theorems : Array Name := #[`Dubon2026.sourceGammaOneHeckeUpper_finiteIndex, `Dubon2026.sourceGammaOneHeckeConjugation_matrix, `Dubon2026.sourceGammaOneConstantHeckeH1_cocycle, `Dubon2026.sourceGammaOneConstantHeckeCocycles_apply, `Dubon2026.sourceNativeInductionCounit_mk, `Dubon2026.sourceNativeCoinductionCounit_eq_cosetTrace, `Dubon2026.sourceShapiroH1_hom_cocycle, `Dubon2026.sourceShapiroH1_evaluation_lift, `Dubon2026.sourceCoinducedCosetTrace_shapiroOne, `Dubon2026.sourceShapiroLiftCocycles_trace, `Dubon2026.sourceHeckeH1Trace_shapiro, `Dubon2026.sourceHeckeH1Trace_eq_subgroupTrace, `Dubon2026.sourceGammaOneConstantH1Iso_factor, `Dubon2026.sourceGammaOneConstantH1Transfer_eq_trace, `Dubon2026.sourceGammaOne_literalTrace_surjective]
  let constructions : Array Name := #[`Dubon2026.sourceGammaOneHeckeUpper, `Dubon2026.sourceGammaOneUpperToGammaZero, `Dubon2026.sourceGammaOneHeckeConjugation, `Dubon2026.sourceGammaOneConstantHeckeMorphism, `Dubon2026.sourceGammaOneConstantHeckeCocycles, `Dubon2026.sourceGammaOneConstantHeckeH1, `Dubon2026.sourceShapiroLiftCocycles, `Dubon2026.sourceShapiroEvaluationCocycles, `Dubon2026.sourceGammaOneSubgroupH1Iso]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HECKE TRACE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
