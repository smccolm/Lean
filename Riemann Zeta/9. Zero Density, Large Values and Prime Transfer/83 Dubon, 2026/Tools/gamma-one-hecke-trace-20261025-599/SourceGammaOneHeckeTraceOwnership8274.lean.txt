open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOneHeckeSubgroup, `Dubon2026.SourceGammaOneConstantHecke, `Dubon2026.SourceNativeCoinductionCosetTrace, `Dubon2026.SourceShapiroH1CocycleComparison, `Dubon2026.SourceShapiroH1CosetTrace, `Dubon2026.SourceGammaOneTransferCosetFormula, `Dubon2026.SourceGammaOneLiteralTransferSurjectivity, `Dubon2026.SourceGammaOneCoverHeckeSquare, `Dubon2026.SourcePrincipalCoprimeReduction, `Dubon2026.SourceGammaOneRaisedCosets, `Dubon2026.SourceGammaOneCoverHeckeCosets, `Dubon2026.SourceGammaOneCoverHeckeRepresentatives, `Dubon2026.SourceGammaOneBinaryRestriction, `Dubon2026.SourceGammaOneBinaryHeckeSum, `Dubon2026.SourceGammaOneBinaryRestrictionHecke, `Dubon2026.SourceCohomologyTracePullbackCoefficients, `Dubon2026.SourceShapiroH1PullbackNaturality, `Dubon2026.SourceH1TraceBaseChange, `Dubon2026.SourceGammaOneCoverImage, `Dubon2026.SourceGammaOneCoverImageHecke, `Dubon2026.SourceGammaOneCoverHeckeTraceCosets, `Dubon2026.SourceGammaOneCoverHeckeTraceBaseChange, `Dubon2026.SourceCoinducedTowerFunctions, `Dubon2026.SourceCoinducedTowerTrace, `Dubon2026.SourceShapiroH1Tower, `Dubon2026.SourceH1TraceTower]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
