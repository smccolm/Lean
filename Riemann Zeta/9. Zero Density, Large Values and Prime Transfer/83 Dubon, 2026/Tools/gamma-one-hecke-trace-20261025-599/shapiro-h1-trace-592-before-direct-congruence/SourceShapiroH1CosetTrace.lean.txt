/-! DUB-17/18 -> original all-weight adic association -> Gamma1 geometric
Hecke comparison. Identify the already proved literal H1 trace with the
native Shapiro/coinduction transfer used to establish surjectivity. -/
namespace Dubon2026
noncomputable section
open CategoryTheory
variable {R G : Type} [CommRing R] [Group G] (A : Rep.{0} R G) (S : Subgroup G)
  [Fintype (G ⧸ S)]

/-- Tracing the existing explicit Shapiro cochain is the original literal
coset trace, with its translated representatives reindexed exactly. -/
theorem sourceCoinducedCosetTrace_shapiroOne (c : S → A) (g : G) :
    (coinducedCosetTrace A S).hom (coinducedShapiroOne A S c g) =
      cosetTraceOne A S c g := by
  change (∑ q : G ⧸ S, coinducedCosetValues A S (coinducedShapiroOne A S c g) q) =
    ∑ q : G ⧸ S, cosetTraceOneTerm A S c g q
  rw [coinducedShapiroOne_values]
  exact Equiv.sum_comp (MulAction.toPerm g⁻¹) (cosetTraceOneTerm A S c g)

/-- The native coefficient map of the explicit lift is the exact cocycle
trace already used in the original Hecke construction. -/
theorem sourceShapiroLiftCocycles_trace
    (c : groupCohomology.cocycles₁ (Rep.res S.subtype A)) :
    groupCohomology.mapCocycles₁ (MonoidHom.id G) (coinducedCosetTrace A S)
      (sourceShapiroLiftCocycles A S c) = sourceHeckeTraceCocycles A S c := by
  ext g
  exact sourceCoinducedCosetTrace_shapiroOne A S c g

/-- The original literal H1 trace after native Shapiro is exactly the
native cohomology map of the actual coinduced coset trace. -/
theorem sourceHeckeH1Trace_shapiro
    (x : groupCohomology.H1 (Rep.coind S.subtype (Rep.res S.subtype A))) :
    sourceHeckeH1Trace A S ((groupCohomology.coindIso (Rep.res S.subtype A) 1).hom x) =
      (groupCohomology.functor R G 1).map (coinducedCosetTrace A S) x := by
  obtain ⟨c, rfl⟩ :=
    (ModuleCat.epi_iff_surjective (groupCohomology.H1π
      (Rep.coind S.subtype (Rep.res S.subtype A)))).mp inferInstance x
  rw [sourceShapiroH1_hom_cocycle, sourceHeckeH1Trace_cocycle]
  calc
    _ = (groupCohomology.functor R G 1).map (coinducedCosetTrace A S)
        (groupCohomology.H1π _
          (sourceShapiroLiftCocycles A S (sourceShapiroEvaluationCocycles A S c))) := by
      change _ = groupCohomology.map (MonoidHom.id G) (coinducedCosetTrace A S) 1 _
      have hm := groupCohomology.H1π_comp_map_apply (MonoidHom.id G)
        (coinducedCosetTrace A S)
        (sourceShapiroLiftCocycles A S (sourceShapiroEvaluationCocycles A S c))
      exact (congrArg (fun z => groupCohomology.H1π A z)
        (sourceShapiroLiftCocycles_trace A S (sourceShapiroEvaluationCocycles A S c))).symm.trans hm.symm
    _ = _ := congrArg
      (fun y => (groupCohomology.functor R G 1).map (coinducedCosetTrace A S) y)
      (sourceShapiroH1_evaluation_lift A S c)

/-- The literal degree-one Hecke trace and the native Shapiro trace are
the same linear map on the full original subgroup H1. -/
theorem sourceHeckeH1Trace_eq_subgroupTrace :
    sourceHeckeH1Trace A S = (subgroupCohomologyTrace A S 1).hom := by
  ext x
  let y := (groupCohomology.coindIso (Rep.res S.subtype A) 1).inv x
  have hy : (groupCohomology.coindIso (Rep.res S.subtype A) 1).hom y = x :=
    (groupCohomology.coindIso (Rep.res S.subtype A) 1).toLinearEquiv.apply_symm_apply x
  rw [← hy, sourceHeckeH1Trace_shapiro]
  change _ = (groupCohomology.functor R G 1).map (coinducedCosetTrace A S)
    ((groupCohomology.coindIso (Rep.res S.subtype A) 1).inv
      ((groupCohomology.coindIso (Rep.res S.subtype A) 1).hom y))
  exact congrArg
    (fun z => (groupCohomology.functor R G 1).map (coinducedCosetTrace A S) z)
    ((groupCohomology.coindIso (Rep.res S.subtype A) 1).toLinearEquiv.symm_apply_apply y).symm

end
end Dubon2026
