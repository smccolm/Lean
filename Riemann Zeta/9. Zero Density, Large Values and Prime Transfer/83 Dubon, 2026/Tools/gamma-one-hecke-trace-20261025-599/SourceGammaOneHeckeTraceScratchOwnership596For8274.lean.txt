import SourceCohomologyTracePullbackCoefficients
import SourceShapiroH1PullbackNaturality
import SourceH1TraceBaseChange
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceCohomologyTracePullbackCoefficients, `SourceShapiroH1PullbackNaturality, `SourceH1TraceBaseChange]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
