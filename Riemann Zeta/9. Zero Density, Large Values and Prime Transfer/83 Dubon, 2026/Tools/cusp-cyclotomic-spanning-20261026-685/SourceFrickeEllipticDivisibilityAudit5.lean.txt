private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFrickeHolomorphicDivisionInvariance, `SourceFrickeHolomorphicDivisionSpan, `SourceFrickeHolomorphicDivisionTwist, `SourceFrickeEllipticDivisibilityTwist, `SourceFrickeEllipticCoordinateTwist]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.principalInvariant_holomorphic_quotient, `Dubon2026.SourceFricke.exists_frickeSpan_laurent, `Dubon2026.SourceFricke.meromorphic_cusp_holomorphic_quotient, `Dubon2026.SourceFricke.mem_frickeSpan_of_holomorphic_quotient, `Dubon2026.SourceFricke.exists_twisted_holomorphic_quotient, `Dubon2026.SourceFricke.frickeCoefficientTwist_transport_elliptic_divisibility, `Dubon2026.SourceFricke.cyclotomicFrickeJShift_ne_zero, `Dubon2026.SourceFricke.frickeCoefficientTwist_E4_divisibility, `Dubon2026.SourceFricke.frickeCoefficientTwist_E6_divisibility]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.cyclotomicFrickeJShift]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Fricke Elliptic Divisibility postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Fricke Elliptic Divisibility theorems discovered"
  logInfo m!"SOURCE FRICKE ELLIPTIC DIVISIBILITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
