private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCuspLaurentLinearRelations, `SourceFrickeLaurentIndependence]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.coeffMap_sum_smul, `Dubon2026.SourceFunctionField.CuspTaylor.Represents.cusp_smul_sum, `Dubon2026.SourceFunctionField.cusp_laurent_sum_smul_eq_zero_iff, `Dubon2026.SourceFunctionField.linearIndependent_functions_of_laurent_coefficients, `Dubon2026.SourceFricke.linearIndependent_holomorphic_cyclotomic_fricke_fractions]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Fricke Laurent Independence postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Fricke Laurent Independence theorems discovered"
  logInfo m!"SOURCE FRICKE LAURENT INDEPENDENCE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
