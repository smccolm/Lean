private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceEisensteinFactorCyclotomicLaurent, `SourceCuspWeightImageFourier, `SourceGammaOneCyclotomicCuspSpan]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.coeffMap_eisensteinLaurentK, `Dubon2026.SourceFricke.periodic_eisensteinFactor, `Dubon2026.SourceFricke.represents_eisensteinFactorLaurentK, `Dubon2026.SourceFricke.eisensteinFactorLaurentK_ne_zero, `Dubon2026.SourceFricke.exists_cuspWeightImage_laurent, `Dubon2026.SourceFricke.cuspWeightImage_qExpansion_coeff_mem, `Dubon2026.SourceFricke.gammaOne_cuspWeightImage_qExpansion_coeff_mem, `Dubon2026.SourceFricke.span_gammaOne_cyclotomic_cuspForms_eq_top, `Dubon2026.SourceFricke.span_gammaOne_weightTwo_cyclotomic_cuspForms_eq_top]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.eisensteinLaurentK]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Cyclotomic Cusp Span postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Cyclotomic Cusp Span theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CYCLOTOMIC CUSP SPAN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
