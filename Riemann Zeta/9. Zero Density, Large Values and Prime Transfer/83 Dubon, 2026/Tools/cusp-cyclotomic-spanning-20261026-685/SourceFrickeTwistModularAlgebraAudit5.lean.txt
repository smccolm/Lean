private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFrickeCyclotomicModularAction, `SourceFrickeTwistModularEquivariance, `SourceFrickeSpanMultiplication, `SourceFrickePrincipalInvariance, `SourceFrickeTwistPowers]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.smul_mem_cyclotomicLevelRing, `Dubon2026.SourceFricke.smul_mem_holomorphicCyclotomicFrickeFractions, `Dubon2026.SourceFricke.smul_mem_frickeSpan, `Dubon2026.SourceFricke.exists_frickeSpan_sum_original, `Dubon2026.SourceFricke.frickeCoefficientTwist_pullback, `Dubon2026.SourceFricke.frickeCoefficientTwist_isZeroAtImInfty_sl_mul_discriminant_iff, `Dubon2026.SourceFricke.mul_mem_frickeSpan, `Dubon2026.SourceFricke.frickeCoefficientTwist_mul, `Dubon2026.SourceFricke.smul_eq_self_cyclotomicLevelRing, `Dubon2026.SourceFricke.smul_eq_self_holomorphicCyclotomicFrickeFractions, `Dubon2026.SourceFricke.smul_eq_self_frickeSpan, `Dubon2026.SourceFricke.principalInvariant_frickeSpan, `Dubon2026.SourceFricke.pow_mem_frickeSpan, `Dubon2026.SourceFricke.frickeCoefficientTwist_pow]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.frickeSpanPullback, `Dubon2026.SourceFricke.frickeSpanMul, `Dubon2026.SourceFricke.frickeSpanPow]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Fricke Twist Modular Algebra postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Fricke Twist Modular Algebra theorems discovered"
  logInfo m!"SOURCE FRICKE TWIST MODULAR ALGEBRA AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
