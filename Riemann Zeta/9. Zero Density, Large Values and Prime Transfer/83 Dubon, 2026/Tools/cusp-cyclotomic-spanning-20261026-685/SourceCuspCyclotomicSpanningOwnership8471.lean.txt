private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceCuspLaurentLinearRelations, `Dubon2026.SourceFrickeLaurentIndependence, `Dubon2026.SourceFrickeCyclotomicHolomorphicAlgebra, `Dubon2026.SourceFrickeScalarExtensionBasis, `Dubon2026.SourceComplexBasisCoefficientTwist, `Dubon2026.SourceFrickeCoefficientTwist, `Dubon2026.SourceFrickeSpanLaurentRealization, `Dubon2026.SourceLaurentTwistOrder, `Dubon2026.SourceCuspVanishingOrder, `Dubon2026.SourceFrickeTwistCuspVanishing, `Dubon2026.SourceFrickeCyclotomicModularAction, `Dubon2026.SourceFrickeTwistModularEquivariance, `Dubon2026.SourceFrickeSpanMultiplication, `Dubon2026.SourceFrickePrincipalInvariance, `Dubon2026.SourceFrickeTwistPowers, `Dubon2026.SourceFrickeHolomorphicDivisionInvariance, `Dubon2026.SourceFrickeHolomorphicDivisionSpan, `Dubon2026.SourceFrickeHolomorphicDivisionTwist, `Dubon2026.SourceFrickeEllipticDivisibilityTwist, `Dubon2026.SourceFrickeEllipticCoordinateTwist, `Dubon2026.SourceHolomorphicCoprimeDivision, `Dubon2026.SourceCuspWeightFactors, `Dubon2026.SourceCuspWeightHolomorphicDivision, `Dubon2026.SourceCuspWeightTransformation, `Dubon2026.SourceCuspWeightCriterion, `Dubon2026.SourceCuspWeightImage, `Dubon2026.SourceFrickeTwistCuspConditions, `Dubon2026.SourceCuspWeightImageSpan, `Dubon2026.SourceCuspWeightImageTwist, `Dubon2026.SourcePureTranscendentalConstantIntersection, `Dubon2026.SourceAlgebraicallyClosedFixedField, `Dubon2026.SourceTwistDescentFiniteSupport, `Dubon2026.SourceTwistStableSubspaceDescent, `Dubon2026.SourceOriginalFrickeSpanDescent, `Dubon2026.SourceCuspWeightImageDescent, `Dubon2026.SourceCuspFrickeRationalSpan, `Dubon2026.SourceEisensteinFactorCyclotomicLaurent, `Dubon2026.SourceCuspWeightImageFourier, `Dubon2026.SourceGammaOneCyclotomicCuspSpan]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
