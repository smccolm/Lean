import SourceHolomorphicCoprimeDivision
import SourceCuspWeightFactors
import SourceCuspWeightHolomorphicDivision
import SourceCuspWeightTransformation
import SourceCuspWeightCriterion
import SourceCuspWeightImage
import SourceFrickeTwistCuspConditions
import SourceCuspWeightImageSpan
import SourceCuspWeightImageTwist
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceHolomorphicCoprimeDivision, `SourceCuspWeightFactors, `SourceCuspWeightHolomorphicDivision, `SourceCuspWeightTransformation, `SourceCuspWeightCriterion, `SourceCuspWeightImage, `SourceFrickeTwistCuspConditions, `SourceCuspWeightImageSpan, `SourceCuspWeightImageTwist]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
