private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourcePureTranscendentalConstantIntersection, `SourceAlgebraicallyClosedFixedField, `SourceTwistDescentFiniteSupport, `SourceTwistStableSubspaceDescent, `SourceOriginalFrickeSpanDescent, `SourceCuspWeightImageDescent, `SourceCuspFrickeRationalSpan]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.ConstantFixedField.mem_bot_of_isAlgebraic_of_mem_adjoin_algebraicIndependent, `Dubon2026.SourceFunctionField.ConstantFixedField.exists_algEquiv_apply_ne_of_exists_notMem, `Dubon2026.SourceFunctionField.ConstantFixedField.exists_intermediateField_isAlgebraic_notMem_of_isAlgebraic, `Dubon2026.SourceFunctionField.ConstantFixedField.notMem_adjoin_sq_of_transcendental, `Dubon2026.SourceFunctionField.ConstantFixedField.exists_algEquiv_apply_ne_of_transcendental, `Dubon2026.SourceFunctionField.ConstantFixedField.exists_algEquiv_apply_ne, `Dubon2026.SourceFunctionField.ConstantFixedField.mem_range_of_fixed, `Dubon2026.SourceFunctionField.TwistDescent.descent_core, `Dubon2026.SourceFunctionField.TwistDescent.descent_generic, `Dubon2026.SourceFricke.mem_originalFrickeSpan_iff, `Dubon2026.SourceFricke.span_originalFrickeSpan_eq_top, `Dubon2026.SourceFricke.frickeCoefficientTwist_fixes_original_submodule, `Dubon2026.SourceFricke.frickeCoefficientTwist_sum_original_submodule, `Dubon2026.SourceFricke.mem_span_originalFrickeSpan_of_twist_stable, `Dubon2026.SourceFricke.cuspWeightImageIntoSpan_injective, `Dubon2026.SourceFricke.cuspWeightImageIntoSpan_twist_mem_range, `Dubon2026.SourceFricke.cuspWeightImageIntoSpan_descent, `Dubon2026.SourceFricke.span_cuspWeightImage_cyclotomic_fractions_eq_top]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.originalFrickeSpan, `Dubon2026.SourceFricke.cuspWeightImageIntoSpan]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Cusp Fricke Rational Span postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Cusp Fricke Rational Span theorems discovered"
  logInfo m!"SOURCE CUSP FRICKE RATIONAL SPAN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
