/-- The actual trace, restricted to the original joint Hecke image. -/
def binaryPrimitiveJointHeckeTrace
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    (g : Qbar ≃ₐ[ℚ] Qbar) : (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq).range :=
  ⟨binaryPrimitiveJointTrace Q p n w g, binaryPrimitiveJointTrace_mem_range Q p n w Hq hH g⟩

/-- The multiplicative original determinant, restricted to the same Hecke image. -/
def binaryPrimitiveJointHeckeDeterminant
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J) :
    (Qbar ≃ₐ[ℚ] Qbar) →* (binaryPrimitiveJointHeckeDiamondWord Q p n w Hq).range where
  toFun g := ⟨binaryPrimitiveJointDeterminant Q p n w g,
    binaryPrimitiveJointDeterminant_mem_range Q p n w Hq g⟩
  map_one' := Subtype.ext (map_one _)
  map_mul' g h := Subtype.ext (map_mul _ g h)

private theorem continuous_discrete_subtype {X Y : Type} [TopologicalSpace X]
    (f : X → Y) (P : Y → Prop) (hf : @Continuous X Y _ ⊥ f) (hP : ∀ x, P (f x)) :
    @Continuous X {y // P y} _ ⊥ (fun x => ⟨f x, hP x⟩) := by
  letI : TopologicalSpace Y := ⊥
  letI : DiscreteTopology Y := ⟨rfl⟩
  have he : (inferInstance : TopologicalSpace {y // P y}) = ⊥ := DiscreteTopology.eq_bot
  exact Eq.mp (congrArg (fun t : TopologicalSpace {y // P y} =>
    @Continuous X {y // P y} _ t (fun x => ⟨f x, hP x⟩)) he) (hf.subtype_mk hP)

/-- Restricting the proved trace image preserves finite discrete continuity. -/
theorem binaryPrimitiveJointHeckeTrace_continuous
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell)) :
    @Continuous _ _ _ ⊥ (binaryPrimitiveJointHeckeTrace Q p n w Hq hH) :=
  continuous_discrete_subtype _ _ (binaryPrimitiveJointTrace_continuous Q p n w) _

/-- The same determinant is continuous in the original finite Hecke image. -/
theorem binaryPrimitiveJointHeckeDeterminant_continuous
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J) :
    @Continuous _ _ _ ⊥ (binaryPrimitiveJointHeckeDeterminant Q p n w Hq) :=
  continuous_discrete_subtype _ _ (binaryPrimitiveJointDeterminant_continuous Q p n w) _

/-- The exact product identity holds inside the original Hecke image. -/
theorem binaryPrimitiveJointHeckeTrace_product
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    (g h : Qbar ≃ₐ[ℚ] Qbar) :
    binaryPrimitiveJointHeckeTrace Q p n w Hq hH g * binaryPrimitiveJointHeckeTrace Q p n w Hq hH h =
      binaryPrimitiveJointHeckeTrace Q p n w Hq hH (g * h) +
        binaryPrimitiveJointHeckeTrace Q p n w Hq hH (g * h⁻¹) *
          binaryPrimitiveJointHeckeDeterminant Q p n w Hq h :=
  Subtype.ext (binaryPrimitiveJointTrace_product Q p n w Hq hH g h)
