private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceAbsoluteNormFrobeniusConjugate, `Dubon2026.SourceGammaOnePrimitiveUnitDiamond, `Dubon2026.SourceGammaOnePrimitiveJointUnitDiamond, `Dubon2026.SourceGammaOnePrimitiveJointDeterminant, `Dubon2026.SourceGammaOnePrimitiveJointTrace, `Dubon2026.SourceGammaOnePrimitiveJointTraceFrobenius, `Dubon2026.SourceUnitTraceConjugation, `Dubon2026.SourceGammaOnePrimitiveJointTraceConjugation, `Dubon2026.SourceGammaOnePrimitiveJointTraceRange, `Dubon2026.SourceUnitTraceProduct, `Dubon2026.SourceGammaOnePrimitiveJointTraceIdentities, `Dubon2026.SourceGammaOnePrimitiveJointHeckeTrace, `Dubon2026.SourceGammaOnePrimitiveJointTraceSpecialization, `Dubon2026.SourceGammaOnePrimitiveJointTraceSpecializedFrobenius, `Dubon2026.SourceGammaOneOriginalAdicTrace, `Dubon2026.SourceAbsoluteContinuousLocalFactor, `Dubon2026.SourceAbsoluteCyclicFrobeniusEquality, `Dubon2026.SourceAbsoluteTracePairComparison, `Dubon2026.SourceGammaOneCompatibleAdicTrace, `Dubon2026.SourcePositiveAdicFamilyLift, `Dubon2026.SourcePositiveAdicReconstruction, `Dubon2026.SourceAdicFunctionContinuity, `Dubon2026.SourceCompatibleTraceLift, `Dubon2026.SourceGammaOneOriginalLocalTrace, `Dubon2026.SourceGammaOneConstructedLocalTrace, `Dubon2026.SourceTraceDeterminantTrilinear, `Dubon2026.SourceGroupAlgebraTrace, `Dubon2026.SourceGroupTraceTrilinear, `Dubon2026.SourceTraceRadical, `Dubon2026.SourceTraceRadicalQuotient, `Dubon2026.SourceTraceQuotientQuadratic, `Dubon2026.SourceTraceIdempotentCorner, `Dubon2026.SourceActualTraceAlgebra, `Dubon2026.SourceGammaOneOriginalTraceAlgebra]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
