private def runOriginalTraceAlgebraDependencyAudit : CoreM Unit := do

  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_original_local_trace_algebra]
  let localModules : Array Name := #[`SourceTraceDeterminantTrilinear, `SourceGroupAlgebraTrace, `SourceGroupTraceTrilinear, `SourceTraceRadical, `SourceTraceRadicalQuotient, `SourceTraceQuotientQuadratic, `SourceTraceIdempotentCorner, `SourceActualTraceAlgebra, `SourceGammaOneOriginalTraceAlgebra]

  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE ORIGINAL TRACE ALGEBRA DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the VALUE closure of the original trace-algebra consumer. Original constructed local traces, native group-algebra induction, actual trace radical and quotient lift are consumed. Rank-two matrix realization and DUB-17--20 remain OPEN."

#eval runOriginalTraceAlgebraDependencyAudit
