private theorem trace_of_quadratic {G R : Type} [Group G] [Ring R]
    (ρ : G →* R) (g : G) (H D : R) (h : ρ g * H = ρ g * ρ g + D) :
    ρ g + ρ g⁻¹ * D = H := by
  apply ((Group.isUnit g).map ρ).mul_left_cancel
  calc
    ρ g * (ρ g + ρ g⁻¹ * D) = ρ g * ρ g + D := by
      rw [mul_add, ← mul_assoc, ← map_mul, mul_inv_cancel, map_one, one_mul]
    _ = ρ g * H := h.symm

/-- The trace expression determined by the actual joint Galois action and
its original multiplicative determinant term. -/
def binaryPrimitiveJointTrace (σ : Qbar ≃ₐ[ℚ] Qbar) :
    binaryPrimitiveJointOperatorRing Q p n w :=
  binaryPrimitiveJointGaloisRep Q p n w σ +
    binaryPrimitiveJointGaloisRep Q p n w σ⁻¹ * binaryPrimitiveJointDeterminant Q p n w σ

/-- The actual candidate trace is continuous in the finite discrete ring. -/
theorem binaryPrimitiveJointTrace_continuous :
    @Continuous _ _ _ ⊥ (binaryPrimitiveJointTrace Q p n w) := by
  letI : TopologicalSpace (binaryPrimitiveJointOperatorRing Q p n w) := ⊥
  letI : DiscreteTopology (binaryPrimitiveJointOperatorRing Q p n w) := ⟨rfl⟩
  exact (binaryPrimitiveJointGaloisRep_continuous Q p n w).add
    (((binaryPrimitiveJointGaloisRep_continuous Q p n w).comp continuous_inv).mul
      (binaryPrimitiveJointDeterminant_continuous Q p n w))

/-- Actual original local Frobenius witnesses evaluate the same trace expression
at the original Hecke generator; all geometric and diamond-label inputs are proved. -/
theorem binaryPrimitiveJointTrace_exists_local_frobenius
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    (ell : sourceBinaryHeckePrime (Q * N))
    (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure ell.val ell.property.1) :
    let A := sourceRationalLocalValuation ell.val ell.property.1 e
    ∃ d : A.decompositionSubgroup ℚ,
      (∀ a : IsLocalRing.ResidueField A, d • a = a ^ ell.val) ∧
      binaryPrimitiveJointTrace Q p n w d.val =
        binaryPrimitiveJointHeckeDiamondWord Q p n w Hq (FreeAlgebra.ι ℤ (Sum.inl ell)) := by
  obtain ⟨g, hg, d, hd, hpoly⟩ :=
    binaryPrimitiveJointGaloisRep_exists_local_frobenius Q p n w Hq hH ell e
  let A := sourceRationalLocalValuation ell.val ell.property.1 e
  letI := GaussReduction.residue_charP_of_prime_mem_nonunits A ell.val ell.property.1
    (sourceRationalLocalValuation_prime ell.val ell.property.1 e)
  have hD := binaryPrimitiveJointDeterminant_frobenius Q p n w A ell Hq g hg d hd
  refine ⟨d, hd, ?_⟩
  exact trace_of_quadratic (binaryPrimitiveJointGaloisRep Q p n w) d.val _ _
    (hpoly.trans (congrArg
      (binaryPrimitiveJointGaloisRep Q p n w d.val * binaryPrimitiveJointGaloisRep Q p n w d.val + ·)
      hD.symm))

end
