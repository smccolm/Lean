/-- Actual Galois operators commute with every value of the original trace. -/
theorem binaryPrimitiveJointGaloisRep_trace_commute
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    (g h : Qbar ≃ₐ[ℚ] Qbar) :
    Commute (binaryPrimitiveJointGaloisRep Q p n w g) (binaryPrimitiveJointTrace Q p n w h) := by
  obtain ⟨a, ha⟩ := binaryPrimitiveJointTrace_mem_range Q p n w Hq hH h

/-- Every original trace value is invariant under actual Galois conjugation. -/

private theorem commute_of_mem_range {A B : Type} [Ring A] [Ring B]
    (θ : A →+* B) [IsMulCommutative θ.range] (x y : B)
    (hx : x ∈ θ.range) (hy : y ∈ θ.range) : Commute x y :=
  congrArg Subtype.val (mul_comm (⟨x, hx⟩ : θ.range) ⟨y, hy⟩)

identity with its actual multiplicative determinant, with centrality proved. -/
theorem binaryPrimitiveJointTrace_product
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) J)
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
      (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))
    (g h : Qbar ≃ₐ[ℚ] Qbar) :
    binaryPrimitiveJointTrace Q p n w g * binaryPrimitiveJointTrace Q p n w h =
      binaryPrimitiveJointTrace Q p n w (g * h) +
        binaryPrimitiveJointTrace Q p n w (g * h⁻¹) * binaryPrimitiveJointDeterminant Q p n w h := by
  let θ := binaryPrimitiveJointHeckeDiamondWord Q p n w Hq
  letI : IsMulCommutative θ.range :=
    binaryPrimitiveJointHeckeDiamondWord_range_isMulCommutative Q p n w Hq hH
  have hDT : Commute (binaryPrimitiveJointDeterminant Q p n w g)
      (binaryPrimitiveJointTrace Q p n w h) := commute_of_mem_range θ _ _
    (binaryPrimitiveJointDeterminant_mem_range Q p n w Hq g)
    (binaryPrimitiveJointTrace_mem_range Q p n w Hq hH h)
  have hDρ : Commute (binaryPrimitiveJointDeterminant Q p n w h)
      (binaryPrimitiveJointGaloisRep Q p n w g⁻¹) := by
    obtain ⟨a, ha⟩ := binaryPrimitiveJointDeterminant_mem_range Q p n w Hq h
    rw [← ha]
    exact (binaryPrimitiveJointGaloisRep_word Q p n w Hq hH g⁻¹ a).symm
  have hDD : Commute (binaryPrimitiveJointDeterminant Q p n w h)
      (binaryPrimitiveJointDeterminant Q p n w g) := commute_of_mem_range θ _ _
    (binaryPrimitiveJointDeterminant_mem_range Q p n w Hq h)
    (binaryPrimitiveJointDeterminant_mem_range Q p n w Hq g)
  exact sourceUnitTrace_product (binaryPrimitiveJointGaloisRep Q p n w)
    (binaryPrimitiveJointDeterminant Q p n w) g h
    ((binaryPrimitiveJointGaloisRep_trace_commute Q p n w Hq hH g⁻¹ h).mul_left hDT) hDρ hDD
