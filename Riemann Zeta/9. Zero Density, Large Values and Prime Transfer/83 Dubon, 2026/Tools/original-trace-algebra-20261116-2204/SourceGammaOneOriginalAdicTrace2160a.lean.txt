/-- The original modular form gives a continuous dimension-two trace system
at every adic precision, with exact source Frobenius values and no supplied
Hecke character, density, centrality, local geometry or finite trace identity. -/
theorem exists_gammaOne_original_adic_trace
    (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k) (hlarge : 3 < p)
    {E : Type} [Field E] [CharZero E] [ValuativeRel E] [CharP 𝓀[E] p]
    (τ : sourcePrimitiveCoefficientField f →ₐ[ℚ] E) :
    let C := 𝒪[E] ⧸ (IsLocalRing.maximalIdeal 𝒪[E]) ^ (n + 1)
    ∃ (t : (Qbar ≃ₐ[ℚ] Qbar) → C) (D : (Qbar ≃ₐ[ℚ] Qbar) →* C),
      @Continuous _ _ _ ⊥ t ∧ @Continuous _ _ _ ⊥ D ∧ t 1 = 2 ∧
      (∀ g h, t (g * h) = t (h * g)) ∧
      (∀ g h, t g * t h = t (g * h) + t (g * h⁻¹) * D h) ∧
      ∀ (q : ℕ) (hq : q.Prime), ¬q ∣ Q * p →
        ∀ (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
          (φ : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq)),
          (∀ z : IsLocalRing.ResidueField (algebraicLocalIntegerRing
            (rationalPrimeLocalClosure q hq)), φ • z = z ^ q) →
          t (embeddedNormalGaloisRestriction e φ) =
            Ideal.Quotient.mk ((IsLocalRing.maximalIdeal 𝒪[E]) ^ (n + 1))
              ⟨τ (sourcePrimitiveFieldCoefficient f q),
                sourcePrimitiveCoefficient_valuation_le_one f hk τ q⟩ ∧
          D (embeddedNormalGaloisRestriction e φ) = (q : C) ^ Int.toNat (k - 1) := by
  have hs := exists_gammaOne_primitive_joint_original_character Q p n f hk hlarge τ
  let Hq := hs.2.choose
  have hH := hs.2.choose_spec.1
  have hchar := hs.2.choose_spec.2
  let χ := hchar.choose
  have hχ := hchar.choose_spec
  let t := binaryPrimitiveJointSpecializedTrace Q p d w Hq hH χ
  let D := binaryPrimitiveJointSpecializedDeterminant Q p d w Hq χ
  refine ⟨t, D,
    binaryPrimitiveJointSpecializedTrace_continuous Q p d w Hq hH χ,
    binaryPrimitiveJointSpecializedDeterminant_continuous Q p d w Hq χ,
    binaryPrimitiveJointSpecializedTrace_one Q p d w Hq hH χ,
    binaryPrimitiveJointSpecializedTrace_mul_comm Q p d w Hq hH χ,
    binaryPrimitiveJointSpecializedTrace_product Q p d w Hq hH χ, ?_⟩
  intro q hq haway e φ hφ
  have hcop : q.Coprime (Q * N) := hq.coprime_iff_not_dvd.mpr (by
    intro h
    rcases hq.dvd_mul.mp h with hQ | hN
    · exact haway (dvd_mul_of_dvd_left hQ p)
    · exact haway (dvd_mul_of_dvd_right (hq.dvd_of_dvd_pow hN) Q))
  let ell : sourceBinaryHeckePrime (Q * N) := ⟨q, hq, hcop⟩
  have ht := binaryPrimitiveJointSpecializedTrace_norm_frobenius Q p d w Hq hH χ ell e φ hφ
  have hc := binaryPrimitiveJoint_character_hecke Q p n f hk τ Hq χ hχ ell
  have hD := binaryPrimitiveJointSpecializedDeterminant_norm_frobenius Q p d w Hq χ
    (binaryPrimitiveJoint_character_diamond Q p n f hk τ Hq χ hχ) ell e φ hφ
  have hw : Int.toNat (k - 2) + 1 = Int.toNat (k - 1) := by omega
  exact ⟨ht.trans hc, by simpa only [hw] using hD⟩

end
