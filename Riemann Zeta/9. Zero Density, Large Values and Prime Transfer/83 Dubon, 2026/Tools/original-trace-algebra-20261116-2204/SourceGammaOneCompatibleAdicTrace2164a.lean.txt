/-- The original form supplies one compatible continuous trace/determinant
system at all adic precisions, with literal original Fourier Frobenius values. -/
theorem exists_gammaOne_compatible_original_adic_trace
    (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k) (hlarge : 3 < p)
    {E : Type} [Field E] [CharZero E] [ValuativeRel E] [CharP 𝓀[E] p]
    (τ : sourcePrimitiveCoefficientField f →ₐ[ℚ] E) :
    let C := fun n : ℕ => 𝒪[E] ⧸ (IsLocalRing.maximalIdeal 𝒪[E]) ^ (n + 1)
    ∃ (t : ∀ n : ℕ, (Qbar ≃ₐ[ℚ] Qbar) → C n)
      (D : ∀ n : ℕ, (Qbar ≃ₐ[ℚ] Qbar) →* C n),
      (∀ n, @Continuous _ _ _ ⊥ (t n) ∧ @Continuous _ _ _ ⊥ (D n) ∧ t n 1 = 2 ∧
        (∀ g h, t n (g * h) = t n (h * g)) ∧
        (∀ g h, t n g * t n h = t n (g * h) + t n (g * h⁻¹) * D n h) ∧
        ∀ (q : ℕ) (hq : q.Prime), ¬q ∣ Q * p →
          ∀ (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
            (φ : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq)),
            (∀ z : IsLocalRing.ResidueField (algebraicLocalIntegerRing
              (rationalPrimeLocalClosure q hq)), φ • z = z ^ q) →
            t n (embeddedNormalGaloisRestriction e φ) =
              Ideal.Quotient.mk ((IsLocalRing.maximalIdeal 𝒪[E]) ^ (n + 1))
                ⟨τ (sourcePrimitiveFieldCoefficient f q),
                  sourcePrimitiveCoefficient_valuation_le_one f hk τ q⟩ ∧
            D n (embeddedNormalGaloisRestriction e φ) = (q : C n) ^ Int.toNat (k - 1)) ∧
      ∀ (m n : ℕ) (hmn : m ≤ n) (g : Qbar ≃ₐ[ℚ] Qbar),
        Ideal.Quotient.factorPow (IsLocalRing.maximalIdeal 𝒪[E]) (Nat.add_le_add_right hmn 1)
            (t n g) = t m g ∧
        Ideal.Quotient.factorPow (IsLocalRing.maximalIdeal 𝒪[E]) (Nat.add_le_add_right hmn 1)
            (D n g) = D m g := by
  let C := fun n : ℕ => 𝒪[E] ⧸ (IsLocalRing.maximalIdeal 𝒪[E]) ^ (n + 1)
  have hs := fun n => exists_gammaOne_original_adic_trace Q p n f hk hlarge τ
  let t : ∀ n, (Qbar ≃ₐ[ℚ] Qbar) → C n := fun n => (hs n).choose
  let D : ∀ n, (Qbar ≃ₐ[ℚ] Qbar) →* C n := fun n => (hs n).choose_spec.choose
  have hdata := fun n => (hs n).choose_spec.choose_spec
  refine ⟨t, D, hdata, ?_⟩
  intro m n hmn g
  letI : TopologicalSpace (C m) := ⊥
  letI : DiscreteTopology (C m) := ⟨rfl⟩
  letI : TopologicalSpace (C n) := ⊥
  letI : DiscreteTopology (C n) := ⟨rfl⟩
  let π : C n →+* C m := Ideal.Quotient.factorPow
    (IsLocalRing.maximalIdeal 𝒪[E]) (Nat.add_le_add_right hmn 1)
  let f₁ : Gal(Qbar/ℚ) → C m × C m := fun a => (π (t n a), π (D n a))
  let f₂ : Gal(Qbar/ℚ) → C m × C m := fun a => (t m a, D m a)
  have hπ : Continuous π := continuous_of_discreteTopology
  have hc₁ : Continuous f₁ :=
    (hπ.comp (hdata n).1).prodMk (hπ.comp (hdata n).2.1)
  have hc₂ : Continuous f₂ := (hdata m).1.prodMk (hdata m).2.1
  have ht₁ (a b : Gal(Qbar/ℚ)) : f₁ (a * b) = f₁ (b * a) := by
    apply Prod.ext
    · exact congrArg π ((hdata n).2.2.2.1 a b)
    · change π (D n (a * b)) = π (D n (b * a))
      simp only [map_mul, mul_comm]
  have ht₂ (a b : Gal(Qbar/ℚ)) : f₂ (a * b) = f₂ (b * a) := by
    apply Prod.ext
    · exact (hdata m).2.2.2.1 a b
    · change D m (a * b) = D m (b * a)
      simp only [map_mul, mul_comm]
  have he : f₁ = f₂ := absolute_continuous_cyclic_eq_of_norm_frobenius
    f₁ f₂ hc₁ hc₂ ht₁ ht₂ (Q * p).primeFactors (by
      intro q hq hqS e φ hφ
      have haway : ¬ q ∣ Q * p := fun h => hqS (hq.mem_primeFactors' h)
      have hn := (hdata n).2.2.2.2.2 q hq haway e φ hφ
      have hm := (hdata m).2.2.2.2.2 q hq haway e φ hφ
      apply Prod.ext
      · change π (t n (embeddedNormalGaloisRestriction e φ)) = t m (embeddedNormalGaloisRestriction e φ)
        rw [hn.1, hm.1]
        exact Ideal.Quotient.factor_mk _ _ _ _
      · change π (D n (embeddedNormalGaloisRestriction e φ)) = D m (embeddedNormalGaloisRestriction e φ)
        rw [hn.2, hm.2, map_pow, map_natCast])
  exact ⟨congrArg Prod.fst (congrFun he g), congrArg Prod.snd (congrFun he g)⟩

end
