import SourceTraceDeterminantTrilinear
import SourceGroupAlgebraTrace
import SourceGroupTraceTrilinear
import SourceTraceRadical
import SourceTraceRadicalQuotient
import SourceTraceQuotientQuadratic
import SourceTraceIdempotentCorner
import SourceActualTraceAlgebra
import SourceGammaOneOriginalTraceAlgebra
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceTraceDeterminantTrilinear, `SourceGroupAlgebraTrace, `SourceGroupTraceTrilinear, `SourceTraceRadical, `SourceTraceRadicalQuotient, `SourceTraceQuotientQuadratic, `SourceTraceIdempotentCorner, `SourceActualTraceAlgebra, `SourceGammaOneOriginalTraceAlgebra]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
