open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceAbsoluteNormFrobeniusConjugate, `SourceGammaOnePrimitiveUnitDiamond, `SourceGammaOnePrimitiveJointUnitDiamond, `SourceGammaOnePrimitiveJointDeterminant, `SourceGammaOnePrimitiveJointTrace, `SourceGammaOnePrimitiveJointTraceFrobenius, `SourceUnitTraceConjugation, `SourceGammaOnePrimitiveJointTraceConjugation, `SourceGammaOnePrimitiveJointTraceRange, `SourceUnitTraceProduct, `SourceGammaOnePrimitiveJointTraceIdentities, `SourceGammaOnePrimitiveJointHeckeTrace, `SourceGammaOnePrimitiveJointTraceSpecialization, `SourceGammaOnePrimitiveJointTraceSpecializedFrobenius, `SourceGammaOneOriginalAdicTrace, `SourceAbsoluteContinuousLocalFactor, `SourceAbsoluteCyclicFrobeniusEquality, `SourceAbsoluteTracePairComparison, `SourceGammaOneCompatibleAdicTrace]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
