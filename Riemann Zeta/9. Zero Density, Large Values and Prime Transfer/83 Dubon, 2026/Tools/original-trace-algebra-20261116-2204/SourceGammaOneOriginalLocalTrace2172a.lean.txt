/-- The original form has a continuous characteristic-zero trace and determinant
in every genuine local coefficient field of residue characteristic greater than three,
with the original Fourier values at every good-prime norm Frobenius. -/
theorem exists_gammaOne_original_local_trace
    (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k) (hlarge : 3 < p)
    {E : Type} [Field E] [CharZero E] [ValuativeRel E] [UniformSpace E]
    [IsUniformAddGroup E] [IsNonarchimedeanLocalField E] [CharP 𝓀[E] p]
    (τ : sourcePrimitiveCoefficientField f →ₐ[ℚ] E) :
    ∃ (t : (Qbar ≃ₐ[ℚ] Qbar) → 𝒪[E])
      (D : (Qbar ≃ₐ[ℚ] Qbar) →* 𝒪[E]),
      Continuous t ∧ Continuous D ∧ t 1 = 2 ∧
      (∀ g h, t (g * h) = t (h * g)) ∧
      (∀ g h, t g * t h = t (g * h) + t (g * h⁻¹) * D h) ∧
      ∀ (q : ℕ) (hq : q.Prime), ¬q ∣ Q * p →
        ∀ (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
          (φ : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq)),
          (∀ z : IsLocalRing.ResidueField (algebraicLocalIntegerRing
            (rationalPrimeLocalClosure q hq)), φ • z = z ^ q) →
          t (embeddedNormalGaloisRestriction e φ) =
            ⟨τ (sourcePrimitiveFieldCoefficient f q),
              sourcePrimitiveCoefficient_valuation_le_one f hk τ q⟩ ∧
          D (embeddedNormalGaloisRestriction e φ) = (q : 𝒪[E]) ^ Int.toNat (k - 1) := by
  obtain ⟨ts, Ds, hdata, hcomp⟩ :=
    exists_gammaOne_compatible_original_adic_trace Q p f hk hlarge τ
  obtain ⟨t, D, ht, hD, hOne, hCyc, hProd, hquot⟩ :=
    exists_complete_trace_of_compatible_positive_quotients
      (IsLocalRing.maximalIdeal 𝒪[E]) compactSeparatedLocalRing_isAdic ts Ds
      (fun n => (hdata n).1) (fun n => (hdata n).2.1)
      (fun n => (hdata n).2.2.1) (fun n => (hdata n).2.2.2.1)
      (fun n => (hdata n).2.2.2.2.1) hcomp
  refine ⟨t, D, ht, hD, hOne, hCyc, hProd, ?_⟩
  intro q hq haway e φ hφ
  constructor
  · apply positiveAdic_ext (IsLocalRing.maximalIdeal 𝒪[E])
    intro n
    exact (hquot n _).1.trans ((hdata n).2.2.2.2.2 q hq haway e φ hφ).1
  · apply positiveAdic_ext (IsLocalRing.maximalIdeal 𝒪[E])
    intro n
    rw [(hquot n _).2, ((hdata n).2.2.2.2.2 q hq haway e φ hφ).2,
      map_pow, map_natCast]

end
