open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceComplexWedgeStokes, `Dubon2026.SourceComplexWedgeLocalLog, `Dubon2026.SourceComplexWedgeDisc, `Dubon2026.SourceComplexWedgeIntegral, `Dubon2026.SourceGammaOneUntwistWedgeIntegral, `Dubon2026.SourceModularSmoothCutoff, `Dubon2026.SourceModularSmoothGeometry, `Dubon2026.SourceModularSmoothCosetSum, `Dubon2026.SourceModularSmoothCover, `Dubon2026.SourceModularSmoothedFundamentalSupport, `Dubon2026.SourceModularSmoothedFundamental, `Dubon2026.SourceGammaOneSmoothPartition, `Dubon2026.SourceModularProjectiveFibers, `Dubon2026.SourceModularSmoothOrbitMass, `Dubon2026.SourceModularInvariantIntegralUnfolding, `Dubon2026.SourceModularSmoothIntegralLimit, `Dubon2026.SourceModularSmoothPlaneIntegral, `Dubon2026.SourceGammaOnePeterssonSmoothLimit, `Dubon2026.SourceComplexWindingLogLift, `Dubon2026.SourceModularWindingCharacter, `Dubon2026.SourceModularWindingCuspLimit, `Dubon2026.SourceModularParabolicWindingLog, `Dubon2026.SourceComplexDipoleLocalModel, `Dubon2026.SourceComplexDipoleLogDerivative, `Dubon2026.SourceComplexDipolePairing, `Dubon2026.SourceUpperHalfPlaneDipole]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
