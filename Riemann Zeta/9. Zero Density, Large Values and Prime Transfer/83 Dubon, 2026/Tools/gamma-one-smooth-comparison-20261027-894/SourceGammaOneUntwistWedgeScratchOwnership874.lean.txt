import SourceComplexWedgeStokes
import SourceComplexWedgeLocalLog
import SourceComplexWedgeDisc
import SourceComplexWedgeIntegral
import SourceGammaOneUntwistWedgeIntegral
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceComplexWedgeStokes, `SourceComplexWedgeLocalLog, `SourceComplexWedgeDisc, `SourceComplexWedgeIntegral, `SourceGammaOneUntwistWedgeIntegral]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
