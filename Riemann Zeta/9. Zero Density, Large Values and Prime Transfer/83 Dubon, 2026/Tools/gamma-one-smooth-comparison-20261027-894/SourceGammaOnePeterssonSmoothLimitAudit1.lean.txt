private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceModularProjectiveFibers, `SourceModularSmoothOrbitMass, `SourceModularInvariantIntegralUnfolding, `SourceModularSmoothIntegralLimit, `SourceModularSmoothPlaneIntegral, `SourceGammaOnePeterssonSmoothLimit]
  let theorems : Array Name := #[`Dubon2026.SourceSmoothedUnfolding.projection_surjective, `Dubon2026.SourceSmoothedUnfolding.mem_projection_ker, `Dubon2026.SourceSmoothedUnfolding.projection_ker_card, `Dubon2026.SourceSmoothedUnfolding.projection_fiber_card, `Dubon2026.SourceSmoothedUnfolding.tsum_projection, `Dubon2026.SourceSmoothedUnfolding.finite_support_projective, `Dubon2026.SourceSmoothedUnfolding.projectiveMass_nonneg, `Dubon2026.SourceSmoothedUnfolding.two_mul_projectiveMass, `Dubon2026.SourceSmoothedUnfolding.projectiveMass_le_half, `Dubon2026.SourceSmoothedUnfolding.projectiveMass_eq_half, `Dubon2026.SourceSmoothedUnfolding.tendsto_projectiveMass, `Dubon2026.SourceSmoothedUnfolding.measurable_projectiveMass, `Dubon2026.SourceSmoothedUnfolding.integral_mul_eq_periodized, `Dubon2026.SourceSmoothedUnfolding.closedDomain_ae_eq, `Dubon2026.SourceSmoothedUnfolding.center_le, `Dubon2026.SourceSmoothedUnfolding.integral_smooth_eq_mass, `Dubon2026.SourceSmoothedUnfolding.tendsto_integral_mass, `Dubon2026.SourceSmoothedUnfolding.tendsto_integral_smooth, `Dubon2026.SourceSmoothedUnfolding.smooth_eq_zero_of_im_nonpos, `Dubon2026.SourceSmoothedUnfolding.integral_plane_eq_integral_uhp, `Dubon2026.SourceSmoothedUnfolding.tendsto_integral_plane, `Dubon2026.sourceCusp_petersson_integrable_closedDomain, `Dubon2026.sourceGamma1_petersson_smoothed_limit]
  let constructions : Array Name := #[`Dubon2026.SourceSmoothedUnfolding.projection, `Dubon2026.SourceSmoothedUnfolding.projectiveMass, `Dubon2026.SourceSmoothedUnfolding.closedDomain]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Petersson Smooth Limit postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Petersson Smooth Limit theorems discovered"
  logInfo m!"SOURCE GAMMA ONE PETERSSON SMOOTH LIMIT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
