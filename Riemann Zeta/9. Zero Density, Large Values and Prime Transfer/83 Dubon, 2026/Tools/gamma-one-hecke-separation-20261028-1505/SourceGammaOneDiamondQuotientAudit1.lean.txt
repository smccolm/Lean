private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceTateDualArithmeticDiamond, `SourceGammaOneArithmeticDiamondPolynomial, `SourceGammaOneGeometricDiamond, `SourceGammaOneGaloisDiamondPolynomial, `SourceGammaOneTateDiamondRelations, `SourceGammaOneDiamondQuotientGalois, `SourceGammaOneDiamondQuotientPolynomial, `SourceGammaOneDiamondQuotientHecke]
  let theorems : Array Name := #[`Dubon2026.sourceTateDual_arithmetic_diamond_quadratic, `Dubon2026.SourceFunctionField.GammaOne.binaryTateArithmeticFrobeniusOperator_diamond_quadratic_of_picard, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_arithmetic_frobenius_diamond_quadratic_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_galois_frobenius_diamond_quadratic_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.binaryTateDiamondRelations_generator, `Dubon2026.SourceFunctionField.GammaOne.binaryTateDiamondOperator_galois_commute, `Dubon2026.SourceFunctionField.GammaOne.binaryTateDiamondRelations_galois, `Dubon2026.SourceFunctionField.GammaOne.binaryTateDiamondRelations_mk_diamond, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientGaloisRep_mk, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotient_finite, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientGaloisRep_continuous, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_diamond_quotient_frobenius_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_diamond_relations_hecke_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_diamond_quotient_hecke_frobenius_of_degree_genus_eq]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.binaryTateDiamondOperator, `Dubon2026.SourceFunctionField.GammaOne.binaryTateDiamondRelations, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientGaloisRep]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIAMOND QUOTIENT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
