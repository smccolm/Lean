private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneIntegralRatioResidue, `SourceGammaOneIntegralJRatio, `SourceGammaOneResidueRatioDegree, `SourceGammaOneIntegralRatioDegreeAssembly]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.intFormRatiosC_subset_residueImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.qExpFunctionFieldC_le_residueImage, `Dubon2026.SourceFunctionField.GammaOne.jqModC_mem_integralRatioField, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residue_coordinate_degree_eq_relfinrank, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.integral_ratio_residue_coordinate_degree_le, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.integral_ratio_coordinate_degree_le_original, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_degree_and_residueField_of_integral_ratio_index]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE INTEGRAL RATIO AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
