private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceTateDualArithmeticDiamond, `Dubon2026.SourceGammaOneArithmeticDiamondPolynomial, `Dubon2026.SourceGammaOneGeometricDiamond, `Dubon2026.SourceGammaOneGaloisDiamondPolynomial, `Dubon2026.SourceGammaOneTateDiamondRelations, `Dubon2026.SourceGammaOneDiamondQuotientGalois, `Dubon2026.SourceGammaOneDiamondQuotientPolynomial, `Dubon2026.SourceGammaOneDiamondQuotientHecke, `Dubon2026.SourceGammaOneTateTransfer, `Dubon2026.SourceGammaOneDiamondQuotientTransfer, `Dubon2026.SourceGammaOneDiamondQuotientNonzero, `Dubon2026.SourceGammaOneDiamondQuotientAssembly, `Dubon2026.SourceRationalLocalFrobenius, `Dubon2026.SourceRationalLocalClosureEmbedding, `Dubon2026.SourceGammaOneDiamondQuotientLocal, `Dubon2026.SourceGammaOneDiamondQuotientHeckeFamily, `Dubon2026.SourceGammaOneDiamondQuotientWords, `Dubon2026.SourceGammaOneDiamondQuotientWordKernel, `Dubon2026.SourceGammaOneDiamondQuotientCharacter, `Dubon2026.SourceGammaOneDiamondQuotientCharacterAssembly, `Dubon2026.SourceGammaOneDiamondQuotientGaloisHecke, `Dubon2026.SourceGammaOneDiamondQuotientChosenFrobenius, `Dubon2026.SourceGammaOneIntegralRatioResidue, `Dubon2026.SourceGammaOneIntegralJRatio, `Dubon2026.SourceGammaOneResidueRatioDegree, `Dubon2026.SourceGammaOneIntegralRatioDegreeAssembly, `Dubon2026.SourceIntegralEisensteinRows, `Dubon2026.SourceIntegralEisensteinHolomorphic, `Dubon2026.SourceIntegralEisensteinModular, `Dubon2026.SourceGammaOneEisensteinSum, `Dubon2026.SourceGammaOneEisensteinLattice, `Dubon2026.SourceGammaOneEisensteinPositive, `Dubon2026.SourceGammaOneEisensteinCoefficients, `Dubon2026.SourceGammaOneIntegralEisenstein, `Dubon2026.SourceGammaOneEisensteinSeparation]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
