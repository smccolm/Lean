private theorem quotient_eq_one {R W G : Type*} [CommRing R] [AddCommGroup W]
    [Module R W] [Monoid G] (ρ : Representation R G W) (K : Submodule R W)
    (hK : ∀ g, K ≤ K.comap (ρ g)) (g : G) (hg : ρ g = 1) :
    ρ.quotient K hK g = 1 := by
  apply Submodule.linearMap_qext
  apply LinearMap.ext
  intro x
  change K.mkQ (ρ g x) = K.mkQ x
  rw [hg]
  rfl

private theorem continuous_quotient_discrete {R W G : Type*} [CommRing R]
    [AddCommGroup W] [Module R W] [Group G] [TopologicalSpace G] [IsTopologicalGroup G]
    (ρ : Representation R G W) (K : Submodule R W) (hK : ∀ g, K ≤ K.comap (ρ g))
    (hρ : ∀ᶠ g in 𝓝 (1 : G), ρ g = 1) :
    @Continuous _ _ _ ⊥ (ρ.quotient K hK).toFun := by
  letI : TopologicalSpace (Module.End R (W ⧸ K)) := ⊥
  letI : DiscreteTopology (Module.End R (W ⧸ K)) := ⟨rfl⟩
  apply continuous_of_tendsto_nhds_one
  apply tendsto_const_nhds.congr'
  exact hρ.mono (fun g hg => (quotient_eq_one ρ K hK g hg).symm)

variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "T" => TateModule p (Pic0 Qbar (functionFieldOver (Gamma1 M) Qbar))

private local instance : Algebra ℤ_[p] (ZMod N) := (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V := Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

/-- The actual Galois action on the quotient imposing all original diamond relations. -/
def binaryDiamondQuotientGaloisRep : (Qbar ≃ₐ[ℚ] Qbar) →*
    Module.End (ZMod N) ((T →ₗ[ℤ_[p]] V) ⧸ binaryTateDiamondRelations Q p n w) :=
  (binaryArithmeticTateGaloisRep Q p n w).quotient
    (binaryTateDiamondRelations Q p n w) (binaryTateDiamondRelations_galois Q p n w)

theorem binaryDiamondQuotientGaloisRep_mk (σ : Qbar ≃ₐ[ℚ] Qbar) (φ : T →ₗ[ℤ_[p]] V) :
    binaryDiamondQuotientGaloisRep Q p n w σ ((binaryTateDiamondRelations Q p n w).mkQ φ) =
      (binaryTateDiamondRelations Q p n w).mkQ (binaryArithmeticTateGaloisRep Q p n w σ φ) := rfl

theorem binaryDiamondQuotient_finite :
    Finite ((T →ₗ[ℤ_[p]] V) ⧸ binaryTateDiamondRelations Q p n w) := by
  letI := binaryArithmeticTateDual_finite Q p n w
  exact Finite.of_surjective (binaryTateDiamondRelations Q p n w).mkQ
    (binaryTateDiamondRelations Q p n w).mkQ_surjective

theorem binaryDiamondQuotientGaloisRep_continuous :
    @Continuous _ _ _ ⊥ (binaryDiamondQuotientGaloisRep Q p n w).toFun :=
  continuous_quotient_discrete (binaryArithmeticTateGaloisRep Q p n w)
    (binaryTateDiamondRelations Q p n w) (binaryTateDiamondRelations_galois Q p n w)
    (binaryArithmeticTateGaloisRep_eventually_one Q p n w)

end
