/-! DUB-17/18 -> original all-weight association -> finite Hecke-character
quotient. Define the actual combined diamond operators and their
diamond-minus-one relation submodule. The proved original Galois/diamond
commutation makes this submodule Galois stable. Nonzero character
realization and geometric degree/genus preservation are separate obligations. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open SourceCurve CongruenceSubgroup
private theorem relations_stable {R W I : Type*} [CommRing R]
    [AddCommGroup W] [Module R W] (D : I → Module.End R W)
    (F : Module.End R W) (hc : ∀ i, Commute F (D i)) :
    Submodule.span R (Set.range fun z : I × W => D z.1 z.2 - z.2) ≤
      (Submodule.span R (Set.range fun z : I × W => D z.1 z.2 - z.2)).comap F := by
  apply Submodule.span_le.mpr
  rintro _ ⟨⟨i, x⟩, rfl⟩
  change F (D i x - x) ∈ Submodule.span R (Set.range fun z : I × W => D z.1 z.2 - z.2)
  rw [map_sub, LinearMap.congr_fun (hc i).eq x]
  exact Submodule.subset_span (Set.mem_range_self (i, F x))

variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "T" => TateModule p (Pic0 Qbar (functionFieldOver (Gamma1 M) Qbar))

private local instance : Algebra ℤ_[p] (ZMod N) := (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V := Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

/-- The original coefficient diamond combined with the arithmetic Picard diamond. -/
def binaryTateDiamondOperator (g : Gamma0 M) : Module.End (ZMod N) (T →ₗ[ℤ_[p]] V) :=
  let b := sourceGammaOneBinaryDiamondCoefficient Q N w g
  { toFun := fun φ => (b.restrictScalars ℤ_[p]).comp (φ.comp (arithmeticDiamondTate M p g))
    map_add' := fun φ ψ => LinearMap.ext fun x => b.map_add (φ (arithmeticDiamondTate M p g x)) (ψ (arithmeticDiamondTate M p g x))
    map_smul' := fun c φ => LinearMap.ext fun x => b.map_smul c (φ (arithmeticDiamondTate M p g x)) }

/-- Relations imposing value one for every original combined diamond. -/
def binaryTateDiamondRelations : Submodule (ZMod N) (T →ₗ[ℤ_[p]] V) :=
  Submodule.span (ZMod N) (Set.range fun z : Gamma0 M × (T →ₗ[ℤ_[p]] V) =>
    binaryTateDiamondOperator Q p n w z.1 z.2 - z.2)

theorem binaryTateDiamondRelations_generator (g : Gamma0 M) (φ : T →ₗ[ℤ_[p]] V) :
    binaryTateDiamondOperator Q p n w g φ - φ ∈ binaryTateDiamondRelations Q p n w :=
  Submodule.subset_span (Set.mem_range_self (g, φ))

theorem binaryTateDiamondOperator_galois_commute (σ : Qbar ≃ₐ[ℚ] Qbar) (g : Gamma0 M) :
    Commute (binaryArithmeticTateGaloisRep Q p n w σ) (binaryTateDiamondOperator Q p n w g) := by
  apply LinearMap.ext
  intro φ
  exact binaryArithmeticTateGaloisRep_diamond Q p n w σ g φ

theorem binaryTateDiamondRelations_galois (σ : Qbar ≃ₐ[ℚ] Qbar) :
    binaryTateDiamondRelations Q p n w ≤
      (binaryTateDiamondRelations Q p n w).comap (binaryArithmeticTateGaloisRep Q p n w σ) := by
  exact relations_stable (binaryTateDiamondOperator Q p n w)
    (binaryArithmeticTateGaloisRep Q p n w σ)
    (binaryTateDiamondOperator_galois_commute Q p n w σ)

theorem binaryTateDiamondRelations_mk_diamond (g : Gamma0 M) (φ : T →ₗ[ℤ_[p]] V) :
    (binaryTateDiamondRelations Q p n w).mkQ (binaryTateDiamondOperator Q p n w g φ) =
      (binaryTateDiamondRelations Q p n w).mkQ φ :=
  (Submodule.Quotient.eq _).mpr (binaryTateDiamondRelations_generator Q p n w g φ)

end
