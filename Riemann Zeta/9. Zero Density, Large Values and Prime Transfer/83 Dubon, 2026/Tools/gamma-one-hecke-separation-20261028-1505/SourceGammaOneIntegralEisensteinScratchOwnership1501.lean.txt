import SourceIntegralEisensteinRows
import SourceIntegralEisensteinHolomorphic
import SourceIntegralEisensteinModular
import SourceGammaOneEisensteinSum
import SourceGammaOneEisensteinLattice
import SourceGammaOneEisensteinPositive
import SourceGammaOneEisensteinCoefficients
import SourceGammaOneIntegralEisenstein
import SourceGammaOneEisensteinSeparation
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceIntegralEisensteinRows, `SourceIntegralEisensteinHolomorphic, `SourceIntegralEisensteinModular, `SourceGammaOneEisensteinSum, `SourceGammaOneEisensteinLattice, `SourceGammaOneEisensteinPositive, `SourceGammaOneEisensteinCoefficients, `SourceGammaOneIntegralEisenstein, `SourceGammaOneEisensteinSeparation]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
