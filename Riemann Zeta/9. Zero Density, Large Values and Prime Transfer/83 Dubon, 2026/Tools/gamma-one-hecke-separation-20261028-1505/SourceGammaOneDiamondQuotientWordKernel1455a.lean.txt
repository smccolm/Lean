theorem binaryDiamondQuotientOriginalTransfer_word_kernel
    (hlarge : 3 < p)
    (Hq : sourceBinaryHeckePrime (Q * N) → Module.End (ZMod N) ((T →ₗ[ℤ_[p]] V) ⧸ binaryTateDiamondRelations Q p n w))
    (hH : ∀ ell, (Hq ell).comp (binaryTateDiamondRelations Q p n w).mkQ =
  (binaryTateDiamondRelations Q p n w).mkQ.comp (binaryTatePrimeOperators Q p n w ell))

    (hlift : sourceGammaOneIntegralParabolicTransfer Q N w)
    (a : FreeAlgebra ℤ (sourceBinaryHeckePrime (Q * N)))
    (ha : FreeAlgebra.lift ℤ Hq a = 0)
    (x : H1 (sourceBinaryCoefficientRep ℤ Q w))
    (hx : x ∈ sourceParabolicH1 (Gamma0 Q) (sourceBinaryCoefficientRep ℤ Q w)) :
    sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w
      ((sourceGammaOneOriginalHeckeWord Q N w a).val x) = 0 := by
  have hs := binaryDiamondQuotientOriginalTransfer_parabolic_preimage Q p n w hlarge
  have h1 := hs hlift
  have h2 := h1 x
  obtain ⟨z, hz⟩ := h2 hx
  let β := binaryDiamondQuotientOriginalTransfer Q p n w hlarge
  let u := fun ell => sourceBinaryH1PrimeOperators (ZMod N) Q w
    (sourceGammaOneHeckePrimeForget Q N ell)
  have hw := binaryDiamondQuotientOriginalTransfer_word Q p n w hlarge Hq hH a z
  have hzero := LinearMap.congr_fun ha z
  have hzword := hw.symm.trans ((congrArg β hzero).trans (map_zero β))
  have hc := sourceGammaOneOriginalHeckeWord_change Q N w a x
  exact hc.trans ((congrArg (FreeAlgebra.lift ℤ u a) hz.symm).trans hzword)

end
