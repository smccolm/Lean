import SourceRationalLocalFrobenius
import SourceRationalLocalClosureEmbedding
import SourceGammaOneDiamondQuotientLocal
import SourceGammaOneDiamondQuotientHeckeFamily
import SourceGammaOneDiamondQuotientWords
import SourceGammaOneDiamondQuotientWordKernel
import SourceGammaOneDiamondQuotientCharacter
import SourceGammaOneDiamondQuotientCharacterAssembly
import SourceGammaOneDiamondQuotientGaloisHecke
import SourceGammaOneDiamondQuotientChosenFrobenius
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceRationalLocalFrobenius, `SourceRationalLocalClosureEmbedding, `SourceGammaOneDiamondQuotientLocal, `SourceGammaOneDiamondQuotientHeckeFamily, `SourceGammaOneDiamondQuotientWords, `SourceGammaOneDiamondQuotientWordKernel, `SourceGammaOneDiamondQuotientCharacter, `SourceGammaOneDiamondQuotientCharacterAssembly, `SourceGammaOneDiamondQuotientGaloisHecke, `SourceGammaOneDiamondQuotientChosenFrobenius]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
