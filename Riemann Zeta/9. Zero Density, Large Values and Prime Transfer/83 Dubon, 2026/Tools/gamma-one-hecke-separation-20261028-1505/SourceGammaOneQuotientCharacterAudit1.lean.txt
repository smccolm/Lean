private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRationalLocalFrobenius, `SourceRationalLocalClosureEmbedding, `SourceGammaOneDiamondQuotientLocal, `SourceGammaOneDiamondQuotientHeckeFamily, `SourceGammaOneDiamondQuotientWords, `SourceGammaOneDiamondQuotientWordKernel, `SourceGammaOneDiamondQuotientCharacter, `SourceGammaOneDiamondQuotientCharacterAssembly, `SourceGammaOneDiamondQuotientGaloisHecke, `SourceGammaOneDiamondQuotientChosenFrobenius]
  let theorems : Array Name := #[`Dubon2026.sourceRationalLocalValuation_exists_frobenius, `Dubon2026.sourceRationalLocalClosure_nonempty_embedding, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_diamond_relations_hecke_at_local_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_diamond_relations_hecke_family_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_diamond_quotient_hecke_family_of_degree_genus_eq, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientOriginalTransfer_word, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientOriginalTransfer_word_kernel, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotient_original_character_of_transfer_away, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_diamond_quotient_original_character_of_degree_genus_transfer, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientGaloisRep_prime_of_intertwining, `Dubon2026.SourceFunctionField.GammaOne.binaryDiamondQuotientGaloisRep_word_of_intertwining, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_diamond_quotient_hecke_frobenius_of_intertwining]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.sourceQuotientOperatorCharacterData]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE QUOTIENT CHARACTER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
