private theorem quotient_polynomial {R W G : Type*} [CommRing R]
    [AddCommGroup W] [Module R W] [Monoid G] (ρ : Representation R G W)
    (K : Submodule R W) (hK : ∀ g, K ≤ K.comap (ρ g))
    (H D : Module.End R W) (g : G) (a : R)
    (hp : ∀ x, ρ g (H x) = ρ g (ρ g x) + a • D x)
    (hd : ∀ x, K.mkQ (D x) = K.mkQ x) (x : W) :
    ρ.quotient K hK g (K.mkQ (H x)) =
      ρ.quotient K hK g (ρ.quotient K hK g (K.mkQ x)) + a • K.mkQ x := by
  change K.mkQ (ρ g (H x)) = K.mkQ (ρ g (ρ g x)) + a • K.mkQ x
  rw [hp, map_add, map_smul, hd]

variable (Q p n w : ℕ) [NeZero Q] [Fact p.Prime]
local notation "N" => p ^ (n + 1)
local notation "M" => Q * N ^ 2
local notation "Qbar" => AlgebraicClosure ℚ
local notation "Γ" => Gamma1 M
local notation "V" => sourceBinaryCoefficientRep (ZMod N) Q w
local notation "T" => TateModule p (Pic0 Qbar (functionFieldOver Γ Qbar))
variable (A : ValuationSubring Qbar)
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Γ
attribute [local instance] specializationResidueAlgebra

private local instance : Algebra ℤ_[p] (ZMod N) := (PadicInt.toZModPow (n + 1)).toAlgebra
private local instance : Module ℤ_[p] V := Module.restrictScalars ℤ_[p] (ZMod N) V
private local instance : IsScalarTower ℤ_[p] (ZMod N) V :=
  IsScalarTower.restrictScalars ℤ_[p] (ZMod N) V

theorem gammaOne_diamond_quotient_frobenius_of_degree_genus_eq
    (ell : sourceBinaryHeckePrime (Q * N)) [CharP kk ell.val]
    (hdeg : Module.finrank Qbar⟮jCoordinateOver Γ Qbar⟯ (functionFieldOver Γ Qbar) =
      Module.finrank kk⟮residueJCoordinate A Γ⟯ Fb)
    (hgen : genusFF kk Fb = genusFF Qbar (functionFieldOver Γ Qbar))
    (d : A.decompositionSubgroup ℚ) (hd : ∀ a : kk, d • a = a ^ ell.val)
    (hp : (p : kk) ≠ 0) :
    ∀ φ : T →ₗ[ℤ_[p]] V,
      binaryDiamondQuotientGaloisRep Q p n w d.val
          ((binaryTateDiamondRelations Q p n w).mkQ (binaryTatePrimeOperators Q p n w ell φ)) =
        binaryDiamondQuotientGaloisRep Q p n w d.val
          (binaryDiamondQuotientGaloisRep Q p n w d.val
            ((binaryTateDiamondRelations Q p n w).mkQ φ)) +
          (ell.val : ZMod N) ^ (w + 1) • (binaryTateDiamondRelations Q p n w).mkQ φ := by
  have hs := exists_gammaOne_galois_frobenius_diamond_quadratic_of_degree_genus_eq Q p n w A ell
  have h1 := hs hdeg
  have h2 := h1 hgen
  have h3 := h2 d hd hp
  obtain ⟨g, _, hq⟩ := h3
  exact quotient_polynomial (binaryArithmeticTateGaloisRep Q p n w)
    (binaryTateDiamondRelations Q p n w) (binaryTateDiamondRelations_galois Q p n w)
    (binaryTatePrimeOperators Q p n w ell) (binaryTateDiamondOperator Q p n w g)
    d.val ((ell.val : ZMod N) ^ (w + 1)) hq
    (binaryTateDiamondRelations_mk_diamond Q p n w g)

end
