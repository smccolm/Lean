private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralEisensteinRows, `SourceIntegralEisensteinHolomorphic, `SourceIntegralEisensteinModular, `SourceGammaOneEisensteinSum, `SourceGammaOneEisensteinLattice, `SourceGammaOneEisensteinPositive, `SourceGammaOneEisensteinCoefficients, `SourceGammaOneIntegralEisenstein, `SourceGammaOneEisensteinSeparation]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.series_slash, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.series_locallyUniform, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.slashInvariant_mdifferentiable, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.slashInvariant_bounded, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.modularForm_coe, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.rowSum_slash, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.gammaOneRowSum_coe, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.rowSum_eq_integer_lattice, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.kappa_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.rowSum_eq_positive_series, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.positiveCoefficient_summable, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.positive_series_eq_coefficients, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.positiveCoefficient_zero, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.positiveCoefficient_one, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.normalizedForm_coe, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.normalizedForm_slash, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.coefficient_zero, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.coefficient_one, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.normalizedForm_hasSum, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.normalizedForm_integral, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.ratio_mem_original_field, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.coefficient_identity_class, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.coefficient_other_class, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.ratio_separates_signed_identity]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.rows, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.series, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.slashInvariant, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.modularForm, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.rowSum, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.gammaOneRowSum, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.kappa, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.positiveCoefficient, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.coefficient, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.integralSeries, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.normalizedForm, `Dubon2026.SourceFunctionField.GammaOne.IntegralEisenstein.ratio]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE INTEGRAL EISENSTEIN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
