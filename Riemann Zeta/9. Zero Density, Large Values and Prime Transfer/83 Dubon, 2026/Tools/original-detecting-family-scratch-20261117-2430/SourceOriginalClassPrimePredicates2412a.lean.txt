regular detecting primes for each original H1 class. Retain the literal

open scoped NumberField
attribute [local instance] rationalArithmeticOriginalFiniteGalois

/-- A regular congruence prime for the original coefficient field and
representation. No splitting over the coefficient field is asserted. -/
def RationalArithmeticOriginalRegularAuxiliaryPrime
    {K ι : Type} [Field K] [TopologicalSpace K] [Fintype ι] [DecidableEq ι]
    (a : 𝓞 ℚ) (p level : ℕ)
    (R : rationalArithmeticGaloisGroup a →* Matrix.GeneralLinearGroup ι K)
    (q : Nat.Primes) : Prop :=
  ¬ (q.val : 𝓞 ℚ) ∣ a ∧ q.val ≡ 1 [MOD p ^ (level + 1)] ∧
  (let U := rationalUnramifiedExtension a
   letI : Algebra ℚ U := U.algebra'
   letI : IsGalois ℚ U := originalUnramifiedGaloisUnion_isGalois a
   ∃ D : FiniteGaloisIntermediateField ℚ U,
     ∃ ψ : Gal(D/ℚ) →* Matrix.GeneralLinearGroup ι K,
       (∀ w : Gal(U/ℚ), ψ (AlgEquiv.restrictNormalHom D w) = R w) ∧
       (letI : NumberField D := NumberField.of_module_finite ℚ D
        Chebotarev.UnramifiedIn ℚ D (Ideal.span {(q.val : 𝓞 ℚ)}) ∧
        ∃ γ : Gal(rationalPrimeLocalClosure q.val q.property /
            rationalPrimeLocalField q.val q.property),
          Chebotarev.frobeniusClass ℚ D (Ideal.span {(q.val : 𝓞 ℚ)}) =
            ConjClasses.mk (AlgEquiv.restrictNormalHom D
              (rationalPrimeArithmeticRestriction a q.val q.property γ)) ∧
          (R (rationalPrimeArithmeticRestriction a q.val q.property γ)).val.charpoly.Separable))

/-- Exact pointwise regular-prime conclusion for an original class. -/
def HasArithmeticRegularClassPrimes
    (a : 𝓞 ℚ) (p level : ℕ)
    {K ι : Type} [Field K] [TopologicalSpace K] [Fintype ι] [DecidableEq ι]
    (A : Rep.{0} K (rationalArithmeticGaloisGroup a))
    [TopologicalSpace A] [DiscreteTopology A] [ContinuousSMul K A]
    (hA : Continuous (fun z : rationalArithmeticGaloisGroup a × A => A.ρ z.1 z.2))
    (R : rationalArithmeticGaloisGroup a →* Matrix.GeneralLinearGroup ι K)
    (x : ContinuousRepresentationH1 A hA) : Prop :=
  ∀ S : Finset ℕ, ∃ q : Nat.Primes, q.val ∉ S ∧
    RationalArithmeticOriginalRegularAuxiliaryPrime a p level R q ∧
    rationalArithmeticH1PrimeRestriction a A hA q.val q.property x ≠ 0

/-- Every nonzero original cocycle class has the native regular-prime output. -/
def HasArithmeticRegularCocycleDetection
    (a : 𝓞 ℚ) (p level : ℕ)
    {K ι : Type} [Field K] [TopologicalSpace K] [Fintype ι] [DecidableEq ι]
    (A : Rep.{0} K (rationalArithmeticGaloisGroup a))
    [TopologicalSpace A] [DiscreteTopology A] [ContinuousSMul K A]
    (hA : Continuous (fun z : rationalArithmeticGaloisGroup a × A => A.ρ z.1 z.2))
    (R : rationalArithmeticGaloisGroup a →* Matrix.GeneralLinearGroup ι K) : Prop :=
  ∀ c : continuousRepresentationCocyclesOne A,
    (Submodule.Quotient.mk c : ContinuousRepresentationH1 A hA) ≠ 0 →
      HasArithmeticRegularDetectingPrimes a p level A hA R c

end
end Dubon2026
