import SourceFrickeStableAlgebraV2
import SourceFrickeStableGroupV2
import SourceFrickeStableFieldV2
import SourceFrickeStableEmbeddingV2
import SourceFrickeStableFractionV2
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceFrickeStableAlgebraV2, `SourceFrickeStableGroupV2, `SourceFrickeStableFieldV2, `SourceFrickeStableEmbeddingV2, `SourceFrickeStableFractionV2]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
