import SourceLaurentAnalyticPolynomial
import SourceLevelOneAnalyticJPolynomial
import SourceInvariantAnalyticJPolynomial
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceLaurentAnalyticPolynomial, `SourceLevelOneAnalyticJPolynomial, `SourceInvariantAnalyticJPolynomial]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
