private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFrickeStableAlgebra, `SourceFrickeStableGroup, `SourceFrickeStableField, `SourceFrickeStableEmbedding, `SourceFrickeStableFraction]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.LevelGens.ring_le_HolFn, `Dubon2026.SourceFricke.LevelGens.levelRing_le_ring, `Dubon2026.SourceFricke.LevelGens.ring_isDomain, `Dubon2026.SourceFricke.LevelGens.smul_mem_ring, `Dubon2026.SourceFricke.LevelGens.ring_coe_smul, `Dubon2026.SourceFricke.LevelGens.ring_smulCommClass, `Dubon2026.SourceFricke.LevelGens.smul_eq_self_of_pm, `Dubon2026.SourceFricke.LevelGens.levelFixer_le_ker, `Dubon2026.SourceFricke.LevelGens.ker_le_levelFixer, `Dubon2026.SourceFricke.LevelGens.grp_mk_smul, `Dubon2026.SourceFricke.LevelGens.grp_smulCommClass, `Dubon2026.SourceFricke.LevelGens.grp_faithful, `Dubon2026.SourceFricke.LevelGens.grp_smulDistribClass, `Dubon2026.SourceFricke.LevelGens.grp_smul_algebraMap, `Dubon2026.SourceFricke.LevelGens.grpField_faithful, `Dubon2026.SourceFricke.LevelGens.grpField_smulCommClass, `Dubon2026.SourceFricke.LevelGens.finrank_field_fixed, `Dubon2026.SourceFricke.LevelGens.coe_aeval_jA, `Dubon2026.SourceFricke.LevelGens.ring_fixed_eq_aeval_j, `Dubon2026.SourceFricke.LevelGens.smul_jK, `Dubon2026.SourceFricke.LevelGens.fixedPoints_field_eq_adjoin_j, `Dubon2026.SourceFricke.LevelGens.finrank_ratJ, `Dubon2026.SourceFricke.LevelGens.incl_injective, `Dubon2026.SourceFricke.LevelGens.coe_incl, `Dubon2026.SourceFricke.LevelGens.toRingField_injective, `Dubon2026.SourceFricke.LevelGens.toField_algebraMap, `Dubon2026.SourceFricke.LevelGens.toField_jK, `Dubon2026.SourceFricke.LevelGens.scalarTowerField, `Dubon2026.SourceFricke.LevelGens.scalarTowerRatJ, `Dubon2026.SourceFricke.LevelGens.ratJ_map_toField, `Dubon2026.SourceFricke.LevelGens.finrank_ratJN_field, `Dubon2026.SourceFricke.LevelGens.finrank_levelField_field, `Dubon2026.SourceFricke.LevelGens.toField_surjective, `Dubon2026.SourceFricke.LevelGens.exists_frac, `Dubon2026.SourceFricke.exists_levelRing_frac]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.LevelGens, `Dubon2026.SourceFricke.LevelGens.std, `Dubon2026.SourceFricke.LevelGens.ring, `Dubon2026.SourceFricke.LevelGens.ringAction, `Dubon2026.SourceFricke.LevelGens.grpRingAction, `Dubon2026.SourceFricke.LevelGens.field, `Dubon2026.SourceFricke.LevelGens.fieldRingSMul, `Dubon2026.SourceFricke.LevelGens.grpFieldAction, `Dubon2026.SourceFricke.LevelGens.fieldActionSMul, `Dubon2026.SourceFricke.LevelGens.fieldMulAction, `Dubon2026.SourceFricke.LevelGens.ringSelfDistribMulAction, `Dubon2026.SourceFricke.LevelGens.jA, `Dubon2026.SourceFricke.LevelGens.jK, `Dubon2026.SourceFricke.LevelGens.ratJ, `Dubon2026.SourceFricke.LevelGens.incl, `Dubon2026.SourceFricke.LevelGens.toField, `Dubon2026.SourceFricke.LevelGens.algebraField, `Dubon2026.SourceFricke.LevelGens.moduleField, `Dubon2026.SourceFricke.LevelGens.smulField, `Dubon2026.SourceFricke.LevelGens.algebraRatJ, `Dubon2026.SourceFricke.LevelGens.moduleRatJ, `Dubon2026.SourceFricke.LevelGens.smulRatJ, `Dubon2026.SourceFricke.LevelGens.ratJEquiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE FRICKE STABLE FAMILY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
