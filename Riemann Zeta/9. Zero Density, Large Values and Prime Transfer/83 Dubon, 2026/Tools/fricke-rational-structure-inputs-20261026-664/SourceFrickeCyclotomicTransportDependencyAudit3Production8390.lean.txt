private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFricke.wpQSeriesK_map_inclusion, `Dubon2026.SourceFricke.wpQSeriesK_map_cyclotomic]
  let localModules : Array Name := #[`Dubon2026.SourceFrickeCyclotomicCoefficients, `Dubon2026.SourceFrickeCyclotomicTransport, `Dubon2026.SourceFrickeTorsionConjugation, `Dubon2026.SourceFrickeCyclotomicField, `Dubon2026.SourceWeierstrassNormalizedTorsion, `Dubon2026.SourceWeierstrassTorsionCuspBound, `Dubon2026.SourceWeierstrassTorsionFourierCoefficients, `Dubon2026.SourceWeierstrassTorsionFourierSum, `Dubon2026.SourceWeierstrassTorsionFourierExpansion, `Dubon2026.SourceFrickeCuspInputs, `Dubon2026.SourceLipschitzTwo, `Dubon2026.SourceLipschitzTwoRealBoundary, `Dubon2026.SourceWeierstrassRowSeries, `Dubon2026.SourceWeierstrassLatticeRows, `Dubon2026.SourceWeierstrassTorsionExpansion, `Dubon2026.SourceWeierstrassTorsionTail, `Dubon2026.SourceWeierstrassTorsionHolomorphy, `Dubon2026.SourceFrickeHolomorphy, `Dubon2026.SourceFrickePeriodHomothetyV2, `Dubon2026.SourceFrickeModularLatticeV2, `Dubon2026.SourceFrickeTorsionTransformationV2, `Dubon2026.SourceFrickeFunctionTransformationV2]
  let required : Array Name := #[`Dubon2026.SourceFricke.isPrimitiveRoot_zetaN, `Dubon2026.SourceFricke.zetaN_pow_N, `Dubon2026.SourceFricke.pow_mul_eq_pow_mod_mul, `Dubon2026.SourceFricke.ringHom_wpMonCoeffK, `Dubon2026.SourceFricke.ringHom_wpQCoeffK, `Dubon2026.SourceFricke.wpQSeriesK_map_inclusion, `Dubon2026.SourceFricke.wpQSeriesK_map_cyclotomic, `Dubon2026.SourceFricke.zetaK, `Dubon2026.SourceFricke.wpMonCoeffK, `Dubon2026.SourceFricke.wpQCoeffK, `Dubon2026.SourceFricke.wpQSeriesK, `Dubon2026.SourceFricke.finite_wpMonExp_fiber, `Dubon2026.SourceFricke.wpMonCoeff_mem_kN, `Dubon2026.SourceFricke.wpQCoeff_mem_kN, `Dubon2026.SourceFricke.qExpansion_wpNorm_coeff, `Complex.isPrimitiveRoot_exp]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE FRICKE CYCLOTOMIC TRANSPORT DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the actual torsion Fourier coefficient-embedding consumers. Only declaration values are traversed. Native root-of-unity powers and finite exponent fibers transport the actual Fourier expansion to the transformed torsion coordinate. Fricke product transport, separation, rational spanning, differential realization and arithmetic comparison remain; 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
