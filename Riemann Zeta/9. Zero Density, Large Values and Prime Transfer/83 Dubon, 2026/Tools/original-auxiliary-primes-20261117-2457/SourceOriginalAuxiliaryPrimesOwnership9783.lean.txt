open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceOriginalArithmeticRootKernel, `Dubon2026.SourceSubfieldProjectorSpan, `Dubon2026.SourceSubfieldRegularTrace, `Dubon2026.SourceSubfieldRegularKernel, `Dubon2026.SourceTraceZeroCoefficientEmbedding, `Dubon2026.SourceTraceZeroConjugacy, `Dubon2026.SourceOriginalExtendedTraceTest, `Dubon2026.SourceFiniteLinearOriginalRegularDetection, `Dubon2026.SourceOriginalArithmeticRegularCocycle, `Dubon2026.SourceOriginalRegularPrimePredicates, `Dubon2026.SourceOriginalRegularPrimeAssembly, `Dubon2026.SourceOriginalArithmeticRegularPrimes, `Dubon2026.SourceOriginalArithmeticRegularPrimeProperty, `Dubon2026.SourceGammaOneOriginalRegularPrimes, `Dubon2026.SourceOriginalCoefficientH1Finite, `Dubon2026.SourceOriginalClassPrimePredicates, `Dubon2026.SourceOriginalCocycleClassPrimes, `Dubon2026.SourceOriginalClassRegularPrimes, `Dubon2026.SourceOriginalDetectingFamilyProperty, `Dubon2026.SourcePrimeIndexedDetectingFamily, `Dubon2026.SourceOriginalFiniteDetectingFamily, `Dubon2026.SourceOriginalArithmeticFamilyProperty, `Dubon2026.SourceGammaOneOriginalDetectingFamilies, `Dubon2026.SourceOriginalLocalCoefficientH1Finite, `Dubon2026.SourceOriginalLocalUnramifiedDimension, `Dubon2026.SourceOriginalPrimeUnramifiedDimension, `Dubon2026.SourceOriginalUnramifiedAdjointProperty, `Dubon2026.SourceGammaOneOriginalUnramifiedDimensions]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
