private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceSupportedCharacterFiber, `SourceSupportedCharacterDivisibility, `SourceParabolicHeckeSpan, `SourceParabolicHeckeCongruence, `SourceParabolicAdicOperatorKernel]
  let theorems : Array Name := #[`Dubon2026.sourceModule_nontrivial_characterFiber_of_annihilator, `Dubon2026.sourceSupportedCharacter_div_isIntegral, `Dubon2026.sourceParabolicHeckeSpan_finite, `Dubon2026.sourcePrimitiveParabolicHeckeSpan_annihilator, `Dubon2026.sourceBinaryH1Change_heckeAlgebra_zero, `Dubon2026.sourceParabolicHeckeSpan_operator_congruence, `Dubon2026.sourcePrimitiveParabolicHecke_congruence_integral_away, `Dubon2026.sourcePrimitiveParabolicH1AdicQuotientCharacter_zero_away]
  let constructions : Array Name := #[`Dubon2026.sourceParabolicHeckeSpan]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Supported Parabolic Congruence postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Supported Parabolic Congruence theorems discovered"
  logInfo m!"SOURCE SUPPORTED PARABOLIC CONGRUENCE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
