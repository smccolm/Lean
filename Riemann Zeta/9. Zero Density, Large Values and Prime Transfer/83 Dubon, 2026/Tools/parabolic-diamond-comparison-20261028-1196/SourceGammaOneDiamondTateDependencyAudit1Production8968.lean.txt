private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.periodQuotientDiamond_eq_quotEnd, `Dubon2026.SourceFunctionField.GammaOne.periodTateTensor_diamond, `Dubon2026.SourceFunctionField.GammaOne.periodQuotientDiamond]
  let localModules : Array Name := #[`Dubon2026.SourceGammaOneDiamondPeriodTate, `Dubon2026.SourceGammaOneDiamondCuspPrimitive, `Dubon2026.SourceGammaOneDiamondPeriodH1, `Dubon2026.SourceGammaOneDiamondPeriodLattice, `Dubon2026.SourceGammaOneDiamondFiniteH1, `Dubon2026.SourceGammaOneDiamondConjugation, `Dubon2026.SourceGammaOneDiamondH1, `Dubon2026.SourceGammaOneDiamondLabels, `Dubon2026.SourceGammaOneUnitDiamondAction]

  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue

  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"

Arithmetic Picard diamond and the original coefficient convention, transfer lifting and exceptional primes remain unresolved; DUB17--20 OPEN."
