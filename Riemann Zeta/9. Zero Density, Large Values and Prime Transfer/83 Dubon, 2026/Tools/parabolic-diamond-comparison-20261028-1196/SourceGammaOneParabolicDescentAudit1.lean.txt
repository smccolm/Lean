private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneParabolicOperatorAlgebra, `SourceGammaOneParabolicTateKernel, `SourceGammaOneParabolicTransferCriterion, `SourceParabolicCongruenceException, `SourceGammaOneParabolicCharacterTransfer]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.binaryParabolicWord_val, `Dubon2026.SourceFunctionField.GammaOne.binaryParabolicHeckeAlgebra_finite, `Dubon2026.SourceFunctionField.GammaOne.binaryParabolicWord_eq_zero_iff, `Dubon2026.SourceFunctionField.GammaOne.binaryTateWord_eq_zero_iff_parabolic, `Dubon2026.sourceGammaOneParabolicTransfer_word_kernel, `Dubon2026.sourcePrimitiveParabolicExceptionalInteger_ne_zero, `Dubon2026.sourcePrimitiveParabolicH1AdicQuotientCharacter_zero, `Dubon2026.sourcePrimitiveGammaOneParabolicCharacter_of_transfer_away]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.binaryParabolicPrimeOperators, `Dubon2026.SourceFunctionField.GammaOne.binaryParabolicHeckeAlgebra, `Dubon2026.sourceGammaOneIntegralParabolicTransfer, `Dubon2026.sourcePrimitiveParabolicExceptionalInteger]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Parabolic Descent postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Parabolic Descent theorems discovered"
  logInfo m!"SOURCE GAMMA ONE PARABOLIC DESCENT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
