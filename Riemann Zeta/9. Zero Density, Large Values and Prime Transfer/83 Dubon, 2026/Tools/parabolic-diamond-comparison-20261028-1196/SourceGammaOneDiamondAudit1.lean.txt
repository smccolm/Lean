private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneDiamondConjugation, `SourceGammaOneDiamondH1, `SourceGammaOneDiamondLabels, `SourceGammaOneUnitDiamondAction]
  let theorems : Array Name := #[`Dubon2026.sourceGammaOneDiamond_conj_mem, `Dubon2026.sourceGammaOneDiamondConjugation_trace, `Dubon2026.sourceGammaOneDiamondConjugation_mul, `Dubon2026.sourceGammaOneDiamondH1_hom_apply, `Dubon2026.sourceGammaOneDiamondH1_comp, `Dubon2026.sourceGammaOneBinaryParabolicH1_diamond_mem, `Dubon2026.sourceGammaOneDiamondH1_lift_independent, `Dubon2026.sourceGammaOneDiamondLabel_lift, `Dubon2026.sourceGammaOneUnitDiamondH1_eq, `Dubon2026.sourceGammaOneUnitDiamondH1_one, `Dubon2026.sourceGammaOneUnitDiamondH1_mul, `Dubon2026.sourceGammaOneBinaryParabolicH1_unitDiamond_mem]
  let constructions : Array Name := #[`Dubon2026.sourceGammaOneDiamondConjugation, `Dubon2026.sourceGammaOneDiamondMorphism, `Dubon2026.sourceGammaOneDiamondH1, `Dubon2026.sourceGammaOneUnitDiamondH1, `Dubon2026.sourceGammaOneUnitDiamondH1Hom]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Diamond postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Diamond theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIAMOND AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
