import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourcePrimitiveParabolicCharacterKernel, `Dubon2026.SourceFiniteIntegralSaturation, `Dubon2026.SourceSupportedCharacterFiber, `Dubon2026.SourceSupportedCharacterDivisibility, `Dubon2026.SourceParabolicHeckeSpan, `Dubon2026.SourceParabolicHeckeCongruence, `Dubon2026.SourceParabolicAdicOperatorKernel, `Dubon2026.SourceGammaOneParabolicOperatorAlgebra, `Dubon2026.SourceGammaOneParabolicTateKernel, `Dubon2026.SourceGammaOneParabolicTransferCriterion, `Dubon2026.SourceParabolicCongruenceException, `Dubon2026.SourceGammaOneParabolicCharacterTransfer, `Dubon2026.SourceGammaOneDiamondConjugation, `Dubon2026.SourceGammaOneDiamondH1, `Dubon2026.SourceGammaOneDiamondLabels, `Dubon2026.SourceGammaOneUnitDiamondAction, `Dubon2026.SourceGammaOneDiamondCuspPrimitive, `Dubon2026.SourceGammaOneDiamondPeriodH1, `Dubon2026.SourceGammaOneDiamondPeriodLattice, `Dubon2026.SourceGammaOneDiamondFiniteH1, `Dubon2026.SourceGammaOneDiamondPeriodTate]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
