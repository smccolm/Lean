import SourceSupportedCharacterFiber
import SourceSupportedCharacterDivisibility
import SourceParabolicHeckeSpan
import SourceParabolicHeckeCongruence
import SourceParabolicAdicOperatorKernel
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceSupportedCharacterFiber, `SourceSupportedCharacterDivisibility, `SourceParabolicHeckeSpan, `SourceParabolicHeckeCongruence, `SourceParabolicAdicOperatorKernel]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
