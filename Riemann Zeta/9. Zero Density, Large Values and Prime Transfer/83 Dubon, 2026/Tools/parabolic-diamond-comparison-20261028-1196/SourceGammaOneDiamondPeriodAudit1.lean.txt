private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneDiamondCuspPrimitive, `SourceGammaOneDiamondPeriodH1, `SourceGammaOneDiamondPeriodLattice, `SourceGammaOneDiamondFiniteH1]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.diamondGL_le_conj, `Dubon2026.SourceFunctionField.GammaOne.cuspDiamond_coe, `Dubon2026.SourceFunctionField.GammaOne.diamondPrimitive_hasDerivAt, `Dubon2026.SourceFunctionField.GammaOne.diamondPrimitive_difference, `Dubon2026.SourceFunctionField.GammaOne.periodValue_cuspDiamond, `Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_cuspDiamond, `Dubon2026.SourceFunctionField.GammaOne.realPeriodH1_cuspDiamond_unit, `Dubon2026.SourceFunctionField.GammaOne.periodDiamondLinear_dualPeriod, `Dubon2026.SourceFunctionField.GammaOne.periodDiamondLinear_preserves_integralLattice, `Dubon2026.SourceFunctionField.GammaOne.periodLatticeDiamond_latticePeriod, `Dubon2026.SourceFunctionField.GammaOne.periodDiamondLinear_along, `Dubon2026.SourceFunctionField.GammaOne.binaryLatticeH1Equiv_diamond, `Dubon2026.SourceFunctionField.GammaOne.binaryLatticeH1Equiv_unitDiamond]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.cuspDiamond, `Dubon2026.SourceFunctionField.GammaOne.cuspDiamondLinear, `Dubon2026.SourceFunctionField.GammaOne.periodDiamondLinear, `Dubon2026.SourceFunctionField.GammaOne.periodLatticeDiamond]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Diamond Period postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Diamond Period theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIAMOND PERIOD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
