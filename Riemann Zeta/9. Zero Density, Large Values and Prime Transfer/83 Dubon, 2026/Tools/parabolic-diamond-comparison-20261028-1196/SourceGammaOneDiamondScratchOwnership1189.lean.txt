import SourceGammaOneDiamondConjugation
import SourceGammaOneDiamondH1
import SourceGammaOneDiamondLabels
import SourceGammaOneUnitDiamondAction
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneDiamondConjugation, `SourceGammaOneDiamondH1, `SourceGammaOneDiamondLabels, `SourceGammaOneUnitDiamondAction]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
