theorem sourceParabolicHeckeSpan_operator_congruence (Q w N : ℕ)
    (T : sourceBinaryH1HeckeAlgebra Q w)
    (hT : ∀ x : H1 (sourceBinaryCoefficientRep ℤ Q w),
      x ∈ sourceParabolicH1 (Gamma0 Q) (sourceBinaryCoefficientRep ℤ Q w) →
        sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w (T.val x) = 0) :
    ∀ x : sourceParabolicHeckeSpan Q w,
      sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w (T.val x.val) = 0 := by
  let A := sourceBinaryH1HeckeAlgebra Q w
  letI : IsMulCommutative A := Algebra.isMulCommutative_adjoin ℤ (by
    rintro _ ⟨p, rfl⟩ _ ⟨q, rfl⟩
    exact (sourceBinaryH1PrimeOperators_commute ℤ Q w p q).eq)
  letI : CommRing A := inferInstance
  intro x
  have hx := x.property
  generalize x.val = y at hx ⊢
  induction hx using Submodule.span_induction with
  | mem y hy => exact hT y hy
  | zero => rw [map_zero, map_zero]
  | add y z _ _ hy hz => rw [map_add, map_add, hy, hz, add_zero]
  | smul a y _ hy =>
    have hc : T.val (a.val y) = a.val (T.val y) :=
      congrArg (fun U : A => U.val y) (mul_comm T a)
    change sourceBinaryH1Change (Int.castRingHom (ZMod N)) Q w (T.val (a.val y)) = 0
    rw [hc]
    exact sourceBinaryH1Change_heckeAlgebra_zero Q w N a (T.val y) hy

end
