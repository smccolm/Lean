/-! DUB17/18 -> original all-weight association -> arithmetic Gamma1
realization -> diamond comparison. Prove independence of the chosen
Gamma0 lift on native H1. Existing integral SL2 reduction supplies every
unit label; no lift, normalizer or character-descent premise is assumed.
The lower-right entry convention agrees with the inspected TauCeti
single-coset diamond construction. -/
namespace Dubon2026
noncomputable section
open CongruenceSubgroup groupCohomology CategoryTheory
open scoped MatrixGroups
variable (R V : Type) [CommRing R] [AddCommGroup V] [Module R V]
    (M : ℕ)

theorem sourceGammaOneDiamondH1_lift_independent (g h : Gamma0 M)
    (hlabel : Gamma0Map M g = Gamma0Map M h) (b : Module.End R V) :
    sourceGammaOneDiamondH1 R V M g b = sourceGammaOneDiamondH1 R V M h b := by
  have hker : g * h⁻¹ ∈ Gamma1' M := by
    change Gamma0Map M (g * h⁻¹) = 1
    rw [map_mul, hlabel, ← map_mul, mul_inv_cancel, map_one]
  let u : Gamma1 M := ⟨g.val * h.val⁻¹,
    (Gamma1_mem M _).mpr ((Gamma1_to_Gamma0_mem _).mp hker)⟩
  apply LinearMap.ext
  intro x
  apply (ConcreteCategory.bijective_of_isIso
    (H1IsoOfIsTrivial (Rep.trivial R (Gamma1 M) V)).hom).injective
  apply AddMonoidHom.ext
  intro γ
  have hγ : γ = Additive.ofMul γ.toMul := rfl
  rw [hγ, sourceGammaOneDiamondH1_hom_apply, sourceGammaOneDiamondH1_hom_apply]
  apply congrArg b
  have he : sourceGammaOneDiamondConjugation M g γ.toMul =
      u * sourceGammaOneDiamondConjugation M h γ.toMul * u⁻¹ := by
    apply Subtype.ext
    change g.val * γ.toMul.val * g.val⁻¹ =
      (g.val * h.val⁻¹) * (h.val * γ.toMul.val * h.val⁻¹) * (g.val * h.val⁻¹)⁻¹
    group
  rw [he, ofMul_mul, ofMul_mul, ofMul_inv, map_add, map_add, map_neg]
  abel

theorem sourceGammaOneDiamondLabel_lift [NeZero M] (d : (ZMod M)ˣ) :
    ∃ g : Gamma0 M, Gamma0Map M g = (d : ZMod M) := by
  let A : SL(2, ZMod M) := ⟨!![(d⁻¹ : (ZMod M)ˣ), 0; 0, (d : ZMod M)], by
    simp [Matrix.det_fin_two]⟩
  obtain ⟨g, hg⟩ := SL2Reduction.SL2_reduction_surjective M A
  have h10 := congrArg (fun B : SL(2, ZMod M) => B 1 0) hg
  have h11 := congrArg (fun B : SL(2, ZMod M) => B 1 1) hg
  have hmem : g ∈ Gamma0 M := by simpa [Gamma0_mem, A] using h10
  exact ⟨⟨g, hmem⟩, by simpa [Gamma0Map, A] using h11⟩

/-- The genuine lower-right-unit-labelled diamond on native constant-coefficient H1. -/
def sourceGammaOneUnitDiamondH1 [NeZero M] (d : (ZMod M)ˣ) :
    Module.End R (H1 (Rep.trivial R (Gamma1 M) V)) :=
  sourceGammaOneDiamondH1 R V M (sourceGammaOneDiamondLabel_lift M d).choose LinearMap.id

theorem sourceGammaOneUnitDiamondH1_eq [NeZero M] (d : (ZMod M)ˣ)
    (g : Gamma0 M) (hg : Gamma0Map M g = (d : ZMod M)) :
    sourceGammaOneUnitDiamondH1 R V M d =
      sourceGammaOneDiamondH1 R V M g LinearMap.id :=
  sourceGammaOneDiamondH1_lift_independent R V M _ g
    ((sourceGammaOneDiamondLabel_lift M d).choose_spec.trans hg.symm) LinearMap.id

end
