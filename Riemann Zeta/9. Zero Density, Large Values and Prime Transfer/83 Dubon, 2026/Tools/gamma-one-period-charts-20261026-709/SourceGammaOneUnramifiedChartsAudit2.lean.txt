private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRamificationIsolation, `SourceGammaOneUnramifiedCharts]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.analyticRamification_dvd_full_stabilizer_half, `Dubon2026.SourceFunctionField.GammaOne.analyticRamification_eq_one_of_generic, `Dubon2026.SourceFunctionField.GammaOne.analyticRamification_eventually_one, `Dubon2026.SourceFunctionField.GammaOne.exists_unramified_analytic_chart, `Dubon2026.SourceFunctionField.GammaOne.exists_generic_analytic_chart]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Unramified Charts postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Unramified Charts theorems discovered"
  logInfo m!"SOURCE GAMMA ONE UNRAMIFIED CHARTS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceGammaOneRamificationIsolation
