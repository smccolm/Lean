private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_generic_analytic_chart]
  let localModules : Array Name := #[`Dubon2026.SourceGammaOneAnalyticEvaluation, `Dubon2026.SourceGammaOneAnalyticGerms, `Dubon2026.SourceGammaOneAnalyticJOrders, `Dubon2026.SourceGammaOneAnalyticOrder, `Dubon2026.SourceGammaOneAnalyticOrderNontrivial, `Dubon2026.SourceGammaOneAnalyticPlaces, `Dubon2026.SourceGammaOneAnalyticResidues, `Dubon2026.SourceGammaOneJRealization, `Dubon2026.SourceGammaOneMeromorphicRealization, `Dubon2026.SourceGammaOneRamificationIsolation, `Dubon2026.SourceGammaOneRatioRealization, `Dubon2026.SourceGammaOneRealizationAlgebra, `Dubon2026.SourceGammaOneUnramifiedCharts]
  let required : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.analyticRamification_dvd_full_stabilizer_half, `Dubon2026.SourceFunctionField.GammaOne.analyticRamification_eq_one_of_generic, `Dubon2026.SourceFunctionField.GammaOne.analyticRamification_eventually_one, `Dubon2026.SourceFunctionField.GammaOne.exists_unramified_analytic_chart, `Dubon2026.SourceFunctionField.GammaOne.exists_generic_analytic_chart, `Dubon2026.SourceFunctionField.GammaOne.analyticPlace_order, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_j_sub, `Dubon2026.SourceFunctionField.GammaOne.placeEval_meromorphicOrderAt, `Dubon2026.SourceFunctionField.GammaOne.placeEval_analyticAt_iff, `Dubon2026.SourceFunctionField.GammaOne.meromorphicNFAt_placeEval, `Dubon2026.SourceFunctionField.JMultiplicity.main, `Dubon2026.SourceCurve.AnalyticDisc.exists_parameter_of_evaluation_family]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE GAMMA ONE UNRAMIFIED CHARTS DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the actual local chart away from E4/E6 zeros. Only declaration values are traversed. Exact j-order and isolated zeros derive local unramifiedness; actual residue evaluation supplies the uniformizer and all normalized orders for the existing parameter theorem. Elliptic centers, global charts/periods, Hecke/diamond/Picard/H1 arithmetic comparison, original all-weight characteristic-zero association and literal higher symmetric-power continuation remain; 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
