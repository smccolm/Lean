private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneAnalyticResidues, `SourceGammaOneAnalyticEvaluation, `SourceGammaOneEvaluationRealizationGerms]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.analyticPlace_mem_iff, `Dubon2026.SourceFunctionField.GammaOne.realize_sub_constant_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.place_residue_bijective, `Dubon2026.SourceFunctionField.GammaOne.realize_tendsto_residue, `Dubon2026.SourceFunctionField.GammaOne.regular_analytic_representative, `Dubon2026.SourceFunctionField.GammaOne.meromorphicNFAt_placeEval, `Dubon2026.SourceFunctionField.GammaOne.placeEval_analyticAt_iff, `Dubon2026.SourceFunctionField.GammaOne.placeEval_meromorphicOrderAt, `Dubon2026.SourceFunctionField.GammaOne.placeEval_eq_realize_of_analyticAt, `Dubon2026.SourceFunctionField.GammaOne.placeEval_eventuallyEq_realize]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.placeEval]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Analytic Residues postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Analytic Residues theorems discovered"
  logInfo m!"SOURCE GAMMA ONE ANALYTIC RESIDUES AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
