private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneUniformCuspPole, `SourceGammaOneHolomorphicModularQuotient, `SourceGammaOneHolomorphicFieldRealization]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.finite_modular_orbit_of_invariant, `Dubon2026.SourceFricke.exists_uniform_discriminant_pole_bound, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_modular_quotient, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_modular_quotient_of_meromorphic_cusps, `Dubon2026.SourceFunctionField.GammaOne.exists_original_field_of_holomorphic]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Holomorphic Field postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Holomorphic Field theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HOLOMORPHIC FIELD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
