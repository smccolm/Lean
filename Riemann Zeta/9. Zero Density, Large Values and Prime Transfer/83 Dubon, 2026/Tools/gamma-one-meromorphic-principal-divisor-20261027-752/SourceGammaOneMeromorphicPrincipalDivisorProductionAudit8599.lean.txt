private def runSourceResidueFunctionalAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`Dubon2026.SourceGammaOneCuspResidues, `Dubon2026.SourceGammaOneCuspBoundedRealization, `Dubon2026.SourceGammaOneCuspPlaceExhaustion, `Dubon2026.SourceGammaOneUniformCuspPole, `Dubon2026.SourceGammaOneHolomorphicModularQuotient, `Dubon2026.SourceGammaOneHolomorphicFieldRealization, `Dubon2026.SourceGammaOneOriginalPoleMultiplier, `Dubon2026.SourceGammaOnePoleClearedNormalization, `Dubon2026.SourceGammaOnePoleClearedProduct, `Dubon2026.SourceGammaOnePoleClearedCuspBounds, `Dubon2026.SourceGammaOneMeromorphicFieldRealization, `Dubon2026.SourceGammaOneMeromorphicCuspLimits, `Dubon2026.SourceGammaOneMeromorphicPrincipalDivisor]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.cuspPlace_mem_iff, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_sub_constant_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_order_eq_zero_of_tendsto, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_tendsto_residue, `Dubon2026.SourceFunctionField.GammaOne.regular_cusp_analytic_representative, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_cusp_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_bounded_of_cusp_regular, `Dubon2026.SourceFunctionField.GammaOne.constant_of_mem_all_analytic_and_cusp_places, `Dubon2026.SourceFunctionField.GammaOne.place_analytic_or_actual_cusp, `Dubon2026.SourceFunctionField.GammaOne.exists_cusp_place_eq_of_j_pole, `Dubon2026.SourceFunctionField.GammaOne.order_eq_zero_at_j_poles_of_cusp_limits, `Dubon2026.SourceFricke.finite_modular_orbit_of_invariant, `Dubon2026.SourceFricke.exists_uniform_discriminant_pole_bound, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_modular_quotient, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_modular_quotient_of_meromorphic_cusps, `Dubon2026.SourceFunctionField.GammaOne.exists_original_field_of_holomorphic, `Dubon2026.SourceFunctionField.GammaOne.exists_original_pole_multiplier, `Dubon2026.SourceFunctionField.GammaOne.exists_original_pole_multiplier_at_analytic_places, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_of_nonnegative_orders, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_invariant_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.boundedAtImInfty_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.exists_pole_cleared_holomorphic_product, `Dubon2026.SourceFunctionField.GammaOne.pole_bounds_of_local_product, `Dubon2026.SourceFunctionField.GammaOne.meromorphic_cusps_of_holomorphic_pole_bounds, `Dubon2026.SourceFunctionField.GammaOne.exists_original_field_of_prescribed_meromorphic_orders, `Dubon2026.SourceFunctionField.GammaOne.tendsto_atImInfty_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.eventually_continuousAt_realize_cusp, `Dubon2026.SourceFunctionField.GammaOne.tendsto_cuspFunction_of_tendsto, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_tendsto_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.isPrincipal_of_prescribed_meromorphic_orders, `Dubon2026.SourceFunctionField.GammaOne.analytic_divisor_isPrincipal_of_meromorphic_function]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue functional postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue functional theorems discovered"
  logInfo m!"SOURCE GAMMA ONE MEROMORPHIC PRINCIPAL DIVISOR AUDIT PASS: {count} theorems including private/generated declarations. The exact meromorphic function gives a principal divisor in the original Gamma1 field with all cusp orders derived. Its construction from periods and original arithmetic comparison remain; 16/20 gates."

#eval runSourceResidueFunctionalAudit
