#print axioms Dubon2026.SourceFunctionField.GammaOne.pole_bounds_of_local_product
#print axioms Dubon2026.SourceFunctionField.GammaOne.meromorphic_cusps_of_holomorphic_pole_bounds
#print axioms Dubon2026.SourceFunctionField.GammaOne.exists_original_field_of_prescribed_meromorphic_orders

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOnePoleClearedProduct, `SourceGammaOnePoleClearedCuspBounds, `SourceGammaOneMeromorphicFieldRealization]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_pole_cleared_holomorphic_product, `Dubon2026.SourceFunctionField.GammaOne.pole_bounds_of_local_product, `Dubon2026.SourceFunctionField.GammaOne.meromorphic_cusps_of_holomorphic_pole_bounds, `Dubon2026.SourceFunctionField.GammaOne.exists_original_field_of_prescribed_meromorphic_orders]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Meromorphic Field postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Meromorphic Field theorems discovered"
  logInfo m!"SOURCE GAMMA ONE MEROMORPHIC FIELD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceGammaOnePoleClearedProduct
#lint in SourceGammaOnePoleClearedCuspBounds
#lint in SourceGammaOneMeromorphicFieldRealization
