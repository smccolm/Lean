private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCuspBoundedRealization, `SourceGammaOneCuspPlaceExhaustion]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_cusp_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_quotient_bounded_of_cusp_regular, `Dubon2026.SourceFunctionField.GammaOne.constant_of_mem_all_analytic_and_cusp_places, `Dubon2026.SourceFunctionField.GammaOne.place_analytic_or_actual_cusp, `Dubon2026.SourceFunctionField.GammaOne.exists_cusp_place_eq_of_j_pole, `Dubon2026.SourceFunctionField.GammaOne.order_eq_zero_at_j_poles_of_cusp_limits]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Cusp Exhaustion postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Cusp Exhaustion theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CUSP EXHAUSTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceGammaOneCuspBoundedRealization
#lint in SourceGammaOneCuspPlaceExhaustion
