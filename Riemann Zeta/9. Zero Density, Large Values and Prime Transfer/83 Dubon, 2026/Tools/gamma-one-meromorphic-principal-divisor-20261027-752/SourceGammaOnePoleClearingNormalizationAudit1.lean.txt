private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneOriginalPoleMultiplier, `SourceGammaOnePoleClearedNormalization]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_original_pole_multiplier, `Dubon2026.SourceFunctionField.GammaOne.exists_original_pole_multiplier_at_analytic_places, `Dubon2026.SourceFunctionField.GammaOne.exists_holomorphic_of_nonnegative_orders, `Dubon2026.SourceFunctionField.GammaOne.holomorphic_invariant_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.boundedAtImInfty_of_local_eq]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Pole Clearing Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Pole Clearing Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE POLE CLEARING NORMALIZATION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceGammaOneOriginalPoleMultiplier
