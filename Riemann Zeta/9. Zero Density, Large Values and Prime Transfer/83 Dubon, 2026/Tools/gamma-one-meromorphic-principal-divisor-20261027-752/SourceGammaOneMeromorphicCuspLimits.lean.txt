theorem tendsto_atImInfty_of_local_eq {F G : ℍ → ℂ} {c : ℂ}
    (hG : ∀ᶠ τ in atImInfty, ContinuousAt G τ)
    (he : ∀ τ : ℍ, G =ᶠ[𝓝[≠] τ] F)
    (hF : Tendsto F atImInfty (𝓝 c)) : Tendsto G atImInfty (𝓝 c) := by
  obtain ⟨A, hA⟩ := (atImInfty_mem _).mp hG
  apply Metric.tendsto_nhds.mpr
  intro ε hε
  obtain ⟨B, hB⟩ := (atImInfty_mem _).mp
    (Metric.tendsto_nhds.mp hF (ε / 2) (half_pos hε))
  apply (atImInfty_mem _).mpr
  refine ⟨max A B + 1, fun τ hτ => ?_⟩
  have hτA : A ≤ τ.im := by linarith [le_max_left A B]
  have hτB : B < τ.im := by linarith [le_max_right A B]
  have hnear : ∀ᶠ w in 𝓝 τ, B < w.im :=
    continuous_im.continuousAt.tendsto.eventually (Ioi_mem_nhds hτB)
  have hbound : dist (G τ) c ≤ ε / 2 := by
    have hd : Tendsto (fun w : ℍ => dist (G w) c) (𝓝 τ) (𝓝 (dist (G τ) c)) :=
      (hA τ hτA).dist tendsto_const_nhds
    apply le_of_tendsto (hd.mono_left
      (show 𝓝[≠] τ ≤ 𝓝 τ from nhdsWithin_le_nhds))
    filter_upwards [he τ, hnear.filter_mono nhdsWithin_le_nhds] with w hew hw
    rw [hew]
    exact (hB w hw.le).le
  exact hbound.trans_lt (half_lt_self hε)

theorem eventually_continuousAt_realize_cusp
    {Γ : Subgroup SL(2, ℤ)} (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) (hT : ModularGroup.T ∈ Γ)
    (x : functionFieldOver Γ ℂ) (γ : SL(2, ℤ)) :
    ∀ᶠ τ in atImInfty,
      ContinuousAt (fun w : ℍ => realize Γ (x : LaurentSeries ℂ) (γ • w)) τ := by
  obtain ⟨k, g, h, hh, hr⟩ := functionFieldOver_modular_ratio hT x
  let gt := ModularForm.translate g (γ : GL (Fin 2) ℝ)
  let ht := ModularForm.translate h (γ : GL (Fin 2) ℝ)
  have hN : (0 : ℝ) < N := by exact_mod_cast NeZero.pos N
  have hp : Function.Periodic ((ht : ℍ → ℂ) ∘ ofComplex) N :=
    SlashInvariantFormClass.periodic_comp_ofComplex ht
      (CuspAuxiliary.natCast_mem_strictPeriods_conj N hΓ γ)
  have hn : ∀ᶠ τ in atImInfty, ht τ ≠ 0 := by
    have hq := (Function.Periodic.qParam_tendsto hN).comp tendsto_coe_atImInfty
    filter_upwards [hq.eventually (eventually_ne_zero_translated_cuspFunction N hΓ h hh γ)]
      with τ hτ
    simpa only [Function.comp_apply, ← eq_cuspFunction τ hN.ne' hp] using hτ
  filter_upwards [hn] with τ hτ
  have hcont : ContinuousAt (fun w : ℍ => gt w / ht w) τ :=
    gt.holo'.continuous.continuousAt.div ht.holo'.continuous.continuousAt hτ
  apply hcont.congr_of_eventuallyEq
  filter_upwards [ht.holo'.continuous.continuousAt.eventually_ne hτ] with w hw
  exact realize_smul_eq_translated_ratio hT g h x hr γ w hw

theorem tendsto_cuspFunction_of_tendsto {N : ℕ} [NeZero N] {F : ℍ → ℂ} {c : ℂ}
    (hF : Tendsto F atImInfty (𝓝 c)) :
    Tendsto (cuspFunction N F) (𝓝[≠] (0 : ℂ)) (𝓝 c) := by
  have hN : (0 : ℝ) < N := by exact_mod_cast NeZero.pos N
  have ht := hF.comp (tendsto_comap_im_ofComplex.comp (Function.Periodic.invQParam_tendsto hN))
  apply ht.congr'
  filter_upwards [self_mem_nhdsWithin] with q hq
  simp only [cuspFunction, Function.Periodic.cuspFunction_eq_of_nonzero _ _ hq,
    Function.comp_apply]

theorem cuspRealize_tendsto_of_local_eq
    {Γ : Subgroup SL(2, ℤ)} (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) (hT : ModularGroup.T ∈ Γ)
    (x : functionFieldOver Γ ℂ) (F : ℍ → ℂ)
    (he : ∀ τ : ℍ,
      (fun z : ℂ => realize Γ (x : LaurentSeries ℂ) (ofComplex z))
        =ᶠ[𝓝[≠] (τ : ℂ)] (F ∘ ofComplex))
    (γ : SL(2, ℤ)) {c : ℂ}
    (hF : Tendsto (F ∘ (γ • ·)) atImInfty (𝓝 c)) :
    Tendsto (cuspRealize Γ N γ (x : LaurentSeries ℂ)) (𝓝[≠] (0 : ℂ)) (𝓝 c) := by
  have heH (τ : ℍ) : (fun w : ℍ => realize Γ (x : LaurentSeries ℂ) w) =ᶠ[𝓝[≠] τ] F :=
    SourceFunctionField.ModularRatios.eventually_nhdsNE_coe_iff.mp (he τ)
  apply tendsto_cuspFunction_of_tendsto
  apply tendsto_atImInfty_of_local_eq (eventually_continuousAt_realize_cusp N hΓ hT x γ) _ hF
  intro τ
  have ht : Tendsto (fun w : ℍ => γ • w) (𝓝[≠] τ) (𝓝[≠] (γ • τ)) :=
    le_of_eq ((Homeomorph.smul (γ : GL (Fin 2) ℝ)).map_punctured_nhds_eq τ)
  exact (heH (γ • τ)).comp_tendsto ht

end
