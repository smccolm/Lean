private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneMeromorphicCuspLimits, `SourceGammaOneMeromorphicPrincipalDivisor]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.tendsto_atImInfty_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.eventually_continuousAt_realize_cusp, `Dubon2026.SourceFunctionField.GammaOne.tendsto_cuspFunction_of_tendsto, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_tendsto_of_local_eq, `Dubon2026.SourceFunctionField.GammaOne.isPrincipal_of_prescribed_meromorphic_orders, `Dubon2026.SourceFunctionField.GammaOne.analytic_divisor_isPrincipal_of_meromorphic_function]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Meromorphic Principal Divisor postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Meromorphic Principal Divisor theorems discovered"
  logInfo m!"SOURCE GAMMA ONE MEROMORPHIC PRINCIPAL DIVISOR AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
