/-! DUB-17/18 -> eventual residual large image -> exceptional image exclusion.
Native Chebotarev at the identity gives an actual norm Frobenius outside every
finite set where the same original residual action is the identity. This will
supply a nonzero actual Fourier coefficient, independently of large image. -/

  rationalPrimeOriginalClosureNorm rationalPrimeOriginalClosureNormedAlgebra

local notation "Qbar" => AlgebraicClosure ℚ
private local instance : IsGalois ℚ Qbar := sourceRationalAlgebraicClosure_isGalois

/-- The actual original arithmetic action has identity norm Frobenius outside any finite set. -/
theorem original_arithmetic_exists_identity_norm_frobenius
    (a : 𝓞 ℚ) {T : Type} [Group T] [TopologicalSpace T] [DiscreteTopology T]
    (ρ : rationalArithmeticGaloisGroup a →* T) (hρ : Continuous ρ) (S : Finset ℕ) :
    ∃ (q : ℕ) (hq : q.Prime), q ∉ S ∧
      ∃ (e : Qbar →ₐ[ℚ] rationalPrimeLocalClosure q hq)
        (φ : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq)),
        (∀ z : IsLocalRing.ResidueField (algebraicLocalIntegerRing
          (rationalPrimeLocalClosure q hq)), φ • z = z ^ q) ∧
        ρ (sourceAbsoluteArithmeticRestriction a (embeddedNormalGaloisRestriction e φ)) = 1 := by
  let χ : Gal(Qbar/ℚ) →ₜ* T :=
    ⟨ρ.comp (sourceAbsoluteArithmeticRestriction a).toMonoidHom,
      hρ.comp (sourceAbsoluteArithmeticRestriction a).continuous⟩
  obtain ⟨q, hq, hqS, e, φ, hφ, τ, hτ⟩ :=
    absolute_continuous_exists_norm_frobenius_conjugate χ 1 S
  refine ⟨q, hq, hqS, e, φ, hφ, ?_⟩
  have he : χ (embeddedNormalGaloisRestriction e φ) = 1 := by
    have hh := congrArg (fun t => (χ τ)⁻¹ * t * χ τ) hτ
    simpa only [map_mul, map_inv, map_one, mul_assoc, inv_mul_cancel_left,
      inv_mul_cancel, mul_one, one_mul] using hh
  exact he

end
end Dubon2026
