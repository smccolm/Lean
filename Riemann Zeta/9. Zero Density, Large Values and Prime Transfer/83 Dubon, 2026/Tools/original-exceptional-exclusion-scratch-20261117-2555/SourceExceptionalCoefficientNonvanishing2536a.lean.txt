    {F : Type*} [Field F] [NumberField F] (v : HeightOneSpectrum (𝓞 F))
    (q : ℕ) (hq : v.valuation F (q : F) = WithZero.exp (-1 : ℤ))
    (hc : (-14400 : 𝓞 F) ∉ v.asIdeal) (a : F) (ha : a ≠ 0)
    (w : ℕ) (hw : Odd w) :
    exceptionalNonzeroTracePolynomial.eval₂ (Int.castRingHom F) (a ^ 2 / (q : F) ^ w) ≠ 0 := by
  intro h
  exact odd_square_ratio_valuation_ne_one (v.valuation F) a (q : F) ha hq w hw
    (exceptional_numberField_root_valuation_one v _ h hc)

end
