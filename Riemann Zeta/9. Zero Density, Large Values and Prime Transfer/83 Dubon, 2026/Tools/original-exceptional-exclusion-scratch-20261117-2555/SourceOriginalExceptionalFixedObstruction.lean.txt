/-- A fixed good prime and fixed nonzero integer constant obstruct exceptional reduction for the actual form. -/
theorem exists_original_exceptional_integer_obstruction
    (Q : ℕ) [NeZero Q] {k : ℤ} (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k) :
    ∃ (q : ℕ) (H : ℤ[X]), q.Prime ∧ ¬q ∣ Q ∧ H.coeff 0 ≠ 0 ∧
      H.eval₂ (Int.castRingHom (sourcePrimitiveCoefficientField f))
        (exceptionalTracePolynomial.eval₂ (Int.castRingHom (sourcePrimitiveCoefficientField f))
          ((sourcePrimitiveFieldCoefficient f q) ^ 2 /
            (q : sourcePrimitiveCoefficientField f) ^ (k - 1).toNat)) = 0 := by
  letI := sourcePrimitiveCoefficientField_numberField f hk
  obtain ⟨q, hq, _hqS, hqQ, _ha, hP⟩ :=
    exists_original_exceptional_prime_obstruction Q f hk ∅
  obtain ⟨H, hH0, hH⟩ := nonzero_numberField_integer_relation _ hP
  exact ⟨q, H, hq, hqQ, hH0, hH⟩

end
end Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction
