namespace Dubon2026.SourceDickson
noncomputable section
open Polynomial
open scoped nonZeroDivisors

/-- Every nonzero number-field value has an integer relation with nonzero constant term. -/
theorem nonzero_numberField_integer_relation
    {F : Type*} [Field F] [NumberField F] (x : F) (hx : x ≠ 0) :
    ∃ P : ℤ[X], P.coeff 0 ≠ 0 ∧ P.eval₂ (Int.castRingHom F) x = 0 := by
  letI : Algebra.IsAlgebraic ℤ ℚ := IsLocalization.isAlgebraic ℚ (nonZeroDivisors ℤ)
  have ha : IsAlgebraic ℤ x :=
    (Algebra.IsAlgebraic.isAlgebraic (R := ℚ) x).restrictScalars ℤ
  exact ha.exists_nonzero_coeff_and_aeval_eq_zero (mem_nonZeroDivisors_iff_ne_zero.mpr hx)

end
end Dubon2026.SourceDickson
