/-- Every actual primitive form has a nonzero original prime coefficient outside any finite set. -/
theorem exists_original_nonzero_prime_coefficient
    (Q : ℕ) [NeZero Q] {k : ℤ} (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k)
    (S : Finset ℕ) :
    ∃ q : ℕ, q.Prime ∧ q ∉ S ∧ sourcePrimitiveFieldCoefficient f q ≠ 0 := by
  letI : Fact (Nat.Prime 5) := ⟨by decide⟩
  letI := sourcePrimitiveCoefficientField_numberField f hk
  letI := rationalPrimeLocalField_charZero 5 (Fact.out : Nat.Prime 5)
  let F := rationalPrimeLocalField 5 (Fact.out : Nat.Prime 5)
  let L := commonCoefficientField (sourcePrimitiveCoefficientField f) F
  have hs := exists_gammaOne_original_residual_rankTwo Q 5 f hk (by omega)
  dsimp only at hs
  obtain ⟨u, v, _hu, _hv, _hext, _hfin, _hfinite, hdiscrete, hchar,
    τ, _ρint, ρres, _hi, hr, _hred, hFrob, _hodd⟩ := hs
  letI : UniformSpace L := u
  letI : TopologicalSpace L := u.toTopologicalSpace
  letI : ValuativeRel L := v
  letI : TopologicalSpace 𝓀[L] :=
    inferInstanceAs (TopologicalSpace (𝒪[L] ⧸ IsLocalRing.maximalIdeal 𝒪[L]))
  letI : DiscreteTopology 𝓀[L] := hdiscrete
  letI : CharP 𝓀[L] 5 := hchar
  obtain ⟨q, hq, hqS, e, φ, hφ, hidentity⟩ :=
    original_arithmetic_exists_identity_norm_frobenius ((Q * 5 : ℕ) : 𝓞 ℚ)
      ρres hr (S ∪ (Q * 5).primeFactors)
  have hqbad : q ∉ (Q * 5).primeFactors := fun h => hqS (Finset.mem_union_right S h)
  have haway : ¬ q ∣ Q * 5 := by
    intro h
    exact hqbad (Nat.mem_primeFactors.mpr ⟨hq, h, mul_ne_zero (NeZero.ne Q) (by omega)⟩)
  refine ⟨q, hq, fun h => hqS (Finset.mem_union_left _ h), ?_⟩
  intro hz
  have hf := (hFrob q hq haway e φ hφ).1
  rw [hidentity] at hf
  have hzint : (⟨τ (sourcePrimitiveFieldCoefficient f q),
      sourcePrimitiveCoefficient_valuation_le_one f hk τ q⟩ : 𝒪[L]) = 0 := by
    apply Subtype.ext
    simpa only [hz, map_zero]
  rw [hzint, map_zero] at hf
  have htwo : (2 : 𝓀[L]) = 0 := by simpa using hf
  have hd := (CharP.cast_eq_zero_iff 𝓀[L] 5 2).mp htwo
  norm_num at hd

end
