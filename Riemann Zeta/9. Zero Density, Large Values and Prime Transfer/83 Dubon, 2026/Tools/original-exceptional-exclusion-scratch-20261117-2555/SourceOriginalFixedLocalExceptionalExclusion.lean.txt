/-- Literal good-prime Frobenius values on the original residual action exclude the exceptional images. -/
theorem original_fixed_local_not_exceptional
    (Q p : ℕ) [NeZero Q] [Fact p.Prime] {k : ℤ}
    (f : PrimitiveCuspForm Q k) (hk : 2 ≤ k)
    (q : ℕ) (hq : q.Prime) (hqp : q < p)
    (H : ℤ[X]) (hH0 : H.coeff 0 ≠ 0) (hpH : (H.coeff 0).natAbs < p)
    (hH : H.eval₂ (Int.castRingHom (sourcePrimitiveCoefficientField f))
      (exceptionalTracePolynomial.eval₂ (Int.castRingHom (sourcePrimitiveCoefficientField f))
        ((sourcePrimitiveFieldCoefficient f q) ^ 2 /
          (q : sourcePrimitiveCoefficientField f) ^ (k - 1).toNat)) = 0)
    {E : Type} [Field E] [CharZero E] [ValuativeRel E] [CharP 𝓀[E] p]
    (τ : sourcePrimitiveCoefficientField f →ₐ[ℚ] E)
    (ρ : rationalArithmeticGaloisGroup ((Q * p : ℕ) : 𝓞 ℚ) →*
      GeneralLinearGroup (Fin 2) 𝓀[E])
    (j : 𝓀[E] →+* K p)
    (hlocal : ∀ φ : Gal(rationalPrimeLocalClosure q hq/rationalPrimeLocalField q hq),
      (∀ z : IsLocalRing.ResidueField (algebraicLocalIntegerRing
        (rationalPrimeLocalClosure q hq)), φ • z = z ^ q) →
      let a := sourcePrimitiveResidualHeckeCharacter f hk τ
        ⟨Dubon2026.cuspHeckeLinear Q k q,
          Algebra.subset_adjoin ⟨(⟨q, hq.pos⟩ : ℕ+), rfl⟩⟩
      let A := (ρ (rationalPrimeArithmeticRestriction ((Q * p : ℕ) : 𝓞 ℚ) q hq φ)).val
      A.trace = a ∧ A.det = (q : 𝓀[E]) ^ (k - 1).toNat) :
    ¬(Nonempty ((residualProjectiveAction p j ρ).range ≃* alternatingGroup (Fin 4)) ∨
      Nonempty ((residualProjectiveAction p j ρ).range ≃* Equiv.Perm (Fin 4)) ∨
      Nonempty ((residualProjectiveAction p j ρ).range ≃* alternatingGroup (Fin 5))) := by
  intro hex
  have hpq : ¬p ∣ q := fun h => (not_le_of_gt hqp) (Nat.le_of_dvd hq.pos h)
  obtain ⟨φ, hφ⟩ := rationalPrimeOriginalNorm_exists_frobenius q hq
  have hf := hlocal φ hφ
  dsimp only at hf
  have ht := hf.1.trans (sourcePrimitiveResidualHeckeCharacter_hecke f hk τ ⟨q, hq.pos⟩)
  have hd := original_exceptional_characteristic_dvd_relation_constant
    p j ρ hex τ.toRingHom (algebraMap 𝒪[E] E) Subtype.val_injective
    (IsLocalRing.residue 𝒪[E]) (sourcePrimitiveFieldCoefficient f q)
    ⟨τ (sourcePrimitiveFieldCoefficient f q), sourcePrimitiveCoefficient_valuation_le_one f hk τ q⟩
    rfl q (k - 1).toNat hpq
    (rationalPrimeArithmeticRestriction ((Q * p : ℕ) : 𝓞 ℚ) q hq φ) ht hf.2 H hH
  exact not_int_dvd_of_natAbs_lt hH0 hpH hd

end
