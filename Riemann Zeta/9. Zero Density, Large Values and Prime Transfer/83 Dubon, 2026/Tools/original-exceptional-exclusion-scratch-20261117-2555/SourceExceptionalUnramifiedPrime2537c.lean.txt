/-! DUB-17/18 -> exceptional original image exclusion -> actual unramified
coefficient-field primes. Reuse the native integral valuation comparison and

namespace Dubon2026.SourceDickson
noncomputable section
open Polynomial IsDedekindDomain
open scoped NumberField WithZero
variable {F : Type*} [Field F] [NumberField F]

/-- At an actual absolutely unramified prime over q, q has normalized valuation minus one. -/
theorem exceptional_unramified_prime_valuation
    (q : ℕ) (hq : q.Prime) (v : HeightOneSpectrum (𝓞 F))
    [v.asIdeal.LiesOver (rationalPrimePlace q hq).asIdeal]
    [Algebra.IsUnramifiedAt ℤ v.asIdeal] :
    v.valuation F (q : F) = WithZero.exp (-1 : ℤ) := by
  have he : (rationalPrimePlace q hq).asIdeal.ramificationIdx v.asIdeal = 1 := by
    rw [Ideal.over_def v.asIdeal (rationalPrimePlace q hq).asIdeal]
    exact Ideal.ramificationIdx_eq_one_of_isUnramifiedAt (R := ℤ) v.ne_bot
  rw [← map_natCast (algebraMap (𝓞 F) F) q, v.valuation_of_algebraMap,
    ← map_natCast (algebraMap ℤ (𝓞 F)) q,
    unramified_intValuation_algebraMap (rationalPrimePlace q hq) v he]
  apply (rationalPrimePlace q hq).intValuation_singleton (by exact_mod_cast hq.ne_zero)
  ext z
  rw [Ideal.mem_span_singleton, rationalPrimePlace_int_mem_iff]

/-- Actual unramified coefficient primes away from the fixed integer constant give the nonvanishing obstruction. -/
theorem exceptional_unramified_coefficient_ratio_nonvanishing
    (q : ℕ) (hq : q.Prime) (v : HeightOneSpectrum (𝓞 F))
    [v.asIdeal.LiesOver (rationalPrimePlace q hq).asIdeal]
    [Algebra.IsUnramifiedAt ℤ v.asIdeal]
    (hqconst : ¬ (q : ℤ) ∣ (-14400 : ℤ)) (a : F) (ha : a ≠ 0)
    (w : ℕ) (hw : Odd w) :
    exceptionalNonzeroTracePolynomial.eval₂ (Int.castRingHom F) (a ^ 2 / (q : F) ^ w) ≠ 0 := by
  apply exceptional_coefficient_ratio_nonvanishing v q
    (exceptional_unramified_prime_valuation q hq v) _ a ha w hw
  intro hm
  apply hqconst
  apply (rationalPrimePlace_int_mem_iff q hq _).mp
  apply (Ideal.mem_of_liesOver v.asIdeal (rationalPrimePlace q hq).asIdeal _).mpr
  simpa only [map_neg, map_ofNat] using hm

end
end Dubon2026.SourceDickson
