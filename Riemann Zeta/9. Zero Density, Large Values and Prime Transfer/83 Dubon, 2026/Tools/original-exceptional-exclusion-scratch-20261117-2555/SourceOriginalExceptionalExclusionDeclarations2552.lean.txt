import SourceGammaOneOriginalNonexceptionalInputs
import Mathlib.Tactic

open Lean
private def inspectDeclarations : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceExceptionalProjectiveExponent, `SourceOriginalExceptionalScalarPower, `SourceExceptionalTraceChebyshev, `SourceExceptionalNormalizedSquare, `SourceOriginalExceptionalTraceEquation, `SourceExceptionalIntegralPolynomial, `SourceExceptionalPolynomialConstant, `SourceOriginalExceptionalPolynomialRoot, `SourceExceptionalRootArithmetic, `SourceExceptionalValuationParity, `SourceExceptionalNumberFieldRoot, `SourceExceptionalCoefficientNonvanishing, `SourceExceptionalUnramifiedPrime, `SourceExceptionalCoefficientPrimeSupport, `SourceOriginalIdentityFrobenius, `SourceOriginalLocalNonzeroPrimeCoefficient, `SourceOriginalNonzeroPrimeCoefficient, `SourceOriginalExceptionalPrimeObstruction, `SourceNonzeroAlgebraicIntegerRelation, `SourceIntegerRelationReduction, `SourceIntegralPolynomialRatio, `SourceOriginalExceptionalIntegerObstruction, `SourceOriginalExceptionalFixedObstruction, `SourceOriginalFixedLocalExceptionalExclusion, `SourceOriginalExceptionalUniformBound, `SourceNonexceptionalSmallAlternatives, `SourceOriginalNonexceptionalInputsProperty, `SourceGammaOneOriginalNonexceptionalInputs]
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      let m := env.allImportedModuleNames[idx.toNat]!
      if mods.contains m then
        if info.isTheorem then logInfo m!"OWNER {m}: {name}"
        if !name.isInternal && !isPrivateName name then
          if !(← Batteries.Tactic.Lint.isAutoDecl name) then
            let kind := if info.isTheorem then "THEOREM" else "CONSTRUCTION"
            logInfo m!"PUBLIC {kind} {m}: {name}"

#eval inspectDeclarations
