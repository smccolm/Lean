/-! DUB-17/18 -> exceptional original image exclusion -> a fixed finite set of
bad rational coefficient primes. Reuse the proved finiteness of actual
absolutely ramified places. Outside its literal contraction, actual primes
over q exist and discharge the unramified coefficient obstruction. -/
namespace Dubon2026.SourceDickson
noncomputable section
open Polynomial IsDedekindDomain
open scoped NumberField Classical
variable (F : Type) [Field F] [NumberField F]

/-- The rational primes below the actual ramified coefficient-field primes. -/
def exceptionalCoefficientRamifiedPrimes : Finset ℕ :=
  (original_finite_absolutely_ramified_places F).toFinset.image
    (fun v => Rat.HeightOneSpectrum.natGenerator (originalIntegerPrimeBelow F v))

/-- The constructed finite set contains every rational prime with an actual ramified prime above it. -/
theorem exceptional_coefficient_prime_unramified
    (q : ℕ) (hq : q.Prime) (v : HeightOneSpectrum (𝓞 F))
    [v.asIdeal.LiesOver (rationalPrimePlace q hq).asIdeal]
    (hs : q ∉ exceptionalCoefficientRamifiedPrimes F) :
    Algebra.IsUnramifiedAt ℤ v.asIdeal := by
  by_contra h
  apply hs
  apply Finset.mem_image.mpr
  refine ⟨v, (Set.Finite.mem_toFinset _).mpr h, ?_⟩
  have hv : originalIntegerPrimeBelow F v = rationalPrimePlace q hq := by
    apply HeightOneSpectrum.asIdeal_injective
    exact (Ideal.over_def v.asIdeal (rationalPrimePlace q hq).asIdeal).symm
  rw [hv, rationalPrimePlace_natGenerator]

/-- Every rational prime outside the constructed support has an actual unramified coefficient-field prime above it. -/
theorem exists_exceptional_unramified_coefficient_prime
    (q : ℕ) (hq : q.Prime) (hs : q ∉ exceptionalCoefficientRamifiedPrimes F) :
    ∃ v : HeightOneSpectrum (𝓞 F),
      v.asIdeal.LiesOver (rationalPrimePlace q hq).asIdeal ∧
      Algebra.IsUnramifiedAt ℤ v.asIdeal := by
  letI : Algebra.IsIntegral ℤ (𝓞 F) := Algebra.IsIntegral.of_finite ℤ (𝓞 F)
  obtain ⟨P, hP, hp⟩ := Ideal.exists_maximal_ideal_liesOver_of_isIntegral
    (S := 𝓞 F) (rationalPrimePlace q hq).asIdeal
  letI : P.IsMaximal := hP
  letI : P.LiesOver (rationalPrimePlace q hq).asIdeal := hp
  have hne : P ≠ ⊥ := Ideal.ne_bot_of_mem_primesOver
    (rationalPrimePlace q hq).ne_bot ⟨hP.isPrime, hp⟩
  let v : HeightOneSpectrum (𝓞 F) := ⟨P, hP.isPrime, hne⟩
  refine ⟨v, hp, ?_⟩
  exact exceptional_coefficient_prime_unramified F q hq v hs

/-- Outside a fixed finite ramification support, every nonzero odd-weight coefficient has a nonzero fixed exceptional polynomial value. -/
theorem exceptional_coefficient_ratio_nonvanishing_outside_support
    (q : ℕ) (hq : q.Prime) (hs : q ∉ exceptionalCoefficientRamifiedPrimes F)
    (hqconst : ¬ (q : ℤ) ∣ (-14400 : ℤ)) (a : F) (ha : a ≠ 0)
    (w : ℕ) (hw : Odd w) :
    exceptionalNonzeroTracePolynomial.eval₂ (Int.castRingHom F) (a ^ 2 / (q : F) ^ w) ≠ 0 := by
  obtain ⟨v, hv, hu⟩ := exists_exceptional_unramified_coefficient_prime F q hq hs
  letI := hv
  letI := hu
  exact exceptional_unramified_coefficient_ratio_nonvanishing q hq v hqconst a ha w hw

end
