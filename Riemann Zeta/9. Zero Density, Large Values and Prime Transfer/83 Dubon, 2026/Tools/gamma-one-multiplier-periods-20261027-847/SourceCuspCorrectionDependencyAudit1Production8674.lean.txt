private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceCuspCorrection.engine]
  let localModules : Array Name := #[`Dubon2026.SourceCuspCorrectionEdgeChains, `Dubon2026.SourceCuspCorrectionCosetChains, `Dubon2026.SourceCuspCorrectionGenerators, `Dubon2026.SourceCuspCorrectionCyclicCosets, `Dubon2026.SourceCuspCorrectionEdgeCharacters, `Dubon2026.SourceCuspCorrectionRealization]
  let required : Array Name := #[`Dubon2026.SourceCuspCorrection.lTransl, `Dubon2026.SourceCuspCorrection.lTranslMulAut, `Dubon2026.SourceCuspCorrection.chainSect, `Dubon2026.SourceCuspCorrection.edgeChain, `Dubon2026.SourceCuspCorrection.edgeChain_mul, `Dubon2026.SourceCuspCorrection.edgeChainHom, `Dubon2026.SourceCuspCorrection.edgeChainHom_conj, `Dubon2026.SourceCuspCorrection.edgeChainHom_genAt_raw, `Dubon2026.SourceCuspCorrection.enum_injOn_range, `Dubon2026.SourceCuspCorrection.exists_enum_eq_of_cusp_eq, `Dubon2026.SourceCuspCorrection.sum_ite_enum_eq, `Dubon2026.SourceCuspCorrection.edgeChainHom_genAt_apply, `Dubon2026.SourceCuspCorrection.edgeChar_genAt, `Dubon2026.SourceCuspCorrection.single_sub_single_mem_span_of_smul, `Dubon2026.SourceCuspCorrection.single_sub_single_mem_span_cuspDiffAt, `Dubon2026.SourceCuspCorrection.exists_addChar_of_mem_span, `Dubon2026.SourceCuspCorrection.closure_sA_τ, `Dubon2026.SourceCuspCorrection.engine]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE CUSP CORRECTION DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the declaration-VALUE closure of the single actual prescribed-cusp-character consumer engine. Actual finite edge chains, parabolic coset cycles, generator values and integer span realization are consumed. Integral character lifting, unitary logarithms, analytic reciprocity and full arithmetic association remain; 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
