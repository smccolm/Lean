import SourceUnitaryCharacterFiniteOrder
import SourceUnitaryCharacterRecurrence
import SourceUnitaryCharacterTraceBound
import SourceUnitaryCharacterAbelianQuotient
import SourceUnitaryCharacterBasisExt
import SourceUnitaryCharacterTorsion
import SourceUnitaryCircleCharacter
import SourceUnitaryCharacterLogarithm
import SourceGammaOneUnitaryCharacterPeriods
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceUnitaryCharacterFiniteOrder, `SourceUnitaryCharacterRecurrence, `SourceUnitaryCharacterTraceBound, `SourceUnitaryCharacterAbelianQuotient, `SourceUnitaryCharacterBasisExt, `SourceUnitaryCharacterTorsion, `SourceUnitaryCircleCharacter, `SourceUnitaryCharacterLogarithm, `SourceGammaOneUnitaryCharacterPeriods]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
