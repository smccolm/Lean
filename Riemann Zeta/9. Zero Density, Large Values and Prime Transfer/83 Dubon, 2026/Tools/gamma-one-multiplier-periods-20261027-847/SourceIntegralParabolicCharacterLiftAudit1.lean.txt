private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceIntegralCharacterCrossed, `SourceIntegralCharacterPotential, `SourceIntegralCharacterShapiro, `SourceIntegralCharacterFreeProduct, `SourceIntegralCharacterInitialLift, `SourceIntegralCharacterCuspConjugacy, `SourceIntegralCharacterCuspSum, `SourceIntegralCharacterCorrection, `SourceIntegralCharacterModularFacts, `SourceIntegralCharacterModularBridge, `SourceIntegralParabolicCharacterLift, `SourceGammaOneIntegralParabolicLift]

`Dubon2026.SourceIntegralCharacterLift.tr_add, `Dubon2026.SourceIntegralCharacterLift.tr_sub, `Dubon2026.SourceIntegralCharacterLift.tr_neg, `Dubon2026.SourceIntegralCharacterLift.tr_zero, `Dubon2026.SourceIntegralCharacterLift.tr_zsmul, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.map_one, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.map_inv, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.map_inv_apply, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.conj_apply, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.add, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.neg, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.sub, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.zsmul, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.map, `Dubon2026.SourceIntegralCharacterLift.cob_apply, `Dubon2026.SourceIntegralCharacterLift.isCrossed_cob, `Dubon2026.SourceIntegralCharacterLift.cob_apply_of_smul_eq, `Dubon2026.SourceIntegralCharacterLift.cob_map, `Dubon2026.SourceIntegralCharacterLift.exists_potential, `Dubon2026.SourceIntegralCharacterLift.trAut_apply, `Dubon2026.SourceIntegralCharacterLift.trHom_apply, `Dubon2026.SourceIntegralCharacterLift.smul_coe, `Dubon2026.SourceIntegralCharacterLift.smul_bs, `Dubon2026.SourceIntegralCharacterLift.smul_bs_eq_bs_iff, `Dubon2026.SourceIntegralCharacterLift.smul_bs_of_mem, `Dubon2026.SourceIntegralCharacterLift.char_congr, `Dubon2026.SourceIntegralCharacterLift.char_mul, `Dubon2026.SourceIntegralCharacterLift.char_inv, `Dubon2026.SourceIntegralCharacterLift.char_conj, `Dubon2026.SourceIntegralCharacterLift.evalBase_apply, `Dubon2026.SourceIntegralCharacterLift.evalBase_map, `Dubon2026.SourceIntegralCharacterLift.evalBase_sub, `Dubon2026.SourceIntegralCharacterLift.evalBase_zsmul, `Dubon2026.SourceIntegralCharacterLift.evalBase_conj, `Dubon2026.SourceIntegralCharacterLift.conj_mem_iff_smul, `Dubon2026.SourceIntegralCharacterLift.shapiro_apply, `Dubon2026.SourceIntegralCharacterLift.isCrossed_shapiro, `Dubon2026.SourceIntegralCharacterLift.evalBase_shapiro, `Dubon2026.SourceIntegralCharacterLift.IsCrossed.ext_of, `Dubon2026.SourceIntegralCharacterLift.totalHom_right, `Dubon2026.SourceIntegralCharacterLift.isCrossed_liftCrossed, `Dubon2026.SourceIntegralCharacterLift.exists_sub_smul_apply_conj_zpow_eq_zero, `Dubon2026.SourceIntegralCharacterLift.neg_one_mem_center, `Dubon2026.SourceIntegralCharacterLift.eq_one_or_eq_neg_one_of_mem_center, `Dubon2026.SourceIntegralCharacterLift.S_mul_S, `Dubon2026.SourceIntegralCharacterLift.trace_conj_T_zpow, `Dubon2026.SourceIntegralCharacterLift.rho_neg, `Dubon2026.SourceIntegralCharacterLift.eq_or_eq_neg_of_rho_eq, `Dubon2026.SourceIntegralCharacterLift.isOfFinOrder_of_rho, `Dubon2026.SourceIntegralCharacterLift.isOfFinOrder_of_rho_eq_conj_of, `Dubon2026.SourceIntegralCharacterLift.exists_rho_eq_conj_tP_zpow,   let constructions : Array Name := #[`Dubon2026.SourceIntegralCharacterLift.tr, `Dubon2026.SourceIntegralCharacterLift.IsCrossed, `Dubon2026.SourceIntegralCharacterLift.cob, `Dubon2026.SourceIntegralCharacterLift.trAut, `Dubon2026.SourceIntegralCharacterLift.trHom, `Dubon2026.SourceIntegralCharacterLift.bs, `Dubon2026.SourceIntegralCharacterLift.evalBase, `Dubon2026.SourceIntegralCharacterLift.shapiro, `Dubon2026.SourceIntegralCharacterLift.genHom, `Dubon2026.SourceIntegralCharacterLift.totalHom, `Dubon2026.SourceIntegralCharacterLift.liftCrossed, `Dubon2026.SourceIntegralCharacterLift.M23, `Dubon2026.SourceIntegralCharacterLift.G23, `Dubon2026.SourceIntegralCharacterLift.tP]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Integral Parabolic Character Lift postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Integral Parabolic Character Lift theorems discovered"
  logInfo m!"SOURCE INTEGRAL PARABOLIC CHARACTER LIFT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
