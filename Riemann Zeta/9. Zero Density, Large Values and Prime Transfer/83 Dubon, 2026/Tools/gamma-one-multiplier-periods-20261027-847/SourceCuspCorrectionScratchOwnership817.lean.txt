import SourceCuspCorrectionEdgeChains
import SourceCuspCorrectionCosetChains
import SourceCuspCorrectionGenerators
import SourceCuspCorrectionCyclicCosets
import SourceCuspCorrectionEdgeCharacters
import SourceCuspCorrectionRealization
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceCuspCorrectionEdgeChains, `SourceCuspCorrectionCosetChains, `SourceCuspCorrectionGenerators, `SourceCuspCorrectionCyclicCosets, `SourceCuspCorrectionEdgeCharacters, `SourceCuspCorrectionRealization]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
