theorem sourceUpperHalfPlane_primitive_difference
    {F G h : ℍ → ℂ}
    (hF : ∀ z : ℍ, HasDerivAt (F ∘ ofComplex) (h z) (z : ℂ))
    (hG : ∀ z : ℍ, HasDerivAt (G ∘ ofComplex) (h z) (z : ℂ))
    (a b : ℍ) : F b - F a = G b - G a := by
  have hlc : IsLocallyConstant (fun z : ℍ => F z - G z) := by
    apply (IsLocallyConstant.iff_eventually_eq _).mpr
    intro z
    have h₁ : ∀ᶠ w : ℂ in 𝓝 (z : ℂ),
        HasDerivAt (F ∘ ofComplex) (h (ofComplex w)) w := by
      filter_upwards [isOpen_upperHalfPlaneSet.mem_nhds z.im_pos] with w hw
      simpa only [ofComplex_apply_of_im_pos hw] using hF (⟨w, hw⟩ : ℍ)
    have h₂ : ∀ᶠ w : ℂ in 𝓝 (z : ℂ),
        HasDerivAt (G ∘ ofComplex) (h (ofComplex w)) w := by
      filter_upwards [isOpen_upperHalfPlaneSet.mem_nhds z.im_pos] with w hw
      simpa only [ofComplex_apply_of_im_pos hw] using hG (⟨w, hw⟩ : ℍ)
    have he := continuous_coe.continuousAt.eventually
      (SourceFunctionField.OriginalPrimitiveGlue.eventually_sub_eq_of_hasDerivAt h₁ h₂)
    simpa only [Function.comp_apply, ofComplex_apply] using he
  have he := hlc.apply_eq_of_preconnectedSpace b a
  change F b - G b = F a - G a at he
  linear_combination he

variable (Γ : Subgroup SL(2, ℤ)) [Γ.FiniteIndex]

/-- Actual endpoint integral with the original weight-two primitive normalization. -/
def sourceWeightTwoAlong (a b : ℍ) :
    Module.Dual ℂ (CuspForm (Γ.map (mapGL ℝ)) 2) where
  toFun f := sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty f b -
    sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty f a
  map_add' f g := by
    have hd := fun f z => sourceWeightTwoPrimitive_hasDerivAt
      (Γ := Γ.map (mapGL ℝ)) f (Γ.map (mapGL ℝ)).strictWidthInfty_pos
      (Γ.map (mapGL ℝ)).strictWidthInfty_mem_strictPeriods z
    have he := sourceUpperHalfPlane_primitive_difference
      (G := fun z => sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty f z +
        sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty g z) (hd (f + g))
      (fun z => (hd f z).add (hd g z)) a b
    convert he using 1 <;> simp only [Function.comp_apply, Pi.add_apply] <;> ring
  map_smul' c f := by
    have hd := fun f z => sourceWeightTwoPrimitive_hasDerivAt
      (Γ := Γ.map (mapGL ℝ)) f (Γ.map (mapGL ℝ)).strictWidthInfty_pos
      (Γ.map (mapGL ℝ)).strictWidthInfty_mem_strictPeriods z
    have he := sourceUpperHalfPlane_primitive_difference
      (G := fun z => c * sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty f z)
      (hd (c • f))
      (fun z => (hd f z).const_mul c) a b
    convert he using 1 <;> simp only [Function.comp_apply, Pi.smul_apply,
      smul_eq_mul, RingHom.id_apply] <;> ring

theorem sourceWeightTwoAlong_apply (a b : ℍ)
    (f : CuspForm (Γ.map (mapGL ℝ)) 2) :
    sourceWeightTwoAlong Γ a b f =
      sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty f b -
        sourceWeightTwoPrimitive (Γ.map (mapGL ℝ)).strictWidthInfty f a := rfl

theorem sourceWeightTwoAlong_smul (γ : Γ) (a : ℍ)
    (f : CuspForm (Γ.map (mapGL ℝ)) 2) :
    sourceWeightTwoAlong Γ a (γ.val • a) f = sourceWeightTwoPeriodValue Γ f γ := by
  rw [sourceWeightTwoAlong_apply, sourceWeightTwoPrimitive_period]
  ring

theorem sourceWeightTwoAlong_eq_sub (a b : ℍ) :
    sourceWeightTwoAlong Γ a b =
      sourceWeightTwoAlong Γ UpperHalfPlane.I b -
        sourceWeightTwoAlong Γ UpperHalfPlane.I a := by
  ext f
  simp only [LinearMap.sub_apply, sourceWeightTwoAlong_apply]
  ring

end
