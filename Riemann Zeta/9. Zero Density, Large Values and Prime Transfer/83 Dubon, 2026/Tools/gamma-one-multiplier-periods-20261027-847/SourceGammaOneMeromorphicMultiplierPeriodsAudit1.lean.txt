private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceModularEllipticFixedPoint, `SourceMeromorphicMultiplierParabolic, `SourceMeromorphicMultiplierGerm, `SourceGammaOneMeromorphicMultiplierElliptic, `SourceGammaOneMeromorphicMultiplierPeriods]
  let theorems : Array Name := #[`Dubon2026.SourceMeromorphicMultiplier.exists_smul_eq_self_of_trace_sq_lt_four, `Dubon2026.SourceMeromorphicMultiplier.tendsto_vadd_atImInfty, `Dubon2026.SourceMeromorphicMultiplier.conj_T_zpow_smul, `Dubon2026.SourceMeromorphicMultiplier.exists_apply_ne_zero, `Dubon2026.SourceMeromorphicMultiplier.multiplier_one, `Dubon2026.SourceMeromorphicMultiplier.multiplier_mul, `Dubon2026.SourceMeromorphicMultiplier.multiplier_eq_one_of_trace_sq, `Dubon2026.SourceMeromorphicMultiplier.multiplier_eq_denom_zpow, `Dubon2026.SourceMeromorphicMultiplier.gammaOne_multiplier_eq_one_of_trace_sq_lt, `Dubon2026.sourceGamma1_meromorphic_multiplier_eq_exp_real_period]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Meromorphic Multiplier Periods postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Meromorphic Multiplier Periods theorems discovered"
  logInfo m!"SOURCE GAMMA ONE MEROMORPHIC MULTIPLIER PERIODS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
