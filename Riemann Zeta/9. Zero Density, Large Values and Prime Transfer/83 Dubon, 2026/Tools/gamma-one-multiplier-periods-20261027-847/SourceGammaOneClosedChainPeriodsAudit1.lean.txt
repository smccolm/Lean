private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceWeightTwoEndpointPeriods, `SourceWeightTwoIntegralPeriods, `SourceWeightTwoClosedChainPeriods, `SourceWeightTwoSegmentPeriods, `SourceGammaOneClosedChainIntegral]
  let theorems : Array Name := #[`Dubon2026.sourceUpperHalfPlane_primitive_difference, `Dubon2026.sourceWeightTwoAlong_apply, `Dubon2026.sourceWeightTwoAlong_smul, `Dubon2026.sourceWeightTwoAlong_eq_sub, `Dubon2026.sourceWeightTwoDualPeriod_apply, `Dubon2026.sourceWeightTwoDualPeriod_mem, `Dubon2026.sourceWeightTwoAlong_smul_sub, `Dubon2026.sourceWeightTwoAlong_quotient_smul, `Dubon2026.sourceWeightTwoOrbitPeriod_mk, `Dubon2026.sourceWeightTwo_chain_period_mem_of_boundary_eq_zero, `Dubon2026.sourceWeightTwoOrbitBoundary_apply, `Dubon2026.sourceWeightTwo_chain_period_mem_of_orbit_boundary, `Dubon2026.sourceUpperHalfPlane_segment_im_pos, `Dubon2026.sourceWeightTwoAlong_eq_segment_integral, `Dubon2026.sourceGamma1_closed_chain_integrals_mem]
  let constructions : Array Name := #[`Dubon2026.sourceWeightTwoAlong, `Dubon2026.sourceWeightTwoDualPeriod, `Dubon2026.sourceWeightTwoIntegralLattice, `Dubon2026.sourceWeightTwoOrbitPeriod, `Dubon2026.sourceWeightTwoOrbitBoundary]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Closed Chain Periods postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Closed Chain Periods theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CLOSED CHAIN PERIODS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
