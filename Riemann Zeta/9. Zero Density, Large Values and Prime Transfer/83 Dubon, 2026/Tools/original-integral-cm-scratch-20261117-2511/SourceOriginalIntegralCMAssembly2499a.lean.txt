/-- The original integral pair and odd cyclotomic determinant construct the
CM field and native complete local deformation rings on literal enlarged support.
No separate field, multiplier, oddness or determinant-lifting input remains. -/
theorem original_integral_constructed_cm_deformations
    {O : Type} [CommRing O] [NoZeroDivisors O] [IsLocalRing O] [IsNoetherianRing O]
    [Finite (ResidueField O)] [TopologicalSpace (ResidueField O)]
    (a : 𝓞 ℚ) (ha : a ≠ 0) (p w : ℕ) [Fact p.Prime]
    [CharP (ResidueField O) p] (hp2 : 2 < p) (hpa : (p : 𝓞 ℚ) ∣ a) (hw : Odd w)
    (ρi : rationalArithmeticGaloisGroup a →* GeneralLinearGroup (Fin 2) O)
    (ρr : rationalArithmeticGaloisGroup a →* GeneralLinearGroup (Fin 2) (ResidueField O))
    (hred : ∀ g i j, (ρr g).val i j = residue O ((ρi g).val i j))
    (hdet : GeneralLinearGroup.det.comp ρr =
      (sourceArithmeticCyclotomicUnitCharacter (F := ResidueField O) p a hpa).toMonoidHom ^ w) :
    HasOriginalIntegralCMDeformations a p ρi ρr := by
  refine ⟨inferInstance, inferInstance, ?_⟩
  intro hDiscrete hρr
  letI := hDiscrete
  have ha4 : (4 * a : 𝓞 ℚ) ≠ 0 := mul_ne_zero (by norm_num) ha
  have hp4 : (p : 𝓞 ℚ) ∣ 4 * a := hpa.trans (dvd_mul_left a 4)
  obtain ⟨C, hC, hComplex⟩ :=
    original_unramified_exists_cm_quadratic_of_four_dvd (4 * a) (dvd_mul_right 4 a)
  letI := hC
  letI := hComplex
  let ρs : rationalArithmeticGaloisGroup (4 * a) →ₜ* GeneralLinearGroup (Fin 2) (ResidueField O) :=
    (show rationalArithmeticGaloisGroup a →ₜ* GeneralLinearGroup (Fin 2) (ResidueField O)
      from ⟨ρr, hρr⟩).comp (rationalArithmeticSupportRestriction a (4 * a) (dvd_mul_left a 4))
  let π := (rationalArithmeticSupportRestriction a (4 * a) (dvd_mul_left a 4)).comp
    (originalRationalBottomRestriction (4 * a))
  let ρib := ρi.comp π.toMonoidHom
  let ρrb := ρs.comp (originalRationalBottomRestriction (4 * a))
  have hdets : GeneralLinearGroup.det.comp ρs.toMonoidHom =
      (sourceArithmeticCyclotomicUnitCharacter (F := ResidueField O) p (4 * a) hp4).toMonoidHom ^ w :=
    original_support_residual_determinant a (4 * a) (dvd_mul_left a 4) p w hpa hp4 ρr hdet
  have hoddb : (ρrb (finiteBaseRealConjugation (4 * a) ⊥)).val.det = -1 :=
    original_rational_bottom_odd_of_cyclotomic (4 * a) p w hp4 hw
      ρs.toMonoidHom ρs.continuous hdets
  have hredb : ∀ g i j, (ρrb g).val i j = residue O ((ρib g).val i j) := by
    intro g i j
    exact hred (π g) i j
  refine ⟨ha4, hp4, C, hC, hComplex, hoddb, ?_⟩
  intro n hn
  exact original_integral_cm_exists_deformation_ring (4 * a) ⊥ C p n hp2 hn
    ρib ρrb hredb hoddb ha4 hp4

end
