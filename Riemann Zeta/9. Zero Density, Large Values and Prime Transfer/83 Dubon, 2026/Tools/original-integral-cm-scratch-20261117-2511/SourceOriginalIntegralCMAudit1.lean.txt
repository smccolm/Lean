private def audit : CoreM Unit := do
  let env ← getEnv
  let modules : Array Name := #[`SourceOriginalIntegralOdd, `SourceOriginalIntegralCMSimilitude, `SourceOriginalIntegralCMDeformationRing, `SourceOriginalIntegralCMRingData, `SourceOriginalSupportBottomOdd, `SourceOriginalIntegralCMSupportField, `SourceOriginalIntegralCMProperty, `SourceOriginalIntegralCMAssembly, `SourceGammaOneOriginalIntegralCM]
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let mut nt := 0
  let mut nd := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Project postulate: {name}"
        let axioms ← Lean.collectAxioms name
        let forbidden := axioms.filter fun a => !allowed.contains a
        unless forbidden.isEmpty do throwError "Forbidden dependencies: {name}: {forbidden}"
        if info.isTheorem then nt := nt + 1 else nd := nd + 1
  logInfo m!"SOURCE ORIGINAL INTEGRAL CM AXIOM AUDIT PASS: {nt} theorems and {nd} other declarations, including private/generated declarations. One actual original residual action has the image alternatives, original symmetric spanning, absolute irreducibility, adjoint H0 vanishing and actual arithmetic twisted-kernel detection. Arithmetic image exclusions and remaining local and auxiliary-prime conditions remain OPEN."
