/-- Preserve the original rational scalar structure. -/
private local instance originalSupportBottomOddAlgebra (a : 𝓞 ℚ) :
    Algebra ℚ (rationalUnramifiedExtension a) := (rationalUnramifiedExtension a).algebra'

/-- The actual rational bottom field is totally real. -/
private local instance originalSupportBottomOddReal (a : 𝓞 ℚ) :
    IsTotallyReal (⊥ : IntermediateField ℚ (rationalUnramifiedExtension a)) :=
  IsTotallyReal.ofRingEquiv (IntermediateField.botEquiv ℚ (rationalUnramifiedExtension a)).symm.toRingEquiv

/-- The literal support and bottom restriction of the original residual action
has the odd real determinant derived from its original cyclotomic determinant. -/
theorem original_support_bottom_cyclotomic_odd
    (a : 𝓞 ℚ) (p w : ℕ) [Fact p.Prime] (hpa : (p : 𝓞 ℚ) ∣ a) (hw : Odd w)
    {L : Type} [Field L] [CharP L p] [TopologicalSpace L] [DiscreteTopology L]
    (ρ : rationalArithmeticGaloisGroup a →* GeneralLinearGroup (Fin 2) L) (hρ : Continuous ρ)
    (hdet : GeneralLinearGroup.det.comp ρ =
      (sourceArithmeticCyclotomicUnitCharacter (F := L) p a hpa).toMonoidHom ^ w) :
    let π := (rationalArithmeticSupportRestriction a (4 * a) (dvd_mul_left a 4)).comp
      (originalRationalBottomRestriction (4 * a))
    let ρb := (show rationalArithmeticGaloisGroup a →ₜ* GeneralLinearGroup (Fin 2) L
      from ⟨ρ, hρ⟩).comp π
    (ρb (finiteBaseRealConjugation (4 * a) ⊥)).val.det = -1 := by
  let ρs : rationalArithmeticGaloisGroup (4 * a) →ₜ* GeneralLinearGroup (Fin 2) L :=
    (show rationalArithmeticGaloisGroup a →ₜ* GeneralLinearGroup (Fin 2) L from ⟨ρ, hρ⟩).comp
      (rationalArithmeticSupportRestriction a (4 * a) (dvd_mul_left a 4))
  have hp4 : (p : 𝓞 ℚ) ∣ 4 * a := hpa.trans (dvd_mul_left a 4)
  have hdets : GeneralLinearGroup.det.comp ρs.toMonoidHom =
      (sourceArithmeticCyclotomicUnitCharacter (F := L) p (4 * a) hp4).toMonoidHom ^ w :=
    original_support_residual_determinant a (4 * a) (dvd_mul_left a 4) p w hpa hp4 ρ hdet
  exact original_rational_bottom_odd_of_cyclotomic (4 * a) p w hp4 hw
    ρs.toMonoidHom ρs.continuous hdets

end
end Dubon2026.SourceDickson
