private def audit : CoreM Unit := do
  let env ← getEnv
  let modules : Array Name := #[`SourceFiniteSubfieldIrreducible, `SourceSubfieldSymmetricAbsolute, `SourceFiniteLinearSymmetric, `SourceGammaOneAbsoluteSymmetricResidual, `SourceSubfieldKernelIrreducible, `SourceFiniteLinearKernelSymmetric, `SourceGammaOneCyclotomicSymmetricResidual, `SourceFiniteLinearAbsoluteGalois, `SourceGammaOneAbsoluteCyclotomicResidual, `SourceResidualConjugateIrreducible, `SourceConjugateSymmetricSpanning, `SourceFiniteLinearOriginalSpanning, `SourceOriginalSymmetricSpanningProperty, `SourceGammaOneSymmetricKernelSpanning, `SourceSpanningIrreducibility, `SourceSpanningCoefficientExtension, `SourceOriginalSymmetricAbsolute, `SourceGammaOneResidueAbsoluteSymmetric]
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let mut nt := 0
  let mut nd := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Project postulate: {name}"
        let axioms ← Lean.collectAxioms name
        let forbidden := axioms.filter fun a => !allowed.contains a
        unless forbidden.isEmpty do throwError "Forbidden dependencies: {name}: {forbidden}"
        if info.isTheorem then nt := nt + 1 else nd := nd + 1
  logInfo m!"SOURCE ORIGINAL SYMMETRIC AXIOM AUDIT PASS: {nt} theorems and {nd} other declarations, including private/generated declarations. The original-form consumer derives absolute symmetric-power irreducibility in the large-image branch over actual finite coefficient extensions and bases, including restriction to the original arithmetic and absolute Galois cyclotomic kernels. Arithmetic large-image exclusions remain OPEN."

#eval audit
