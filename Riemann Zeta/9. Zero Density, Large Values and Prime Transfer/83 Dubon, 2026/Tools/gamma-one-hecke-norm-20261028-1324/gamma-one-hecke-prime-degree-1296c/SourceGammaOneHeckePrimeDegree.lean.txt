variable (L : Type*) [Field L] [Algebra ℚ L]
variable (M p : ℕ) [NeZero M] [Fact p.Prime]

theorem gammaOne_over_prime_relfinrank (hcop : Nat.Coprime p M) :
    relfinrank (functionFieldOver (Gamma1 M) L)
      (functionFieldOver (Gamma1 M ⊓ Gamma0 (p * M)) L) = p + 1 := by
  have hj : DegreeTransport.K L jq ≤ functionFieldOver (Gamma1 M) L :=
    DegreeTransport.K_le L (jq_mem_qExpFunctionField (Gamma1 M))
  have hinc : functionFieldOver (Gamma1 M) L ≤
      functionFieldOver (Gamma1 M ⊓ Gamma0 (p * M)) L :=
    gamma1_hecke_baseChange_le M (p * M) L
  have h := relfinrank_mul_relfinrank hj hinc
  rw [gammaOne_coordinate_relfinrank_eq_index M L,
    gammaOne_hecke_coordinate_relfinrank_eq_index M p L,
    gammaOne_prime_signed_hecke_index M p hcop,
    Nat.mul_comm (p + 1)] at h
  exact Nat.eq_of_mul_eq_mul_left
    (Nat.pos_of_ne_zero (Subgroup.FiniteIndex.index_ne_zero (H := plusMinus (Gamma1 M)))) h

theorem heckeAlpha_finrank (hcop : Nat.Coprime p M) :
    letI := SourceCurve.algebraAlong (heckeAlpha L M p)
    letI : Module (laurentBaseChange L (x1FunctionField M))
        (laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))) := Algebra.toModule
    Module.finrank (laurentBaseChange L (x1FunctionField M))
      (laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))) = p + 1 := by
  letI := SourceCurve.algebraAlong (heckeAlpha L M p)
  letI : Module (laurentBaseChange L (x1FunctionField M))
      (laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))) := Algebra.toModule
  have h := gammaOne_over_prime_relfinrank L M p hcop
  rw [relfinrank_eq_finrank_of_le (gamma1_hecke_baseChange_le M (p * M) L)] at h
  exact h

end
