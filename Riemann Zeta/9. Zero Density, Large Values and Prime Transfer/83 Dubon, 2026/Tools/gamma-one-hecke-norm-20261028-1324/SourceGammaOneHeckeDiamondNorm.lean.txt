/-! DUB-17/18 -> original Gamma1 geometric Tate Frobenius -> norm
reduction. Assemble the actual reverse automorphism, its precise original

namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open CongruenceSubgroup
open scoped MatrixGroups
local notation "Qbar" => AlgebraicClosure ℚ
local instance qbarNormAlgebraic : Algebra.IsAlgebraic ℚ Qbar :=
  AlgebraicClosure.isAlgebraic ℚ
variable (M p : ℕ) [NeZero M] [Fact p.Prime]
local notation "FF" => laurentBaseChange Qbar (x1FunctionField M)
local notation "GG" => laurentBaseChange Qbar (x1x0FunctionFieldC ℚ M (p * M))

theorem exists_gammaOne_adjacentJ_arithmetic_diamond (hcop : Nat.Coprime M p) :
    ∃ g : Gamma0 M, ∃ w : GG ≃ₐ[Qbar] GG,
      (p : ZMod M) * (g.val 1 1 : ZMod M) = 1 ∧
      (∀ x : FF, w (heckeAlpha Qbar M p x) = heckeBeta Qbar M p x) ∧
      ∀ x : FF, w (heckeBeta Qbar M p x) =
        heckeAlpha Qbar M p (ConstantField.arithmeticDiamondAut M g x) := by
  obtain ⟨γ, hd⟩ := exists_adjacentJ_diamond_lift M p hcop
  obtain ⟨σ, hbase, hj⟩ := exists_gammaOne_adjacentJ_automorphism M p hcop
  exact ⟨⟨adjacentJDiamondMatrix p γ.val hd, adjacentJDiamondMatrix_mem M p γ hd⟩,
    geomAut Qbar (x1x0FunctionFieldC ℚ M (p * M)) σ,
    adjacentJDiamondMatrix_label M p γ hd,
    geomAut_adjacentJ_forward M p Qbar σ hbase,
    geomAut_adjacentJ_reverse M p hcop σ hbase hj γ hd⟩

attribute [local instance] HeckeNorm.alphaAlgebra HeckeNorm.alphaModule

theorem exists_gammaOne_hecke_diamond_norm_product (hcop : Nat.Coprime M p)
    (ζ : Qbarˣ) (hζ : IsPrimitiveRoot (ζ : Qbar) p) :
    ∃ g : Gamma0 M, ∃ w : GG ≃ₐ[Qbar] GG,
      (p : ZMod M) * (g.val 1 1 : ZMod M) = 1 ∧
      (∀ x : FF, w (heckeAlpha Qbar M p x) = heckeBeta Qbar M p x) ∧
      (∀ x : FF, w (heckeBeta Qbar M p x) =
        heckeAlpha Qbar M p (ConstantField.arithmeticDiamondAut M g x)) ∧
      ∀ v : GG, qExpand Qbar p (Algebra.norm FF v : LaurentSeries Qbar) =
        (∏ b ∈ Finset.range p, qTwist (ζ ^ b) (w v : LaurentSeries Qbar)) *
          qExpand Qbar p (v : LaurentSeries Qbar) := by
  obtain ⟨g, w, hlabel, hf, hb⟩ := exists_gammaOne_adjacentJ_arithmetic_diamond M p hcop
  have hj : w (heckeBeta Qbar M p (jCoordinateOver (Gamma1 M) Qbar)) =
      heckeAlpha Qbar M p (jCoordinateOver (Gamma1 M) Qbar) := by
    rw [hb]
    exact congrArg (heckeAlpha Qbar M p) (ConstantField.arithmeticDiamondHom_j M g)
  exact ⟨g, w, hlabel, hf, hb, HeckeNorm.qExpand_norm Qbar M p hcop ζ hζ w hf hj⟩

theorem exists_gammaOne_hecke_norm_beta_product (hcop : Nat.Coprime M p)
    (ζ : Qbarˣ) (hζ : IsPrimitiveRoot (ζ : Qbar) p) :
    ∃ g : Gamma0 M, (p : ZMod M) * (g.val 1 1 : ZMod M) = 1 ∧
      ∀ x : FF, qExpand Qbar p
          (Algebra.norm FF (heckeBeta Qbar M p x) : LaurentSeries Qbar) =
        (∏ b ∈ Finset.range p, qTwist (ζ ^ b)
          (ConstantField.arithmeticDiamondAut M g x : LaurentSeries Qbar)) *
            qExpand Qbar p (qExpand Qbar p (x : LaurentSeries Qbar)) := by
  obtain ⟨g, w, hlabel, _, hb, hn⟩ :=
    exists_gammaOne_hecke_diamond_norm_product M p hcop ζ hζ
  refine ⟨g, hlabel, fun x => ?_⟩
  simpa only [hb, heckeAlpha_coe, heckeBeta_coe] using hn (heckeBeta Qbar M p x)

end
