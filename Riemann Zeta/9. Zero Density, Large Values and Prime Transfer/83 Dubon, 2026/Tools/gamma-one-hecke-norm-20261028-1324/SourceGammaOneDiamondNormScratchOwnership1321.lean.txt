import SourceGammaOneAdjacentJDiamondMatrix
import SourceGammaOneAdjacentJDiamondAction
import SourceGammaOneRationalAnalyticGerms
import SourceGammaOneUpperHalfPlaneHeckeGerms
import SourceGammaOneAdjacentJAnalyticAction
import SourceGammaOneUpperHalfPlaneGermFaithfulness
import SourceGammaOneAdjacentJReverseDiamond
import SourceGammaOneAdjacentJArithmeticDiamond
import SourceGammaOneHeckeDiamondNorm
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneAdjacentJDiamondMatrix, `SourceGammaOneAdjacentJDiamondAction, `SourceGammaOneRationalAnalyticGerms, `SourceGammaOneUpperHalfPlaneHeckeGerms, `SourceGammaOneAdjacentJAnalyticAction, `SourceGammaOneUpperHalfPlaneGermFaithfulness, `SourceGammaOneAdjacentJReverseDiamond, `SourceGammaOneAdjacentJArithmeticDiamond, `SourceGammaOneHeckeDiamondNorm]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
