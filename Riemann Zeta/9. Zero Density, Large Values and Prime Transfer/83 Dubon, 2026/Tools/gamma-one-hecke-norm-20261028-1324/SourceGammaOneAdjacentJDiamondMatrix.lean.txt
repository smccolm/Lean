/-! DUB-17/18 -> original Gamma1 Frobenius norm reduction -> reverse-map
diamond identity. A coprime lift gamma in Gamma1(M) with p dividing its
lower-right entry gives the exact integral matrix delta representing
D_p gamma D_p / p. Its lower-right diamond label satisfies p*d = 1.
The analytic reverse-map comparison is the next required consumer. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open CongruenceSubgroup Matrix.SpecialLinearGroup
open scoped MatrixGroups

/-- The integral determinant-one matrix D_p gamma D_p / p. -/
def adjacentJDiamondMatrix (p : ℕ) (γ : SL(2, ℤ)) (hd : (p : ℤ) ∣ γ 1 1) : SL(2, ℤ) :=
  ⟨!![(p : ℤ) * γ 0 0, γ 0 1; γ 1 0, γ 1 1 / (p : ℤ)], by
    rw [Matrix.det_fin_two_of]
    linear_combination (Matrix.det_fin_two γ.val).symm.trans γ.property +
      γ 0 0 * Int.ediv_mul_cancel hd⟩

theorem adjacentJDiamondMatrix_mem (M p : ℕ) (γ : Gamma1 M)
    (hd : (p : ℤ) ∣ γ.val 1 1) : adjacentJDiamondMatrix p γ.val hd ∈ Gamma0 M := by
  rw [Gamma0_mem]
  exact (Gamma1_mem M γ.val).mp γ.property |>.2.2

theorem adjacentJDiamondMatrix_label (M p : ℕ) (γ : Gamma1 M)
    (hd : (p : ℤ) ∣ γ.val 1 1) :
    (p : ZMod M) * (adjacentJDiamondMatrix p γ.val hd 1 1 : ZMod M) = 1 := by
  have he : (p : ℤ) * (γ.val 1 1 / (p : ℤ)) = γ.val 1 1 := by
    rw [mul_comm, Int.ediv_mul_cancel hd]
  change (p : ZMod M) * ((γ.val 1 1 / (p : ℤ) : ℤ) : ZMod M) = 1
  have hc := congrArg (Int.castRingHom (ZMod M)) he
  simp only [map_mul, Int.coe_castRingHom, Int.cast_natCast] at hc
  rw [hc]
  exact (Gamma1_mem M γ.val).mp γ.property |>.2.1

theorem exists_adjacentJ_diamond_lift (M p : ℕ) [NeZero M] [NeZero p]
    (hcop : Nat.Coprime M p) : ∃ γ : Gamma1 M, (p : ℤ) ∣ γ.val 1 1 := by
  let red := Matrix.SpecialLinearGroup.map (n := Fin 2) (Int.castRingHom (ZMod p))
  obtain ⟨γ, hγ⟩ := gammaOneReduction_surjective M p hcop (red ModularGroup.S)
  have hc := congrArg (fun A : SL(2, ZMod p) => A 1 1) hγ
  have hz : (γ.val 1 1 : ZMod p) = 0 := by simpa [red, ModularGroup.S] using hc
  exact ⟨γ, (ZMod.intCast_zmod_eq_zero_iff_dvd _ _).mp hz⟩

end
end Dubon2026.SourceFunctionField.GammaOne
