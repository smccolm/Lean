import SourceGammaOneHeckePrimeDegree
import Dubon2026.SourcePrimeConjugateRoots
import Mathlib.FieldTheory.AlgebraicClosure

/-! DUB-17/18 -> original Gamma1 Frobenius norm -> complete embeddings.
Adapt the native actual-field norm embeddings to the original Gamma1
correspondence. Only the proved forward equation and reverse j-coordinate
equation are used. No involution or reverse equation on arbitrary lower
elements is presumed. The native source credits anthropics/fermats-last-theorem,
Apache-2.0, commit 6e837e75355538c7f80bab5b956861e86c4eacc2. -/
namespace Dubon2026.SourceFunctionField.GammaOne.HeckeNorm
noncomputable section
open IntermediateField PrimeDegreeAux
variable (L : Type*) [Field L] [Algebra ℚ L]
variable (M p : ℕ) [Fact p.Prime]
local notation "FF" => laurentBaseChange L (x1FunctionField M)
local notation "GG" => laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))
local notation "EE" => AlgebraicClosure (LaurentSeries L)

/-- The actual first degeneracy map supplies the source extension algebra. -/
@[reducible]
def alphaAlgebra : Algebra FF GG := SourceCurve.algebraAlong (heckeAlpha L M p)

/-- The actual second degeneracy map supplies the Laurent closure's lower-field algebra. -/
@[reducible]
def targetAlgebra : Algebra FF EE :=
  ((algebraMap (LaurentSeries L) EE).comp
    ((laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))).toSubfield.subtype.comp (heckeBeta L M p).toRingHom)).toAlgebra

attribute [local instance] alphaAlgebra

/-- The module induced by the actual alpha algebra. -/
@[reducible]
def alphaModule : Module FF GG := Algebra.toModule

attribute [local instance] alphaModule

/-- Twist the actual Atkin map to obtain an embedding over the original inclusion. -/
def twistEmbedding (w : GG ≃ₐ[L] GG)
    (hforward : ∀ x, w (heckeAlpha L M p x) = heckeBeta L M p x)
    (c : Lˣ) (hc : c ^ p = 1) :
    letI := targetAlgebra L M p; GG →ₐ[FF] EE := by
  letI := targetAlgebra L M p
  exact {
    toRingHom := (algebraMap (LaurentSeries L) EE).comp
      ((qTwist c).comp ((laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))).toSubfield.subtype.comp w.toAlgHom.toRingHom))
    commutes' x := by
      change algebraMap (LaurentSeries L) EE (qTwist c (w (heckeAlpha L M p x))) =
        algebraMap (LaurentSeries L) EE (heckeBeta L M p x)
      rw [hforward x, heckeBeta_coe, qTwist_qExpand_of_pow_eq_one p c hc] }

theorem lastEmbedding_commutes (x : FF) :
    ((algebraMap (LaurentSeries L) EE).comp
      ((qExpand L p).comp (laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))).toSubfield.subtype))
        (heckeAlpha L M p x) =
      algebraMap (LaurentSeries L) EE (heckeBeta L M p x) :=
  congrArg (algebraMap (LaurentSeries L) EE)
    ((congrArg (qExpand L p) (heckeAlpha_coe L M p x)).trans
      (heckeBeta_coe L M p x).symm)

/-- The remaining embedding is literal Laurent q-expansion. -/
def lastEmbedding : letI := targetAlgebra L M p; GG →ₐ[FF] EE := by
  letI := targetAlgebra L M p
  exact {
    toRingHom := (algebraMap (LaurentSeries L) EE).comp
      ((qExpand L p).comp (laurentBaseChange L (x1x0FunctionFieldC ℚ M (p * M))).toSubfield.subtype)
    commutes' := lastEmbedding_commutes L M p }

/-- The p twists and one expansion, indexed by the existing prime-conjugate indexing set. -/
def embedding (ζ : Lˣ) (hζ : IsPrimitiveRoot (ζ : L) p)
    (w : GG ≃ₐ[L] GG)
    (hforward : ∀ x, w (heckeAlpha L M p x) = heckeBeta L M p x) :
    Fin (p + 1) → (letI := targetAlgebra L M p; GG →ₐ[FF] EE) :=
  Fin.cases (lastEmbedding L M p) (fun b =>
    twistEmbedding L M p w hforward (ζ ^ (b : ℕ)) (by
      rw [← pow_mul, Nat.mul_comm (b : ℕ) p, pow_mul,
        units_pow_eq_one_of_isPrimitiveRoot hζ, one_pow]))

theorem embedding_zero_apply (ζ : Lˣ) (hζ : IsPrimitiveRoot (ζ : L) p)
    (w : GG ≃ₐ[L] GG)
    (hforward : ∀ x, w (heckeAlpha L M p x) = heckeBeta L M p x)
    (v : GG) : embedding L M p ζ hζ w hforward 0 v =
      algebraMap (LaurentSeries L) EE (qExpand L p v) := rfl

theorem embedding_succ_apply (ζ : Lˣ) (hζ : IsPrimitiveRoot (ζ : L) p)
    (w : GG ≃ₐ[L] GG)
    (hforward : ∀ x, w (heckeAlpha L M p x) = heckeBeta L M p x)
    (b : Fin p) (v : GG) : embedding L M p ζ hζ w hforward b.succ v =
      algebraMap (LaurentSeries L) EE (qTwist (ζ ^ (b : ℕ)) (w v)) := rfl

theorem embedding_jBeta (ζ : Lˣ) (hζ : IsPrimitiveRoot (ζ : L) p)
    (w : GG ≃ₐ[L] GG)
    (hforward : ∀ x, w (heckeAlpha L M p x) = heckeBeta L M p x)
    (hjback : w (heckeBeta L M p (jCoordinateOver (CongruenceSubgroup.Gamma1 M) L)) =
      heckeAlpha L M p (jCoordinateOver (CongruenceSubgroup.Gamma1 M) L))
    (i : Fin (p + 1)) :
    embedding L M p ζ hζ w hforward i (heckeBeta L M p (jCoordinateOver (CongruenceSubgroup.Gamma1 M) L)) =
      algebraMap (LaurentSeries L) EE (PhiGen.conj p ζ i) := by
  refine Fin.cases ?_ (fun b => ?_) i
  · rw [embedding_zero_apply, heckeBeta_coe, PhiGen.conj_zero, qExpand_qExpand]
    rfl
  · rw [embedding_succ_apply, hjback, heckeAlpha_coe, PhiGen.conj_succ]
    rfl

end
end Dubon2026.SourceFunctionField.GammaOne.HeckeNorm
