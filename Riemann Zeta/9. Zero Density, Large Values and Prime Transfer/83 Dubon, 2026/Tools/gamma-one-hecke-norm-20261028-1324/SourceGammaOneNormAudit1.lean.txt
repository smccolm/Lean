private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCoprimeGeneration, `SourceGammaOneAdjacentJRatio, `SourceGammaOneRealizationRestriction, `SourceGammaOneAdjacentJNonmembership, `SourceGammaOneAdjacentJGeneration, `SourceGammaOneAdjacentJPolynomial, `SourceGammaOneAdjacentJEmbedding, `SourceGammaOneAdjacentJPreservation, `SourceGammaOneAdjacentJAutomorphism, `SourceGammaOneAdjacentJCoefficients, `SourceGammaOneHeckeNormEmbeddings, `SourceGammaOneHeckeEmbeddingCensus, `SourceGammaOneHeckeNormProduct]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.gammaOne_sup_gamma0_coprime, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_modular_ratio, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_holomorphic, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_represents, `Dubon2026.SourceFunctionField.GammaOne.realize_restrict_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.continuous_invariant_of_realize_germs, `Dubon2026.SourceFunctionField.GammaOne.prime_j_not_mem_gammaOne_over_complex, `Dubon2026.SourceFunctionField.GammaOne.prime_j_not_mem_gammaOne, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_adjacentJ_finrank, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_adjacentJ_generation, `Dubon2026.SourceFunctionField.GammaOne.adjacentJPolynomial_root, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_isIntegral, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_minpoly_natDegree, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_minpoly_eq, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_minpoly_reverse_root, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_mem_correspondence, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_adjacentJ_embedding, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_embedding_preserves, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_adjacentJ_endomorphism, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_endomorphism_bijective, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_adjacentJ_automorphism, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_adjacentJ_over, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.lastEmbedding_commutes, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.embedding_zero_apply, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.embedding_succ_apply, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.embedding_jBeta, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.embedding_injective, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.embedding_bijective, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.qExpand_norm, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.exists_gammaOne_hecke_norm_product]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.adjacentJBase, `Dubon2026.SourceFunctionField.GammaOne.adjacentJPolynomial, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.alphaAlgebra, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.targetAlgebra, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.alphaModule, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.twistEmbedding, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.lastEmbedding, `Dubon2026.SourceFunctionField.GammaOne.HeckeNorm.embedding]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE NORM AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
