private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneAdjacentJDiamondMatrix, `SourceGammaOneAdjacentJDiamondAction, `SourceGammaOneRationalAnalyticGerms, `SourceGammaOneUpperHalfPlaneHeckeGerms, `SourceGammaOneAdjacentJAnalyticAction, `SourceGammaOneUpperHalfPlaneGermFaithfulness, `SourceGammaOneAdjacentJReverseDiamond, `SourceGammaOneAdjacentJArithmeticDiamond, `SourceGammaOneHeckeDiamondNorm]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.adjacentJDiamondMatrix_mem, `Dubon2026.SourceFunctionField.GammaOne.adjacentJDiamondMatrix_label, `Dubon2026.SourceFunctionField.GammaOne.exists_adjacentJ_diamond_lift, `Dubon2026.SourceFunctionField.GammaOne.adjacentJDiamondMatrix_gl, `Dubon2026.SourceFunctionField.GammaOne.adjacentJDiamondMatrix_smul, `Dubon2026.SourceFunctionField.GammaOne.rationalAnalyticGerm_constants, `Dubon2026.SourceFunctionField.GammaOne.realize_restrict_upperHalfPlane_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.realize_smul_upperHalfPlane_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.realize_heckeBeta_upperHalfPlane_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.realize_adjacentJ_upperHalfPlane_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_realize_base_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_realize_j_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_realize_upperHalfPlane_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.eq_of_realize_upperHalfPlane_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_reverse_diamond_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.adjacentJ_reverse_complex_diamond, `Dubon2026.SourceFunctionField.GammaOne.geomAut_adjacentJ_forward, `Dubon2026.SourceFunctionField.GammaOne.geomAut_adjacentJ_reverse_rational, `Dubon2026.SourceFunctionField.GammaOne.geomAut_adjacentJ_reverse, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_adjacentJ_arithmetic_diamond, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_hecke_diamond_norm_product, `Dubon2026.SourceFunctionField.GammaOne.exists_gammaOne_hecke_norm_beta_product, `Dubon2026.SourceFunctionField.GammaOne.qbarDiamondAlgebraic, `Dubon2026.SourceFunctionField.GammaOne.qbarNormAlgebraic]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.adjacentJDiamondMatrix, `Dubon2026.SourceFunctionField.GammaOne.upperHalfPlaneGermRatAlgebra, `Dubon2026.SourceFunctionField.GammaOne.rationalComplexMap, `Dubon2026.SourceFunctionField.GammaOne.upperHalfPlaneRealizeGerm, `Dubon2026.SourceFunctionField.GammaOne.rationalAnalyticGerm]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIAMOND NORM AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
