private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOnePrimeHeckeIndex, `SourceGammaOneSignedHeckeIndex, `SourceGammaOneHeckeCoordinateDegree, `SourceGammaOneHeckePrimeDegree]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.gammaOneReduction_surjective, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_lower_subgroup_eq_borel_comap, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_prime_lower_relIndex, `Dubon2026.SourceFunctionField.GammaOne.plusMinus_gammaOne_hecke_eq_inf, `Dubon2026.SourceFunctionField.GammaOne.gammaOneSignedReduction_surjective, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_signed_lower_subgroup_eq_borel_comap, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_prime_signed_lower_relIndex, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_prime_signed_hecke_index, `Dubon2026.SourceFunctionField.GammaOne.coordinate_finrank_eq_relfinrank, `Dubon2026.SourceFunctionField.GammaOne.coordinate_relfinrank_baseChange, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_coordinate_relfinrank_eq_index, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_hecke_coordinate_relfinrank_eq_index, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_over_prime_relfinrank, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha_finrank]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HECKE DEGREE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
