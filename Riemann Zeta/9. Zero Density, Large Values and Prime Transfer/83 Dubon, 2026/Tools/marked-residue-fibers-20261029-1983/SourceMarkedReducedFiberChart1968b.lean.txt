/-! DUB-17/18 -> reverse genus -> finite elliptic fibers. Transport the
actual inertia-invariant marked chart through the native torsion reduction
bijection. Full inertia image up to sign makes every actual exceptional
curve automorphism preserve the resulting residue-point fiber map.
The finite torsion-field constructions provide precisely these inputs. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open WeierstrassCurve SourceEllipticReduction SourceEllipticStabilizer SourceEllipticVariableChange

theorem exists_reduced_marked_fiber_chart
    {F K M Y : Type*} [Field F] [Field K] [Field M] [Algebra F M]
    [IsAlgClosed K] [DecidableEq K] [DecidableEq M]
    (E : WeierstrassCurve F) (E₀ : WeierstrassCurve K) [E₀.IsElliptic]
    (N : ℕ) (hNK : (N : K) ≠ 0) (w : ValuationSubring M)
    (θ : Submodule.torsionBy ℤ (E.baseChange M).toAffine.Point (N : ℤ) →+ E₀.toAffine.Point)
    (hθ : Function.Injective θ)
    (hcard : Nat.card {P : (E.baseChange M).toAffine.Point // N • P = 0} = N ^ 2)
    (f : {P : (E.baseChange M).toAffine.Point // addOrderOf P = N} → Y)
    (hf : Function.Surjective f)
    (hinertia : ∀ P Q, (∃ σ : w.inertiaSubgroup F,
      Q.val = Affine.Point.map (W' := E) σ.val.val.toAlgHom P.val ∨
        Q.val = -(Affine.Point.map (W' := E) σ.val.val.toAlgHom P.val)) → f Q = f P)
    (hlift : ∀ δ : VariableChange K, δ • E₀ = E₀ →
      ∃ σ : w.inertiaSubgroup F,
        (∀ T, HEq (vcInvFun δ E₀.toAffine (θ T)) (θ (galoisTorsionMap E N σ.val.val T))) ∨
        (∀ T, HEq (vcInvFun δ E₀.toAffine (θ T)) (-(θ (galoisTorsionMap E N σ.val.val T))))) :
    ∃ g : {P : E₀.toAffine.Point // addOrderOf P = N} → Y,
      Function.Surjective g ∧
      (∀ P Q, Q.val = -P.val → g Q = g P) ∧
      ∀ (δ : VariableChange K) (hδ : δ • E₀ = E₀) P Q,
        Q.val = stabilizerPointEquiv E₀ δ hδ P.val → g Q = g P := by
  obtain ⟨ψ, hψ⟩ := exists_exactOrder_reduction_equiv E₀ N hNK θ hθ hcard
  let t (P : {P : (E.baseChange M).toAffine.Point // addOrderOf P = N}) :
      Submodule.torsionBy ℤ (E.baseChange M).toAffine.Point (N : ℤ) :=
    ⟨P.val, by
      rw [Submodule.mem_torsionBy_iff, natCast_zsmul]
      simpa only [P.property] using addOrderOf_nsmul_eq_zero P.val⟩
  have hred (P : {P : E₀.toAffine.Point // addOrderOf P = N}) :
      θ (t (ψ.symm P)) = P.val :=
    (hψ (ψ.symm P) (t (ψ.symm P)) rfl).symm.trans (congrArg Subtype.val (ψ.apply_symm_apply P))
  let g := f ∘ ψ.symm
  refine ⟨g, hf.comp ψ.symm.surjective, ?_, ?_⟩
  · intro P Q hQ
    have he : θ (t (ψ.symm Q)) = θ (-(t (ψ.symm P))) := by
      rw [map_neg, hred, hred, hQ]
    have hval := congrArg (fun T : Submodule.torsionBy ℤ (E.baseChange M).toAffine.Point (N : ℤ) => T.val)
      (hθ he)
    apply hinertia (ψ.symm P) (ψ.symm Q)
    refine ⟨1, Or.inr ?_⟩
    have hid (R : (E.baseChange M).toAffine.Point) :
        Affine.Point.map (W' := E) (1 : M ≃ₐ[F] M).toAlgHom R = R := by
      cases R <;> rfl
    simpa only [hid] using hval
  · intro δ hδ P Q hQ
    obtain ⟨σ, hσ⟩ := hlift δ hδ
    have hpoint : stabilizerPointEquiv E₀ δ hδ (θ (t (ψ.symm P))) = θ (t (ψ.symm Q)) := by
      rw [hred, hred, hQ]
    apply hinertia (ψ.symm P) (ψ.symm Q)
    refine ⟨σ, ?_⟩
    rcases hσ with hσ | hσ
    · have he := (stabilizerPointEquiv_eq_iff_heq E₀ δ hδ _ _).mpr (hσ (t (ψ.symm P)))
      have ht := congrArg (fun T : Submodule.torsionBy ℤ (E.baseChange M).toAffine.Point (N : ℤ) => T.val)
        (hθ (hpoint.symm.trans he))
      exact Or.inl ht
    · have he := (stabilizerPointEquiv_eq_iff_heq E₀ δ hδ _ _).mpr (hσ (t (ψ.symm P)))
      have he' : θ (t (ψ.symm Q)) = θ (-(galoisTorsionMap E N σ.val.val (t (ψ.symm P)))) := by
        rw [map_neg]
        exact hpoint.symm.trans he
      have ht := congrArg (fun T : Submodule.torsionBy ℤ (E.baseChange M).toAffine.Point (N : ℤ) => T.val)
        (hθ he')
      exact Or.inr ht

end
end Dubon2026.SourceFunctionField.MarkedChart
