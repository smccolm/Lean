import Mathlib.AlgebraicGeometry.EllipticCurve.Affine.Point

/-! DUB-17/18 -> reverse genus -> finite elliptic fibers -> exhaustion of
the original marked chart. In the actual normal torsion field, native
conjugacy of normalized embeddings and chart covariance reach every
embedding from one exact-order point. This avoids another point count,
quotient cardinality argument, or full matrix monodromy construction. -/
namespace Dubon2026.SourceFunctionField.MarkedChart
noncomputable section
open WeierstrassCurve

theorem natural_marked_chart_surjective_of_normal
    {K k F Ω : Type*} [Field K] [Field k] [Field F] [Field Ω] [DecidableEq Ω]
    [Algebra K k] [Algebra K F] [Algebra K Ω] [Algebra k Ω]
    [IsScalarTower K k Ω] [Normal k Ω]
    (E : WeierstrassCurve k) [E.IsElliptic] (N : ℕ) (x : F)
    (hE : Transcendental K E.j)
    (hgen : IntermediateField.adjoin K ({E.j} : Set k) = ⊤)
    (P : (E.baseChange Ω).toAffine.Point) (hP : addOrderOf P = N)
    (Ψ : {Q : (E.baseChange Ω).toAffine.Point // addOrderOf Q = N} →
      {ψ : F →ₐ[K] Ω // ψ x = algebraMap k Ω E.j})
    (hnat : ∀ (σ : Ω ≃ₐ[k] Ω) Q Q',
      Q'.val = Affine.Point.map (W' := E) σ.toAlgHom Q.val →
      (Ψ Q').val = (σ.toAlgHom.restrictScalars K).comp (Ψ Q).val) :
    Function.Surjective Ψ := by
  intro φ
  obtain ⟨σ, hσ⟩ := exists_normalized_embedding_conjugacy E.j hE hgen x
    (Ψ ⟨P, hP⟩).val φ.val (Ψ ⟨P, hP⟩).property φ.property
  let Q : {Q : (E.baseChange Ω).toAffine.Point // addOrderOf Q = N} :=
    ⟨Affine.Point.map (W' := E) σ.toAlgHom P,
      (addOrderOf_injective _ (Affine.Point.map_injective σ.toAlgHom) P).trans hP⟩
  exact ⟨Q, Subtype.ext ((hnat σ ⟨P, hP⟩ Q rfl).trans hσ)⟩

end
end Dubon2026.SourceFunctionField.MarkedChart
