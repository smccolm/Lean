private def runMarkedAlgebraicChartAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceMarkedOriginalSeparability, `SourceMarkedSeparableChartValues, `SourceMarkedSeparableEmbeddingDescent, `SourceMarkedRationalPointDescent, `SourceMarkedOriginalAlgebraicChart, `SourceMarkedFullTorsionPointLift, `SourceMarkedChartNormalExhaustion, `SourceMarkedOriginalFullTorsionChart]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.MarkedChart.original_coordinate_isSeparable, `Dubon2026.SourceFunctionField.MarkedChart.exists_separable_polynomial_for_coordinate_embedding, `Dubon2026.SourceFunctionField.MarkedChart.exists_descended_separable_coordinate_embedding, `Dubon2026.SourceFunctionField.MarkedChart.exists_descended_marked_chart, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_natural_marked_chart_algebraic, `Dubon2026.SourceFunctionField.MarkedChart.exists_exactOrder_baseChange_preimage_of_full_torsion, `Dubon2026.SourceFunctionField.MarkedChart.natural_marked_chart_surjective_of_normal, `Dubon2026.SourceFunctionField.MarkedChart.exists_original_natural_marked_chart_full_torsion]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE MARKED ALGEBRAIC CHART AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runMarkedAlgebraicChartAudit
