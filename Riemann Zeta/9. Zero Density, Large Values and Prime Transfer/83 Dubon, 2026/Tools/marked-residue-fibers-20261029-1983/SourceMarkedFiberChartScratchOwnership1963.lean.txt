import SourceMarkedChartPlaceOrbits
import SourceMarkedOriginalOfJChart
import SourceMarkedOriginalPlaceChart
import SourceMarkedFullTorsionNonempty
import SourceMarkedChartFiberSurjection
import SourceMarkedOriginalFiberChart
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceMarkedChartPlaceOrbits, `SourceMarkedOriginalOfJChart, `SourceMarkedOriginalPlaceChart, `SourceMarkedFullTorsionNonempty, `SourceMarkedChartFiberSurjection, `SourceMarkedOriginalFiberChart]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
