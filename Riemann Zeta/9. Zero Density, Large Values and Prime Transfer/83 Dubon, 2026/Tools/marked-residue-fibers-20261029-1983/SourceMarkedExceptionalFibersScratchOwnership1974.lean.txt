import SourceMarkedSignPermutation
import SourceMarkedSignFreeOrbits
import SourceMarkedSignFiberFactor
import SourceMarkedPeriodicFiberBound
import SourceMarkedTorsionReductionEquiv
import SourceMarkedQuadraticFiberBounds
import SourceMarkedReducedFiberChart
import SourceMarkedOriginalResidueFiberFive
import SourceMarkedOriginalTameFiberBounds
import SourceMarkedFullInertiaLifts
import SourceMarkedOriginalResidueFiberTwo
import SourceMarkedOriginalResidueFiberThree
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceMarkedSignPermutation, `SourceMarkedSignFreeOrbits, `SourceMarkedSignFiberFactor, `SourceMarkedPeriodicFiberBound, `SourceMarkedTorsionReductionEquiv, `SourceMarkedQuadraticFiberBounds, `SourceMarkedReducedFiberChart, `SourceMarkedOriginalResidueFiberFive, `SourceMarkedOriginalTameFiberBounds, `SourceMarkedFullInertiaLifts, `SourceMarkedOriginalResidueFiberTwo, `SourceMarkedOriginalResidueFiberThree]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
