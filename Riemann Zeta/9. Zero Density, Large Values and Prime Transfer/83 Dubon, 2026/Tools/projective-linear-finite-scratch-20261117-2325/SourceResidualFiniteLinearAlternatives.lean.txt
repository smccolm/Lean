/-! DUB-17/18 -> original residual large image -> actual finite coefficient bounds.
The two finite coefficient embeddings, change of basis, SL₂ lower bound and
scalar-GL₂ upper bound are now realized in one finite field. -/
namespace Dubon2026.SourceDickson
noncomputable section
open Matrix
variable (p : ℕ) [Fact p.Prime]

/-- Linear subfield bounds over an actual finite extension of the original residual coefficients. -/
def HasFiniteLinearSubfieldBounds {F : Type*} [Field F] {Γ : Type*} [Group Γ]
    (j : F →+* K p) (ρ : Γ →* GL (Fin 2) F) : Prop :=
    ∃ m : ℕ, 1 ≤ m ∧ ∃ E : IntermediateField (ZMod p) (K p), Finite E ∧
      ∃ (jE : F →+* E) (θ : GaloisField p m →+* E) (B : GL (Fin 2) E),
        E.subtype.comp jE = j ∧ E.subtype.comp θ = galoisFieldRingHom (p := p) m ∧
        ((GeneralLinearGroup.map θ).comp
          (SpecialLinearGroup.toGL (n := Fin 2) (R := GaloisField p m))).range ≤
          (((GeneralLinearGroup.map jE).comp ρ).range).map
            (MulEquiv.toMonoidHom (MulAut.conj B)) ∧
        ∀ x ∈ (((GeneralLinearGroup.map jE).comp ρ).range).map
            (MulEquiv.toMonoidHom (MulAut.conj B)),
          ∃ (u : Eˣ) (a : GL (Fin 2) (GaloisField p m)),
            x = GeneralLinearGroup.scalar (Fin 2) u * GeneralLinearGroup.map θ a

/-- Algebraic-closure bounds for the actual residual image descend to a finite coefficient extension. -/
theorem finite_linear_subfield_bounds_of_bounds {F : Type*} [Field F] [Finite F]
    {Γ : Type*} [Group Γ] (j : F →+* K p) (ρ : Γ →* GL (Fin 2) F)
    (h : HasLinearSubfieldBounds p ((GeneralLinearGroup.map j).comp ρ).range) :
    HasFiniteLinearSubfieldBounds p j ρ := by
  obtain ⟨m, hm, A, hlo, hhi⟩ := h
  obtain ⟨E, hE, jE, θ, B, hj, hθ, hB⟩ := exists_residual_finite_descent p j m A
  have hImage : (((GeneralLinearGroup.map jE).comp ρ).range).map
      (GeneralLinearGroup.map E.subtype) = ((GeneralLinearGroup.map j).comp ρ).range := by
    rw [MonoidHom.map_range]
    apply congrArg MonoidHom.range
    apply MonoidHom.ext
    intro g
    change GeneralLinearGroup.map (E.subtype.comp jE) (ρ g) = GeneralLinearGroup.map j (ρ g)
    rw [hj]
  refine ⟨m, hm, E, hE, jE, θ, B, hj, hθ, ?_⟩
  apply linear_subfield_bounds_descend E.subtype θ ((GeneralLinearGroup.map jE).comp ρ).range B
  · rw [hθ, hImage, hB]
    exact hlo
  · rw [hθ, hImage, hB]
    exact hhi

/-- The actual residual action has either a visible small projective image or finite-field linear bounds. -/
theorem residual_small_image_or_finite_linear_bounds [Fact (2 < p)] (hp : 3 < p)
    {F : Type*} [Field F] [Finite F] {Γ : Type*} [Group Γ]
    (j : F →+* K p) (ρ : Γ →* GL (Fin 2) F) (hρ : (Set.range ρ).Finite) :
    HasSmallProjectiveImage p (residualProjectiveAction p j ρ).range ∨
      HasFiniteLinearSubfieldBounds p j ρ := by
  rcases residual_small_image_or_linear_bounds p hp j ρ hρ with hs | hl
  · exact Or.inl hs
  · exact Or.inr (finite_linear_subfield_bounds_of_bounds p j ρ hl)

end
end Dubon2026.SourceDickson
