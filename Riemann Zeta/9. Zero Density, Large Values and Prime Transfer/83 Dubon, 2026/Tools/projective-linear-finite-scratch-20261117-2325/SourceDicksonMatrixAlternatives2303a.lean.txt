variable (p : ℕ) [Fact p.Prime] [Fact (2 < p)]

/-- The already proved index-two comparison determines the cardinality of PSL₂. -/
theorem card_PSL2_of_five_le (F : Type*) [Field F] [CharP F p] [Fintype F]
    (hcard : 5 ≤ Fintype.card F) :
    Nat.card (Matrix.ProjectiveSpecialLinearGroup (Fin 2) F) =
      Fintype.card F * (Fintype.card F ^ 2 - 1) / 2 := by
  have hi := PSLImageInPGL_index p F hcard
  obtain ⟨e⟩ := index_two_subgroup_iso_PSL p F hcard (PSLImageInPGL F) hi
  have h := Subgroup.card_mul_index (PSLImageInPGL F)
  rw [hi, card_PGL2, Nat.card_congr e.toEquiv] at h
  omega

/-- Dickson alternatives with the two large cases identified as actual matrix subgroups. -/
def HasProjectiveMatrixAlternatives (G : Subgroup (PGLOf (K p))) : Prop :=
    (IsCyclic G) ∨
    (∃ n : ℕ, n ≥ 2 ∧ Nonempty (G ≃* DihedralGroup n)) ∨
    (Nonempty (G ≃* alternatingGroup (Fin 4))) ∨
    (Nonempty (G ≃* Equiv.Perm (Fin 4))) ∨
    (Nonempty (G ≃* alternatingGroup (Fin 5))) ∨
    (∃ (m t : ℕ) (_ : m ≥ 1) (_ : Nat.Coprime t p) (_ : t ∣ p ^ m - 1)
      (φ : Multiplicative (ZMod t) →* MulAut (Multiplicative (Fin m → ZMod p))),
      Nonempty (G ≃* (Multiplicative (Fin m → ZMod p)) ⋊[φ] Multiplicative (ZMod t))) ∨
    (∃ m : ℕ, m ≥ 1 ∧ ∃ g : PGLOf (K p),
      G.map (MulEquiv.toMonoidHom (MulAut.conj g)) =
        (PSLImageInPGL (GaloisField p m)).map (pglMap (galoisFieldRingHom (p := p) m))) ∨
    (∃ m : ℕ, m ≥ 1 ∧ ∃ g : PGLOf (K p),
      G.map (MulEquiv.toMonoidHom (MulAut.conj g)) =
        (pglMap (galoisFieldRingHom (p := p) m)).range)

/-- Over characteristic greater than three, the abstract classification yields matrix conjugacy. -/
theorem hasProjectiveMatrixAlternatives_of_dickson (hp : 3 < p)
    (G : Subgroup (PGLOf (K p))) [Finite G] (h : HasDicksonAlternatives p G) :
    HasProjectiveMatrixAlternatives p G := by
  rcases h with hc | hd | ha4 | hs4 | ha5 | hs | ⟨m, hm, ⟨e⟩⟩ | ⟨m, hm, ⟨e⟩⟩
  · exact Or.inl hc
  · exact Or.inr (Or.inl hd)
  · exact Or.inr (Or.inr (Or.inl ha4))
  · exact Or.inr (Or.inr (Or.inr (Or.inl hs4)))
  · exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inl ha5))))
  · exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inl hs)))))
  · have hpm : 5 ≤ p ^ m := by
      have hne : p ≠ 4 := by intro he; have hh := (Fact.out : p.Prime); norm_num [he] at hh
      exact le_trans (by omega : 5 ≤ p) (Nat.le_self_pow (by omega) p)
    have hcard : 5 ≤ Fintype.card (GaloisField p m) := by
      rw [Fintype.card_eq_nat_card, GaloisField.card p m (by omega)]
      exact hpm
    have hn : Nat.card G = p ^ m * (p ^ (2 * m) - 1) / 2 := by
      rw [Nat.card_congr e.toEquiv, card_PSL2_of_five_le p _ hcard,
        Fintype.card_eq_nat_card, GaloisField.card p m (by omega), pow_mul']
    exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inl
      ⟨m, hm, psl_conjugate_to_standard p G m hpm hn⟩))))))
  · have hn : Nat.card G = p ^ m * (p ^ (2 * m) - 1) := by
      rw [Nat.card_congr e.toEquiv, card_PGL2,
        Fintype.card_eq_nat_card, GaloisField.card p m (by omega), pow_mul']
    exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inr
      ⟨m, hm, pgl_conjugate_to_standard p G m hm hn⟩))))))

/-- The actual finite projective image satisfies the matrix-subgroup classification. -/
theorem residual_projective_matrix_classification (hp : 3 < p)
    {F : Type*} [Field F] {Γ : Type*} [Group Γ]
    (j : F →+* K p) (ρ : Γ →* GL (Fin 2) F) (hρ : (Set.range ρ).Finite) :
    HasProjectiveMatrixAlternatives p (residualProjectiveAction p j ρ).range := by
  letI := residualProjectiveAction_finite p j ρ hρ
  exact hasProjectiveMatrixAlternatives_of_dickson p hp _
    (residual_projective_classification p j ρ hρ)

end
