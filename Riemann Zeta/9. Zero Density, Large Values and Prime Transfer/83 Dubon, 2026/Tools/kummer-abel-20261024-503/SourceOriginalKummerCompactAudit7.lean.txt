private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceFiniteFiberOpenMap, `SourceCurveUnramifiedFiberCard, `SourceCurveUnramifiedChartRestriction, `SourceAnalyticRestrictionTopology, `SourceOriginalWeilRestrictionComparison, `SourceOriginalWeilRestrictionTopology, `SourceOriginalWeilCoverCompact]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.isClosedMap_of_open_constant_finite_fibers, `Dubon2026.SourceCurve.compactSpace_of_open_constant_finite_fibers, `Dubon2026.SourceCurve.Place.card_unramified_restriction_fiber, `Dubon2026.SourceCurve.AnalyticChart.Orders.restrict_unramified, `Dubon2026.SourceCurve.map_nhds_eq_of_analytic_chart_comparison, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.restriction_family_orders, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.restriction_chart_comparison, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.restriction_map_nhds, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.continuous_restriction, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.isOpenMap_restriction, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.restrictionIntegral, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.card_restriction_fiber, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.place_compactSpace, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.compactCoverFinite, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.compactCoverIntegral]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.restrictionComparison]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Compact postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Compact theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER COMPACT AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
