private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverLocalDifferentials, `SourceOriginalWeilCoverDifferentialReadings, `SourceOriginalWeilCoverChartGermHom, `SourceOriginalWeilCoverChartDerivativeGerms, `SourceOriginalWeilCoverChartGermDerivation, `SourceOriginalWeilCoverDifferentialRegularity, `SourceOriginalWeilCoverDifferentialTransitions, `SourceOriginalWeilCoverChartPrimitives, `SourceOriginalWeilCoverPathLocalPrimitives, `SourceOriginalWeilCoverPathPrimitives]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_place_dCoordGenerates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_differentials_nontrivial, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.mem_cover_regularDifferentials_iff, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.mem_cover_regularDifferentials_iff_ord_nonneg, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_ordDifferential_D_eq_ord_sub_one, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartRead_uniformizer, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.deriv_chartRead_uniformizer, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_eq_coefficient, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analyticAt_readDifferential, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartReadHom_apply, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.eventually_differentiableAt_chartRead, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.deriv_chartRead_add, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.deriv_chartRead_mul, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.deriv_chartRead_smul, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.deriv_chartRead_one, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartGerm_smul, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartGerm_scalarTower, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartReadDerivation_apply, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartReadDerivation_lift_smul_D, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_eq_expression_germ, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chart_dCoord_order_zero, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chart_differentialCoeff_regular, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.analyticOnNhd_readDifferential, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.uniformizer_regular_on_chart, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_eq_expression_nhds, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential_change, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_readDifferential_primitive_on_chart, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.continuousAt_placeChart, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.placeChart_source_mem_nhds, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_path_local_primitive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_isPrimitiveAlong, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.IsPrimitiveAlong.sub_eq_initial]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.readDifferential, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartReadHom, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartGermAlgebra, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.chartReadDerivation, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.IsPrimitiveAlong]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Differential postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Differential theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER DIFFERENTIAL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
