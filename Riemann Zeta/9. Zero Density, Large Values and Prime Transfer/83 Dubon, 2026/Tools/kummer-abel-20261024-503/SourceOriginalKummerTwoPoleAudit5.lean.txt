private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverRiemannRoch, `SourceCurveTwoPoleDimensions, `SourceOriginalWeilCoverTwoPoleDimensions, `SourceCurveTwoPoleDifferential, `SourceOriginalWeilCoverTwoPoleDifferential]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverCanonicalDivisor_apply, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_riemann_roch_constant, `Dubon2026.SourceCurve.two_pole_dimensions_of_riemann_roch, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_two_pole_dimensions, `Dubon2026.SourceCurve.exists_two_pole_function_of_dimensions, `Dubon2026.SourceCurve.exists_two_pole_differential_of_function, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_two_pole_function, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_two_pole_differential]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.coverCanonicalDivisor]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Two Pole postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Two Pole theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER TWO POLE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
