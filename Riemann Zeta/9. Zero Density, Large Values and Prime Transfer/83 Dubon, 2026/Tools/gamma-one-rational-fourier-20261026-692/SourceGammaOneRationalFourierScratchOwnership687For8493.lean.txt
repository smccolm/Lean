import SourceCyclotomicFrickeIndex
import SourceFrickeLaurentCyclotomicConjugation
import SourceFrickeLaurentFieldConjugation
import SourceOriginalFrickeLaurentField
import SourceFrickeCyclotomicAnalyticTransport
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceCyclotomicFrickeIndex, `SourceFrickeLaurentCyclotomicConjugation, `SourceFrickeLaurentFieldConjugation, `SourceOriginalFrickeLaurentField, `SourceFrickeCyclotomicAnalyticTransport]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
