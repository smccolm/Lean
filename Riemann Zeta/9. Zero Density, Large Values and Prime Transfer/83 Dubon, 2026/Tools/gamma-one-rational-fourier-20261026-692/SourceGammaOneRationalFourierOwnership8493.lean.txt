import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceCyclotomicFrickeIndex, `Dubon2026.SourceFrickeLaurentCyclotomicConjugation, `Dubon2026.SourceFrickeLaurentFieldConjugation, `Dubon2026.SourceOriginalFrickeLaurentField, `Dubon2026.SourceFrickeCyclotomicAnalyticTransport, `Dubon2026.SourceFrickeCyclotomicConjugateRelation, `Dubon2026.SourceCyclotomicModularLift, `Dubon2026.SourceCyclotomicGammaOneLift, `Dubon2026.SourceCyclotomicLevelRingCovariance, `Dubon2026.SourceCyclotomicConjugateCancellation, `Dubon2026.SourceCyclotomicFractionCovariance, `Dubon2026.SourceCyclotomicConjugateCuspOrder, `Dubon2026.SourceCyclotomicHolomorphicDivision, `Dubon2026.SourceCyclotomicEllipticDivisibility, `Dubon2026.SourceCyclotomicEllipticCoordinates, `Dubon2026.SourceGammaOneCuspCyclotomicConjugation, `Dubon2026.SourceCyclotomicCuspFourierTransport, `Dubon2026.SourceGammaOneCyclotomicFourierConjugation, `Dubon2026.SourceCyclotomicTraceStructure, `Dubon2026.SourceGammaOneFourierLinear, `Dubon2026.SourceGammaOneRationalTraceSpan, `Dubon2026.SourceGammaOneRationalCuspSpan]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
