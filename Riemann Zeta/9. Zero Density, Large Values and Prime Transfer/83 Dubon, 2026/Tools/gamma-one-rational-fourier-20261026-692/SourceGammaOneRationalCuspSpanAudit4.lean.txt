private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCyclotomicTraceStructure, `SourceGammaOneFourierLinear, `SourceGammaOneRationalTraceSpan, `SourceGammaOneRationalCuspSpan]
  let theorems : Array Name := #[`Dubon2026.SourceFricke.isCyclotomicExtension_kN, `Dubon2026.SourceFricke.finiteDimensional_kN, `Dubon2026.SourceFricke.isGalois_kN, `Dubon2026.SourceFricke.gammaOne_one_mem_strictPeriods, `Dubon2026.SourceFricke.gammaOne_eq_of_qExpansion_coeff_eq, `Dubon2026.SourceFricke.gammaOne_mem_span_rational_of_conjugates, `Dubon2026.SourceFricke.span_gammaOne_rational_cuspForms_eq_top, `Dubon2026.SourceFricke.span_gammaOne_weightTwo_rational_cuspForms_eq_top]
  let constructions : Array Name := #[`Dubon2026.SourceFricke.gammaOneQCoeffLinear]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Rational Cusp Span postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Rational Cusp Span theorems discovered"
  logInfo m!"SOURCE GAMMA ONE RATIONAL CUSP SPAN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
