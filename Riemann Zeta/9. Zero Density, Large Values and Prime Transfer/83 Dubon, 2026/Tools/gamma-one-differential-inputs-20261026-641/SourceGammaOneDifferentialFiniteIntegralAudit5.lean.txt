private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceModularQDerivative, `SourceModularQDerivativeExpansion, `SourceJEulerModularIdentity, `SourceJEulerPolynomial, `SourceGammaOneDifferentialFiniteIntegral]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.ModularQDerivative.norm_qParam_one_lt_one, `Dubon2026.SourceFunctionField.ModularQDerivative.hasDerivAt_qParam_one, `Dubon2026.SourceFunctionField.ModularQDerivative.normalizedDerivOfComplex_eq_qParam_mul_deriv, `Dubon2026.SourceFunctionField.ModularQDerivative.periodic_normalizedDerivOfComplex, `Dubon2026.SourceFunctionField.ModularQDerivative.qExpansion_coeff_unique', `Dubon2026.SourceFunctionField.ModularQDerivative.hasSum_qExpansion_normalizedDerivOfComplex, `Dubon2026.SourceFunctionField.ModularQDerivative.qExpansion_coeff_normalizedDerivOfComplex, `Dubon2026.SourceFunctionField.ModularQDerivative.coe_qExpansion_normalizedDerivOfComplex, `Dubon2026.SourceFunctionField.ModularQDerivative.qExpansion_derivative_bracket, `Dubon2026.SourceFunctionField.ModularQDerivative.qEuler_jqModC_mul_discriminant, `Dubon2026.SourceFunctionField.ModularQDerivative.laurent_discriminant_relation, `Dubon2026.SourceFunctionField.ModularQDerivative.exists_qEuler_jqModC_formula, `Dubon2026.SourceFunctionField.ModularQDerivative.exists_qEuler_jqModC_power_six, `Dubon2026.SourceFunctionField.GammaOne.isIntegral_cusp_differential_finite_expression]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIFFERENTIAL FINITE INTEGRAL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
