private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneModularRatios, `SourceGammaOneFieldModularRatio]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.ModularRatios.qL_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.qL_mul, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.qL_add, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.qL_one, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.qL_const, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.one_ne_zero', `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.mul_ne_zero', `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.isRatio_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.IsRatio.mul, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.IsRatio.add, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.IsRatio.inv, `Dubon2026.SourceFunctionField.GammaOne.isRatio_of_mem_qExpFunctionFieldC, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_modular_ratio]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.ModularRatios.qL, `Dubon2026.SourceFunctionField.GammaOne.ModularRatios.IsRatio]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE FIELD MODULAR RATIO AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
