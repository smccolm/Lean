private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceInverseJLevelOneIntegral, `SourceIntegralHomogeneousRelation, `SourceModularInverseJIntegral, `SourceGammaOneCuspAuxiliaryForm, `SourceGammaOneCuspAuxiliaryExpansion, `SourceGammaOneDifferentialCuspIntegral]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.InverseJIntegral.eisenstein_four_expansion_ne_zero, `Dubon2026.SourceFunctionField.InverseJIntegral.levelOne_isIntegral, `Dubon2026.SourceFunctionField.InverseJIntegral.isIntegral_div_of_relation, `Dubon2026.SourceFunctionField.InverseJIntegral.isIntegral_modular_quotient_E4_cube, `Dubon2026.SourceFunctionField.InverseJIntegral.isIntegral_modular_quotient_E4, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.conj_T_pow_mem, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.mapGL_T_zpow, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.natCast_mem_strictPeriods_conj, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.pow_slash, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.num_slash, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.qExpansion_Gform_mul, `Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.cusp_expression_identity, `Dubon2026.SourceFunctionField.GammaOne.isIntegral_cusp_differential_cusp_expression]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CuspAuxiliary.Gform]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIFFERENTIAL CUSP INTEGRAL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
