import SourceInverseJLevelOneIntegral
import SourceIntegralHomogeneousRelation
import SourceModularInverseJIntegral
import SourceGammaOneCuspAuxiliaryForm
import SourceGammaOneCuspAuxiliaryExpansion
import SourceGammaOneDifferentialCuspIntegral
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceInverseJLevelOneIntegral, `SourceIntegralHomogeneousRelation, `SourceModularInverseJIntegral, `SourceGammaOneCuspAuxiliaryForm, `SourceGammaOneCuspAuxiliaryExpansion, `SourceGammaOneDifferentialCuspIntegral]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
