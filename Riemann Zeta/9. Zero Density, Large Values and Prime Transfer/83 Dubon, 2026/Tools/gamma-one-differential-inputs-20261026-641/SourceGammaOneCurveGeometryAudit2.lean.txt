private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCurveGeometry, `SourceGammaOneCurveInputs]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.D_jCoordinateOver_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_diffQExp_injective, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_dCoordGenerates, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_localDifferentials, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_isCurveOver, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_hasCanonicalDivisor, `Dubon2026.SourceFunctionField.GammaOne.gamma1_diffQExp_injective, `Dubon2026.SourceFunctionField.GammaOne.gamma1_localDifferentials, `Dubon2026.SourceFunctionField.GammaOne.gamma1_isCurveOver, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hasCanonicalDivisor, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_diffQExp_injective, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_localDifferentials, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_isCurveOver, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_hasCanonicalDivisor]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CURVE GEOMETRY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
