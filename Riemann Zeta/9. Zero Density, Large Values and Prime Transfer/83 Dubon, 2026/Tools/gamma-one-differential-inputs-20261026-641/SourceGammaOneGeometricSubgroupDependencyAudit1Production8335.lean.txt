import Dubon2026.SourceGammaOneHeckeGeometricSubgroup
import Mathlib.Tactic

open Lean

private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.heckeUpperGeometricEquiv, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_conjugation, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_matrix, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_smul]
  let localModules : Array Name := #[`Dubon2026.SourceGammaOneHeckeGeometricSubgroup, `Dubon2026.SourceGammaOneModularStretch]
  let required : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.geometricToUpper_upperToGeometric, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_geometricToUpper, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_conjugation, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_matrix, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric_smul, `Dubon2026.SourceFunctionField.GammaOne.upperToGeometric, `Dubon2026.SourceFunctionField.GammaOne.geometricToUpper, `Dubon2026.SourceFunctionField.GammaOne.heckeUpperGeometricEquiv, `Dubon2026.originalHeckeRaisedToUpper_toRaised, `Dubon2026.originalHeckeUpperToRaised_toUpper, `Dubon2026.originalHeckeUpperToRaised_matrix, `Dubon2026.originalHeckeUpperToRaised_smul, `Dubon2026.sourceGammaOneHeckeConjugation, `Dubon2026.sourceGammaOneUpperToGammaZero, `Dubon2026.SourceFunctionField.GammaOne.levelRaiseConj_mem_Gamma1]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE GAMMA ONE GEOMETRIC SUBGROUP DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the actual upper/geometric subgroup isomorphism, its literal original-conjugation identity or analytic matrix/action equation. Only declaration values are traversed. Both original inverse identities, Gamma1-preserving level raising and original integral conjugation are reused. H1/Picard transpose and diamond conventions, original homogeneous coefficient action, full/parabolic comparison, arithmetic Galois realization and higher continuation remain required. 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
