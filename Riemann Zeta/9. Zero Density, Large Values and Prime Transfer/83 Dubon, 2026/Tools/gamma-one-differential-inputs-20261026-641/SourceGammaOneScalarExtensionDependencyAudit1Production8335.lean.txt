import Dubon2026.SourceGammaOneScalarExtension

private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_eq_qExpFunctionFieldC]
  let localModules : Array Name := #[`Dubon2026.SourceGammaOneScalarExtension, `Dubon2026.SourceGammaOneFunctionFieldBaseChange, `Dubon2026.SourceGammaOneFunctionFieldFinite, `Dubon2026.SourceGammaOneRationalRelation, `Dubon2026.SourceGammaOneJCoordinate, `Dubon2026.SourceGammaOneFunctionField, `Dubon2026.SourceGammaOneComplexRatio, `Dubon2026.SourceGammaOneComplexRelation, `Dubon2026.SourceGammaOneNormRelation, `Dubon2026.SourceGammaOneRatioClosure]
  let required : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.coeffEmb_intSeriesC, `Dubon2026.SourceFunctionField.GammaOne.coeffEmb_image_intFormRatiosC, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_eq_qExpFunctionFieldC, `Dubon2026.SourceFunctionField.coeffEmb_algebraMap, `Dubon2026.SourceFunctionField.coeffEmb_mem_laurentBaseChange, `RingHom.map_field_closure, `Dubon2026.SourceFunctionField.GammaOne.mem_intFormRatiosC]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE GAMMA ONE SCALAR EXTENSION DEPENDENCY AUDIT PASS: all {required.size} required dependencies occur in the actual field equality. Only declaration values are traversed. Integral ratio images and original coefficient/field closure proofs identify the literal scalar extension with the required modular q-expansion field. Regular-cusp, Picard/H1 and diamond comparison, all-weight arithmetic association and higher continuation remain required. 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
