import SourcePrescribedReductionPrime
import SourceAdicCoefficientResidueComparison
import SourcePrescribedCompletionResidue
import SourceReductionCompletionBasePrime
import SourceReductionCompletionPadicMap
import SourceReductionCompletionFinite
import SourcePadicCompletionValuation
import SourceReductionCompletionLocalStructures
import SourceOriginalPrescribedReductionResidual
import SourceOriginalExactReductionResidual
import SourceOriginalAlgebraicIntegerResidual
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourcePrescribedReductionPrime, `SourceAdicCoefficientResidueComparison, `SourcePrescribedCompletionResidue, `SourceReductionCompletionBasePrime, `SourceReductionCompletionPadicMap, `SourceReductionCompletionFinite, `SourcePadicCompletionValuation, `SourceReductionCompletionLocalStructures, `SourceOriginalPrescribedReductionResidual, `SourceOriginalExactReductionResidual, `SourceOriginalAlgebraicIntegerResidual]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
