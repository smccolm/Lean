private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceValuationResidualAction, `SourceNormValuationRestriction, `SourceRationalLocalValuation, `SourceOriginalNormLocalResidual, `SourceNumberFieldNormInertiaLift, `SourceAbsoluteFiniteUnramified, `SourceNumberFieldSupportRamification, `SourceAbsoluteArithmeticRestriction, `SourceAbsoluteArithmeticDescent, `SourceArithmeticLocalEmbeddingExtension, `SourceOriginalArithmeticResidual]
  let theorems : Array Name := #[`Dubon2026.valuation_decomposition_of_mem_iff, `Dubon2026.valuation_residue_power_iff, `Dubon2026.valuation_comap_nonunits_iff, `Dubon2026.norm_restriction_mem_decomposition, `Dubon2026.norm_restriction_frobenius, `Dubon2026.norm_restriction_inertia, `Dubon2026.sourceRationalAlgebraicClosure_isGalois, `Dubon2026.sourceRationalLocalValuation_prime, `Dubon2026.sourceRationalLocalValuation_frobenius, `Dubon2026.sourceRationalLocalValuation_inertia, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_prescribed_normLocal_residual_realization, `Dubon2026.numberField_embedded_inertia_exists_norm_lift, `Dubon2026.absolute_faithful_factor_unramified, `Dubon2026.numberField_support_ramification_of_rational_unramified, `Dubon2026.sourceAbsoluteArithmeticRestriction_commutes, `Dubon2026.absolute_continuous_normInertia_arithmetic_descent, `Dubon2026.sourceArithmeticLocalEmbedding_exists_extension, `Dubon2026.sourceAbsoluteArithmeticRestriction_local, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_prescribed_arithmetic_residual_realization]
  let constructions : Array Name := #[`Dubon2026.sourceRationalLocalValuation, `Dubon2026.sourceAbsoluteArithmeticRestriction]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Arithmetic Residual postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Arithmetic Residual theorems discovered"
  logInfo m!"SOURCE ORIGINAL ARITHMETIC RESIDUAL AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
