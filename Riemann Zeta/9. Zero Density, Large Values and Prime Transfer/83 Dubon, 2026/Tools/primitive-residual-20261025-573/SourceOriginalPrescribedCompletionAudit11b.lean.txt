private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourcePrescribedReductionPrime, `SourceAdicCoefficientResidueComparison, `SourcePrescribedCompletionResidue, `SourceReductionCompletionBasePrime, `SourceReductionCompletionPadicMap, `SourceReductionCompletionFinite, `SourcePadicCompletionValuation, `SourceReductionCompletionLocalStructures, `SourceOriginalPrescribedReductionResidual, `SourceOriginalExactReductionResidual, `SourceOriginalAlgebraicIntegerResidual]
  let theorems : Array Name := #[`Dubon2026.sourceReductionPrime_prime_mem, `Dubon2026.sourceReductionPrime_below, `Dubon2026.sourceAdicCoefficientResidueEquiv_mk, `Dubon2026.sourcePrescribedCompletionResidueMap_integer, `Dubon2026.sourcePrescribedCompletionResidueMap_injective, `Dubon2026.sourceReductionPrime_rationalBelow, `Dubon2026.sourceReductionPrime_liesOver_rational, `Dubon2026.sourceReductionCompletionPadicMap_continuous, `Dubon2026.sourceReductionCompletionPadicMap_rational, `Dubon2026.sourceReductionCompletion_finite, `Dubon2026.sourceReductionCompletion_isModuleTopology, `Dubon2026.sourcePadicCompletionEquiv_mem_integer, `Dubon2026.sourceReductionCompletion_hasExtension, `Dubon2026.sourceReductionCompletion_isNonarchimedeanLocalField, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_numberField_reduction_residual_realization, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_prescribed_reduction_residual_realization, `Dubon2026.SourceFunctionField.OriginalChart.originalPrimitive_algebraicInteger_reduction_residual_realization]
  let constructions : Array Name := #[`Dubon2026.sourceReductionPrime, `Dubon2026.sourceAdicCoefficientResidueEquiv, `Dubon2026.sourcePrescribedCompletionResidueMap, `Dubon2026.sourceReductionCompletionPadicMap]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Prescribed Completion postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Prescribed Completion theorems discovered"
  logInfo m!"SOURCE ORIGINAL PRESCRIBED COMPLETION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
