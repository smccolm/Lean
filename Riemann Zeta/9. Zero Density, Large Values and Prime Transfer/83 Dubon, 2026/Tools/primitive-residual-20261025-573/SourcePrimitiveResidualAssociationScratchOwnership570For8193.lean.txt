import SourceValuationResidualAction
import SourceNormValuationRestriction
import SourceRationalLocalValuation
import SourceOriginalNormLocalResidual
import SourceNumberFieldNormInertiaLift
import SourceAbsoluteFiniteUnramified
import SourceNumberFieldSupportRamification
import SourceAbsoluteArithmeticRestriction
import SourceAbsoluteArithmeticDescent
import SourceArithmeticLocalEmbeddingExtension
import SourceOriginalArithmeticResidual
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceValuationResidualAction, `SourceNormValuationRestriction, `SourceRationalLocalValuation, `SourceOriginalNormLocalResidual, `SourceNumberFieldNormInertiaLift, `SourceAbsoluteFiniteUnramified, `SourceNumberFieldSupportRamification, `SourceAbsoluteArithmeticRestriction, `SourceAbsoluteArithmeticDescent, `SourceArithmeticLocalEmbeddingExtension, `SourceOriginalArithmeticResidual]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
