import SourcePrescribedReductionPrime
import SourceAdicCoefficientResidueComparison

/-! DUB-17/18 -> prescribed residual association -> the chosen completion
has exactly the source coefficient reduction. Its actual residue field is
the quotient by the kernel of that reduction. This constructs the residue
embedding and proves agreement on every original algebraic integer. -/
namespace Dubon2026
noncomputable section
open IsDedekindDomain
open scoped NumberField ValuativeRel
variable (F : Type) [Field F] [NumberField F]
variable {k : Type*} [Field k] (ell : ℕ) (hell : ell.Prime) [CharP k ell]
variable (φ : 𝓞 F →+* k)
local notation "w" => sourceReductionPrime F ell hell φ
private local instance : ValuativeRel ((w).adicCompletion F) :=
  ValuativeRel.ofValuation (NormedField.valuation (K := (w).adicCompletion F))

def sourcePrescribedCompletionResidueMap : 𝓀[(w).adicCompletion F] →+* k :=
  (Ideal.Quotient.lift (w).asIdeal φ (fun _ h => h)).comp
    (sourceAdicCoefficientResidueEquiv F w).symm.toRingHom

theorem sourcePrescribedCompletionResidueMap_integer (x : 𝓞 F) :
    sourcePrescribedCompletionResidueMap F ell hell φ
      (IsLocalRing.residue 𝒪[(w).adicCompletion F]
        (originalIntegerToValuationIntegerMap F ((w).adicCompletion F) x)) = φ x := by
  rw [← sourceAdicCoefficientResidueEquiv_mk F w x]
  change (Ideal.Quotient.lift (w).asIdeal φ (fun _ h => h))
    ((sourceAdicCoefficientResidueEquiv F w).symm
      (sourceAdicCoefficientResidueEquiv F w (Ideal.Quotient.mk (w).asIdeal x))) = φ x
  rw [RingEquiv.symm_apply_apply]
  rfl

theorem sourcePrescribedCompletionResidueMap_injective :
    Function.Injective (sourcePrescribedCompletionResidueMap F ell hell φ) :=
  (sourcePrescribedCompletionResidueMap F ell hell φ).injective

end
end Dubon2026
