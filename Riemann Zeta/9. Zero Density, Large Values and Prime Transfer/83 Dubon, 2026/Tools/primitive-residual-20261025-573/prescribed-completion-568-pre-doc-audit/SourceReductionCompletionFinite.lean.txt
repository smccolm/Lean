/-! DUB-17/18 -> prescribed residual association -> the completion selected
by the original reduction is finite over the Tate-prime field. Reuse the
native finite-completion theorem and transport scalars through the native
rational p-adic equivalence; retain the actual completion topology. -/
namespace Dubon2026
noncomputable section
open IsDedekindDomain
open scoped NumberField
variable (F : Type) [Field F] [NumberField F]
variable {k : Type*} [Field k] (ell : ℕ) (hell : ell.Prime) [Fact ell.Prime] [CharP k ell]
variable (φ : 𝓞 F →+* k)
local notation "w" => sourceReductionPrime F ell hell φ
local notation "v" => Equiv.symm (Rat.HeightOneSpectrum.primesEquiv (R := 𝓞 ℚ)) (Subtype.mk ell hell)
local notation "E" => HeightOneSpectrum.adicCompletion F w
local notation "B" => HeightOneSpectrum.adicCompletion ℚ v
private local instance : (w).asIdeal.LiesOver (v).asIdeal :=
  sourceReductionPrime_liesOver_rational F ell hell φ
private local instance : Algebra B E := (originalAdicBaseChangeMap ℚ F v w).toAlgebra
private local instance : ContinuousSMul B E :=
  continuousSMul_of_algebraMap B E (originalAdicBaseChangeMap_continuous ℚ F v w)
private local instance : IsScalarTower ℚ B E :=
  IsScalarTower.of_algebraMap_eq fun x => by
    change algebraMap ℚ E x = originalAdicBaseChangeMap ℚ F v w (algebraMap ℚ B x)
    rw [originalAdicBaseChangeMap_algebraMap]
    exact IsScalarTower.algebraMap_apply ℚ F E x
private local instance : Algebra ℚ_[ell] E :=
  (sourceReductionCompletionPadicMap F ell hell φ).toAlgebra

attribute [-instance] UniformSpace.Completion.instSMul in
theorem sourceReductionCompletion_finite : Module.Finite ℚ_[ell] E := by
  let e := Padic.adicCompletionEquiv (𝓞 ℚ) ⟨ell, hell⟩
  letI : Module.Finite B E := inferInstance
  let f : E →ₛₗ[e.symm.toRingHom] E := {
    toFun := id
    map_add' _ _ := rfl
    map_smul' a x := by
      change originalAdicBaseChangeMap ℚ F v w a * x =
        sourceReductionCompletionPadicMap F ell hell φ (e.symm a) * x
      change originalAdicBaseChangeMap ℚ F v w a * x =
        originalAdicBaseChangeMap ℚ F v w (e (e.symm a)) * x
      rw [e.apply_symm_apply] }
  letI : RingHomSurjective e.symm.toRingHom := ⟨e.symm.surjective⟩
  exact Module.Finite.of_surjective f Function.surjective_id

attribute [-instance] UniformSpace.Completion.instSMul in
theorem sourceReductionCompletion_isModuleTopology : IsModuleTopology ℚ_[ell] E := by
  letI : ContinuousSMul ℚ_[ell] E := continuousSMul_of_algebraMap ℚ_[ell] E
    (sourceReductionCompletionPadicMap_continuous F ell hell φ)
  letI : Module.Finite ℚ_[ell] E := sourceReductionCompletion_finite F ell hell φ
  exact isModuleTopologyOfFiniteDimensional

end
end Dubon2026
