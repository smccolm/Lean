  let constructions : Array Name := #[`Dubon2026.sourceReductionPrime, `Dubon2026.sourceAdicCoefficientResidueEquiv, `Dubon2026.sourcePrescribedCompletionResidueMap, `Dubon2026.sourceReductionCompletionPadicMap, `Dubon2026.sourceRationalLocalValuation, `Dubon2026.sourceAbsoluteArithmeticRestriction]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original residue functional postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original residue functional theorems discovered"
  logInfo m!"SOURCE PRIMITIVE RESIDUAL ASSOCIATION AUDIT PASS: {count} theorems including private/generated declarations. The actual original primitive form in every weight at least two supplies its prescribed-characteristic residual representation on the original support, with exact original coefficient reduction, weight determinant and good-prime inertia. Global determinant, adic realization, local/image properties, automorphy and all-weight continuation remain open; 16/20 gates."
