import SourceWeightTwoDegeneracyAdjoint
import SourceWeightTwoEigenLevelDescent
import SourceWeightTwoNewEigenNormalization
import SourceParabolicPrimitiveDivisor
import SourceArithmeticSupportInflationValues
import SourceOriginalDivisorArithmeticResidual
import SourceParabolicArithmeticResidual
import SourceResidualGaloisRealization
import SourcePrimitiveWeightTwoGalois
import SourcePrimitiveRestoredGalois
import SourcePrimitiveOriginalResidualGalois
import SourcePrimitiveResidualGaloisCharP
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceWeightTwoDegeneracyAdjoint, `SourceWeightTwoEigenLevelDescent, `SourceWeightTwoNewEigenNormalization, `SourceParabolicPrimitiveDivisor, `SourceArithmeticSupportInflationValues, `SourceOriginalDivisorArithmeticResidual, `SourceParabolicArithmeticResidual, `SourceResidualGaloisRealization, `SourcePrimitiveWeightTwoGalois, `SourcePrimitiveRestoredGalois, `SourcePrimitiveOriginalResidualGalois, `SourcePrimitiveResidualGaloisCharP]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
