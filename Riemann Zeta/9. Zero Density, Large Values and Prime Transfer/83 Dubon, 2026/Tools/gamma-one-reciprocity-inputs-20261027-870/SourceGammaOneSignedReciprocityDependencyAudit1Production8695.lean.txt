import Dubon2026.SourceWeightTwoSignedForms
import Dubon2026.SourceWeightTwoSignedReciprocity
import Mathlib.Tactic

open Lean

private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.SignedForms.integralLattice_comp_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.multiplier_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.orbit_iff]
  let localModules : Array Name := #[`Dubon2026.SourceWeightTwoSignedForms, `Dubon2026.SourceWeightTwoSignedReciprocity]
  let required : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.SignedForms.weightTwo_slash_neg, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.along_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.period_extend_representative, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.dualPeriod_comp_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.integralLattice_comp_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.multiplier_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.orbit_iff, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.extend, `Dubon2026.sourceUpperHalfPlane_primitive_difference, `Dubon2026.sourceWeightTwoPrimitive_hasDerivAt, `Dubon2026.sourceWeightTwoDualPeriod_mem, `Dubon2026.SourceFunctionField.GammaOne.CentralSign.representative_spec, `Submodule.span_induction]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE GAMMA ONE SIGNED RECIPROCITY DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the declaration-VALUE closure of the signed-to-original lattice, multiplier and orbit consumers. Original periods and central-sign representatives are consumed. Analytic reciprocity and full all-weight goal remain; 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
