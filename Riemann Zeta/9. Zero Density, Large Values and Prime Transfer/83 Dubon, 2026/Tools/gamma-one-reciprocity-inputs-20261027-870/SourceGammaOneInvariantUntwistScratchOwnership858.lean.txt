import SourceWeightTwoPeriodUntwist
import SourceMeromorphicNormalUntwist
import SourceMeromorphicPeriodUntwist
import SourceMeromorphicTranslatedGerms
import SourceMeromorphicCuspUntwist
import SourceCuspDerivativeDecay
import SourceMeromorphicUntwistDecay
import SourceMeromorphicUntwistDerivative
import SourceGammaOneInvariantUntwist
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceWeightTwoPeriodUntwist, `SourceMeromorphicNormalUntwist, `SourceMeromorphicPeriodUntwist, `SourceMeromorphicTranslatedGerms, `SourceMeromorphicCuspUntwist, `SourceCuspDerivativeDecay, `SourceMeromorphicUntwistDecay, `SourceMeromorphicUntwistDerivative, `SourceGammaOneInvariantUntwist]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
