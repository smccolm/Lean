private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceWeightTwoSignedForms, `SourceWeightTwoSignedReciprocity]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.SignedForms.weightTwo_slash_neg, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.extend_apply, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.extend_injective, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.along_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.period_extend_representative, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.dualPeriod_comp_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.integralLattice_comp_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.multiplier_extend, `Dubon2026.SourceFunctionField.GammaOne.SignedForms.orbit_iff]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.SignedForms.extend]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Signed Reciprocity postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Signed Reciprocity theorems discovered"
  logInfo m!"SOURCE GAMMA ONE SIGNED RECIPROCITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
