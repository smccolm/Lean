open Lean

#print axioms Dubon2026.SourceFunctionField.GammaOne.exists_analytic_support_representatives
#print axioms Dubon2026.SourceFunctionField.GammaOne.sum_on_analytic_representatives_eq_degree
#print axioms Dubon2026.SourceFunctionField.GammaOne.weighted_analytic_order_eq_half_coefficient
#print axioms Dubon2026.SourceFunctionField.GammaOne.exists_degree_zero_reciprocity_data

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneDivisorReciprocityData]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_analytic_support_representatives, `Dubon2026.SourceFunctionField.GammaOne.sum_on_analytic_representatives_eq_degree, `Dubon2026.SourceFunctionField.GammaOne.weighted_analytic_order_eq_half_coefficient, `Dubon2026.SourceFunctionField.GammaOne.exists_degree_zero_reciprocity_data]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Divisor Reciprocity Data postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Divisor Reciprocity Data theorems discovered"
  logInfo m!"SOURCE GAMMA ONE DIVISOR RECIPROCITY DATA AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceGammaOneDivisorReciprocityData
