private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceWeightTwoPeriodUntwist, `SourceMeromorphicNormalUntwist, `SourceMeromorphicPeriodUntwist, `SourceMeromorphicTranslatedGerms, `SourceMeromorphicCuspUntwist, `SourceCuspDerivativeDecay, `SourceMeromorphicUntwistDecay, `SourceMeromorphicUntwistDerivative, `SourceGammaOneInvariantUntwist]
  let theorems : Array Name := #[`Dubon2026.SourcePeriodUntwist.Ek_hasDerivAt, `Dubon2026.SourcePeriodUntwist.Ek_smul, `Dubon2026.SourcePeriodUntwist.Ek_exists_cusp_limit, `Dubon2026.SourcePeriodUntwist.differentiableOn_Ek, `Dubon2026.SourcePeriodUntwist.analyticAt_Ek, `Dubon2026.SourcePeriodUntwist.R_ne_zero, `Dubon2026.SourcePeriodUntwist.norm_R, `Dubon2026.SourcePeriodUntwist.multiplier_mul_R_smul, `Dubon2026.SourcePeriodUntwist.hasFDerivAt_R, `Dubon2026.SourcePeriodUntwist.fderiv_R_apply, `Dubon2026.SourcePeriodUntwist.contDiffAt_R, `Dubon2026.SourcePeriodUntwist.normalized_eq_of_analytic, `Dubon2026.SourcePeriodUntwist.normalized_eq_of_tendsto, `Dubon2026.SourcePeriodUntwist.normalized_eq_zero_of_order_ne_zero, `Dubon2026.SourcePeriodUntwist.normalized_local_form, `Dubon2026.SourcePeriodUntwist.normalized_eventually_regular, `Dubon2026.SourcePeriodUntwist.order_smul_of_multiplier, `Dubon2026.SourcePeriodUntwist.reg_smul_of_multiplier, `Dubon2026.SourcePeriodUntwist.Φ_smul, `Dubon2026.SourcePeriodUntwist.Φ_local, `Dubon2026.SourcePeriodUntwist.order_translate, `Dubon2026.SourcePeriodUntwist.meromorphicAt_translate, `Dubon2026.SourcePeriodUntwist.reg_eq_nhdsNE, `Dubon2026.SourcePeriodUntwist.analyticAt_reg_of_order_zero, `Dubon2026.SourcePeriodUntwist.meromorphicOrderAt_eq_zero_of_near, `Dubon2026.SourcePeriodUntwist.near_cusp_translated, `Dubon2026.SourcePeriodUntwist.reg_near_cusp, `Dubon2026.SourcePeriodUntwist.tendsto_reg_cusp, `Dubon2026.SourcePeriodUntwist.Φ_cusp, `Dubon2026.SourcePeriodUntwist.eventually_imInf_iff, `Dubon2026.SourcePeriodUntwist.tendsto_deriv_zero, `Dubon2026.SourcePeriodUntwist.analyticAt_translate, `Dubon2026.SourcePeriodUntwist.analyticAt_reg_translate, `Dubon2026.SourcePeriodUntwist.Φ_decay, `Dubon2026.SourcePeriodUntwist.countable_bad, `Dubon2026.SourcePeriodUntwist.Φ_ae, `Dubon2026.sourceMeromorphic_exists_invariant_untwist, `Dubon2026.sourceGamma1_meromorphic_invariant_untwist]
  let constructions : Array Name := #[`Dubon2026.SourcePeriodUntwist.Ek, `Dubon2026.SourcePeriodUntwist.R, `Dubon2026.SourcePeriodUntwist.reg, `Dubon2026.SourcePeriodUntwist.Φ, `Dubon2026.SourcePeriodUntwist.Good]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Invariant Untwist postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Invariant Untwist theorems discovered"
  logInfo m!"SOURCE GAMMA ONE INVARIANT UNTWIST AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
