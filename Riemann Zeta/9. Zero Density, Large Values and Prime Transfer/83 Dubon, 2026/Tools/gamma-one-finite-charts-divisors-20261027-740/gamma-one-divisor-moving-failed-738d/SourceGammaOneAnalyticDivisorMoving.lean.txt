/-! DUB-17/18 -> actual Gamma1 Picard/period/Hecke comparison -> analytic
representatives of every divisor class. Prescribe orders on the finite
support of the original j coordinate and subtract the actual principal
divisor. Proved finite-place exhaustion then realizes the remaining support.
This reuses SourceOriginalAnalyticDivisorMoving and the existing curve
geometry, without requiring cusp charts or a new divisor framework. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane SourceCurve
open scoped MatrixGroups
variable {Γ : Subgroup SL(2, ℤ)}
local notation "F" => functionFieldOver Γ ℂ

theorem exists_divisor_analytic_decomposition (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) (hT : ModularGroup.T ∈ Γ)
    (D : Divisor ℂ F) :
    ∃ (E : Divisor ℂ F) (c : ℍ →₀ ℤ),
      Divisor.IsPrincipal E ∧ Divisor.degree E = 0 ∧
      D = E + Finsupp.mapDomain (analyticPlace hT) c := by
  classical
  letI : Γ.FiniteIndex := Subgroup.finiteIndex_of_le hΓ
  letI := finiteDimensional_jCoordinateOver hT ℂ
  letI : IsCurveOver ℂ F := functionFieldOver_isCurveOver Γ ℂ hT
  let S := (finite_function_support_of_separating_coordinate (jCoordinateOver Γ ℂ)
    (transcendental_jCoordinateOver Γ ℂ) (jCoordinateOver Γ ℂ)).toFinset
  obtain ⟨x, hx0, hx⟩ := exists_prescribed_place_orders hT S D
  obtain ⟨E, hE, hE0⟩ := HasPrincipalDivisors.exists_divisor (K := ℂ) («F» := F) x hx0
  let D' := D - E
  have hfinite : ∀ v ∈ D'.support, jCoordinateOver Γ ℂ ∈ v.toValuationSubring := by
    intro v hv
    apply v.mem_of_ord_nonneg
    have hnot : v ∉ S := by
      intro hS
      apply Finsupp.mem_support_iff.mp hv
      change D v - E v = 0
      rw [hE, hx v hS, sub_self]
    have he : v.ord (jCoordinateOver Γ ℂ) = 0 := by
      simpa only [S, Set.Finite.mem_toFinset, Set.mem_setOf_eq, not_not] using hnot
    exact he ▸ le_rfl
  let lift : Place ℂ F → ℍ := fun v =>
    if h : jCoordinateOver Γ ℂ ∈ v.toValuationSubring then
      (exists_analytic_place_eq N hΓ hT v h).choose else I
  have hlift : ∀ v ∈ D'.support, analyticPlace hT (lift v) = v := by
    intro v hv
    simp only [lift, dif_pos (hfinite v hv)]
    exact (exists_analytic_place_eq N hΓ hT v (hfinite v hv)).choose_spec
  have hcomp : Finsupp.mapDomain (analyticPlace hT) (Finsupp.mapDomain lift D') = D' := by
    rw [← Finsupp.mapDomain_comp]
    calc
      Finsupp.mapDomain (analyticPlace hT ∘ lift) D' = Finsupp.mapDomain id D' :=
        Finsupp.mapDomain_congr (fun v hv => hlift v hv)
      _ = D' := by rw [Finsupp.mapDomain_id]
  refine ⟨E, Finsupp.mapDomain lift D', ⟨x, hx0, hE⟩, hE0, ?_⟩
  rw [hcomp]
  dsimp only [D']
  abel

theorem degree_mapDomain_analyticPlace [Γ.FiniteIndex]
    (hT : ModularGroup.T ∈ Γ) (c : ℍ →₀ ℤ) :
    Divisor.degree (Finsupp.mapDomain (analyticPlace hT) c) =
      c.sum (fun _ n => n) := by
  classical
  have hr : ∀ P : Place ℂ F, Module.finrank ℂ P.ResidueField = 1 := fun P =>
    Module.finrank_of_bijective_algebraMap (place_residue_bijective hT P)
  induction c using Finsupp.induction with
  | zero => simp
  | single_add a b f _ _ ih =>
    rw [Finsupp.mapDomain_add, map_add, Finsupp.mapDomain_single, ih,
      Finsupp.sum_add_index' (fun _ => rfl) (fun _ _ _ => rfl), Finsupp.sum_single_index rfl,
      Divisor.degree_single, hr]
    simp

theorem exists_degree_zero_analytic_representative (N : ℕ) [NeZero N]
    (hΓ : CongruenceSubgroup.Gamma N ≤ Γ) (hT : ModularGroup.T ∈ Γ)
    (D : Divisor ℂ F) (hD : Divisor.degree D = 0) :
    ∃ (E : Divisor ℂ F) (c : ℍ →₀ ℤ),
      Divisor.IsPrincipal E ∧ Divisor.degree E = 0 ∧ c.sum (fun _ n => n) = 0 ∧
      D = E + Finsupp.mapDomain (analyticPlace hT) c := by
  letI : Γ.FiniteIndex := Subgroup.finiteIndex_of_le hΓ
  obtain ⟨E, c, hE, hE0, he⟩ := exists_divisor_analytic_decomposition N hΓ hT D
  refine ⟨E, c, hE, hE0, ?_, he⟩
  rw [he, map_add, hE0, zero_add, degree_mapDomain_analyticPlace] at hD
  exact hD

end
end Dubon2026.SourceFunctionField.GammaOne
