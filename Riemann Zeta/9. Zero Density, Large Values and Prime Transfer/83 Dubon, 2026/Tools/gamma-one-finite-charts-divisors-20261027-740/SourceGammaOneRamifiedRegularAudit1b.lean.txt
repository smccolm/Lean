private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneEllipticFibers, `SourceGammaOneRamifiedRegularParameter, `SourceGammaOneRamifiedRegularOrder]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.exists_stabilizer_multiplier, `Dubon2026.SourceFunctionField.GammaOne.place_eq_of_coord_power_eq, `Dubon2026.SourceFunctionField.GammaOne.ramifiedPlace_zero, `Dubon2026.SourceFunctionField.GammaOne.ramifiedPlace_regular_analytic, `Dubon2026.SourceFunctionField.GammaOne.ramifiedPlace_regular_order]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.ramifiedPlace]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Ramified Regular postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Ramified Regular theorems discovered"
  logInfo m!"SOURCE GAMMA ONE RAMIFIED REGULAR AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
