import Dubon2026.SourceResidualFiniteLinearAlternatives
import Mathlib.LinearAlgebra.Matrix.Trace

/-! DUB-17/18 -> original regular-prime local invariant dimensions ->
original-field eigenvalues. The full SL2 lower bound forces every subfield
coefficient to lie in the original residue field, by prescribed SL2 traces.
This avoids assuming that the auxiliary extension is the original field. -/
namespace Dubon2026.SourceDickson
noncomputable section
open Matrix

/-- Every field element is the trace of an actual determinant-one matrix. -/
theorem sl2_exists_prescribed_trace {F : Type} [Field F] (t : F) :
    ∃ A : SpecialLinearGroup (Fin 2) F, Matrix.trace A.val = t := by
  refine ⟨⟨!![t, -1; 1, 0], by simp [Matrix.det_fin_two]⟩, ?_⟩
  simp [Matrix.trace, Fin.sum_univ_two]

/-- A full conjugated subfield SL2 in the original image forces its entire
coefficient field into the original coefficient embedding, using actual traces. -/
theorem conjugated_subfield_sl2_coefficients_descend
    {F E k G : Type} [Field F] [Field E] [Field k] [Group G]
    (j : F →+* E) (θ : k →+* E) (ρ : G →* GeneralLinearGroup (Fin 2) F)
    (B : GeneralLinearGroup (Fin 2) E)
    (hlo : ((GeneralLinearGroup.map θ).comp
      (SpecialLinearGroup.toGL (n := Fin 2) (R := k))).range ≤
      ((MulEquiv.toMonoidHom (MulAut.conj B)).comp
        ((GeneralLinearGroup.map j).comp ρ)).range) :
    ∀ t : k, ∃ r : F, j r = θ t := by
  intro t
  obtain ⟨A, hA⟩ := sl2_exists_prescribed_trace t
  obtain ⟨g, hg⟩ := hlo ⟨A, rfl⟩
  have ht := congrArg (fun M : GeneralLinearGroup (Fin 2) E => Matrix.trace M.val) hg
  change Matrix.trace (B.val * (ρ g).val.map j * (B⁻¹).val) =
    Matrix.trace (A.val.map θ) at ht
  rw [Matrix.trace_units_conj] at ht
  refine ⟨Matrix.trace (ρ g).val, ?_⟩
  rw [AddMonoidHom.map_trace, ht, ← AddMonoidHom.map_trace, hA]

end
end Dubon2026.SourceDickson
