/-- The original torus prime output supplies every input to the native
cyclic-image calculation over the same original coefficient field. -/
theorem rationalArithmetic_torus_detection_supplies_local_dimensions
    (a : 𝓞 ℚ) (p n level : ℕ) [Fact p.Prime] (hn : 2 * n < p - 1)
    {K : Type} [Field K] [Finite K] [CharP K p]
    [TopologicalSpace K] [DiscreteTopology K] [IsTopologicalRing K]
    (A : Rep.{0} K (rationalArithmeticGaloisGroup a))
    [TopologicalSpace A] [DiscreteTopology A] [ContinuousSMul K A]
    (hA : Continuous (fun z : rationalArithmeticGaloisGroup a × A => A.ρ z.1 z.2))
    (ρ : rationalArithmeticGaloisGroup a →* GeneralLinearGroup (Fin (n + 1)) K)
    (hρ : Continuous ρ) (c : continuousRepresentationCocyclesOne A)
    (hprimes : HasArithmeticTorusDetectingPrimes a p n level A hA ρ c) :
    HasArithmeticTorusDimensionalPrimes a p n level A hA ρ hρ c := by
  dsimp only [HasArithmeticTorusDimensionalPrimes]
  dsimp only [HasArithmeticTorusDetectingPrimes] at hprimes
  let U := rationalUnramifiedExtension a
  letI : Algebra ℚ U := U.algebra'
  letI : IsGalois ℚ U := originalUnramifiedGaloisUnion_isGalois a
  obtain ⟨u, hu, D, ψ, hψ, hD⟩ := hprimes
  letI : Algebra ℚ D := D.algebra'
  letI : NumberField D := NumberField.of_module_finite ℚ D
  exact ⟨u, hu, D, ψ, hψ, rationalArithmetic_torus_finite_factor_dimensions
    a p n level hn A hA ρ hρ c u hu D ψ hψ hD⟩

end
end Dubon2026
