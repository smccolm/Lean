variable {R A : Type} [CommRing R] [AddCommGroup A] [Module R A]

/-- The genuine integer-row boundary action, with the full supplied
coefficient endomorphism applied to each original cusp value. -/
def primitiveBoundaryHecke (b : Module.End R A) :
    Module.End R ((SL(2, ℤ) ⧸ Gamma1 M) → A) where
  toFun f q := ∑ o : orbitRel.Quotient (Subgroup.zpowers (primitiveCuspGenerator M q))
      (Gamma1 M ⧸ sourceGammaOneHeckeUpper M d),
    boundaryMultiplicity M d q o • b (f (boundaryTarget M d q o))
  map_add' f g := by
    funext q
    simp only [Pi.add_apply, map_add, smul_add, Finset.sum_add_distrib]
  map_smul' a f := by
    funext q
    change (∑ o, boundaryMultiplicity M d q o • b (a • f (boundaryTarget M d q o))) =
      a • ∑ o, boundaryMultiplicity M d q o • b (f (boundaryTarget M d q o))
    rw [Finset.smul_sum]
    apply Finset.sum_congr rfl
    intro o _
    rw [map_smul]
    exact smul_comm _ _ _

    (primitiveBoundaryHecke_exists_integer_rows M d q))) b x
