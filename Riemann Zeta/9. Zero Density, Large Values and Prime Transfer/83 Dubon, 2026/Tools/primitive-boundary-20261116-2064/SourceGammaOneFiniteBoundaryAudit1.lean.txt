private def runGammaOneFiniteBoundaryAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOnePrimitiveBoundaryOperators, `SourceGammaOnePrimitiveBoundaryWords, `SourceGammaOneFiniteBoundaryCharacter]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.primitiveBoundaryHecke_hom_values, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveCuspBoundary_prime, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveCuspBoundary_word, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveBoundary_word_zero_parabolic, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitivePrimePowerBoundary_finite, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_gammaOne_diamond_quotient_original_character_of_finite_boundary_word]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.primitiveBoundaryHecke, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveBoundaryPrimeOperators]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE GAMMAONE FINITE BOUNDARY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runGammaOneFiniteBoundaryAudit
