private def runGammaOnePrimitiveBoundaryAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOnePrimitiveCuspBoundary, `SourceGammaOneFiniteBinaryBoundary]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.primitiveCuspWidth_pos, `Dubon2026.SourceFunctionField.GammaOne.primitiveCuspGenerator_trace, `Dubon2026.SourceFunctionField.GammaOne.character_parabolic_of_primitive_boundary, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveCuspBoundary_eq_zero_iff, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveCuspBoundary_ker, `Dubon2026.SourceFunctionField.GammaOne.binaryHeckeWord_parabolic_iff_primitive_boundary_zero, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimePowerHeckeWord_parabolic_iff_primitive_boundary_zero]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.primitiveCuspWidth, `Dubon2026.SourceFunctionField.GammaOne.primitiveCuspGenerator, `Dubon2026.SourceFunctionField.GammaOne.binaryPrimitiveCuspBoundary]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Marked algebraic chart postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Marked algebraic chart theorems discovered"
  logInfo m!"SOURCE GAMMAONE PRIMITIVE BOUNDARY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runGammaOnePrimitiveBoundaryAudit
