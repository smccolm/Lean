private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneHeckePointTransition, `SourceGammaOnePeriodNormalization, `SourceGammaOneScalarHeckePeriods, `SourceGammaOneHeckeCharacters, `SourceGammaOneHolomorphicH1Hecke]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.heckeCosetPoint_transition, `Dubon2026.SourceFunctionField.GammaOne.along_eq_primitive_one, `Dubon2026.SourceFunctionField.GammaOne.primitive_one_period, `Dubon2026.SourceFunctionField.GammaOne.periodValue_cuspHeckeTrace, `Dubon2026.sourceGammaOneConstantHeckeH1_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.holomorphicPeriodH1_hom_apply, `Dubon2026.SourceFunctionField.GammaOne.holomorphicPeriodH1_cuspHeckeTrace]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.holomorphicPeriodH1]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Holomorphic H1 Hecke postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Holomorphic H1 Hecke theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HOLOMORPHIC H1 HECKE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
