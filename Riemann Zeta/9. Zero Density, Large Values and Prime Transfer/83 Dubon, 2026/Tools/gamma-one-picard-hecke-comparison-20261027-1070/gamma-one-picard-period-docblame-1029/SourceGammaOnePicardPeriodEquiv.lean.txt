/-! DUB17/18 -> actual Gamma1 arithmetic realization -> Picard/H1 comparison.
The literal original complex Picard group is identified with the original
integer-period quotient by the two actual surjective maps with equal kernels.
The endpoint formula verifies that this equivalence is the intended Abel map.
Hecke, diamond, transpose and arithmetic compatibility remain separate work. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane SourceCurve CongruenceSubgroup SourceWeightTwoJacobi
open Matrix.SpecialLinearGroup
open scoped MatrixGroups GammaOne
variable (N : ℕ) [NeZero N] (hT : ModularGroup.T ∈ Gamma1 N)
local notation "V" => CuspForm (Subgroup.map (mapGL ℝ) (Gamma1 N)) 2
local notation "F" => functionFieldOver (Gamma1 N) ℂ

def picardPeriodEquiv : Pic0 ℂ F ≃+
    Module.Dual ℂ V ⧸ sourceWeightTwoIntegralLattice (Gamma1 N) :=
  (QuotientAddGroup.quotientKerEquivOfSurjective
    (analyticPicardHom N hT) (analyticPicardHom_surjective N hT)).symm.trans
      ((QuotientAddGroup.quotientAddEquivOfEq (analyticPicardHom_ker_eq_periodHom_ker N hT)).trans
        (QuotientAddGroup.quotientKerEquivOfSurjective
          (analyticPeriodHom N) (analyticPeriodHom_surjective N)))

theorem picardPeriodEquiv_analytic_divisor (c : analyticDegree.ker) :
    picardPeriodEquiv N hT (Pic0.mk (analyticDivisorHom N hT c)) =
      (sourceWeightTwoIntegralLattice (Gamma1 N)).mkQ
        (c.val.sum (fun τ a => a • sourceWeightTwoAlong (Gamma1 N) UpperHalfPlane.I τ)) := by
  let ePicard := QuotientAddGroup.quotientKerEquivOfSurjective
    (analyticPicardHom N hT) (analyticPicardHom_surjective N hT)
  let ePeriod := QuotientAddGroup.quotientKerEquivOfSurjective
    (analyticPeriodHom N) (analyticPeriodHom_surjective N)
  have hpre : ePicard.symm (analyticPicardHom N hT c) =
      (QuotientAddGroup.mk c : analyticDegree.ker ⧸ (analyticPicardHom N hT).ker) := by
    apply ePicard.injective
    rw [ePicard.apply_symm_apply]
    rfl
  change ePeriod ((QuotientAddGroup.quotientAddEquivOfEq
    (analyticPicardHom_ker_eq_periodHom_ker N hT))
      (ePicard.symm (analyticPicardHom N hT c))) = analyticPeriodHom N c
  rw [hpre]
  rfl

end
end Dubon2026.SourceFunctionField.GammaOne
