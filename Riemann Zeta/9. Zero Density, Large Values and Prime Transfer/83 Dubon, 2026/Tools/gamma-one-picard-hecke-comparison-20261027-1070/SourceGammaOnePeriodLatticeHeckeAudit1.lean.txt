private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCuspHeckeLinear, `SourceGammaOnePeriodHeckeLattice]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.cuspHeckeTrace_add, `Dubon2026.SourceFunctionField.GammaOne.cuspHeckeTrace_smul, `Dubon2026.SourceFunctionField.GammaOne.periodHeckeLinear_apply, `Dubon2026.SourceFunctionField.GammaOne.periodHeckeLinear_dualPeriod, `Dubon2026.SourceFunctionField.GammaOne.periodHeckeLinear_preserves_integralLattice, `Dubon2026.SourceFunctionField.GammaOne.periodQuotientHecke_mk, `Dubon2026.SourceFunctionField.GammaOne.periodHeckeLinear_along]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.cuspHeckeLinear, `Dubon2026.SourceFunctionField.GammaOne.periodHeckeLinear, `Dubon2026.SourceFunctionField.GammaOne.periodQuotientHecke]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Period Lattice Hecke postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Period Lattice Hecke theorems discovered"
  logInfo m!"SOURCE GAMMA ONE PERIOD LATTICE HECKE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
