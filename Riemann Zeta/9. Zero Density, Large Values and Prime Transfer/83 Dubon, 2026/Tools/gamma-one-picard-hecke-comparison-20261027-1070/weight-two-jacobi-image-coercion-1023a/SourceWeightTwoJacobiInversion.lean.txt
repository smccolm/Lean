/-! DUB17/18 -> actual Gamma1 Picard-period comparison -> Jacobi inversion.
The actual finite-divisor endpoint image contains a neighborhood of zero,
hence is an open and closed subgroup of connected complex coordinate space.
Every original dual cusp functional is represented by a finite degree-zero
integer divisor. No period-lattice or surjectivity certificate is assumed. -/
namespace Dubon2026.SourceWeightTwoJacobi

variable (Γ : Subgroup SL(2, ℤ)) [Γ.FiniteIndex]

theorem exists_degree_zero_endpoint_coordinates {n : ℕ}
    (b : Module.Basis (Fin n) ℂ (CuspForm (Γ.map (mapGL ℝ)) 2)) (u : Fin n → ℂ) :
    ∃ c : ℍ →₀ ℤ, c.sum (fun _ a => a) = 0 ∧ endpointCoordinates Γ b c = u := by
  classical
  let v : ℍ → (Fin n → ℂ) := fun τ i => b i τ
  obtain ⟨κ, a, -, hspan, hli⟩ := exists_linearIndependent' ℂ v
  haveI : Finite κ := hli.finite
  letI : Fintype κ := Fintype.ofFinite κ
  have htop : ⊤ ≤ Submodule.span ℂ (range (v ∘ a)) := by
    rw [hspan, show Submodule.span ℂ (range v) = ⊤ from span_cusp_evaluation_eq_top Γ b]
  let β : Module.Basis κ ℂ (Fin n → ℂ) := Module.Basis.mk hli htop
  have hβ : ∀ k, β k = fun i => b i (a k) := by
    intro k
    simp only [β, Module.Basis.mk_apply]
    rfl
  let S := endpointImage Γ b
  have hopen : IsOpen (S : Set (Fin n → ℂ)) :=
    S.isOpen_of_mem_nhds (endpointImage_mem_nhds_of_frame Γ b a β hβ)
  have hclosed : IsClosed (S : Set (Fin n → ℂ)) := S.isClosed_of_isOpen hopen
  have hS : (S : Set (Fin n → ℂ)) = univ := IsClopen.eq_univ ⟨hclosed, hopen⟩ ⟨0, S.zero_mem⟩
  have hu : u ∈ S := by
    rw [hS]
    exact mem_univ u
  exact (mem_endpointImage_iff Γ b u).mp hu

theorem exists_degree_zero_endpoint_functional {n : ℕ}
    (b : Module.Basis (Fin n) ℂ (CuspForm (Γ.map (mapGL ℝ)) 2))
    (u : Module.Dual ℂ (CuspForm (Γ.map (mapGL ℝ)) 2)) :
    ∃ c : ℍ →₀ ℤ, c.sum (fun _ a => a) = 0 ∧
      c.sum (fun τ a => a • sourceWeightTwoAlong Γ UpperHalfPlane.I τ) = u := by
  obtain ⟨c, hc, he⟩ := exists_degree_zero_endpoint_coordinates Γ b (fun i => u (b i))
  refine ⟨c, hc, b.ext fun i => ?_⟩
  simpa only [endpointCoordinates, Finsupp.linearCombination_apply, Finsupp.sum,
    Finset.sum_apply, Pi.smul_apply, LinearMap.sum_apply, LinearMap.smul_apply] using congrFun he i

end
