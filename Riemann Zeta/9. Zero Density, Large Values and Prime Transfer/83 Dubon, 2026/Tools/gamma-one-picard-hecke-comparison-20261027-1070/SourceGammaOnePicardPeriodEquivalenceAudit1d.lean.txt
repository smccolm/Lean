private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceWeightTwoEvaluationFrame, `SourceWeightTwoEndpointImage, `SourceWeightTwoEndpointOpenImage, `SourceWeightTwoJacobiInversion, `SourceGammaOneJacobiInversion, `SourceGammaOneAnalyticPicardPresentation, `SourceGammaOnePicardPeriodKernel, `SourceGammaOnePicardPeriodEquiv]
  let theorems : Array Name := #[`Dubon2026.SourceWeightTwoJacobi.span_cusp_evaluation_eq_top, `Dubon2026.SourceWeightTwoJacobi.analyticDegree_apply, `Dubon2026.SourceWeightTwoJacobi.mem_endpointImage_iff, `Dubon2026.SourceWeightTwoJacobi.endpoint_difference_mem_image, `Dubon2026.SourceWeightTwoJacobi.endpointImage_mem_nhds_of_frame, `Dubon2026.SourceWeightTwoJacobi.exists_degree_zero_endpoint_coordinates, `Dubon2026.SourceWeightTwoJacobi.exists_degree_zero_endpoint_functional, `Dubon2026.SourceFunctionField.GammaOne.exists_analytic_divisor_endpoint_functional, `Dubon2026.SourceFunctionField.GammaOne.analyticPicardHom_surjective, `Dubon2026.SourceFunctionField.GammaOne.analyticPeriodHom_surjective, `Dubon2026.SourceFunctionField.GammaOne.analyticPicardHom_ker_eq_periodHom_ker, `Dubon2026.SourceFunctionField.GammaOne.picardPeriodEquiv_analytic_divisor]
  let constructions : Array Name := #[`Dubon2026.SourceWeightTwoJacobi.analyticDegree, `Dubon2026.SourceWeightTwoJacobi.endpointCoordinates, `Dubon2026.SourceWeightTwoJacobi.endpointImage, `Dubon2026.SourceFunctionField.GammaOne.analyticDivisorHom, `Dubon2026.SourceFunctionField.GammaOne.analyticPicardHom, `Dubon2026.SourceFunctionField.GammaOne.analyticEndpointHom, `Dubon2026.SourceFunctionField.GammaOne.analyticPeriodHom, `Dubon2026.SourceFunctionField.GammaOne.picardPeriodEquiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Picard Period Equivalence postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Picard Period Equivalence theorems discovered"
  logInfo m!"SOURCE GAMMA ONE PICARD PERIOD EQUIVALENCE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
