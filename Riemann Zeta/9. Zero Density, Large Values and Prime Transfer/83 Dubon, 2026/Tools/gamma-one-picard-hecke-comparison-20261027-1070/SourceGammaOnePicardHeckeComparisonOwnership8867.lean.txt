private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOneInvariantDivisorPeriod, `Dubon2026.SourceGammaOnePrincipalMeromorphicRealization, `Dubon2026.SourceGammaOneAnalyticAbelKernel, `Dubon2026.SourceWeightTwoEvaluationFrame, `Dubon2026.SourceWeightTwoEndpointImage, `Dubon2026.SourceWeightTwoEndpointOpenImage, `Dubon2026.SourceWeightTwoJacobiInversion, `Dubon2026.SourceGammaOneJacobiInversion, `Dubon2026.SourceGammaOneAnalyticPicardPresentation, `Dubon2026.SourceGammaOnePicardPeriodKernel, `Dubon2026.SourceGammaOnePicardPeriodEquiv, `Dubon2026.SourceGammaOneHeckeAlphaGerms, `Dubon2026.SourceGammaOneHeckeStretchLaurent, `Dubon2026.SourceGammaOneHeckeBetaGerms, `Dubon2026.SourceGammaOneHeckeAnalyticPlaces, `Dubon2026.SourceGammaOneHeckeAnalyticFiber, `Dubon2026.SourceGammaOneWeightTwoHeckeCosets, `Dubon2026.SourceGammaOneHeckePrimitiveDerivative, `Dubon2026.SourceGammaOneHeckeTraceCosets, `Dubon2026.SourceGammaOneCuspHeckeTrace, `Dubon2026.SourceGammaOneHeckePrimitiveTrace, `Dubon2026.SourceGammaOneHeckePointTransition, `Dubon2026.SourceGammaOnePeriodNormalization, `Dubon2026.SourceGammaOneScalarHeckePeriods, `Dubon2026.SourceGammaOneHeckeCharacters, `Dubon2026.SourceGammaOneHolomorphicH1Hecke, `Dubon2026.SourceGammaOneCuspHeckeLinear, `Dubon2026.SourceGammaOnePeriodHeckeLattice, `Dubon2026.SourceGammaOneHeckeGammaZeroMap, `Dubon2026.SourceGammaOneHeckeGammaZeroPlaces, `Dubon2026.SourceGammaOneHeckeGenericOrbits, `Dubon2026.SourceGammaOneHeckeFiberPresentation, `Dubon2026.SourceGammaOneHeckeGenericFiber, `Dubon2026.SourceGammaOneHeckeJDegree, `Dubon2026.SourceGammaOneHeckeAllFibers, `Dubon2026.SourceGammaOneHeckeExactRamification, `Dubon2026.SourceGammaOneHeckePlaceOrbits, `Dubon2026.SourceGammaOneCentralStabilizerCard, `Dubon2026.SourceGammaOneHeckeActualStabilizers, `Dubon2026.SourceGammaOneHeckeUpperOrbits, `Dubon2026.SourceGammaOneHeckeCosetMultiplicity, `Dubon2026.SourceGammaOneHeckeAnalyticDivisor, `Dubon2026.SourceGammaOneAnalyticHeckeDivisors, `Dubon2026.SourceGammaOnePicardHeckeDivisors, `Dubon2026.SourceGammaOneEndpointHecke, `Dubon2026.SourceGammaOnePicardPeriodHecke]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
