/-! DUB17/18 -> actual Gamma1 Picard/Hecke comparison -> exact geometric
fibers. At a point with two-element full stabilizer, the already proved
Gamma1 and original Gamma0 orbit identifications combine to give the exact
intersection orbit. This generic statement is an input to the fixed-degree
argument; it does not assume or assert the elliptic case. -/

theorem stabilizer_eq_one_or_neg_one_of_card_two (τ : ℍ)
    (hc : Nat.card (MulAction.stabilizer SL(2, ℤ) τ) = 2)
    (g : SL(2, ℤ)) (hg : g • τ = τ) : g = 1 ∨ g = -1 := by
  classical
  by_contra h
  push Not at h
  letI := Fintype.ofFinite (MulAction.stabilizer SL(2, ℤ) τ)
  have h3 : 2 < Nat.card (MulAction.stabilizer SL(2, ℤ) τ) := by
    rw [Nat.card_eq_fintype_card, Fintype.two_lt_card_iff]
    refine ⟨1, StabilizerCharacter.negOne τ,
      ⟨g, MulAction.mem_stabilizer_iff.mpr hg⟩, ?_, ?_, ?_⟩
    · intro he
      have hv := congrArg (fun z : MulAction.stabilizer SL(2, ℤ) τ => z.val 0 0) he
      norm_num [StabilizerCharacter.negOne] at hv
    · exact fun he => h.1 (congrArg Subtype.val he).symm
    · exact fun he => h.2 (congrArg Subtype.val he).symm
  omega

theorem analyticHeckePlace_eq_iff_orbit_of_generic (M d : ℕ) [NeZero M] [NeZero d]
    (hH : ModularGroup.T ∈ Gamma1 M ⊓ Gamma0 (d * M)) (τ τ' : ℍ)
    (hc : Nat.card (MulAction.stabilizer SL(2, ℤ) τ) = 2) :
    analyticPlace hH τ = analyticPlace hH τ' ↔
      ∃ γ : Gamma1 M ⊓ Gamma0 (d * M), (γ : SL(2, ℤ)) • τ = τ' := by
  constructor
  · intro he
    have hT : ModularGroup.T ∈ Gamma1 M := by simp [Gamma1_mem, ModularGroup.T]
    have hα := congrArg
      (Place.restrictAlong (heckeAlpha ℂ M d) (heckeAlpha_finite ℂ M d)) he
    rw [analyticPlace_restrict_heckeAlpha M d hT hH,
      analyticPlace_restrict_heckeAlpha M d hT hH] at hα
    obtain ⟨g₁, hg₁⟩ := (analyticPlace_eq_iff_gamma1_orbit M hT τ τ').mp hα
    obtain ⟨g₀, hg₀⟩ := (originalAnalyticPlace_eq_iff_gamma0_orbit (d * M) τ τ').mp
      (originalGammaZeroPlace_eq_of_heckePlace_eq M d hH τ τ' he)
    have hfix : (g₀.val⁻¹ * g₁.val) • τ = τ := by
      rw [mul_smul, hg₁, ← hg₀, inv_smul_smul]
    have hm : g₀.val⁻¹ * g₁.val ∈ Gamma0 (d * M) := by
      rcases stabilizer_eq_one_or_neg_one_of_card_two τ hc _ hfix with h | h
      · rw [h]
        exact (Gamma0 (d * M)).one_mem
      · rw [h]
        simp [Gamma0_mem]
    have hmem : g₁.val ∈ Gamma0 (d * M) := by
      simpa only [mul_inv_cancel_left] using (Gamma0 (d * M)).mul_mem g₀.property hm
    exact ⟨⟨g₁.val, g₁.property, hmem⟩, hg₁⟩
  · rintro ⟨γ, rfl⟩
    exact (analyticPlace_smul hH γ τ).symm

end
