import SourceWeightTwoEndpointImage
import Mathlib.Analysis.Calculus.InverseFunctionTheorem.FDeriv

/-! DUB17/18 -> actual Gamma1 Picard-period comparison -> Jacobi inversion.
An evaluation frame makes a sum of the existing normalized cusp primitives
locally invertible. Its local image is represented by explicit finite
degree-zero upper-half-plane divisors. No algebraic-place chart is assumed. -/
namespace Dubon2026.SourceWeightTwoJacobi
noncomputable section
open UpperHalfPlane Filter Set Matrix.SpecialLinearGroup SourcePeriodUntwist
open scoped MatrixGroups Topology
variable (Γ : Subgroup SL(2, ℤ)) [Γ.FiniteIndex]

theorem endpointImage_mem_nhds_of_frame {n : ℕ}
    (b : Fin n → CuspForm (Γ.map (mapGL ℝ)) 2) {κ : Type*} [Fintype κ]
    (a : κ → ℍ) (β : Module.Basis κ ℂ (Fin n → ℂ))
    (hβ : ∀ k, β k = fun i => b i (a k)) :
    (endpointImage Γ b : Set (Fin n → ℂ)) ∈ 𝓝 (0 : Fin n → ℂ) := by
  classical
  let Φ : κ → Fin n → ℂ → ℂ := fun k i z => Ek (b i) (z + (a k : ℂ))
  let G : (κ → ℂ) → (Fin n → ℂ) := fun w => ∑ k, fun i => Φ k i (w k) - Φ k i 0
  have hG0 : G 0 = 0 := by
    ext i
    simp only [G, Finset.sum_apply, Pi.zero_apply, sub_self, Finset.sum_const_zero]
  have hderiv : ∀ k i, HasStrictDerivAt (Φ k i) (b i (a k)) 0 := by
    intro k i
    have ha : AnalyticAt ℂ (Φ k i) 0 :=
      (analyticAt_Ek (b i) (by simpa only [id_eq, zero_add] using (a k).im_pos)).comp
        (analyticAt_id.add (analyticAt_const (v := (a k : ℂ))))
    have hd : HasDerivAt (Φ k i) (b i (a k)) 0 := by
      simpa only [Φ, Function.comp_def, id_eq, zero_add, ofComplex_apply, mul_one] using
        (Ek_hasDerivAt (b i) (by simpa only [id_eq, zero_add] using (a k).im_pos)).comp 0
          ((hasDerivAt_id 0).add_const (a k : ℂ))
    have hs := ha.hasStrictDerivAt
    rwa [hd.deriv] at hs
  let e : (κ → ℂ) ≃L[ℂ] (Fin n → ℂ) := β.equivFun.symm.toContinuousLinearEquiv
  have he : ∀ w : κ → ℂ, e w = ∑ k, w k • (fun i => b i (a k)) := by
    intro w
    simp only [e, LinearEquiv.coe_toContinuousLinearEquiv', Module.Basis.equivFun_symm_apply, hβ]
  have hGderiv : HasStrictFDerivAt G (e : (κ → ℂ) →L[ℂ] (Fin n → ℂ)) 0 := by
    rw [hasStrictFDerivAt_pi']
    intro i
    have hsum : HasStrictFDerivAt (fun w : κ → ℂ => ∑ k, (Φ k i (w k) - Φ k i 0))
        (∑ k, (ContinuousLinearMap.toSpanSingleton ℂ (b i (a k))).comp
          (ContinuousLinearMap.proj k)) 0 := by
      refine HasStrictFDerivAt.fun_sum fun k _ => ?_
      exact ((hderiv k i).hasStrictFDerivAt.comp (f := fun w : κ → ℂ => w k)
        (0 : κ → ℂ) (hasStrictFDerivAt_apply k (0 : κ → ℂ))).sub_const _
    have hfun : (fun w : κ → ℂ => G w i) = fun w => ∑ k, (Φ k i (w k) - Φ k i 0) := by
      funext w
      simp only [G, Finset.sum_apply]
    rw [hfun]
    refine hsum.congr_fderiv ?_
    ext w
    simp [he, ContinuousLinearMap.sum_apply]
  have hmap := hGderiv.map_nhds_eq_of_equiv
  rw [hG0] at hmap
  have hGU : G '' Set.univ ∈ 𝓝 (0 : Fin n → ℂ) := by
    rw [← hmap]
    exact Filter.image_mem_map Filter.univ_mem
  apply Filter.mem_of_superset hGU
  rintro _ ⟨w, _, rfl⟩
  apply (endpointImage Γ b).sum_mem
  intro k _
  simpa only [Φ, Function.comp_def, id_eq, zero_add, Ek, ofComplex_apply] using
    endpoint_difference_mem_image Γ b (ofComplex (w k + (a k : ℂ))) (a k)

end
end Dubon2026.SourceWeightTwoJacobi
