private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCentralStabilizerCard, `SourceGammaOneHeckeActualStabilizers, `SourceGammaOneHeckeUpperOrbits, `SourceGammaOneHeckeCosetMultiplicity, `SourceGammaOneHeckeAnalyticDivisor, `SourceGammaOneAnalyticHeckeDivisors, `SourceGammaOnePicardHeckeDivisors, `SourceGammaOneEndpointHecke, `SourceGammaOnePicardPeriodHecke]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.card_stabilizer_plusMinus, `Dubon2026.SourceFunctionField.GammaOne.hecke_stabilizer_card, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_stabilizer_ramification, `Dubon2026.SourceFunctionField.GammaOne.analyticHeckePlace_eq_iff_upper_orbit, `Dubon2026.SourceFunctionField.GammaOne.heckeCosetPoint_restrict_beta, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_ramification_eq_coset_card, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_pullback_single_analytic, `Dubon2026.SourceFunctionField.GammaOne.heckeDivisor_single_analytic_cosets, `Dubon2026.SourceFunctionField.GammaOne.analyticHeckeLinear_apply, `Dubon2026.SourceFunctionField.GammaOne.analyticHeckeLinear_single, `Dubon2026.SourceFunctionField.GammaOne.analyticDegree_mapDomain, `Dubon2026.SourceFunctionField.GammaOne.analyticDegree_hecke, `Dubon2026.SourceFunctionField.GammaOne.heckeDivisor_mapDomain_analytic, `Dubon2026.SourceFunctionField.GammaOne.analyticPicardHom_hecke, `Dubon2026.SourceFunctionField.GammaOne.analyticEndpointHom_hecke, `Dubon2026.SourceFunctionField.GammaOne.analyticPeriodHom_hecke, `Dubon2026.SourceFunctionField.GammaOne.picardPeriodEquiv_hecke]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.analyticHeckeLinear, `Dubon2026.SourceFunctionField.GammaOne.analyticHeckeDegreeZero]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Picard Period Hecke postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Picard Period Hecke theorems discovered"
  logInfo m!"SOURCE GAMMA ONE PICARD PERIOD HECKE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
