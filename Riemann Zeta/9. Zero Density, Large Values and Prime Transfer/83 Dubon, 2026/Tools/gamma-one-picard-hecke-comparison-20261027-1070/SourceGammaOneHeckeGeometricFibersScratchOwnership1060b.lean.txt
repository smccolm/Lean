import SourceGammaOneHeckeGammaZeroMap
import SourceGammaOneHeckeGammaZeroPlaces
import SourceGammaOneHeckeGenericOrbits
import SourceGammaOneHeckeFiberPresentation
import SourceGammaOneHeckeGenericFiber
import SourceGammaOneHeckeJDegree
import SourceGammaOneHeckeAllFibers
import SourceGammaOneHeckeExactRamification
import SourceGammaOneHeckePlaceOrbits
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneHeckeGammaZeroMap, `SourceGammaOneHeckeGammaZeroPlaces, `SourceGammaOneHeckeGenericOrbits, `SourceGammaOneHeckeFiberPresentation, `SourceGammaOneHeckeGenericFiber, `SourceGammaOneHeckeJDegree, `SourceGammaOneHeckeAllFibers, `SourceGammaOneHeckeExactRamification, `SourceGammaOneHeckePlaceOrbits]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
