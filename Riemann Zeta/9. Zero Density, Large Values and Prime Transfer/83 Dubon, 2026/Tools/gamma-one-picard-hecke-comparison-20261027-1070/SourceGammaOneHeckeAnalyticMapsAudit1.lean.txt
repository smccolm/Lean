private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneHeckeAlphaGerms, `SourceGammaOneHeckeStretchLaurent, `SourceGammaOneHeckeBetaGerms, `SourceGammaOneHeckeAnalyticPlaces, `SourceGammaOneHeckeAnalyticFiber]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.realize_heckeAlpha_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.modularStretch_qL, `Dubon2026.SourceFunctionField.GammaOne.modularStretch_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.realize_heckeBeta_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_heckeAlpha, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_heckeBeta, `Dubon2026.SourceFunctionField.GammaOne.analyticPlace_restrict_heckeAlpha, `Dubon2026.SourceFunctionField.GammaOne.analyticPlace_restrict_heckeBeta, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha_analytic_ramification, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_analytic_ramification, `Dubon2026.SourceFunctionField.GammaOne.Gamma_le_hecke_geometric, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_reverse_j_relation, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_j_mem_of_restrict_analytic, `Dubon2026.SourceFunctionField.GammaOne.exists_analytic_place_of_restrict_heckeBeta]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Hecke Analytic Maps postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Hecke Analytic Maps theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HECKE ANALYTIC MAPS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
