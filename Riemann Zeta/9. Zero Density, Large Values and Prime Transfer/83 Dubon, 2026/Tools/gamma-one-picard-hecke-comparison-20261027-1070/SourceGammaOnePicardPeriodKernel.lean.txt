/-! DUB17/18 -> actual Gamma1 Picard-period comparison. The original endpoint
map onto the original integer-period quotient is surjective by the proved
Jacobi inversion. Its kernel equals the kernel of the actual Picard-class
map, by both directions of the finite-divisor Abel-Jacobi theorem. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane SourceCurve CongruenceSubgroup SourceWeightTwoJacobi
open Matrix.SpecialLinearGroup
open scoped MatrixGroups GammaOne
variable (N : ℕ) [NeZero N]
local notation "V" => CuspForm (Subgroup.map (mapGL ℝ) (Gamma1 N)) 2

/-- The original endpoint functional associated to a finite integer divisor. -/
def analyticEndpointHom : (ℍ →₀ ℤ) →ₗ[ℤ] Module.Dual ℂ V :=
  Finsupp.linearCombination ℤ (fun τ => sourceWeightTwoAlong (Gamma1 N) UpperHalfPlane.I τ)

/-- The class of a degree-zero endpoint functional modulo the original integral periods. -/
def analyticPeriodHom : analyticDegree.ker →+
    Module.Dual ℂ V ⧸ sourceWeightTwoIntegralLattice (Gamma1 N) :=
  ((sourceWeightTwoIntegralLattice (Gamma1 N)).mkQ.comp
    ((analyticEndpointHom N).comp analyticDegree.ker.subtype)).toAddMonoidHom

theorem analyticPeriodHom_surjective : Function.Surjective (analyticPeriodHom N) := by
  intro q
  obtain ⟨u, rfl⟩ := (sourceWeightTwoIntegralLattice (Gamma1 N)).mkQ_surjective q
  obtain ⟨c, hc, he⟩ := exists_analytic_divisor_endpoint_functional N u
  refine ⟨⟨c, by simpa only [LinearMap.mem_ker, analyticDegree_apply] using hc⟩, ?_⟩
  change (sourceWeightTwoIntegralLattice (Gamma1 N)).mkQ
    (c.sum (fun τ a => a • sourceWeightTwoAlong (Gamma1 N) UpperHalfPlane.I τ)) = _
  rw [he]

theorem analyticPicardHom_ker_eq_periodHom_ker (hT : ModularGroup.T ∈ Gamma1 N) :
    (analyticPicardHom N hT).ker = (analyticPeriodHom N).ker := by
  ext c
  change analyticPicardHom N hT c = 0 ↔ analyticPeriodHom N c = 0
  change Pic0.mk (analyticDivisorHom N hT c) = 0 ↔
    (sourceWeightTwoIntegralLattice (Gamma1 N)).mkQ
      (c.val.sum (fun τ a => a • sourceWeightTwoAlong (Gamma1 N) UpperHalfPlane.I τ)) = 0
  rw [Pic0.mk_eq_zero_iff, Submodule.mkQ_apply, Submodule.Quotient.mk_eq_zero]
  exact analytic_divisor_isPrincipal_iff_period_mem N hT c.val
    (by simpa only [LinearMap.mem_ker, analyticDegree_apply] using c.property)

end
end Dubon2026.SourceFunctionField.GammaOne
