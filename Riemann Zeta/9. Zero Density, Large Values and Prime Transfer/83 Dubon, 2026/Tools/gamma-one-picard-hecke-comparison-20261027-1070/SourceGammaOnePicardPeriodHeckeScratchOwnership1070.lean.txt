import SourceGammaOneCentralStabilizerCard
import SourceGammaOneHeckeActualStabilizers
import SourceGammaOneHeckeUpperOrbits
import SourceGammaOneHeckeCosetMultiplicity
import SourceGammaOneHeckeAnalyticDivisor
import SourceGammaOneAnalyticHeckeDivisors
import SourceGammaOnePicardHeckeDivisors
import SourceGammaOneEndpointHecke
import SourceGammaOnePicardPeriodHecke
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneCentralStabilizerCard, `SourceGammaOneHeckeActualStabilizers, `SourceGammaOneHeckeUpperOrbits, `SourceGammaOneHeckeCosetMultiplicity, `SourceGammaOneHeckeAnalyticDivisor, `SourceGammaOneAnalyticHeckeDivisors, `SourceGammaOnePicardHeckeDivisors, `SourceGammaOneEndpointHecke, `SourceGammaOnePicardPeriodHecke]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
