private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneHeckeGammaZeroMap, `SourceGammaOneHeckeGammaZeroPlaces, `SourceGammaOneHeckeGenericOrbits, `SourceGammaOneHeckeFiberPresentation, `SourceGammaOneHeckeGenericFiber, `SourceGammaOneHeckeJDegree, `SourceGammaOneHeckeAllFibers, `SourceGammaOneHeckeExactRamification, `SourceGammaOneHeckePlaceOrbits]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.heckeGammaZeroField_le, `Dubon2026.SourceFunctionField.GammaOne.realize_heckeGammaZero_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.analyticOrder_heckeGammaZero, `Dubon2026.SourceFunctionField.GammaOne.mem_heckeGammaZero_analyticPlace, `Dubon2026.SourceFunctionField.GammaOne.originalGammaZeroPlace_eq_of_heckePlace_eq, `Dubon2026.SourceFunctionField.GammaOne.stabilizer_eq_one_or_neg_one_of_card_two, `Dubon2026.SourceFunctionField.GammaOne.analyticHeckePlace_eq_iff_orbit_of_generic, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.j_rep, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.card_stabilizer_rep, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.finite_orbitQuot, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.toFiber_surjective, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.toFiber_injective_of_generic, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.sum_orders_of_generic, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.sum_relative_indices_eq_index, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.exists_fiber_order_sum_eq_index, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.finrank_over_j_eq_index, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.toFiber_bijective, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.sum_orders, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.sum_analytic_fiber_orders_eq_index, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.two_mul_ramification_eq_stabilizer_card, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.toFiber_mk_val, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.analyticHeckePlace_eq_iff_orbit]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.heckeGammaZeroMap, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.OrbitQuot, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.rep, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.Fiber, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.toFiber, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.equivOfGeneric, `Dubon2026.SourceFunctionField.GammaOne.HeckeFiberOrbits.equiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Hecke Geometric Fibers postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Hecke Geometric Fibers theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HECKE GEOMETRIC FIBERS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
