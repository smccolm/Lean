/-! DUB17/18 -> actual Gamma1 Picard/H1 comparison -> exact Abel-Jacobi
kernel for finite analytic divisors. Both directions refer to literal
principal divisors in the original integral-ratio field and the original
integer period lattice. The forward function is an actual modular quotient;
the converse function is constructed from the actual prescribed residues. -/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open UpperHalfPlane SourceCurve CongruenceSubgroup
open scoped MatrixGroups GammaOne

theorem analytic_divisor_period_mem_of_isPrincipal (N : ℕ) [NeZero N]
    (hT : ModularGroup.T ∈ Gamma1 N) (c : ℍ →₀ ℤ) (hc : c.sum (fun _ n => n) = 0)
    (hprincipal : Divisor.IsPrincipal (Finsupp.mapDomain (analyticPlace hT) c)) :
    c.sum (fun τ n => n • sourceWeightTwoAlong (Gamma1 N) UpperHalfPlane.I τ) ∈
      sourceWeightTwoIntegralLattice (Gamma1 N) := by
  have hΓ : Gamma N ≤ Gamma1 N := by
    intro A hA
    rw [Gamma_mem] at hA
    rw [Gamma1_mem]
    exact ⟨hA.1, hA.2.2.2, hA.2.2.1⟩
  obtain ⟨F, hF, hInv, hcusp, ho⟩ :=
    exists_invariant_meromorphic_of_principal N hΓ hT c hprincipal
  exact period_mem_of_invariant_meromorphic_function N hT c hc F hF hInv hcusp ho

theorem analytic_divisor_isPrincipal_iff_period_mem (N : ℕ) [NeZero N]
    (hT : ModularGroup.T ∈ Gamma1 N) (c : ℍ →₀ ℤ) (hc : c.sum (fun _ n => n) = 0) :
    Divisor.IsPrincipal (Finsupp.mapDomain (analyticPlace hT) c) ↔
      c.sum (fun τ n => n • sourceWeightTwoAlong (Gamma1 N) UpperHalfPlane.I τ) ∈
        sourceWeightTwoIntegralLattice (Gamma1 N) :=
  ⟨analytic_divisor_period_mem_of_isPrincipal N hT c hc,
    analytic_divisor_isPrincipal_of_period_mem N hT c hc⟩

end
end Dubon2026.SourceFunctionField.GammaOne
