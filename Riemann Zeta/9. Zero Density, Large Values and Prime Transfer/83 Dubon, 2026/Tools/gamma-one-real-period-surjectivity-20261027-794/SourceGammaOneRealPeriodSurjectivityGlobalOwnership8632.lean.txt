    unless (axioms.filter fun a => !allowed.contains a).isEmpty do
      throwError "Forbidden global dependencies: {name}: {axioms}"
    logInfo m!"GLOBAL {env.allImportedModuleNames[idx.toNat]!}: {name} {hash th.type} {hash th.value}"
