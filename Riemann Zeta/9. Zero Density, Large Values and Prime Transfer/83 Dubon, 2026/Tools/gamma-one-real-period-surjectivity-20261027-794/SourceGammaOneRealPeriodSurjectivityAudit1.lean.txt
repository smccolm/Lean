private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneEllipticCosetFiber, `SourceGammaOneEllipticFiberBounds, `SourceGammaOneRealPeriodSurjectivity]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.point_eq_of_leftRel, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.cosetToFiber_mk, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.cosetToFiber_surjective, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.cosetToFiber_stabilizer_smul, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.cyclicToFiber_surjective, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.card_fiber_le_cyclic_orbits, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.two_mul_card_fiber_I_le, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.three_mul_card_fiber_rho_le, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.two_mul_card_j_1728_fiber_le, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.three_mul_card_j_zero_fiber_le, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_real_parabolic_dimension_le_twice_cusp, `Dubon2026.sourceGamma1_real_parabolic_dimension_eq, `Dubon2026.sourceGamma1RealParabolicPeriodHom_surjective, `Dubon2026.sourceGamma1RealPeriodEquiv_apply, `Dubon2026.sourceGamma1_exists_cuspForm_of_real_parabolic_character]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.point, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.cosetToFiber, `Dubon2026.SourceFunctionField.GammaOne.EllipticCounting.cyclicToFiber, `Dubon2026.sourceGamma1RealPeriodEquiv]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Real Period Surjectivity postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Real Period Surjectivity theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REAL PERIOD SURJECTIVITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
