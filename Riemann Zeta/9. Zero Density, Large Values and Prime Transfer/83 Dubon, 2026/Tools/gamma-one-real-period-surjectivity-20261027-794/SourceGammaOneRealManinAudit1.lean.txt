private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRealManinCharacter, `SourceGammaOneRealCuspPotential, `SourceGammaOneRealManinDimension]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_of_mem, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_mul, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_one, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_inv, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_add, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_smul, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_parabolic, `Dubon2026.SourceFunctionField.GammaOne.RealManin.mk_rep, `Dubon2026.SourceFunctionField.GammaOne.RealManin.exists_gam, `Dubon2026.SourceFunctionField.GammaOne.RealManin.gam_mem, `Dubon2026.SourceFunctionField.GammaOne.RealManin.gam_smul, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi_eq_zero_of_fix, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Gpot_smul, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Ppot_mul_left, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Ppot_mul_right, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nX_mk, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nSL_two_term, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nSL_three_term, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nX_add, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nX_smul, `Dubon2026.SourceFunctionField.GammaOne.RealManin.sig2_apply, `Dubon2026.SourceFunctionField.GammaOne.RealManin.sig3_apply, `Dubon2026.SourceFunctionField.GammaOne.RealManin.mk_neg_one_mul, `Dubon2026.SourceFunctionField.GammaOne.RealManin.sig2_sq, `Dubon2026.SourceFunctionField.GammaOne.RealManin.sig3_cube, `Dubon2026.SourceFunctionField.GammaOne.RealManin.connected, `Dubon2026.SourceFunctionField.GammaOne.RealManin.theta_apply, `Dubon2026.SourceFunctionField.GammaOne.RealManin.theta_mem, `Dubon2026.SourceFunctionField.GammaOne.RealManin.thetaPlus_injective, `Dubon2026.SourceFunctionField.GammaOne.RealManin.finrank_W_le, `Dubon2026.SourceFunctionField.GammaOne.RealManin.parabolic_dimension_bound]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.RealManin.Phi, `Dubon2026.SourceFunctionField.GammaOne.RealManin.rep, `Dubon2026.SourceFunctionField.GammaOne.RealManin.gam, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Gpot, `Dubon2026.SourceFunctionField.GammaOne.RealManin.Ppot, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nSL, `Dubon2026.SourceFunctionField.GammaOne.RealManin.nX, `Dubon2026.SourceFunctionField.GammaOne.RealManin.X, `Dubon2026.SourceFunctionField.GammaOne.RealManin.sig2, `Dubon2026.SourceFunctionField.GammaOne.RealManin.sig3, `Dubon2026.SourceFunctionField.GammaOne.RealManin.W, `Dubon2026.SourceFunctionField.GammaOne.RealManin.theta, `Dubon2026.SourceFunctionField.GammaOne.RealManin.thetaPlus]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Real Manin postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Real Manin theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REAL MANIN AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
