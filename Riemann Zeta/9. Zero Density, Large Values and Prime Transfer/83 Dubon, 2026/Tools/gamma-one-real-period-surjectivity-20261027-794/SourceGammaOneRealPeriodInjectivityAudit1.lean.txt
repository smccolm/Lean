import SourceGammaOneRealPeriodInjectivity

open Lean

#print axioms Dubon2026.sourceGamma1_exists_tile_translate
#print axioms Dubon2026.sourceGamma1_realPart_constant
#print axioms Dubon2026.sourceGamma1_eq_zero_of_real_periods_zero
#print axioms Dubon2026.sourceGamma1_real_periods_injective

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneInvariantRealPart, `SourceGammaOneRealPeriodInjectivity]
  let theorems : Array Name := #[`Dubon2026.sourceGamma1_exists_tile_translate, `Dubon2026.sourceGamma1_realPart_constant, `Dubon2026.sourceGamma1_eq_zero_of_real_periods_zero, `Dubon2026.sourceGamma1_real_periods_injective]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Real Period Injectivity postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Real Period Injectivity theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REAL PERIOD INJECTIVITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
