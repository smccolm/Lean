private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCanonicalCoordinates, `SourceGammaOneJDegreeFinsets, `SourceGammaOneCanonicalFiberLower, `SourceGammaOneGenusFiberLower]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.jCanonicalDivisor_apply, `Dubon2026.SourceFunctionField.GammaOne.jCanonicalDivisor_degree, `Dubon2026.SourceFunctionField.GammaOne.ordDifferential_D_eq_order_sub_one, `Dubon2026.SourceFunctionField.GammaOne.jCanonicalDivisor_at_pole, `Dubon2026.SourceFunctionField.GammaOne.jCanonicalDivisor_at_value, `Dubon2026.SourceFunctionField.GammaOne.jCanonicalDivisor_nonneg_of_j_nonneg, `Dubon2026.SourceFunctionField.GammaOne.sum_j_fiber_orders_finset, `Dubon2026.SourceFunctionField.GammaOne.sum_j_pole_orders_finset, `Dubon2026.SourceFunctionField.GammaOne.j_fiber_nonneg, `Dubon2026.SourceFunctionField.GammaOne.j_fiber_value_unique, `Dubon2026.SourceFunctionField.GammaOne.jCanonical_degree_lower_finsets, `Dubon2026.SourceFunctionField.GammaOne.regular_dimension_lower_fiber_cards, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_cusp_dimension_lower_fiber_cards]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.jCanonicalDivisor]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Canonical Genus Lower postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Canonical Genus Lower theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CANONICAL GENUS LOWER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
