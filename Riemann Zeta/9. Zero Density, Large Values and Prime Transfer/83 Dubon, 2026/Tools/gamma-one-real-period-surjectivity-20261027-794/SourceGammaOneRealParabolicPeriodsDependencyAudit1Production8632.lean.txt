import Dubon2026.SourceGammaOneRealPeriodMap
import Dubon2026.SourceGammaOneRealParabolicPeriods

open Lean

private def runPlanarPairingDependencyAudit : CoreM Unit := do
  let env ← getEnv
  let roots : Array Name := #[`Dubon2026.sourceGamma1RealCharacter_finite, `Dubon2026.sourceGamma1RealPeriodHom_apply, `Dubon2026.sourceGamma1RealPeriodHom_injective, `Dubon2026.sourceGamma1CuspForm_finite_real, `Dubon2026.sourceGamma1_period_eq_zero_of_trace, `Dubon2026.sourceGamma1RealPeriodHom_parabolic, `Dubon2026.sourceGamma1RealParabolicPeriodHom_injective, `Dubon2026.sourceGamma1RealParabolicHoms_finite, `Dubon2026.sourceGamma1RealPeriodHom, `Dubon2026.sourceGamma1RealParabolicHoms, `Dubon2026.sourceGamma1RealParabolicPeriodHom]
  let localModules : Array Name := #[`Dubon2026.SourceGammaOneRealPeriodMap, `Dubon2026.SourceGammaOneRealParabolicPeriods, `Dubon2026.SourceGammaOneRealPeriodInjectivity, `Dubon2026.SourceGammaOneInvariantRealPart]
  let required : Array Name := #[`Dubon2026.sourceGamma1RealCharacter_finite, `Dubon2026.sourceGamma1RealPeriodHom_apply, `Dubon2026.sourceGamma1RealPeriodHom_injective, `Dubon2026.sourceGamma1CuspForm_finite_real, `Dubon2026.sourceGamma1_period_eq_zero_of_trace, `Dubon2026.sourceGamma1RealPeriodHom_parabolic, `Dubon2026.sourceGamma1RealParabolicPeriodHom_injective, `Dubon2026.sourceGamma1RealParabolicHoms_finite, `Dubon2026.sourceGamma1RealPeriodHom, `Dubon2026.sourceGamma1RealParabolicHoms, `Dubon2026.sourceGamma1RealParabolicPeriodHom, `Dubon2026.sourceGamma1_real_periods_injective, `Dubon2026.sourceH1Finite, `Dubon2026.sourceWeightTwoPrimitive_period, `Dubon2026.sourceWeightTwoPrimitive_exists_cusp_limit, `Dubon2026.SourceParabolic.classification]
  let mut pending := roots
  let mut seen : Std.HashSet Name := {}
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    if let some info := env.find? name then
      if let some idx := env.getModuleIdxFor? name then
        let mod := env.allImportedModuleNames[idx.toNat]!
        if mod.getRoot == `Dubon2026 || localModules.contains mod then
          if let some value := info.value? (allowOpaque := true) then
            pending := pending ++ value.getUsedConstants
  let missing := required.filter fun name => !seen.contains name
  unless missing.isEmpty do throwError "Missing original source proof dependencies: {missing}"
  logInfo m!"SOURCE GAMMA ONE REAL PARABOLIC PERIODS DEPENDENCY AUDIT PASS: all {required.size} required proof dependencies occur in the actual injective real parabolic period map and its finite target. Only declaration values are traversed. The upper dimension inequality, real period surjectivity, reciprocity, original arithmetic comparison and higher continuation remain; 16/20; DUB-17--20 OPEN."

#eval runPlanarPairingDependencyAudit
