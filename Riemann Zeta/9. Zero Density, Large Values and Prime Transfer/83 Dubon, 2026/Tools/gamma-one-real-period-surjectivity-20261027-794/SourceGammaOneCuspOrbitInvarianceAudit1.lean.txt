private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCuspPlaceLeftInvariance, `SourceGammaOneCuspTranslations, `SourceGammaOneCuspPlaceRightInvariance, `SourceGammaOneRationalCuspPlace]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.cuspRealize_mul_left_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_mul_left, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_mul_left, `Dubon2026.SourceFunctionField.GammaOne.cuspFunction_vadd_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_neg, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_neg, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_neg, `Dubon2026.SourceFunctionField.GammaOne.translated_modular_T_zpow, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_mul_T_zpow_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_mul_T_zpow, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_mul_T_zpow, `Dubon2026.SourceFunctionField.GammaOne.exists_T_zpow_or_neg_of_lower_left, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_eq_of_same_rational_cusp, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_eq_of_rational_cusp_orbit]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Cusp Orbit Invariance postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Cusp Orbit Invariance theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CUSP ORBIT INVARIANCE AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
