private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRationalCyclicCusps, `SourceGammaOneRealPeriodDimensionUpper]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CuspCounting.rationalCuspOrbitFromCoset_translation, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.exists_translation_of_same_rational_cusp, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitToRationalCusp_bijective, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.card_rationalCuspOrbit_eq_translation_orbits, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_rational_cusp_card_eq_j_poles, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_real_parabolic_dimension_upper]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitToRationalCusp]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Real Period Upper postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Real Period Upper theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REAL PERIOD UPPER AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
