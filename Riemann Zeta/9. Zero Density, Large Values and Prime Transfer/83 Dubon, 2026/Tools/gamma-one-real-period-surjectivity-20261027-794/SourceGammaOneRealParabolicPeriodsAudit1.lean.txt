private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRealPeriodMap, `SourceGammaOneRealParabolicPeriods]
  let theorems : Array Name := #[`Dubon2026.sourceGamma1RealCharacter_finite, `Dubon2026.sourceGamma1RealPeriodHom_apply, `Dubon2026.sourceGamma1RealPeriodHom_injective, `Dubon2026.sourceGamma1CuspForm_finite_real, `Dubon2026.sourceGamma1_period_eq_zero_of_trace, `Dubon2026.sourceGamma1RealPeriodHom_parabolic, `Dubon2026.sourceGamma1RealParabolicPeriodHom_injective, `Dubon2026.sourceGamma1RealParabolicHoms_finite]
  let constructions : Array Name := #[`Dubon2026.sourceGamma1RealPeriodHom, `Dubon2026.sourceGamma1RealParabolicHoms, `Dubon2026.sourceGamma1RealParabolicPeriodHom]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Real Parabolic Periods postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Real Parabolic Periods theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REAL PARABOLIC PERIODS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
