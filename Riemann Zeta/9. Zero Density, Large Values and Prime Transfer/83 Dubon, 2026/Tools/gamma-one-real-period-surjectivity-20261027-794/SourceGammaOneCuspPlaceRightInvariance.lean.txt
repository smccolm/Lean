Right integral translation rotates the actual meromorphic cusp germ by a

No periodicity of an arbitrary representative is assumed: the actual
translated modular numerator and denominator supply it. -/

theorem translated_modular_T_zpow {k : ℤ}
    (f : ModularForm (Γ : Subgroup (GL (Fin 2) ℝ)) k)
    (δ : SL(2, ℤ)) (n : ℤ) :
    (ModularForm.translate f ((δ * ModularGroup.T ^ n : SL(2, ℤ)) : GL (Fin 2) ℝ) : ℍ → ℂ) =
      fun τ => ModularForm.translate f (δ : GL (Fin 2) ℝ) ((n : ℝ) +ᵥ τ) := by
  funext τ
  change ((f : ℍ → ℂ) ∣[k] (δ * ModularGroup.T ^ n)) τ =
    ((f : ℍ → ℂ) ∣[k] δ) ((n : ℝ) +ᵥ τ)
  rw [SlashAction.slash_mul, ModularForm.SL_slash_apply, modular_T_zpow_smul]
  rw [ModularGroup.denom_apply]
  simp [ModularGroup.coe_T_zpow, -map_zpow]

theorem cuspRealize_mul_T_zpow_eventuallyEq
    (N : ℕ) [NeZero N] (hΓ : CongruenceSubgroup.Gamma N ≤ Γ)
    (hT : ModularGroup.T ∈ Γ) (x : functionFieldOver Γ ℂ)
    (δ : SL(2, ℤ)) (n : ℤ) :
    cuspRealize Γ N (δ * ModularGroup.T ^ n) (x : LaurentSeries ℂ) =ᶠ[𝓝[≠] (0 : ℂ)]
      fun q => cuspRealize Γ N δ (x : LaurentSeries ℂ)
        (Periodic.qParam N (n : ℝ) * q) := by
  obtain ⟨k, g, h, hh, hx⟩ := functionFieldOver_modular_ratio hT x
  have hN : (0 : ℝ) < N := by exact_mod_cast NeZero.pos N
  let ζ := Periodic.qParam N (n : ℝ)
  have hζ : ζ ≠ 0 := Periodic.qParam_ne_zero (h := (N : ℝ)) (n : ℝ)
  have ht : Tendsto (fun q : ℂ => ζ * q) (𝓝[≠] (0 : ℂ)) (𝓝[≠] (0 : ℂ)) := by
    rw [tendsto_nhdsWithin_iff]
    constructor
    · have hc : Continuous (fun q : ℂ => ζ * q) := continuous_const.mul continuous_id
      simpa only [mul_zero] using (hc.tendsto (0 : ℂ)).mono_left nhdsWithin_le_nhds
    · filter_upwards [self_mem_nhdsWithin] with q hq
      exact mul_ne_zero hζ hq
  have hs (f : ModularForm (Γ : Subgroup (GL (Fin 2) ℝ)) k) :
      cuspFunction N (ModularForm.translate f ((δ * ModularGroup.T ^ n : SL(2, ℤ)) : GL (Fin 2) ℝ)) =ᶠ[𝓝[≠] (0 : ℂ)]
        fun q => cuspFunction N (ModularForm.translate f (δ : GL (Fin 2) ℝ)) (ζ * q) := by
    rw [translated_modular_T_zpow]
    exact cuspFunction_vadd_eventuallyEq hN _
      (SlashInvariantFormClass.periodic_comp_ofComplex
        (ModularForm.translate f (δ : GL (Fin 2) ℝ))
        (CuspAuxiliary.natCast_mem_strictPeriods_conj N hΓ δ)) (n : ℝ)
  have he := (cuspRealize_eventuallyEq_of_ratio N hΓ hT g h x hh hx δ).comp_tendsto ht
  filter_upwards [cuspRealize_eventuallyEq_of_ratio N hΓ hT g h x hh hx
    (δ * ModularGroup.T ^ n), he, hs g, hs h] with q hq heq hgq hhq
  change cuspRealize Γ N (δ * ModularGroup.T ^ n) (x : LaurentSeries ℂ) q =
    cuspRealize Γ N δ (x : LaurentSeries ℂ) (ζ * q)
  simp only [Function.comp_apply, Pi.div_apply] at hq heq
  rw [hq, heq]
  exact congrArg₂ (· / ·) hgq hhq

theorem cuspOrder_mul_T_zpow
    (N : ℕ) [NeZero N] (hΓ : CongruenceSubgroup.Gamma N ≤ Γ)
    (hT : ModularGroup.T ∈ Γ) (x : functionFieldOver Γ ℂ)
    (δ : SL(2, ℤ)) (n : ℤ) :
    cuspOrder Γ N (δ * ModularGroup.T ^ n) x = cuspOrder Γ N δ x := by
  let ζ := Periodic.qParam N (n : ℝ)
  have hζ : ζ ≠ 0 := Periodic.qParam_ne_zero (h := (N : ℝ)) (n : ℝ)
  have ha : AnalyticAt ℂ (fun q : ℂ => ζ * q) 0 := analyticAt_const.mul analyticAt_id
  have hd : deriv (fun q : ℂ => ζ * q) 0 ≠ 0 := by simpa using hζ
  unfold cuspOrder
  rw [meromorphicOrderAt_congr (cuspRealize_mul_T_zpow_eventuallyEq N hΓ hT x δ n)]
  simpa only [Function.comp_def, mul_zero] using
    (meromorphicOrderAt_comp_of_deriv_ne_zero
      (f := cuspRealize Γ N δ (x : LaurentSeries ℂ)) ha hd)

theorem cuspPlace_mul_T_zpow
    (N : ℕ) [NeZero N] (hΓ : CongruenceSubgroup.Gamma N ≤ Γ)
    (hT : ModularGroup.T ∈ Γ) (δ : SL(2, ℤ)) (n : ℤ) :
    cuspPlace N hΓ hT (δ * ModularGroup.T ^ n) = cuspPlace N hΓ hT δ := by
  apply Place.ext
  apply SetLike.ext
  intro x
  rw [cuspPlace_mem_iff, cuspPlace_mem_iff, cuspOrder_mul_T_zpow N hΓ hT]

end
