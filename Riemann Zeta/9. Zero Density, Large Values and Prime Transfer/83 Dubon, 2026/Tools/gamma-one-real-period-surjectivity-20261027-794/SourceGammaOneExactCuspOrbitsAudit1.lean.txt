private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneCuspCyclicOrbit, `SourceGammaOneCuspCosetPlaces, `SourceGammaOneCuspCosetBound, `SourceGammaOneCuspPoleDegree, `SourceGammaOneCuspPositiveWeight, `SourceGammaOneCuspOrbitBijection, `SourceGammaOneExactCuspOrbits]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CuspCounting.pow_card_cyclic_orbit, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.card_cyclic_orbit_pos, `Dubon2026.SourceFunctionField.GammaOne.cuspPlace_mul_left_plusMinus, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.translation_zpow_apply, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.cosetPlace_mk, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.cosetPlace_j_neg, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.cosetPlace_translation_zpow, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.exists_cosetPlace_of_j_pole, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitPlace_mk, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitPlace_j_neg, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.exists_orbitPlace_of_j_pole, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.j_pole_order_le_coset_orbit_card, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.sum_cyclic_orbit_cards_eq_index, `Dubon2026.SourceFunctionField.GammaOne.sum_j_pole_orders_eq_finrank, `Dubon2026.SourceFunctionField.GammaOne.sum_j_pole_orders_eq_index, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.bijective_and_eq_of_positive_weight_bound, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitToPole_surjective, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.polePlaces_finite, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitPlace_out, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitToPole_order_le_card, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.gammaOne_orbitToPole_bijective_and_order, `Dubon2026.SourceFunctionField.GammaOne.gammaOneCuspOrbitPlaceMap_bijective, `Dubon2026.SourceFunctionField.GammaOne.gammaOneCuspOrbitPlaceMap_order, `Dubon2026.SourceFunctionField.GammaOne.gammaOne_j_pole_card_eq_translation_orbits]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CuspCounting.Cosets, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.translation, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.cosetPlace, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.Orbits, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitPlace, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.PolePlaces, `Dubon2026.SourceFunctionField.GammaOne.CuspCounting.orbitToPole, `Dubon2026.SourceFunctionField.GammaOne.gammaOneCuspOrbitPlaceMap]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Exact Cusp Orbits postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Exact Cusp Orbits theorems discovered"
  logInfo m!"SOURCE GAMMA ONE EXACT CUSP ORBITS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
