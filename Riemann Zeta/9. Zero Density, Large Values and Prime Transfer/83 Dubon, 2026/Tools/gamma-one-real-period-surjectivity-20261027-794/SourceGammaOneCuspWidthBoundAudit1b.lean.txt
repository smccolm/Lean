private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneRationalCuspOrbit, `SourceGammaOneCuspJExactOrder, `SourceGammaOneCuspRotationOrder, `SourceGammaOneCuspTranslationRamification]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitMk_eq_iff, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitMk_smul, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitFromCoset_surjective, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbit_finite, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspPlace_smul_infty, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspPlace_smul, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitPlace_mk, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitPlace_j_neg, `Dubon2026.SourceFunctionField.GammaOne.exists_rationalCuspOrbitPlace_of_j_pole, `Dubon2026.SourceFunctionField.GammaOne.cuspOrder_jCoordinate_eq_neg_level, `Dubon2026.SourceFunctionField.GammaOne.cuspRamification_mul_j_pole_order, `Dubon2026.SourceFunctionField.GammaOne.rotation_zpow_eq_one_of_meromorphic_order, `Dubon2026.SourceFunctionField.GammaOne.level_dvd_mul_of_qParam_zpow_eq_one, `Dubon2026.SourceFunctionField.GammaOne.cuspRealize_mul_left_plusMinus_eventuallyEq, `Dubon2026.SourceFunctionField.GammaOne.level_dvd_translation_mul_cuspRamification, `Dubon2026.SourceFunctionField.GammaOne.j_pole_order_le_positive_translation]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.RationalCuspOrbit, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitMk, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitFromCoset, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspPlace, `Dubon2026.SourceFunctionField.GammaOne.rationalCuspOrbitPlace]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Gamma One Cusp Width Bound postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Gamma One Cusp Width Bound theorems discovered"
  logInfo m!"SOURCE GAMMA ONE CUSP WIDTH BOUND AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
