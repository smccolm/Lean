open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`Dubon2026.SourceGammaOneGaussCriterion, `Dubon2026.SourceGammaOneGaussValuation, `Dubon2026.SourceGammaOneGaussResidueHom, `Dubon2026.SourceGammaOneGaussResidueKernel, `Dubon2026.SourceGammaOneGaussConstants, `Dubon2026.SourceGammaOneGaussResidueImage, `Dubon2026.SourceGammaOneRegularProlongation, `Dubon2026.SourceGammaOneResidueJCoordinate, `Dubon2026.SourceReductionResidueCoordinateFinite, `Dubon2026.SourceGammaOneResidueCoordinateFinite, `Dubon2026.SourceGammaOneResidueSeparability, `Dubon2026.SourceGammaOneDecompositionGauss, `Dubon2026.SourceGammaOneResidueArithmetic, `Dubon2026.SourceGammaOneHeckeGauss, `Dubon2026.SourceGammaOneResidueHecke, `Dubon2026.SourceGammaOneResidueHeckeFinite, `Dubon2026.SourceGammaOneResidueFrobenius, `Dubon2026.SourceGammaOneResiduePlaces, `Dubon2026.SourceGammaOneResiduePlaceOrders, `Dubon2026.SourceGammaOneResidueFrobeniusPlaces]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then
          logInfo m!"OWNER {m}: {name}"
          match info with
          | .thmInfo th => logInfo m!"FINGERPRINT {name}: {hash th.type} {hash th.value}"
          | _ => throwError "Expected theorem"

#eval inspectOwnership
