private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneGaussCriterion, `SourceGammaOneGaussValuation, `SourceGammaOneGaussResidueHom, `SourceGammaOneGaussResidueKernel, `SourceGammaOneGaussConstants, `SourceGammaOneGaussResidueImage, `SourceGammaOneRegularProlongation, `SourceGammaOneResidueJCoordinate, `SourceReductionResidueCoordinateFinite, `SourceGammaOneResidueCoordinateFinite]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.qExpFunctionField_exists_primitive_fraction, `Dubon2026.SourceFunctionField.GammaOne.qExpFunctionField_isInt_or_inv, `Dubon2026.SourceFunctionField.GammaOne.mem_coefficientGaussRing_iff, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_witness, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.wit_spec, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resL_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_apply, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.ι_mem_gaussRing, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_ι, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_eq_zero_iff, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.coe_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.algebraMap_mem_gaussRing_iff, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_algebraMap, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.coe_smul', `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.exists_smul_mem_gaussRing, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueEmbedding_residue, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_mem_residueImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueToImage_surjective, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueToImage_ker, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.regularProlongation_residue_coe, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.integral_jCoordinate, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.jCoordinateOver_mem_gaussRing, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom_jCoordinateOver, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.jq_mem_residueImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.transcendental_residueJCoordinate, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.regularProlongation_jCoordinateOver, `Dubon2026.SourceCurve.RegularProlongation.residue_coordinate_rank_le, `Dubon2026.SourceCurve.RegularProlongation.residue_coordinate_finite_and_finrank_le, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.finite_residueCoordinate_and_degree_le, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_finite_residueCoordinate_and_degree_le]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.coefficientGaussRing, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.wit, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resL, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.resHom, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueEmbedding, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueToImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.regularProlongation, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueJCoordinate]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE REDUCTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
