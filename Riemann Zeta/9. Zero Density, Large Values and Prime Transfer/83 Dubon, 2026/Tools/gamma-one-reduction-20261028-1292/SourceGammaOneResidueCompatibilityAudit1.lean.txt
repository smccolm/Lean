private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneResidueSeparability, `SourceGammaOneDecompositionGauss, `SourceGammaOneResidueArithmetic, `SourceGammaOneHeckeGauss, `SourceGammaOneResidueHecke, `SourceGammaOneResidueHeckeFinite]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.pow_ne_residueJCoordinate, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.isSeparable_residueCoordinate, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.gammaOne_isSeparable_residueCoordinate, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionGaussHom_coe, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionGaussHom_resHom, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.coeffMap_decomposition_mem_residueImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionGaussHom_residueToImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResidueAut_eq_one_of_inertia, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueToImage_arithmetic_inertia, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueToImage_arithmetic_frobenius_coeff, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.heckeAlpha_mem_gaussRing, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.heckeAlphaGauss_resHom, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.isInt_qExpand, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.heckeBeta_mem_gaussRing, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.heckeBetaGauss_resHom, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueImage_heckeAlpha_le, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueHeckeAlpha_reduction, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.qExpand_mem_residueImage, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueHeckeBeta_reduction, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residue_hecke_hom_finite, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueHeckeAlpha_finite, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueHeckeBeta_finite]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionGaussHom, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResidueAut, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.heckeAlphaGauss, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.heckeBetaGauss, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueHeckeAlpha, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.residueHeckeBeta]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE RESIDUE COMPATIBILITY AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
