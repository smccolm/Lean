import SourceGammaOneGaussCriterion
import SourceGammaOneGaussValuation
import SourceGammaOneGaussResidueHom
import SourceGammaOneGaussResidueKernel
import SourceGammaOneGaussConstants
import SourceGammaOneGaussResidueImage
import SourceGammaOneRegularProlongation
import SourceGammaOneResidueJCoordinate
import SourceReductionResidueCoordinateFinite
import SourceGammaOneResidueCoordinateFinite
import Mathlib.Tactic

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneGaussCriterion, `SourceGammaOneGaussValuation, `SourceGammaOneGaussResidueHom, `SourceGammaOneGaussResidueKernel, `SourceGammaOneGaussConstants, `SourceGammaOneGaussResidueImage, `SourceGammaOneRegularProlongation, `SourceGammaOneResidueJCoordinate, `SourceReductionResidueCoordinateFinite, `SourceGammaOneResidueCoordinateFinite]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
