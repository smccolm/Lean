private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneResiduePlaces, `SourceGammaOneResiduePlaceOrders, `SourceGammaOneResidueFrobeniusPlaces]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResidueAut_constants, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResidueAut_inv, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.mem_decompositionResiduePlace_iff, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResiduePlace_ord, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResiduePlace_ord_action, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResiduePlace_eq_self_of_inertia, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.geometricFrobenius_eq_inverse_decomposition_pow, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.geometricFrobenius_comap_eq, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.geometricFrobeniusPlace_eq_decomposition, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.geometricFrobenius_place_order]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResidueValuationEquiv, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResiduePlace, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.decompositionResiduePlaceEquiv, `Dubon2026.SourceFunctionField.GammaOne.CoefficientReduction.geometricFrobeniusPlace]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Frobenius Normalization postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Frobenius Normalization theorems discovered"
  logInfo m!"SOURCE GAMMA ONE RESIDUE PLACES AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
