/-! DUB-17/18 -> original Gamma1 Tate Frobenius/inertia -> original good
reduction -> literal coefficient valuation. Reuse native primitive scaling
on finite integer coefficient spans. The field generated by integral series
is used only to prove the valuation dichotomy on the actual modular-ratio
field; it does not replace the original Gamma1 curve.
-/
namespace Dubon2026.SourceFunctionField.GammaOne
noncomputable section
open IntermediateField GaussReduction
open scoped MatrixGroups Classical
variable {L : Type*} [Field L] (A : ValuationSubring L)

private theorem integralSeries_finiteCoefficientSpan {x : LaurentSeries L}
    (hx : x ∈ Algebra.adjoin L (Set.range (intSeriesC L))) :
    ∃ M : Submodule ℤ L, M.FG ∧ ∀ n, x.coeff n ∈ M := by
  induction hx using Algebra.adjoin_induction with
  | mem x hx =>
    obtain ⟨p, rfl⟩ := hx
    refine ⟨Submodule.span ℤ {(1 : L)}, Submodule.fg_span (Set.finite_singleton _), fun n => ?_⟩
    have hm : (intSeriesC L p).coeff n =
        (((HahnSeries.ofPowerSeries ℤ ℤ p).coeff n : ℤ) : L) := by
      simp only [intSeriesC, PowerSeries.coeff_coe]
      split_ifs <;> simp [PowerSeries.coeff_map]
    rw [hm, ← zsmul_one]
    exact Submodule.smul_mem _ _ (Submodule.subset_span rfl)
  | algebraMap a =>
    refine ⟨Submodule.span ℤ {a}, Submodule.fg_span (Set.finite_singleton _), fun n => ?_⟩
    rw [SourceDifferential.algebraMap_laurentSeries_apply_alt, HahnSeries.coeff_single]
    split_ifs
    · exact Submodule.subset_span rfl
    · exact Submodule.zero_mem _
  | add x y _ _ hx hy =>
    obtain ⟨M₁, hM₁, h₁⟩ := hx
    obtain ⟨M₂, hM₂, h₂⟩ := hy
    refine ⟨M₁ ⊔ M₂, hM₁.sup hM₂, fun n => ?_⟩
    rw [HahnSeries.coeff_add]
    exact Submodule.add_mem _ (Submodule.mem_sup_left (h₁ n)) (Submodule.mem_sup_right (h₂ n))
  | mul x y _ _ hx hy =>
    obtain ⟨M₁, hM₁, h₁⟩ := hx
    obtain ⟨M₂, hM₂, h₂⟩ := hy
    refine ⟨M₁ * M₂, hM₁.mul hM₂, fun n => ?_⟩
    rw [HahnSeries.coeff_mul]
    exact Submodule.sum_mem _ fun ij _ => Submodule.mul_mem_mul (h₁ _) (h₂ _)

theorem qExpFunctionField_isInt_or_inv
    (Γ : Subgroup (Matrix.SpecialLinearGroup (Fin 2) ℤ))
    {x : LaurentSeries L} (hx : x ∈ qExpFunctionFieldC L Γ) :
    IsInt A x ∨ IsInt A x⁻¹ := by
  have hle : qExpFunctionFieldC L Γ ≤
      IntermediateField.adjoin L (Set.range (intSeriesC L)) := by
    rw [qExpFunctionFieldC, adjoin_le_iff]
    rintro y ⟨k, f, g, pf, pg, _hf, _hg, _hg0, rfl⟩
    exact div_mem (subset_adjoin L _ ⟨pf, rfl⟩) (subset_adjoin L _ ⟨pg, rfl⟩)
  by_cases hx0 : x = 0
  · left
    rw [hx0]
    exact IsInt.zero
  obtain ⟨r, hr, s, hs, hrs⟩ := (mem_adjoin_iff_div (F := L)).mp (hle hx)
  have hr0 : r ≠ 0 := by rintro rfl; exact hx0 (by rw [hrs, zero_div])
  have hs0 : s ≠ 0 := by rintro rfl; exact hx0 (by rw [hrs, div_zero])
  obtain ⟨Mr, hMr, hrM⟩ := integralSeries_finiteCoefficientSpan hr
  obtain ⟨Ms, hMs, hsM⟩ := integralSeries_finiteCoefficientSpan hs
  obtain ⟨cr, hcr, r₀, hr₀, hr'⟩ := exists_primitive_form A hr0 Mr hMr hrM
  obtain ⟨cs, hcs, s₀, hs₀, hs'⟩ := exists_primitive_form A hs0 Ms hMs hsM
  have hιs : ι A s₀ ≠ 0 := by rw [Ne, ι_eq_zero_iff]; exact ϖ_ne_zero_ne_zero hs₀
  have h : x * ι A s₀ = algebraMap L (LaurentSeries L) (cr / cs) * ι A r₀ := by
    rw [hrs, hr', hs', map_div₀]
    field_simp
  have hc0 : cr / cs ≠ 0 := div_ne_zero hcr hcs
  rcases A.mem_or_inv_mem (cr / cs) with hc | hc
  · left
    refine ⟨HahnSeries.C ⟨cr / cs, hc⟩ * r₀, s₀, hs₀, ?_⟩
    rw [map_mul, ι_C]
    exact h
  · right
    refine ⟨HahnSeries.C ⟨(cr / cs)⁻¹, hc⟩ * s₀, r₀, hr₀, ?_⟩
    have hcC : algebraMap L (LaurentSeries L) (cr / cs) ≠ 0 :=
      (_root_.map_ne_zero _).mpr hc0
    rw [map_mul, ι_C]
    calc x⁻¹ * ι A r₀ = x⁻¹ * (algebraMap L (LaurentSeries L) (cr / cs))⁻¹ *
        (x * ι A s₀) := by rw [h]; field_simp
      _ = algebraMap L (LaurentSeries L) ((cr / cs)⁻¹) * ι A s₀ := by
        rw [map_inv₀]
        field_simp

end
