noncomputable section
variable {L : Type*} [Field L] [Algebra ℚ L] (A : ValuationSubring L)
  (Γ : Subgroup (Matrix.SpecialLinearGroup (Fin 2) ℤ)) [Γ.FiniteIndex]
local notation "FF" => functionFieldOver Γ L
local notation "kk" => IsLocalRing.ResidueField A
local notation "Fb" => residueImage A Γ

theorem finite_residueCoordinate_and_degree_le (hT : ModularGroup.T ∈ Γ) :
    FiniteDimensional (IntermediateField.adjoin kk ({residueJCoordinate A Γ} : Set Fb)) Fb ∧
      Module.finrank (IntermediateField.adjoin kk ({residueJCoordinate A Γ} : Set Fb)) Fb ≤
        Module.finrank (IntermediateField.adjoin L ({jCoordinateOver Γ L} : Set FF)) FF := by
  letI := finiteDimensional_jCoordinateOver hT L
  have hx : Transcendental kk
      ((regularProlongation A Γ).residue
        ⟨jCoordinateOver Γ L, jCoordinateOver_mem_gaussRing A Γ⟩) := by
    rw [regularProlongation_jCoordinateOver]
    exact transcendental_residueJCoordinate A Γ
  have h := SourceCurve.RegularProlongation.residue_coordinate_finite_and_finrank_le
    (regularProlongation A Γ)
    ⟨jCoordinateOver Γ L, jCoordinateOver_mem_gaussRing A Γ⟩ hx
  simpa only [regularProlongation_jCoordinateOver] using h

end
