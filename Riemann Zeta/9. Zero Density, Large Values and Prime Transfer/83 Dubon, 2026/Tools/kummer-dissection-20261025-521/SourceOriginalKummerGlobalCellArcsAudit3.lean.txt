import SourceOriginalWeilCoverLiftedArcEquality
import SourceOriginalWeilCoverLiftedArcMatching
import SourceOriginalWeilCoverGlobalCellCoordinates
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.lifted_boundary_arcs_eq_of_endpoint
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.existsUnique_lifted_boundary_arc
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_global_cell_coordinates

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalWeilCoverLiftedArcEquality, `SourceOriginalWeilCoverLiftedArcMatching, `SourceOriginalWeilCoverGlobalCellCoordinates]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.OriginalWeilCoverChart.lifted_boundary_arcs_eq_of_endpoint, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.existsUnique_lifted_boundary_arc, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_global_cell_coordinates]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Global Cell Arcs postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Global Cell Arcs theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER GLOBAL CELL ARCS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceOriginalWeilCoverLiftedArcEquality
#lint in SourceOriginalWeilCoverLiftedArcMatching
