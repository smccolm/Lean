private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalCellPairingLiftIndices, `SourceOriginalWeilCoverLiftedArcPairing, `SourceOriginalWeilCoverGlobalArcPairing]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedSheetSwap_involutive, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair_involutive, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair_base, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair_orientation, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedSignedEquiv_apply_positive, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair_positive, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_lifted_pairing_transports, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_global_paired_coordinates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_global_paired_coordinates]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedSignedEquiv, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedSheetSwap, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverGlobalPairedCoordinates]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Global Arc Pairing postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Global Arc Pairing theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER GLOBAL ARC PAIRING AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
