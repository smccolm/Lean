private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalLiftedVertexIncidence, `SourceCoverLiftedMarkedCells, `SourceOriginalLiftedDissectionIncidence, `SourceOriginalWeilCoverCenteredDissection]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.CellPairing.lifted_vertex_incident, `Dubon2026.SourceFunctionField.SheetCover.lifted_carrier_injOn, `Dubon2026.SourceFunctionField.SheetCover.lifted_cell_center, `Dubon2026.SourceFunctionField.SheetCover.lifted_marked_separation, `Dubon2026.SourceFunctionField.CellGeometry.OriginalGlobalCellCover.lifted_dissection_vertex_incidence, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_centered_cell_dissection]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialCell.interior']
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Centered Dissection postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Centered Dissection theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER CENTERED DISSECTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
