import SourceCurveFiniteCoordinateRegularGenus
import SourceCurveFiniteCoordinateUnramifiedGenus
import SourceOriginalWeilCoverUnramifiedGenus
import Mathlib.Tactic

open Lean

#print axioms Dubon2026.SourceCurve.finite_coordinate_finrank_regular_eq_genusFF
#print axioms Dubon2026.SourceCurve.finite_coordinate_canonical_degree_eq_regular_dimension
#print axioms Dubon2026.SourceCurve.finite_coordinate_euler_genus_of_unramified
#print axioms Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_euler_genus_multiplicity

private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceCurveFiniteCoordinateRegularGenus, `SourceCurveFiniteCoordinateUnramifiedGenus, `SourceOriginalWeilCoverUnramifiedGenus]
  let theorems : Array Name := #[`Dubon2026.SourceCurve.finite_coordinate_finrank_regular_eq_genusFF, `Dubon2026.SourceCurve.finite_coordinate_canonical_degree_eq_regular_dimension, `Dubon2026.SourceCurve.finite_coordinate_euler_genus_of_unramified, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_euler_genus_multiplicity]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Unramified Genus postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Unramified Genus theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER UNRAMIFIED GENUS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit

#lint in SourceCurveFiniteCoordinateRegularGenus
#lint in SourceCurveFiniteCoordinateUnramifiedGenus
#lint in SourceOriginalWeilCoverUnramifiedGenus
