private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceOriginalLiftedVertexFibers, `SourceFiniteFiberPreimageCard, `SourceOriginalWeilCoverLiftedVertexCount, `SourceOriginalLiftedEulerMultiplicity, `SourceOriginalWeilCoverLiftedEuler, `SourceOriginalLiftedMarkedEulerAssembly, `SourceOriginalWeilCoverEulerDissectionData, `SourceOriginalWeilCoverEulerDissection]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.CellPairing.lifted_endpoint_projects, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.lifted_vertices_eq_preimage, `Dubon2026.SourceFunctionField.SheetCover.card_preimage_of_constant_fibers, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_lifted_vertex_count, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.lifted_euler_multiplicity, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_lifted_euler_identity, `Dubon2026.SourceFunctionField.CellGeometry.OriginalGlobalCellCover.exists_lifted_marked_euler_dissection, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_euler_dissection_of_centered_base, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_euler_cell_dissection]
  let constructions : Array Name := #[]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Euler Dissection postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Euler Dissection theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER EULER DISSECTION AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
