private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceRadialReversedArcImages, `SourceOriginalLiftedPairArcImages, `SourceCoverLiftedCellIncidence, `SourceOriginalCoverLiftedIntersectionChains, `SourceOriginalWeilCoverGlobalIntersection]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.RadialRegion.reversed_arc_images, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair_arc_images, `Dubon2026.SourceFunctionField.CellGeometry.CellPairing.liftedPair_arc_images_of_reversal, `Dubon2026.SourceFunctionField.SheetCover.lifted_carrier_projects, `Dubon2026.SourceFunctionField.SheetCover.lifted_arc_iff_over_arc, `Dubon2026.SourceFunctionField.SheetCover.lifted_carrier_unique, `Dubon2026.SourceFunctionField.CellGeometry.OriginalGlobalCellCover.lifted_intersection_chain, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.cover_global_intersection_coordinates, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.exists_cover_global_intersection_coordinates]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.CellGeometry.OriginalGlobalCellCover.LiftedPointStep, `Dubon2026.SourceFunctionField.OriginalWeilCoverChart.CoverGlobalIntersectionCoordinates]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Kummer Intersection Chains postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Kummer Intersection Chains theorems discovered"
  logInfo m!"SOURCE ORIGINAL KUMMER INTERSECTION CHAINS AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
