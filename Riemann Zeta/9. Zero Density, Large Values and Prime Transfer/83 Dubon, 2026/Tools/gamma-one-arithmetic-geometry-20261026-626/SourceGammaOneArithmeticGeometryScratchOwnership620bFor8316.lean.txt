import SourceGammaOneFunctionField
import SourceGammaOneJCoordinate
import SourceGammaOneRatioClosure
import SourceGammaOneNormRelation
import SourceGammaOneComplexRelation
import SourceGammaOneComplexRatio
import SourceGammaOneRationalRelation
import SourceGammaOneFunctionFieldFinite
import SourceGammaOneFunctionFieldBaseChange
import SourceGammaOneCurveFinite
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneFunctionField, `SourceGammaOneJCoordinate, `SourceGammaOneRatioClosure, `SourceGammaOneNormRelation, `SourceGammaOneComplexRelation, `SourceGammaOneComplexRatio, `SourceGammaOneRationalRelation, `SourceGammaOneFunctionFieldFinite, `SourceGammaOneFunctionFieldBaseChange, `SourceGammaOneCurveFinite]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
