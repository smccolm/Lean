private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneFieldCoefficients, `SourceGammaOneHeckeFieldNaturality, `SourceGammaOneArithmeticField, `SourceGammaOneArithmeticPlaces, `SourceGammaOneArithmeticDivisors, `SourceGammaOneArithmeticPicard, `SourceGammaOneHeckeArithmeticPlaces, `SourceGammaOneHeckeArithmeticDivisors, `SourceGammaOneHeckeArithmetic]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.coeffMap_mem_functionFieldOver, `Dubon2026.SourceFunctionField.GammaOne.fieldCoefficientMap_coe, `Dubon2026.SourceFunctionField.GammaOne.fieldCoefficientMap_constants, `Dubon2026.SourceFunctionField.GammaOne.fieldCoefficientMap_j, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha_coefficientMap, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_coefficientMap, `Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut_constants, `Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut_j, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_ord, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_ord_action, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver_residue_bijective, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisorEquiv_apply, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisor_degree, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisor_principal, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_one, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_mul, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisor_one, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisor_mul, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPicard_mk, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPicard_torsion, `Dubon2026.SourceFunctionField.GammaOne.mem_arithmeticPlace_iff, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_restrictAlong, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace_ramificationAlong, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisorEquiv_single, `Dubon2026.SourceFunctionField.GammaOne.finitePullbackAlong_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.geometricPushforwardAlong_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.heckeDivisor_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.heckePicard_arithmetic, `Dubon2026.SourceFunctionField.GammaOne.gamma1ArithmeticPicardRep_torsion]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.fieldCoefficientMap, `Dubon2026.SourceFunctionField.GammaOne.arithmeticRingAut, `Dubon2026.SourceFunctionField.GammaOne.arithmeticValuationEquiv, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlace, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPlaceEquiv, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDivisorEquiv, `Dubon2026.SourceFunctionField.GammaOne.arithmeticDegZero, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPicard, `Dubon2026.SourceFunctionField.GammaOne.arithmeticPicardRep, `Dubon2026.SourceFunctionField.GammaOne.gamma1ArithmeticPicardRep]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE ARITHMETIC AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
