import SourceGammaOneFieldCoefficients
import SourceGammaOneHeckeFieldNaturality
import SourceGammaOneArithmeticField
import SourceGammaOneArithmeticPlaces
import SourceGammaOneArithmeticDivisors
import SourceGammaOneArithmeticPicard
import SourceGammaOneHeckeArithmeticPlaces
import SourceGammaOneHeckeArithmeticDivisors
import SourceGammaOneHeckeArithmetic
import Lean

open Lean

private def inspectOwnership : CoreM Unit := do
  let env ← getEnv
  let mods : Array Name := #[`SourceGammaOneFieldCoefficients, `SourceGammaOneHeckeFieldNaturality, `SourceGammaOneArithmeticField, `SourceGammaOneArithmeticPlaces, `SourceGammaOneArithmeticDivisors, `SourceGammaOneArithmeticPicard, `SourceGammaOneHeckeArithmeticPlaces, `SourceGammaOneHeckeArithmeticDivisors, `SourceGammaOneHeckeArithmetic]
  for (name, info) in env.constants.toList do
    if info.isTheorem then
      if let some idx := env.getModuleIdxFor? name then
        let m := env.allImportedModuleNames[idx.toNat]!
        if mods.contains m then logInfo m!"OWNER {m}: {name}"

#eval inspectOwnership
