variable {𝒢 : Subgroup (GL (Fin 2) ℝ)} [𝒢.IsArithmetic] [𝒢.HasDetOne]

theorem qExpansion_Aform_top_ne_zero {w : ℤ} (f g : ModularForm 𝒢 w)
    (hg : g ≠ 0) :
    qExpansion 1 (Aform f g (Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ)) : ℍ → ℂ) ≠ 0 := by
  classical
  letI : Fintype (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ) := Fintype.ofFinite _
  have hc : (Finset.univ : Finset (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ)).card =
      Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ) := by
    rw [Finset.card_univ, Nat.card_eq_fintype_card]
  have htop : Aform f g (Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ)) =
      ((-1 : ℂ) ^ Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ)) • ModularForm.norm 𝒮ℒ g := by
    ext τ
    rw [Aform_apply, ModularForm.IsGLPos.smul_apply, norm_apply_eq, smul_eq_mul]
    unfold Pp
    conv_lhs => rw [← hc]
    rw [coeff_card_prod_linear, Finset.prod_neg, hc]
  have hn : ModularForm.norm 𝒮ℒ g ≠ 0 :=
    ModularForm.norm_ne_zero 𝒮ℒ (by
      intro hzero
      exact hg (DFunLike.coe_injective (by simpa using hzero)))
  intro hz
  rw [ModularForm.qExpansion_eq_zero_iff one_pos one_mem_strictPeriods_SL] at hz
  rw [htop] at hz
  exact (smul_ne_zero (pow_ne_zero _ (neg_ne_zero.mpr one_ne_zero)) hn) hz

theorem norm_qExpansion_relation {K : ℕ}
    (h1 : (1 : ℝ) ∈ 𝒢.strictPeriods)
    (f g : ModularForm 𝒢 (12 * (K : ℤ))) :
    ∑ r : Fin (Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ) + 1),
      qExpansion 1 (Aform f g (r : ℕ) : ℍ → ℂ) *
        (qExpansion 1 (f : ℍ → ℂ)) ^ (r : ℕ) *
        (qExpansion 1 (g : ℍ → ℂ)) ^
          (Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ) - (r : ℕ)) = 0 := by
  classical
  letI : Fintype (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ) := Fintype.ofFinite _
  rw [← qExpansion_Sform (one_mem_strictPeriods_inf h1),
    Sform_eq_zero, ModularForm.coe_zero, qExpansion_zero]

theorem norm_coefficient_j_polynomial {K : ℕ}
    (f g : ModularForm 𝒢 (12 * (K : ℤ))) (r : ℕ) :
    ∃ P : Polynomial ℂ,
      P.natDegree ≤ K * Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ) ∧
      (qExpansion 1 (Aform f g r : ℍ → ℂ) : LaurentSeries ℂ) =
        Polynomial.aeval (jqModC ℂ) P *
          (qExpansion 1 ModularForm.discriminant : LaurentSeries ℂ) ^
            (K * Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ)) := by
  obtain ⟨P, hdeg, heq⟩ := exists_levelOne_j_polynomial
    (K * Nat.card (𝒮ℒ ⧸ 𝒢.subgroupOf 𝒮ℒ))
    (ModularForm.mcast (by push_cast; ring) (Aform f g r))
  refine ⟨P, hdeg, ?_⟩
  rw [ModularForm.qExpansion_mcast, ← jqModC_eq_modular_quotient] at heq
  exact heq

end
