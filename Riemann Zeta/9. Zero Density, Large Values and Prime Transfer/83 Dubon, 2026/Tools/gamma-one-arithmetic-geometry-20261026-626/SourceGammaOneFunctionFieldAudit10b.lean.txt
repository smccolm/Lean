private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneFunctionField, `SourceGammaOneJCoordinate, `SourceGammaOneRatioClosure, `SourceGammaOneNormRelation, `SourceGammaOneComplexRelation, `SourceGammaOneComplexRatio, `SourceGammaOneRationalRelation, `SourceGammaOneFunctionFieldFinite, `SourceGammaOneFunctionFieldBaseChange, `SourceGammaOneCurveFinite]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.intSeriesC_zero, `Dubon2026.SourceFunctionField.GammaOne.intSeriesC_mul, `Dubon2026.SourceFunctionField.GammaOne.mem_intFormRatiosC, `Dubon2026.SourceFunctionField.GammaOne.intFormRatiosC_subset, `Dubon2026.SourceFunctionField.GammaOne.div_mem_qExpFunctionFieldC, `Dubon2026.SourceFunctionField.GammaOne.intFormRatiosC_mono, `Dubon2026.SourceFunctionField.GammaOne.qExpFunctionFieldC_mono, `Dubon2026.SourceFunctionField.GammaOne.x1FunctionFieldC_le_x1x0, `Dubon2026.SourceFunctionField.GammaOne.coeffEmb_intSeries, `Dubon2026.SourceFunctionField.GammaOne.jq_eq_integral_modular_ratio, `Dubon2026.SourceFunctionField.GammaOne.jq_mem_qExpFunctionField, `Dubon2026.SourceFunctionField.GammaOne.transcendental_jCoordinate, `Dubon2026.SourceFunctionField.GammaOne.hper, `Dubon2026.SourceFunctionField.GammaOne.mem_qExpFunctionFieldC_iff, `Dubon2026.SourceFunctionField.GammaOne.qExpansion_Aform_top_ne_zero, `Dubon2026.SourceFunctionField.GammaOne.norm_qExpansion_relation, `Dubon2026.SourceFunctionField.GammaOne.norm_coefficient_j_polynomial, `Dubon2026.SourceFunctionField.GammaOne.exists_complex_j_relation, `Dubon2026.SourceFunctionField.GammaOne.exists_complex_j_relation_of_ratio, `Dubon2026.SourceFunctionField.GammaOne.exists_rational_j_relation, `Dubon2026.SourceFunctionField.GammaOne.finiteDimensional_jCoordinate, `Dubon2026.SourceFunctionField.GammaOne.finite_rational_coordinate, `Dubon2026.SourceFunctionField.GammaOne.transcendental_jCoordinateOver, `Dubon2026.SourceFunctionField.GammaOne.finiteDimensional_jCoordinateOver, `Dubon2026.SourceFunctionField.GammaOne.gamma1_finite_coordinate, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_finite_coordinate, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_baseChange_le]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.IsIntegralQExp, `Dubon2026.SourceFunctionField.GammaOne.intSeriesC, `Dubon2026.SourceFunctionField.GammaOne.intFormRatiosC, `Dubon2026.SourceFunctionField.GammaOne.qExpFunctionFieldC, `Dubon2026.SourceFunctionField.GammaOne.x1FunctionFieldC, `Dubon2026.SourceFunctionField.GammaOne.x1FunctionField, `Dubon2026.SourceFunctionField.GammaOne.x1x0FunctionFieldC, `Dubon2026.SourceFunctionField.GammaOne.jCoordinate, `Dubon2026.SourceFunctionField.GammaOne.functionFieldOver, `Dubon2026.SourceFunctionField.GammaOne.jCoordinateOver]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE FUNCTION FIELD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
