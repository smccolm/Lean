private def runPlanarPairingAudit : CoreM Unit := do
  let env ← getEnv
  let allowed : Array Name := #[``propext, ``Classical.choice, ``Quot.sound]
  let modules : Array Name := #[`SourceGammaOneModularStretch, `SourceGammaOneModularStretchCoefficients, `SourceGammaOneHeckeFieldInclusion, `SourceGammaOneHeckeDegeneracy, `SourceGammaOneHeckeFieldFinite]
  let theorems : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.levelRaiseConj_mem_Gamma1, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_le_conj, `Dubon2026.SourceFunctionField.GammaOne.modularStretch_apply, `Dubon2026.SourceFunctionField.GammaOne.modularStretch_coeff, `Dubon2026.SourceFunctionField.GammaOne.modularStretch_integral, `Dubon2026.SourceFunctionField.GammaOne.intSeriesC_expand, `Dubon2026.SourceFunctionField.GammaOne.qExpand_intFormRatiosC, `Dubon2026.SourceFunctionField.GammaOne.gamma1_degeneracy_map_le, `Dubon2026.SourceFunctionField.GammaOne.gamma1_over_degeneracy_map_le, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha_coe, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_coe, `Dubon2026.SourceFunctionField.GammaOne.gamma1_hecke_hom_finite, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha_finite, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta_finite]
  let constructions : Array Name := #[`Dubon2026.SourceFunctionField.GammaOne.modularStretch, `Dubon2026.SourceFunctionField.GammaOne.heckeAlpha, `Dubon2026.SourceFunctionField.GammaOne.heckeBeta]
  for name in theorems do
    match env.find? name with
    | none => throwError "Missing theorem: {name}"
    | some info => if !info.isTheorem then throwError "Incorrect theorem kind: {name}"
  for name in constructions do
    match env.find? name with
    | none => throwError "Missing construction: {name}"
    | some info =>
      if info.isTheorem || info.isAxiom then throwError "Incorrect construction kind: {name}"
      let axioms ← Lean.collectAxioms name
      if !(axioms.filter fun a => !allowed.contains a).isEmpty then
        throwError "Forbidden construction dependencies: {name}: {axioms}"
  let mut count := 0
  for (name, info) in env.constants.toList do
    if let some idx := env.getModuleIdxFor? name then
      if modules.contains env.allImportedModuleNames[idx.toNat]! then
        if info.isAxiom then throwError "Original Primitive Tate Restriction postulate: {name}"
        if info.isTheorem then
          let axioms ← Lean.collectAxioms name
          if !(axioms.filter fun a => !allowed.contains a).isEmpty then
            throwError "Forbidden dependencies: {name}: {axioms}"
          count := count + 1
  if count == 0 then throwError "No Original Primitive Tate Restriction theorems discovered"
  logInfo m!"SOURCE GAMMA ONE HECKE FIELD AUDIT PASS: {count} theorems, {theorems.size} registered consumers and {constructions.size} constructions; standard logical axioms only."

#eval runPlanarPairingAudit
